## 第一步：实现 heap_get_stats

`heap_get_stats` 函数可以获取堆的统计信息，帮助分析内存使用情况。这在调试内存泄漏、分析内存分配模式时非常有用。

在 `heap.c` 中添加（如果还没有实现的话）：

```c
/**
 * heap_get_stats - Get heap statistics
 *
 * @param stats Pointer to stats structure to fill
 * @return HEAP_OK on success, error code on failure
 */
heap_result_t heap_get_stats(heap_stats_t* stats) {
    spinlock_flags_t flags;
    spin_lock_irqsave(&s_heap_lock, &flags);

    if (!s_heap.initialized) {
        spin_unlock_irqrestore(&s_heap_lock, flags);
        return HEAP_ERR_NOT_INIT;
    }

    if (stats == NULL) {
        spin_unlock_irqrestore(&s_heap_lock, flags);
        return HEAP_ERR_INVALID;
    }

    /* Copy basic cached statistics */
    stats->alloc_count = s_heap.stats.alloc_count;
    stats->free_count = s_heap.stats.free_count;
    stats->expand_count = s_heap.stats.expand_count;

    /* Recalculate accurate counts by walking the heap */
    stats->total_blocks = 0;
    stats->used_blocks = 0;
    stats->free_blocks = 0;
    stats->used_bytes = 0;
    stats->free_bytes = 0;
    stats->total_bytes = 0;

    heap_block_t* block = (heap_block_t*)s_heap.heap_start;
    while ((virtual_addr_t)block < s_heap.heap_brk) {
        if (validate_block(block)) {
            size_t user_bytes = block->size - sizeof(heap_block_t);
            stats->total_blocks++;
            stats->total_bytes += user_bytes;

            if (block->used) {
                stats->used_blocks++;
                stats->used_bytes += user_bytes;
            } else {
                stats->free_blocks++;
                stats->free_bytes += user_bytes;
            }
        }
        block = next_block(block);
    }

    spin_unlock_irqrestore(&s_heap_lock, flags);
    return HEAP_OK;
}
```

`heap_get_stats` 的实现有个值得注意的地方。我们维护了一些缓存的统计信息（比如 `alloc_count`、`free_count`），但块数量和字节数是通过遍历堆实时计算的。这是因为块的状态会频繁变化（分配、释放、合并、分割），如果每次都更新统计信息，会有很大的开销。而遍历堆只在需要统计时才做，是个更好的权衡。

遍历堆的逻辑是：从堆起始地址开始，通过 `next_block` 函数找到下一个块，直到达到堆顶。对于每个块，我们验证它的有效性，然后根据使用状态更新统计信息。用户数据大小是块大小减去头部大小。

## 第二步：实现 heap_dump

`heap_dump` 函数可以转储整个堆的状态，打印每个块的详细信息。这在调试内存问题时非常有用，可以直观地看到堆的布局。

在 `heap.c` 中添加（如果还没有实现的话）：

```c
/**
 * heap_dump - Dump heap state for debugging
 *
 * Prints current heap status and block information.
 */
void heap_dump(void) {
    spinlock_flags_t flags;
    spin_lock_irqsave(&s_heap_lock, &flags);

    if (!s_heap.initialized) {
        spin_unlock_irqrestore(&s_heap_lock, flags);
        klog_error("[HEAP] Not initialized\n");
        return;
    }

    klog_info("[HEAP] Heap State:\n");
    klog_info("[HEAP]   Region:  0x%llX - 0x%llX\n",
              s_heap.heap_start, s_heap.heap_end);
    klog_info("[HEAP]   Break:   0x%llX\n", s_heap.heap_brk);
    klog_info("[HEAP]   Total:   %llu bytes (%llu MB)\n",
              s_heap.stats.total_bytes,
              s_heap.stats.total_bytes / (1024 * 1024));
    klog_info("[HEAP]   Used:    %llu bytes (%.1f%%)\n",
              s_heap.stats.used_bytes,
              s_heap.stats.total_bytes > 0
                  ? (100.0 * s_heap.stats.used_bytes / s_heap.stats.total_bytes)
                  : 0.0);
    klog_info("[HEAP]   Free:    %llu bytes\n", s_heap.stats.free_bytes);
    klog_info("[HEAP]   Blocks:  %u used, %u free\n",
              s_heap.stats.used_blocks, s_heap.stats.free_blocks);
    klog_info("[HEAP]   Ops:     %lu allocs, %lu frees, %lu expansions\n",
              s_heap.stats.alloc_count, s_heap.stats.free_count,
              s_heap.stats.expand_count);

    /* Dump free list */
    klog_info("[HEAP] Free List:\n");
    heap_block_t* curr = s_heap.free_list;
    int count = 0;
    while (curr && count < 16) {  /* Limit to first 16 entries */
        klog_info("[HEAP]   [%d] 0x%llX: %llu bytes\n",
                  count++, (virtual_addr_t)curr, curr->size);
        curr = curr->next;
    }
    if (curr) {
        klog_info("[HEAP]   ... and more\n");
    }

    spin_unlock_irqrestore(&s_heap_lock, flags);
}
```

`heap_dump` 打印堆的基本信息，包括区域范围、堆顶位置、总字节数、已用字节数、空闲字节数、块数量、操作次数等。然后打印空闲链表的前 16 个块，限制输出数量避免刷屏。

这个函数在调试时非常有用。比如你怀疑内存泄漏，可以调用 `heap_dump` 查看已用块的数量和大小。比如你想确认堆是否正确扩展，可以查看总字节数和扩展次数。

现在我们来创建一个完整的演示程序，包含各种测试用例。这个演示程序会测试堆分配器的各种功能，验证正确性。

创建 `kernel/demo/heap/heap_demo.c`：

```c
/**
 * @file heap_demo.c
 * @brief Heap Allocator Demo - Demonstrates kmalloc/kfree functionality
 */

#include "heap_demo.h"
#include "mm/heap/heap.h"
#include "base/string.h"
#include "base/memory.h"
#include "klogs/kprintf.h"

/* ============================================================================
 * Test Result Tracking
 * ============================================================================ */

static int tests_passed = 0;
static int tests_failed = 0;

#define TEST_ASSERT(condition, message) \
    do { \
        if (!(condition)) { \
            klog_error("[HEAP_DEMO] FAILED: %s\n", message); \
            tests_failed++; \
            return false; \
        } \
    } while(0)

#define TEST_START(name) \
    klog_info("[HEAP_DEMO] Test: %s...\n", name)

#define TEST_PASS() \
    do { \
        klog_info("[HEAP_DEMO] PASSED\n"); \
        tests_passed++; \
        return true; \
    } while(0)

/* ============================================================================
 * Test 1: Basic Allocation and Deallocation
 * ============================================================================ */

static bool test_basic_alloc_free(void) {
    TEST_START("Basic Alloc/Free");

    void* ptr = kmalloc(64);
    TEST_ASSERT(ptr != NULL, "kmalloc(64) returned NULL");

    /* Write to ensure memory is writable */
    memset(ptr, 0xAA, 64);
    uint8_t* bytes = (uint8_t*)ptr;
    TEST_ASSERT(bytes[0] == 0xAA && bytes[63] == 0xAA,
                "Memory content check failed");

    kfree(ptr);

    TEST_PASS();
}

/* ============================================================================
 * Test 2: Multiple Allocations
 * ============================================================================ */

static bool test_multiple_allocations(void) {
    TEST_START("Multiple Allocations");

    void* ptrs[10];
    size_t sizes[] = {16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192};

    for (int i = 0; i < 10; i++) {
        ptrs[i] = kmalloc(sizes[i]);
        TEST_ASSERT(ptrs[i] != NULL, "kmalloc failed");
        memset(ptrs[i], i, sizes[i]);
    }

    /* Verify all pointers are unique */
    for (int i = 0; i < 10; i++) {
        for (int j = i + 1; j < 10; j++) {
            TEST_ASSERT(ptrs[i] != ptrs[j], "Duplicate pointer returned");
        }
    }

    /* Free all */
    for (int i = 0; i < 10; i++) {
        kfree(ptrs[i]);
    }

    TEST_PASS();
}

/* ============================================================================
 * Test 3: Zero Size Allocation
 * ============================================================================ */

static bool test_zero_size(void) {
    TEST_START("Zero Size Allocation");

    void* ptr = kmalloc(0);
    TEST_ASSERT(ptr == NULL, "kmalloc(0) should return NULL");

    kfree(NULL);  /* Should be safe */

    TEST_PASS();
}

/* ============================================================================
 * Test 4: Large Allocation (triggers expansion)
 * ============================================================================ */

static bool test_large_allocation(void) {
    TEST_START("Large Allocation (Expansion)");

    heap_stats_t stats_before, stats_after;
    heap_get_stats(&stats_before);

    /* Allocate more than initial 64KB */
    void* ptr = kmalloc(128 * 1024);  /* 128KB */
    TEST_ASSERT(ptr != NULL, "kmalloc(128KB) failed");

    memset(ptr, 0xBB, 128 * 1024);

    heap_get_stats(&stats_after);
    TEST_ASSERT(stats_after.total_bytes >= stats_before.total_bytes,
                "Heap should have expanded");

    kfree(ptr);

    TEST_PASS();
}

/* ============================================================================
 * Test 5: Aligned Allocation
 * ============================================================================ */

static bool test_aligned_allocation(void) {
    TEST_START("Aligned Allocation");

    void* ptr16 = kmalloc_aligned(64, 16);
    TEST_ASSERT(ptr16 != NULL, "kmalloc_aligned failed");
    TEST_ASSERT(((virtual_addr_t)ptr16 & 0xF) == 0, "16-byte alignment failed");
    kfree(ptr16);

    void* ptr64 = kmalloc_aligned(128, 64);
    TEST_ASSERT(ptr64 != NULL, "kmalloc_aligned(64) failed");
    TEST_ASSERT(((virtual_addr_t)ptr64 & 0x3F) == 0, "64-byte alignment failed");
    kfree(ptr64);

    void* ptr4096 = kmalloc_aligned(256, 4096);
    TEST_ASSERT(ptr4096 != NULL, "kmalloc_aligned(4096) failed");
    TEST_ASSERT(((virtual_addr_t)ptr4096 & 0xFFF) == 0,
                "4096-byte alignment failed");
    kfree(ptr4096);

    TEST_PASS();
}

/* ============================================================================
 * Demo Runner
 * ============================================================================ */

int heap_run_demo(void) {
    klog_info("[HEAP_DEMO] ===============================================\n");
    klog_info("[HEAP_DEMO] Heap Allocator Demo Starting\n");
    klog_info("[HEAP_DEMO] ===============================================\n");

    tests_passed = 0;
    tests_failed = 0;

    /* Run all tests */
    test_basic_alloc_free();
    test_multiple_allocations();
    test_zero_size();
    test_large_allocation();
    test_aligned_allocation();

    /* Print summary */
    klog_info("[HEAP_DEMO] ===============================================\n");
    klog_info("[HEAP_DEMO] Demo Complete: %d passed, %d failed\n",
              tests_passed, tests_failed);
    klog_info("[HEAP_DEMO] ===============================================\n");

    /* Print final heap stats */
    heap_dump();

    return (tests_failed == 0) ? 0 : -1;
}

void heap_stop_demo(void) {
    klog_info("[HEAP_DEMO] Demo stopped\n");
}
```

这个演示程序包含 5 个测试用例，覆盖了堆分配器的主要功能。每个测试用例都有明确的开始和结束，使用宏来简化测试逻辑。`TEST_ASSERT` 检查条件，如果失败就记录错误并返回。`TEST_START` 和 `TEST_PASS` 标记测试的开始和通过。

# ============================================================================

# Check if demo is enabled
option(ENABLE_HEAP_DEMO "Enable heap allocator demo" OFF)

if(ENABLE_HEAP_DEMO)
    add_library(heap_demo STATIC

    )

    target_include_directories(heap_demo PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../..

    target_compile_options(heap_demo PUBLIC ${CCOS_FREESTANDING_FLAGS})

    if(CCOS_ENABLE_FRAME_POINTER)
        target_compile_options(heap_demo PUBLIC ${CCOS_FRAME_POINTER_FLAGS})
    endif()

        kernel_heap
        klogs
        base

    set_target_properties(heap_demo PROPERTIES
        POSITION_INDEPENDENT_CODE FALSE

注意这里我们使用了一个 CMake 选项 `ENABLE_HEAP_DEMO`，默认是关闭的。如果需要启用演示，可以在配置时传入 `-DENABLE_HEAP_DEMO=ON`。

## 第六步：调试技巧

在开发堆分配器的过程中，有一些调试技巧非常有用。

首先，充分利用 `heap_dump` 函数。当你怀疑堆的状态有问题时，调用 `heap_dump` 查看堆的布局。比如你怀疑内存泄漏，可以查看已用块的数量和大小；比如你想确认块合并是否正常，可以查看空闲块的数量和大小。

其次，使用魔数检测。魔数能帮你快速发现内存损坏的问题。如果某个操作后魔数不对，说明这个操作有问题。你可以用 GDB 设置断点，在魔数被破坏的地方停下。

第三，使用统计信息。`heap_get_stats` 可以告诉你很多关于堆使用情况的信息。比如你可以定期调用它，记录已用字节数的变化趋势。如果已用字节数持续增长但分配次数和释放次数差不多，说明可能有内存泄漏。

第四，编写针对性的测试用例。当你发现一个 bug，写一个测试用例来重现它。这样不仅可以修复当前的 bug，还可以防止以后引入类似的 bug。
