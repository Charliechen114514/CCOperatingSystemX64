# 06 - 类型定义与基础库

没有标准库，连个 `uint8_t` 都没有。这一篇我们自己定义。

---

## 为什么需要自己定义类型

在 Freestanding 环境下，我们没有 `<stdint.h>`、`<stddef.h>` 这些标准头文件。

**但是我们的代码需要**：

```c
uint8_t byte = 0xFF;
uint64_t address = 0x10000;
size_t len = 1024;
NULL  // 空指针
```

**怎么办？自己定义！**

---

## LP64 数据模型

x86_64 Linux 使用 LP64 数据模型：

| 类型 | 位数 | 说明 |
|------|------|------|
| `char` | 8 | 字符 |
| `short` | 16 | 短整数 |
| `int` | 32 | 整数 |
| `long` | 64 | 长整数 |
| `long long` | 64 | 超长整数 |
| `void*` | 64 | 指针 |
| `size_t` | 64 | 大小类型 |

**LP = Long and Pointer are 64-bit**

所以我们可以用 C 内置类型来定义固定宽度类型：

```c
typedef unsigned char      uint8_t;   // 8 位
typedef unsigned short     uint16_t;  // 16 位
typedef unsigned int       uint32_t;  // 32 位
typedef unsigned long long uint64_t;  // 64 位
```

## types.h 实现

我们来看完整的 types.h（参考 [kernel/defines/types.h](../kernel/defines/types.h)）：

```c
#ifndef CCOS_TYPES_H
#define CCOS_TYPES_H

/* Fixed-width integer types */
typedef unsigned char      uint8_t;
typedef unsigned short     uint16_t;
typedef unsigned int       uint32_t;
typedef unsigned long long uint64_t;

typedef signed char        int8_t;
typedef signed short       int16_t;
typedef signed int         int32_t;
typedef signed long long   int64_t;

/* Size type */
typedef unsigned long      size_t;
typedef signed long        ssize_t;

/* Boolean type */
typedef enum {
    false = 0,
    true  = 1
} bool;

/* NULL pointer constant */
#define NULL ((void*)0)

/* Offset type */
typedef signed long        ptrdiff_t;

/* Byte count */
typedef unsigned long      uintptr_t;

#endif /* CCOS_TYPES_H */
```

**逐段解释**：

### 固定宽度整数类型

```c
typedef unsigned char      uint8_t;   // 0 ~ 255
typedef unsigned short     uint16_t;  // 0 ~ 65535
typedef unsigned int       uint32_t;  // 0 ~ 4294967295
typedef unsigned long long uint64_t;  // 0 ~ 18446744073709551615
```

**为什么用 `unsigned long long` 而不是 `unsigned long`？**

在 x86_64 上，`long` 和 `long long` 都是 64 位，但用 `long long` 更明确。

### 大小类型

```c
typedef unsigned long      size_t;
typedef signed long        ssize_t;
```

`size_t` 用于表示大小、长度、索引，保证足够大能放下任何对象大小。

`ssize_t` 是带符号版本，用于可能返回错误码的函数（如 `-1` 表示错误）。

### 布尔类型

```c
typedef enum {
    false = 0,
    true  = 1
} bool;
```

C99 有 `_Bool` 类型，但用 enum 更清晰。

### NULL 指针

```c
#define NULL ((void*)0)
```

标准库的 `<stddef.h>` 定义了 `NULL`，我们手动定义。

## 简单的 string 操作

没有 `<string.h>`，但我们需要一些基本操作。

### memcpy 实现

```c
static inline void* memcpy(void* dest, const void* src, size_t n) {
    uint8_t* d = (uint8_t*)dest;
    const uint8_t* s = (const uint8_t*)src;

    while (n--) {
        *d++ = *s++;
    }

    return dest;
}
```

**`static inline` 的好处**：

- `static` — 内部链接，不与其他文件冲突
- `inline` — 建议编译器内联，减少函数调用开销

### memset 实现

```c
static inline void* memset(void* s, int c, size_t n) {
    uint8_t* p = (uint8_t*)s;

    while (n--) {
        *p++ = (uint8_t)c;
    }

    return s;
}
```

### strlen 实现

```c
static inline size_t strlen(const char* s) {
    const char* p = s;
    while (*p) p++;
    return p - s;
}
```

## 调试辅助宏

没有 `<assert.h>`，我们实现自己的断言：

```c
/* Simple assert */
#define CCOS_ASSERT(cond) \
    do { \
        if (!(cond)) { \
            volatile uint16_t* vga = (volatile uint16_t*)0xB8000; \
            vga[0] = 0x1F41;  /* 'A' for Assert */ \
            while (1) { __asm__ volatile("hlt"); } \
        } \
    } while (0)
```

**使用**：

```c
int ptr = get_pointer();
CCOS_ASSERT(ptr != NULL);  // 失败时显示 'A' 并停机
```

**为什么用 `do { ... } while (0)`？**

这是宏定义的常见技巧，确保宏在使用时像单个语句：

```c
if (condition)
    CCOS_ASSERT(x);  // 没有 do-while 会出语法错误
```

## BSS 清零验证

我们来写个测试验证 BSS 清零：

```c
/* 未初始化的全局变量（在 BSS） */
static int uninitialized_var;
static char buffer[16];

void test_bss(void) {
    /* 检查变量是否为 0 */
    int all_zero = 1;

    if (uninitialized_var != 0) {
        all_zero = 0;
    }

    for (int i = 0; i < 16; i++) {
        if (buffer[i] != 0) {
            all_zero = 0;
            break;
        }
    }

    /* VGA 显示结果 */
    volatile uint16_t* vga = (volatile uint16_t*)0xB8000;
    if (all_zero) {
        vga[0] = 0x1F4F;  /* 'O' for OK */
    } else {
        vga[0] = 0x1F45;  /* 'E' for Error */
    }
}
```

**预期结果**：

- BSS 清零成功 → 显示 'O'
- BSS 未清零 → 显示 'E'

## 类型定义的常见问题

### 问题 1：类型大小不确定

```c
// 错误！int 的大小不固定
typedef int int32_t;  // 在某些平台上 int 可能是 16 位

// 正确
typedef signed int int32_t;  // 但这仍然依赖平台
```

**更好的做法**：用 `<stdint.h>`（如果有），或者用编译器内置类型检查：

```c
_Static_assert(sizeof(int) == 4, "int must be 32-bit");
```

### 问题 2：对齐问题

某些结构体需要特定对齐：

```c
/* 强制 16 字节对齐 */
struct __attribute__((aligned(16))) align_struct {
    uint64_t a;
    uint64_t b;
};
```

### 问题 3：volatile 使用不当

```c
volatile int* ptr = (volatile int*)0x1000;
*ptr = 42;       // 正确，写入 volatile 内存
int x = *ptr;    // 正确，读取 volatile 内存

/* 但这样不行 */
int y = *ptr + *ptr;  // 可能读两次，值可能变化！
```

## ⚠️ 注意

**类型定义放在哪里？**

创建 `kernel/defines/` 目录，把 `types.h` 放进去。然后在需要的地方：

```c
#include "defines/types.h"
```

**不要用系统的 `<stdint.h>`**：

```c
// ❌ 错误！这会包含系统头文件
#include <stdint.h>

// ✅ 正确！用自己的定义
#include "defines/types.h"
```

## 实战示例：完整的小型库

我们来看一个完整的基础库文件：

```c
#ifndef CCOS_MINILIB_H
#define CCOS_MINILIB_H

#include "defines/types.h"

/* Memory operations */
static inline void* memcpy(void* dest, const void* src, size_t n) {
    uint8_t* d = (uint8_t*)dest;
    const uint8_t* s = (const uint8_t*)src;
    while (n--) *d++ = *s++;
    return dest;
}

static inline void* memset(void* s, int c, size_t n) {
    uint8_t* p = (uint8_t*)s;
    while (n--) *p++ = (uint8_t)c;
    return s;
}

static inline int memcmp(const void* s1, const void* s2, size_t n) {
    const uint8_t* p1 = (const uint8_t*)s1;
    const uint8_t* p2 = (const uint8_t*)s2;
    while (n--) {
        if (*p1 != *p2) return *p1 - *p2;
        p1++; p2++;
    }
    return 0;
}

/* String operations */
static inline size_t strlen(const char* s) {
    const char* p = s;
    while (*p) p++;
    return p - s;
}

static inline char* strcpy(char* dest, const char* src) {
    char* d = dest;
    while ((*d++ = *src++));
    return dest;
}

/* Utility */
#define CCOS_MIN(a, b) ((a) < (b) ? (a) : (b))
#define CCOS_MAX(a, b) ((a) > (b) ? (a) : (b))
#define CCOS_ARRAY_SIZE(arr) (sizeof(arr) / sizeof((arr)[0]))

#endif /* CCOS_MINILIB_H */
```
