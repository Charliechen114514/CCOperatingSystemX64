# 第一步 - 合并两个文件

好，现在我们开始动手。

---

## 我们的起点

首先看一下我们现在有什么。如果你跟着之前的教程做下来，你的 `boot/` 目录应该是这样的：

```
boot/
├── boot.asm      # Stage 1 MBR
├── boot2.asm     # Stage 2 加载器
└── lib/
    ├── bios_screen.asm
    ├── bios_string.asm
    ├── disk_io.asm
    ├── pmode.asm
    └── lmode.asm
```

我们的目标是：
1. 创建一个新的 `bootloader.asm`
2. 把 `boot.asm` 和 `boot2.asm` 的内容合并进去
3. 使用 `section` 指令分离两个阶段
4. 删除旧的 `boot.asm` 和 `boot2.asm`

---

## 第一步：创建新文件

先别急着复制代码，我们先把文件创建出来：

```bash
cd /path/to/CCOperatingSystemX64/boot
touch bootloader.asm
```

现在打开你最喜欢的编辑器（我用的是 VSCode + vim 插件），编辑这个文件。

---

## 第二步：写入 Stage 1 代码

我们从 Stage 1 开始。这部分代码基本来自原来的 `boot.asm`：

```asm
; ==============================================================================
; CCOS Unified Bootloader
; ==============================================================================
; This single file contains both Stage 1 (MBR) and Stage 2
; - Stage 1 (0x7C00): Loads the rest of bootloader from disk to 0x7E00
; - Stage 2 (0x7E00): Switches to 64-bit long mode and loads kernel
; ==============================================================================

; ==============================================================================
; Stage 1: MBR (0x7C00)
; ==============================================================================
section .mbr
    org 0x7c00
    bits 16

start:
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7c00
    sti

    ; clear screen
    mov ah, 0x00
    mov al, 0x03
    int 0x10

    ; print header
    mov si, header_msg
    call print_string

    ; print welcome
    mov si, welcome_msg
    call print_string

    ; Load Stage 2 from disk (sectors 2-3)
    mov ax, 0x7E0
    mov es, ax
    xor bx, bx
    mov ah, 0x02
    mov al, 0x02
    mov ch, 0x00
    mov cl, 0x02
    mov dh, 0x00
    mov dl, 0x80
    int 0x13
    jc load_error
    cmp al, 0x02
    jne load_error

    ; jump to Stage 2
    jmp 0x7E00

load_error:
    mov si, load_error_msg
    call print_string
.hang:
    hlt
    jmp .hang

print_string:
    pusha
.loop:
    lodsb
    test al, al
    jz .done
    mov ah, 0x0e
    mov bh, 0
    mov bl, 0x07
    int 0x10
    jmp .loop
.done:
    popa
    ret

header_msg:
    db "READY TO BOOT CCOS", 0x0d, 0x0a, 0

welcome_msg:
    db "[1] Stage 1: Loading second stage...", 0x0d, 0x0a, 0

load_error_msg:
    db "[E1] Failed to load Stage 2", 0x0d, 0x0a, 0

; Pad to exactly 510 bytes (before signature)
times 510-($-$$) db 0

; MBR signature at offset 510-511
dw 0xaa55
```

注意这里的关键点：
- `section .mbr` - 定义了一个名为 `.mbr` 的段
- `org 0x7c00` - 设置起始地址为 0x7C00
- `times 510-($-$$) db 0` - 填充到 510 字节
- `dw 0xaa55` - MBR 签名

## 第三步：添加 Stage 2 代码

现在我们在同一个文件里添加 Stage 2。**注意这里有个坑**，我们用 `vstart` 而不是 `org`：

```asm
; ==============================================================================
; Stage 2: Starts at offset 512 in file, loads at 0x7E00
; ==============================================================================
section .stage2 vstart=0x7E00
    bits 16

stage2_entry:
    jmp short stage2_main
    nop

; GDT
gdt_start:
    dq 0x0000000000000000
gdt_code:
    dq 0x00CF9A000000FFFF
gdt_data:
    dq 0x00CF92000000FFFF
gdt_code64:
    dq 0x00AF9A000000FFFF
gdt_data64:
    dq 0x00CF92000000FFFF
gdt_end:

gdt_ptr:
    dw 5 * 8 - 1
    dd 0x00007E03

stage2_main:
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7E00

    ; Print message
    mov si, msg_stage2
    call print_bios

    ; Load kernel (sector 3 now, since bootloader is sectors 1-2)
    mov ax, 0x1000
    mov es, ax
    xor bx, bx
    mov ah, 0x02
    mov al, 0x01
    mov ch, 0x00
    mov cl, 0x03            ; sector 3 (kernel)
    mov dh, 0x00
    mov dl, 0x80
    int 0x13
    jc kernel_error

    ; Load GDT and switch to protected mode
    lgdt [gdt_ptr]
    mov eax, cr0
    or eax, 1
    mov cr0, eax

    ; Far jump to protected mode
    db 0x66
    db 0xEA
    dd pm_entry
    dw 0x08

kernel_error:
    mov si, msg_kernel_error
    call print_bios
.hang:
    hlt
    jmp .hang

print_bios:
    pusha
.loop:
    lodsb
    test al, al
    jz .done
    mov ah, 0x0e
    int 0x10
    jmp .loop
.done:
    popa
    ret

msg_stage2:
    db "[2] Stage 2: Loading kernel...", 0x0d, 0x0a, 0

msg_kernel_error:
    db "[E2] Failed to load kernel", 0x0d, 0x0a, 0
```

⚠️ **注意！**

这里的 `vstart=0x7E00` 非常重要。如果你写成 `org 0x7E00`，你会发现 GDT 指针 `gdt_ptr` 里的地址会错得一塌糊涂。

我们会在下一篇教程里详细解释为什么。

## 第四步：添加保护模式代码

继续添加保护模式部分：

```asm
; ==============================================================================
; 32-bit Protected Mode
; ==============================================================================
bits 32

pm_entry:
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x7E00

    ; Setup page tables
    call setup_page_tables

    ; Enable PAE
    mov eax, cr4
    or eax, 1 << 5
    mov cr4, eax

    ; Load PML4
    mov eax, 0x9000
    mov cr3, eax

    ; Set LME bit
    mov ecx, 0xC0000080
    rdmsr
    or eax, 1 << 8
    wrmsr

    ; Enable paging
    mov eax, cr0
    or eax, 1 << 31
    mov cr0, eax

    ; Jump to 64-bit
    jmp 0x18:long_mode

.hang:
    hlt
    jmp .hang

setup_page_tables:
    pusha

    ; Clear PML4
    mov edi, 0x9000
    mov ecx, 4096 / 4
    xor eax, eax
    rep stosd

    ; Clear PDPT
    mov edi, 0xA000
    mov ecx, 4096 / 4
    xor eax, eax
    rep stosd

    ; Clear PD
    mov edi, 0xB000
    mov ecx, 4096 / 4
    xor eax, eax
    rep stosd

    ; Setup entries
    mov dword [0x9000], 0x0000A003    ; PML4[0] -> PDPT
    mov dword [0x9FF8], 0x0000A003    ; PML4[511] -> PDPT
    mov dword [0xA000], 0x0000B003    ; PDPT[0] -> PD
    mov dword [0xB000], 0x00000083    ; PD[0] -> 2MB page
    mov dword [0xB004], 0x00000000
    mov dword [0xBFF0], 0x00000083    ; PD[511] -> 2MB page
    mov dword [0xBFF4], 0x00000000

    popa
    ret
```

注意这里我们用了 `bits 32`，告诉 NASM 接下来生成 32 位指令。

## 第五步：添加长模式代码

最后是 64 位长模式部分：

```asm
; ==============================================================================
; 64-bit Long Mode
; ==============================================================================
bits 64

long_mode:
    mov ax, 0x20
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov rsp, 0x7E00

    ; Clear screen
    push rax
    push rdi
    push rcx
    mov rdi, 0xB8000
    mov rcx, 80 * 25
    mov rax, 0x0720
.clear_loop:
    mov [rdi], ax
    add rdi, 2
    loop .clear_loop
    pop rcx
    pop rdi
    pop rax

    ; Print ready message
    mov rsi, msg_ready
    mov rdi, 0xB8000
    mov ah, 0x0F
.print_loop:
    lodsb
    test al, al
    jz .print_done
    stosw
    jmp .print_loop
.print_done:

    ; Jump to kernel
    mov rdi, 0x10000
    call rdi

    ; Halt if kernel returns
kernel_halt:
    hlt
    jmp kernel_halt

msg_ready:
    db "READY TO BOOT KERNEL", 0
```

## 第六步：编译验证

现在我们来验证一下代码是否正确：

```bash
nasm -f bin bootloader.asm -o bootloader.bin
ls -lh bootloader.bin
```

你应该看到类似这样的输出：

```
-rwxr-xr-x 1 user user 1.2K Feb 16 23:45 bootloader.bin
```

如果编译成功，恭喜你！如果失败，检查一下：
- 是不是漏掉了 `bits 16/32/64` 声明
- 是不是 `vstart` 写错了
- 段结束的填充是否正确

---

## 第七步：反汇编验证

这一步很重要，我们要确认生成的二进制文件是正确的：

```bash
ndisasm -b 16 -o 0x7C00 bootloader.bin | head -30
```

你应该能看到 Stage 1 的代码从 0x7C00 开始。

然后检查 Stage 2 的位置：

```bash
ndisasm -b 16 -o 0x7E00 bootloader.bin | head -30
```

等等，这样不对。`ndisasm` 默认从头开始反汇编，我们需要用 offset 参数：

```bash
# 检查 0x200 偏移处（应该是 Stage 2 开始）
ndisasm -b 16 -o 0x7E00 bootloader.bin -k 0x200,0,$ | head -30
```

或者用更简单的方法，直接看二进制：

```bash
hexdump -C bootloader.bin | head -20
```

你应该能看到：
- 偏移 0x0000: Stage 1 代码
- 偏移 0x0200: Stage 2 代码开始
- 偏移 0x01FE: MBR 签名 55 AA

## 常见问题

### 问题 1：编译后文件大小不对

如果编译后的文件不是预期的 1024 字节左右，检查：
- `times 510-($-$$) db 0` 这行是在 `.mbr` section 里的
- MBR 签名 `dw 0xaa55` 在正确的位置

### 问题 2：Stage 2 跳转失败

如果 Stage 1 跳转到 Stage 2 后系统挂死，检查：
- `jmp 0x7E00` 的目标地址是否正确
- Stage 2 的 `vstart` 是否设置为 0x7E00

### 问题 3：GDT 指针错误

如果加载 GDT 时出错，很可能是 `gdt_ptr` 里的地址计算错误。这个问题我们下一篇详细讲。
