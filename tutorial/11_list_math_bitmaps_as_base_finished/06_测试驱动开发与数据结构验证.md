## 第一步：理解测试驱动开发

测试驱动开发（Test-Driven Development, TDD）是一种开发方法论，它的核心思想是：先写测试，再写代码。具体流程是这样的：

1. **写一个失败的测试** — 测试你即将实现的功能
2. **运行测试，确认它失败** — 这证明测试确实在检测什么
3. **写最少的代码让测试通过** — 不要多写，刚好能让测试通过就行
4. **重构代码** — 在测试保护下优化代码结构
5. **重复** — 继续下一个功能

这个流程的好处是：
- 你永远有测试覆盖你的代码
- 重构的时候不容易引入 bug
- 测试本身就是文档，展示了代码应该怎么用

但说实话，在实际开发中完全遵循 TDD 是很困难的。有时候你根本不知道要实现什么功能，写测试就是空谈。有时候写测试比写代码还花时间。我的建议是：对于核心的数据结构和算法，一定要写完整的测试；对于简单的逻辑，可以适当简化。

## 第二步：创建测试框架

我们先来创建一个简单的测试框架。这个框架会提供一些基本的测试宏，比如断言相等、断言不等、断言真值等。

在 `test/` 目录下创建 `host_support.h`：

```c
/**
 * @file host_support.h
 * @brief 测试框架支持（宿主机环境）
 * @date 2026-02-17
 */
#pragma once

#include <stdio.h>
#include <stdbool.h>

// 全局测试计数器
extern int g_test_passed;
extern int g_test_failed;

// 测试信息输出
#define TEST_INFO(msg) printf("  [TEST] %s\n", msg)

// 断言宏
#define TEST_ASSERT_EQ(actual, expected, msg) \
    do { \
        if ((actual) != (expected)) { \
            printf("  [FAIL] %s (expected: %p, got: %p)\n", msg, (void*)(expected), (void*)(actual)); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while(0)

#define TEST_ASSERT_TRUE(cond, msg) \
    do { \
        if (!(cond)) { \
            printf("  [FAIL] %s (condition is false)\n", msg); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while(0)

#define TEST_ASSERT_FALSE(cond, msg) \
    do { \
        if (cond) { \
            printf("  [FAIL] %s (condition is true)\n", msg); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while(0)

// 测试通过标记
#define TEST_PASS(msg) \
    do { \
        printf("  [PASS] %s\n", msg); \
    } while(0)

// 测试套件开始/结束
#define TEST_SUITE_BEGIN(name) \
    do { \
        printf("\n=== Test Suite: %s ===\n", name); \
    } while(0)

#define TEST_SUITE_END() \
    do { \
        printf("\nSuite Complete: %d passed, %d failed\n", g_test_passed, g_test_failed); \
    } while(0)
```

这个测试框架非常简单，但已经够用了。`TEST_ASSERT_EQ` 断言两个值相等，`TEST_ASSERT_TRUE` 断言条件为真，`TEST_PASS` 标记测试通过。每个测试结果都会更新全局计数器，最后可以统计通过和失败的数量。

## 第三步：编写链表测试

现在我们来编写链表的单元测试。链表的测试应该覆盖：初始化、插入、删除、遍历、边界情况等。

在 `test/test_list.c` 中编写：

```c
/**
 * @file test_list.c
 * @brief 链表模块的单元测试
 */

#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include "defines/types.h"
#include "list/list.h"

// 全局测试计数器
int g_test_passed = 0;
int g_test_failed = 0;

#include "host_support.h"

// 测试数据结构
typedef struct test_node {
    int value;
    list_head list;  // 嵌入式链表节点
} test_node;

// 节点池（避免动态分配）
#define MAX_NODES 256
static test_node g_node_pool[MAX_NODES];
static int g_node_idx = 0;

static void reset_nodes(void) {
    g_node_idx = 0;
}

static test_node* create_node(int value) {
    if (g_node_idx < MAX_NODES) {
        g_node_pool[g_node_idx].value = value;
        INIT_LIST_HEAD(&g_node_pool[g_node_idx].list);
        return &g_node_pool[g_node_idx++];
    }
    return NULL;
}

// 初始化测试
static void test_init(void) {
    TEST_INFO("Testing list initialization");

    // 测试静态初始化宏
    LIST_HEAD(static_head);
    TEST_ASSERT_EQ(static_head.next, &static_head, "LIST_HEAD: next points to self");
    TEST_ASSERT_EQ(static_head.prev, &static_head, "LIST_HEAD: prev points to self");

    // 测试动态初始化
    list_head dynamic_head;
    INIT_LIST_HEAD(&dynamic_head);
    TEST_ASSERT_EQ(dynamic_head.next, &dynamic_head, "INIT_LIST_HEAD: next points to self");
    TEST_ASSERT_EQ(dynamic_head.prev, &dynamic_head, "INIT_LIST_HEAD: prev points to self");

    // 测试空链表检测
    TEST_ASSERT_TRUE(list_is_empty(&static_head), "list_is_empty: newly initialized list is empty");
    TEST_ASSERT_TRUE(list_is_empty(&dynamic_head), "list_is_empty: dynamic list is empty");

    TEST_PASS("list initialization");
}

// 插入测试
static void test_insertion(void) {
    TEST_INFO("Testing basic insertion operations");

    reset_nodes();

    list_head head;
    INIT_LIST_HEAD(&head);

    test_node* node1 = create_node(1);
    test_node* node2 = create_node(2);
    test_node* node3 = create_node(3);

    // 测试 list_add（头插法）
    list_add(&node1->list, &head);
    TEST_ASSERT_FALSE(list_is_empty(&head), "list_add: list is not empty after add");
    TEST_ASSERT_EQ(head.next, &node1->list, "list_add: head.next points to inserted node");

    list_add(&node2->list, &head);
    list_add(&node3->list, &head);

    // 验证顺序：head -> node3 -> node2 -> node1 -> head
    TEST_ASSERT_EQ(head.next, &node3->list, "list_add: order verified (head->3)");
    TEST_ASSERT_EQ(node3->list.next, &node2->list, "list_add: order verified (3->2)");
    TEST_ASSERT_EQ(node2->list.next, &node1->list, "list_add: order verified (2->1)");

    TEST_PASS("basic insertion");
}

// 尾插法测试
static void test_tail_insertion(void) {
    TEST_INFO("Testing tail insertion operations");

    reset_nodes();

    list_head head;
    INIT_LIST_HEAD(&head);

    test_node* node1 = create_node(10);
    test_node* node2 = create_node(20);
    test_node* node3 = create_node(30);

    // 测试 list_add_tail（尾插法）
    list_add_tail(&node1->list, &head);
    list_add_tail(&node2->list, &head);
    list_add_tail(&node3->list, &head);

    // 验证顺序：head -> node1 -> node2 -> node3 -> head
    TEST_ASSERT_EQ(head.next, &node1->list, "list_add_tail: order verified (head->1)");
    TEST_ASSERT_EQ(node1->list.next, &node2->list, "list_add_tail: order verified (1->2)");
    TEST_ASSERT_EQ(node2->list.next, &node3->list, "list_add_tail: order verified (2->3)");
    TEST_ASSERT_EQ(head.prev, &node3->list, "list_add_tail: order verified (3->head)");

    TEST_PASS("tail insertion");
}

// 删除测试
static void test_deletion(void) {
    TEST_INFO("Testing deletion operations");

    reset_nodes();

    list_head head;
    INIT_LIST_HEAD(&head);

    test_node* node1 = create_node(1);
    test_node* node2 = create_node(2);
    test_node* node3 = create_node(3);

    // 构建链表：head -> 1 -> 2 -> 3 -> head
    list_add_tail(&node1->list, &head);
    list_add_tail(&node2->list, &head);
    list_add_tail(&node3->list, &head);

    // 删除中间节点
    list_del(&node2->list);

    // 验证顺序：head -> 1 -> 3 -> head
    TEST_ASSERT_EQ(node1->list.next, &node3->list, "list_del: middle node removed");
    TEST_ASSERT_EQ(node3->list.prev, &node1->list, "list_del: links updated");

    TEST_PASS("deletion");
}

// 遍历测试
static void test_traversal(void) {
    TEST_INFO("Testing list traversal");

    reset_nodes();

    list_head head;
    INIT_LIST_HEAD(&head);

    test_node* node1 = create_node(1);
    test_node* node2 = create_node(2);
    test_node* node3 = create_node(3);

    list_add_tail(&node1->list, &head);
    list_add_tail(&node2->list, &head);
    list_add_tail(&node3->list, &head);

    // 测试 list_for_each_entry
    int sum = 0;
    int count = 0;
    test_node* pos;
    list_for_each_entry(pos, &head, list) {
        sum += pos->value;
        count++;
    }

    TEST_ASSERT_EQ(count, 3, "list_for_each_entry: correct count");
    TEST_ASSERT_EQ(sum, 6, "list_for_each_entry: correct sum");

    TEST_PASS("list traversal");
}

// list_entry 测试
static void test_list_entry(void) {
    TEST_INFO("Testing list_entry macro");

    reset_nodes();

    test_node* node = create_node(42);
    INIT_LIST_HEAD(&node->list);

    // 从 list_head 获取包含结构体
    test_node* recovered = list_entry(&node->list, test_node, list);

    TEST_ASSERT_EQ(recovered, node, "list_entry: recovered correct pointer");
    TEST_ASSERT_EQ(recovered->value, 42, "list_entry: value preserved");

    TEST_PASS("list_entry macro");
}

// 主测试函数
int main(void) {
    TEST_SUITE_BEGIN("List Tests");

    test_init();
    test_insertion();
    test_tail_insertion();
    test_deletion();
    test_traversal();
    test_list_entry();

    TEST_SUITE_END();

    return (g_test_failed == 0) ? 0 : 1;
}
```

这个测试文件覆盖了链表的主要功能：初始化、插入（头插和尾插）、删除、遍历、`list_entry` 宏。每个测试都验证了特定的功能边界。

## 第四步：编写位图测试

位图的测试应该覆盖：初始化、单比特操作、范围操作、位扫描、边界情况等。

在 `test/test_bitmap.c` 中编写类似的测试代码。由于篇幅原因，这里只展示关键部分：

```c
/**
 * @file test_bitmap.c
 * @brief 位图模块的单元测试
 */

#include <stdio.h>
#include <string.h>
#include "defines/types.h"
#include "bitmap/bitmap.h"
#include "bitmap/bitmap_helper.h"

int g_test_passed = 0;
int g_test_failed = 0;

#include "host_support.h"

static void test_bitmap_init(void) {
    TEST_INFO("Testing bitmap initialization");

    uint8_t buffer[32];
    bitmap bm;

    bitmap_init(&bm, buffer, 256);

    TEST_ASSERT_EQ(bm.nbits, 256, "bitmap_init: nbits set correctly");
    TEST_ASSERT_EQ(bm.bits, buffer, "bitmap_init: bits pointer set correctly");

    // 验证缓冲区被清零
    bool all_zero = true;
    for (int i = 0; i < 32; i++) {
        if (buffer[i] != 0) {
            all_zero = false;
            break;
        }
    }
    TEST_ASSERT_TRUE(all_zero, "bitmap_init: buffer cleared");

    TEST_PASS("bitmap initialization");
}

static void test_bitmap_single_bit(void) {
    TEST_INFO("Testing single bit operations");

    uint8_t buffer[4];
    bitmap bm;
    bitmap_init(&bm, buffer, 32);

    // 测试 set 和 test
    bitmap_set(&bm, 5);
    bitmap_set(&bm, 10);
    bitmap_set(&bm, 15);

    TEST_ASSERT_TRUE(bitmap_test(&bm, 5), "bitmap_set/test: bit 5 is set");
    TEST_ASSERT_TRUE(bitmap_test(&bm, 10), "bitmap_set/test: bit 10 is set");
    TEST_ASSERT_TRUE(bitmap_test(&bm, 15), "bitmap_set/test: bit 15 is set");
    TEST_ASSERT_FALSE(bitmap_test(&bm, 0), "bitmap_set/test: bit 0 is not set");

    // 测试 clear
    bitmap_clear(&bm, 10);
    TEST_ASSERT_FALSE(bitmap_test(&bm, 10), "bitmap_clear: bit 10 is cleared");
    TEST_ASSERT_TRUE(bitmap_test(&bm, 5), "bitmap_clear: other bits unchanged");

    // 测试 flip
    bitmap_flip(&bm, 5);
    TEST_ASSERT_FALSE(bitmap_test(&bm, 5), "bitmap_flip: bit 5 flipped to 0");
    bitmap_flip(&bm, 5);
    TEST_ASSERT_TRUE(bitmap_test(&bm, 5), "bitmap_flip: bit 5 flipped back to 1");

    TEST_PASS("single bit operations");
}

static void test_bitmap_range(void) {
    TEST_INFO("Testing range operations");

    uint8_t buffer[4];
    bitmap bm;
    bitmap_init(&bm, buffer, 32);

    bitmap_set_range(&bm, 10, 5);

    // 验证比特 10-14 被设置
    for (int i = 10; i < 15; i++) {
        TEST_ASSERT_TRUE(bitmap_test(&bm, i), "bitmap_set_range: bits set correctly");
    }

    // 验证其他比特未被设置
    TEST_ASSERT_FALSE(bitmap_test(&bm, 9), "bitmap_set_range: bit before range not set");
    TEST_ASSERT_FALSE(bitmap_test(&bm, 15), "bitmap_set_range: bit after range not set");

    TEST_PASS("range operations");
}

static void test_bitmap_scan(void) {
    TEST_INFO("Testing bit scanning");

    uint8_t buffer[4];
    bitmap bm;
    bitmap_init(&bm, buffer, 32);

    bitmap_set(&bm, 5);
    bitmap_set(&bm, 10);
    bitmap_set(&bm, 15);

    ssize_t first_set = bitmap_find_first_set(&bm);
    TEST_ASSERT_EQ(first_set, 5, "bitmap_find_first_set: correct");

    ssize_t first_zero = bitmap_find_first_zero(&bm);
    TEST_ASSERT_EQ(first_zero, 0, "bitmap_find_first_zero: correct");

    ssize_t next_zero = bitmap_find_next_zero(&bm, 5);
    TEST_ASSERT_EQ(next_zero, 6, "bitmap_find_next_zero: correct");

    TEST_PASS("bit scanning");
}

int main(void) {
    TEST_SUITE_BEGIN("Bitmap Tests");

    test_bitmap_init();
    test_bitmap_single_bit();
    test_bitmap_range();
    test_bitmap_scan();

    TEST_SUITE_END();

    return (g_test_failed == 0) ? 0 : 1;
}
```

## 第六步：调试数据结构问题的技巧

即使有测试，有时候还是会遇到 bug。这里分享一些调试数据结构问题的技巧：

### 1. 使用 GDB 的可视化命令

GDB 有一些命令可以可视化数据结构：

```bash
# 查看链表
(gdb) p *head
(gdb) p head->next
(gdb) p *(test_node*)((char*)head->next - offsetof(test_node, list))

# 设置条件断点
(gdb) break list_del if entry->next == entry
```

### 2. 添加断言检查链表完整性

在开发阶段可以添加一些调试函数：

```c
#ifdef DEBUG
bool list_verify_integrity(list_head* head) {
    list_head* pos = head->next;
    int count = 0;

    while (pos != head) {
        // 检查指针不为空
        if (!pos) return false;

        // 检查双向链表的指针一致性
        if (pos->next->prev != pos) return false;
        if (pos->prev->next != pos) return false;

        pos = pos->next;
        count++;

        // 防止无限循环
        if (count > 10000) return false;
    }

    return true;
}
#endif
```

### 3. 使用栈回溯定位问题

当遇到崩溃时，在关键位置调用 `dump_stack()` 可以帮助你理解代码的执行路径。

```c
void some_problematic_function(void) {
    if (weird_condition) {
        klog_error("Weird condition detected!\n");
        dump_stack(16);
    }
}
```

## 第七步：Stage 11 总结

到这里我们就完成了 Stage 11 的所有内容。让我们回顾一下这一阶段实现了什么：

### 侵入式双向链表 (`kernel/list/`)
- Linux kernel 风格的 `list_head` 结构
- O(1) 的插入、删除操作
- 安全的遍历宏（支持遍历中删除节点）
- `container_of` 宏获取包含结构体
- 链表拼接和切割操作

### 位操作数学库 (`kernel/math/`)
- 2 的幂次检测和计算
- 内存对齐操作
- 整数除法舍入
- 数学工具函数（min、max、abs、clamp）

### 位图数据结构 (`kernel/bitmap/`)
- 单比特操作（set、clear、test、flip）
- 范围操作
- 位扫描功能
- 位图逻辑运算

### 栈回溯支持 (`kernel/stacktrace/`)
- 基于 RBP 的栈帧遍历
- 帧指针验证
- 美观的栈回溯输出

这些基础数据结构是后续开发物理内存管理、进程管理等功能的基础。有了这些工具，我们就可以高效地管理内存页、跟踪进程状态、调试复杂的问题了。

说实话，实现 Stage 11 的过程中我学到了很多。侵入式链表的设计让我重新思考了数据结构和内存的关系，位操作技巧让我看到了底层优化的魅力，位图和栈回溯则让我体会到了系统编程的实用性。这些知识不仅对内核开发有用，对任何需要高效数据结构的项目都有参考价值。
