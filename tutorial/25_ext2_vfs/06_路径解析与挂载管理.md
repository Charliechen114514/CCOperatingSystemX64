# 路径解析与挂载管理 —— 把文件系统挂起来

有了 inode 缓存和 dentry 缓存，我们现在可以实现路径解析和文件系统挂载功能了。这两个功能是 VFS 的"胶水代码"，把所有组件串联起来，让文件系统真正可用。

路径解析负责把像 `/usr/bin/ls` 这样的路径转换成对应的 inode。挂载管理负责把文件系统安装到目录树中，让用户可以访问。两者都需要协调多个组件，是理解 VFS 工作原理的关键。

## 文件系统注册机制

首先，我们需要一种机制让不同的文件系统可以注册自己。VFS 维护一个文件系统类型表，每种文件系统类型在其中注册自己的挂载函数。

```c
/* 文件系统类型 */
typedef struct fs_type {
    const char* name;              /* 文件系统名称 */
    fs_mount_fn mount_fn;           /* 挂载函数 */
    list_head list;                 /* 注册链表 */
} fs_type_t;

/* 全局文件系统类型列表 */
static LIST_HEAD(fs_types);

/* 注册文件系统类型 */
int vfs_register_fs_type(const char* name, fs_mount_fn mount_fn) {
    fs_type_t* fst = kmalloc(sizeof(fs_type_t));
    if (!fst) {
        return -1;
    }

    fst->name = name;
    fst->mount_fn = mount_fn;
    list_add(&fst->list, &fs_types);

    klog_info("vfs: Registered filesystem type '%s'\n", name);
    return 0;
}
```

EXT2 文件系统在初始化时会调用这个函数注册自己：

```c
/* EXT2 初始化 */
static int __init ext2_init(void) {
    return vfs_register_fs_type("ext2", ext2_mount);
}
```

这样，当用户请求挂载 "ext2" 类型的文件系统时，VFS 就能找到对应的挂载函数。

## 挂载文件系统

挂载是让文件系统可用的关键操作。`vfs_mount()` 函数负责：

```c
int vfs_mount(const char* device, const char* path, const char* fstype) {
    if (!device || !path || !fstype) {
        return -1;
    }

    /* 查找文件系统类型 */
    fs_type_t* fst;
    fs_mount_fn mount_fn = NULL;
    list_for_each_entry(fst, &fs_types, list) {
        if (strcmp(fst->name, fstype) == 0) {
            mount_fn = fst->mount_fn;
            break;
        }
    }

    if (!mount_fn) {
        klog_error("vfs: Filesystem type '%s' not found\n", fstype);
        return -1;
    }

    /* 解析设备名 */
    int dev_id = -1;
    if (strcmp(device, "hda") == 0) dev_id = 0;
    else if (strcmp(device, "hdb") == 0) dev_id = 1;
    else if (strcmp(device, "hdc") == 0) dev_id = 2;
    else if (strcmp(device, "hdd") == 0) dev_id = 3;

    if (dev_id < 0) {
        klog_error("vfs: Invalid device '%s'\n", device);
        return -1;
    }

    /* 分配超级块 */
    vfs_superblock_t* sb = vfs_alloc_super();
    if (!sb) {
        return -1;
    }

    sb->s_dev = dev_id;

    /* 调用文件系统的挂载函数 */
    int result = mount_fn(sb, NULL);
    if (result != 0) {
        vfs_free_super(sb);
        return result;
    }

    /* 添加到超级块列表 */
    vfs_add_super(sb);

    /* 如果是根文件系统，设置为根 */
    if (strcmp(path, "/") == 0) {
        vfs_set_root_sb(sb);
        klog_info("vfs: Set root filesystem\n");
    }

    klog_info("vfs: Mounted %s on %s\n", fstype, path);
    return 0;
}
```

这个函数首先查找文件系统类型，然后解析设备名，分配超级块，最后调用文件系统的挂载函数。文件系统的挂载函数负责读取超级块、初始化文件系统结构等具体工作。

## EXT2 的挂载实现

EXT2 的挂载函数是文件系统挂载的核心，它读取并解析 EXT2 超级块：

```c
int ext2_mount(vfs_superblock_t* sb, const char* data) {
    /* 获取块设备 */
    block_device_t* dev = block_device_get(sb->s_dev);
    if (!dev) {
        return -1;
    }

    /* 分配 EXT2 文件系统信息 */
    ext2_fs_info_t* fs_info = kmalloc(sizeof(ext2_fs_info_t));
    if (!fs_info) {
        block_device_put(dev);
        return -1;
    }

    /* 分配超级块副本 */
    ext2_superblock_t* ext2_sb = kmalloc(sizeof(ext2_superblock_t));
    if (!ext2_sb) {
        kfree(fs_info);
        block_device_put(dev);
        return -1;
    }

    /* 从磁盘读取超级块 */
    if (ext2_read_sb_raw(dev, ext2_sb) != 0) {
        kfree(ext2_sb);
        kfree(fs_info);
        block_device_put(dev);
        return -1;
    }

    fs_info->sb = ext2_sb;
    fs_info->device = dev;
    fs_info->block_size = EXT2_BLOCK_SIZE(ext2_sb);

    /* 计算块组数量 */
    uint32_t blocks_per_group = ext2_sb->s_blocks_per_group;
    uint32_t blocks_count = ext2_sb->s_blocks_count;
    fs_info->ngroups = (blocks_count + blocks_per_group - 1) / blocks_per_group;

    /* 读取块组描述符 */
    if (ext2_read_bg_desc(fs_info) != 0) {
        kfree(ext2_sb);
        kfree(fs_info);
        block_device_put(dev);
        return -1;
    }

    /* 填充 VFS 超级块 */
    sb->s_blocksize = fs_info->block_size;
    sb->s_magic = EXT2_SUPERBLOCK_MAGIC;
    strcpy(sb->s_fsname, "ext2");
    sb->s_fs_info = fs_info;
    sb->s_op = &ext2_super_ops;

    /* 读取根 inode */
    vfs_inode_t* root_inode = vfs_iget(sb, EXT2_ROOT_INO);
    if (!root_inode) {
        kfree(ext2_sb);
        kfree(fs_info);
        block_device_put(dev);
        return -1;
    }

    sb->s_root = root_inode;
    return 0;
}
```

EXT2 超级块位于磁盘的 1024 字节偏移处，这是 EXT2 的设计规范。我们读取这个位置的数据，验证魔数（0xEF53），然后解析文件系统的参数。

## 路径解析

路径解析是 VFS 的核心功能之一，负责把路径字符串转换成对应的 inode。`vfs_path_lookup()` 函数实现这个功能：

```c
int vfs_path_lookup(const char* path, vfs_inode_t** result) {
    if (!path || !result) {
        return -1;
    }

    /* 从根目录开始 */
    vfs_inode_t* current = vfs_get_root_sb()->s_root;
    vfs_inode_get(current);

    /* 跳过开头的 '/' */
    while (*path == '/') path++;

    /* 逐级解析 */
    while (*path) {
        /* 提取下一个组件 */
        char component[256];
        int i = 0;
        while (*path && *path != '/' && i < 255) {
            component[i++] = *path++;
        }
        component[i] = '\0';

        /* 跳过分隔符 */
        while (*path == '/') path++;

        /* 在当前目录中查找组件 */
        vfs_inode_t* next = NULL;
        int ret = current->i_op->lookup(current, component, &next);

        vfs_inode_put(current);

        if (ret != 0) {
            return ret;  /* 查找失败 */
        }

        current = next;
    }

    *result = current;
    return 0;
}
```

这个函数从根目录开始，逐级解析路径的每个组件。对于每个组件（如 "usr"、"bin"、"ls"），调用当前目录的 `lookup` 操作查找对应的 inode。

## 打开文件

有了路径解析，打开文件就很简单了：

```c
int vfs_open(const char* path, int flags, uint32_t mode, file_t** file_out) {
    if (!path || !file_out) {
        return -1;
    }

    /* 解析路径 */
    vfs_inode_t* inode = NULL;
    int ret = vfs_path_lookup(path, &inode);
    if (ret != 0) {
        /* 文件不存在，如果设置了 O_CREAT 则创建 */
        if (flags & O_CREAT) {
            /* TODO: 实现文件创建 */
            return -1;
        }
        return ret;
    }

    /* 检查是否是目录 */
    if (vfs_is_dir(inode) && (flags & O_WRONLY)) {
        vfs_iput(inode);
        return -1;
    }

    /* 分配 file 结构 */
    file_t* file = kmalloc(sizeof(file_t));
    if (!file) {
        vfs_iput(inode);
        return -1;
    }

    file->f_inode = inode;
    file->f_pos = 0;
    file->f_flags = flags;
    file->f_count = 1;
    file->f_op = inode->i_fop;

    /* 调用文件系统的 open 操作 */
    if (file->f_op && file->f_op->open) {
        ret = file->f_op->open(inode, file);
        if (ret != 0) {
            kfree(file);
            vfs_iput(inode);
            return ret;
        }
    }

    *file_out = file;
    return 0;
}
```

这个函数首先解析路径获取 inode，然后分配 file 结构并初始化。注意 file 结构持有了 inode 的引用（通过 `vfs_inode_get()` 或直接设置），在关闭文件时需要释放。

## 初始化流程

VFS 子系统需要在内核初始化时按顺序初始化各个组件：

```c
int vfs_init(void) {
    /* 初始化缓存 */
    vfs_inode_cache_init();
    vfs_dentry_cache_init();
    vfs_file_cache_init();
    vfs_mount_table_init();

    klog_info("vfs: VFS subsystem initialized\n");
    return 0;
}

/* 在内核初始化中调用 */
void kernel_init(void) {
    /* ... */

    /* 初始化块设备 */
    block_init();

    /* 初始化 VFS */
    vfs_init();

    /* 初始化 EXT2 */
    ext2_init();

    /* 挂载根文件系统 */
    vfs_mount("hda", "/", "ext2");

    /* ... */
}
```

初始化顺序很重要：先初始化块设备，再初始化 VFS，然后注册文件系统类型，最后挂载根文件系统。任何步骤出错都会导致后续步骤无法进行。

## 小结

路径解析和文件系统挂载是 VFS 的关键功能。路径解析通过逐级查找目录项，把路径字符串转换成对应的 inode。文件系统挂载则把具体的文件系统安装到目录树中，让用户可以访问。

这两个功能需要协调多个组件：块设备层提供 I/O，inode 缓存管理元数据，dentry 缓存加速查找，文件系统实现具体的操作。所有组件协同工作，构成了一个完整的文件系统子系统。
