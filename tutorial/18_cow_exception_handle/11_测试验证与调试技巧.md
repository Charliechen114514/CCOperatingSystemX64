## COW 功能测试

### 测试一：基本 COW 操作

这个测试验证最基本的 COW 功能：两个虚拟地址映射同一物理页，写入时触发 COW。

```c
void test_cow_basic(void) {
    /* 分配两个虚拟页 */
    virtual_addr_t page1 = vmm_alloc_pages(1, VMAP_FLAG_WRITE);
    virtual_addr_t page2 = vmm_alloc_pages(1, VMAP_FLAG_WRITE);

    /* 写入数据到 page1 */
    volatile uint32_t* p1 = (volatile uint32_t*)page1;
    p1[0] = 0xDEADBEEF;

    /* 获取 page1 的物理地址 */
    page_query_result_t query;
    page_query(vmm_get_current_pml4(), page1, &query);
    physical_addr_t phys = query.phys_addr;

    /* 让 page2 也映射到同一物理页 */
    page_unmap_page(vmm_get_current_pml4(), page2, false);
    page_map_page(vmm_get_current_pml4(), page2, phys, VMAP_FLAG_WRITE, false);

    /* 添加到 COW 跟踪 */
    cow_add_page(phys);
    cow_inc_refcount(phys);

    /* 标记为只读 COW */
    cow_mark_page_readonly(vmm_get_current_pml4(), page1);
    cow_mark_page_readonly(vmm_get_current_pml4(), page2);

    /* 验证点1：读取应该成功 */
    uint32_t val1 = p1[0];
    uint32_t val2 = ((volatile uint32_t*)page2)[0];
    assert(val1 == 0xDEADBEEF);
    assert(val2 == 0xDEADBEEF);

    /* 验证点2：写入 page1 应该触发 COW */
    p1[0] = 0xCAFEBABE;
    assert(p1[0] == 0xCAFEBABE);      /* page1 已修改 */
    assert(((volatile uint32_t*)page2)[0] == 0xDEADBEEF);  /* page2 未修改 */
}
```

### 测试二：引用计数 = 1 优化

这个测试验证当引用计数为 1 时，COW 不需要复制：

```c
void test_cow_refcount_one(void) {
    /* 分配虚拟页 */
    virtual_addr_t page = vmm_alloc_pages(1, VMAP_FLAG_WRITE);
    volatile uint32_t* ptr = (volatile uint32_t*)page;

    /* 获取物理地址 */
    page_query_result_t query;
    page_query(vmm_get_current_pml4(), page, &query);
    physical_addr_t phys = query.phys_addr;

    /* 添加到 COW（refcount = 1） */
    cow_add_page(phys);

    /* 标记为只读 COW */
    cow_mark_page_readonly(vmm_get_current_pml4(), page);

    /* 写入应该只设置写权限，不需要复制 */
    ptr[0] = 0x12345678;

    /* 验证：检查统计信息 */
    cow_stats_t stats;
    cow_get_stats(&stats);
    assert(stats.cow_coalesced > 0);  /* 应该有优化的页 */
}
```

### 测试三：COW 区域注册

这个测试验证批量 COW 注册：

```c
void test_cow_region(void) {
    /* 分配 4 页区域 */
    virtual_addr_t region = vmm_alloc_pages(4, VMAP_FLAG_WRITE);

    /* 写入数据 */
    volatile uint32_t* ptr = (volatile uint32_t*)region;
    for (int i = 0; i < 16; i++) {
        ptr[i] = 0x1000 + i;
    }

    /* 注册为 COW 区域 */
    cow_result_t result = cow_register_region(
        vmm_get_current_pml4(),
        region,
        4 * PAGE_SIZE
    );

    assert(result == COW_OK);

    /* 每一页写入都应该触发 COW */
    for (int i = 0; i < 16; i++) {
        ptr[i] = 0x2000 + i;
    }

    /* 验证统计 */
    cow_stats_t stats;
    cow_get_stats(&stats);
    assert(stats.cow_write_faults >= 4);  /* 至少 4 个 COW 错误 */
}
```

## GDB 调试技巧

### 断点设置

在关键函数设置断点：

```gdb
# COW 相关断点
break cow_handle_fault
break cow_copy_on_write
break cow_make_writable

# 异常处理器断点
break double_fault_handler
break gp_fault_handler
break pf_handler
```

### 检查 COW 状态

在 GDB 中检查 COW 内部状态：

```gdb
# 打印 COW 统计
print s_cow_state.stats

# 检查哈希表
print s_cow_state.page_map

# 检查特定 COW 块
print *(cow_block_t*)0xXXXXX
```

### 检查页表项

```gdb
# 定义一个宏来打印页表项
define print_pte
    set $pte = $arg0
    printf "PTE: 0x%016llX\n", $pte
    if ($pte & 1)
        printf "  Present: Yes\n"
    else
        printf "  Present: No\n"
    end
    if ($pte & 2)
        printf "  Writable: Yes\n"
    else
        printf "  Writable: No\n"
    end
    if ($pte & (1 << 9))
        printf "  COW: Yes\n"
    else
        printf "  COW: No\n"
    end
    printf "  Physical: 0x%012llX\n", ($pte >> 12) * 4096
end

# 使用示例
# 先查询页表获取 PTE，然后
print_pte $pte_value
```

## QEMU Monitor 使用

QEMU Monitor 提供了很多有用的调试命令：

### 查看页表

```
(qemu) info pg
```

这会打印当前页表的内容，可以用来验证 COW 标志是否正确设置。

```
(qemu) info tss
```

## 常见问题排查

### 问题一：COW 页错误不触发

**症状**：写入 COW 页没有任何反应，也没有触发页错误

**可能原因**：
1. 页表项没有正确设置为只读
2. COW 标志没有设置
3. 地址映射问题

**排查步骤**：
```gdb
# 1. 检查页表项
break cow_mark_page_readonly
commands
    printf "Flags before: 0x%llX\n", new_flags
    continue
end

# 2. 检查 COW 标志
print cow_is_cow_page(pte_flags)

# 3. 检查页错误是否被触发
break pf_handler
```

### 问题二：Triple Fault（系统重启）

**症状**：QEMU 直接退出或重启

**可能原因**：
1. Double Fault 处理器没有正确设置 IST 栈
2. IST 栈溢出
3. TSS 配置错误

**排查步骤**：
```bash
# 1. 检查 TSS 初始化日志
[TSS] IST1 (DF): 0xXXXXX  <- 应该有有效地址

# 2. 检查 IDT 配置
# 确认 #DF 和 #SS 的 IST 索引已设置

# 3. 在 QEMU 中使用 -d int,cpu_reset 查看详细信息
qemu-system-x86_64 -d int,cpu_reset ...
```

### 问题三：引用计数泄漏

**症状**：COW 统计中的 `cow_current_blocks` 持续增长

**可能原因**：
1. 进程退出时没有调用 `cow_unregister_region`
2. 引用计数管理错误

**排查步骤**：
```c
// 添加调试代码
void cow_debug_dump(void) {
    klog_info("[COW Debug] Current blocks:\n");
    hashmap_foreach(s_cow_state.page_map, debug_print_block, NULL);
}

// 在适当位置调用
cow_debug_dump();
```

### 问题四：GPF 频繁触发

**症状**：系统正常运行但频繁触发 GPF

**可能原因**：
1. 段选择器配置错误
2. 权限问题
3. 内存访问越界

**排查步骤**：
```gdb
# 在 GPF 处理器设置断点
break gp_fault_handler

# 检查错误码
print error_code

# 检查 RIP（故障指令地址）
print $rip

# 反汇编故障指令
x/5i $rip
```

## 性能分析

### COW 性能指标

```c
void cow_print_performance(void) {
    cow_stats_t stats;
    cow_get_stats(&stats);

    klog_info("=== COW Performance ===\n");
    klog_info("Total write faults:     %llu\n", stats.cow_write_faults);
    klog_info("Pages actually copied:   %llu\n",
              stats.cow_pages_allocated - stats.cow_coalesced);
    klog_info("Pages coalesced (optimized): %llu\n", stats.cow_coalesced);

    if (stats.cow_write_faults > 0) {
        uint64_t copy_ratio = (stats.cow_pages_allocated - stats.cow_coalesced) * 100 / stats.cow_write_faults;
        klog_info("Copy ratio: %llu%%\n", copy_ratio);
    }
}
```

### 哈希表性能

```c
void hashmap_print_stats(hashmap_t* map) {
    size_t size = hashmap_size(map);
    size_t buckets = map->num_buckets;

    klog_info("Hashmap: %zu entries, %zu buckets\n", size, buckets);

    if (buckets > 0) {
        klog_info("Load factor: %.2f\n", (double)size / buckets);
    }

    // 统计最大链长
    size_t max_chain = 0;
    for (size_t i = 0; i < buckets; i++) {
        size_t chain_len = 0;
        hash_entry_t* entry = map->buckets[i];
        while (entry) {
            chain_len++;
            entry = entry->next;
        }
        if (chain_len > max_chain) {
            max_chain = chain_len;
        }
    }

    klog_info("Max chain length: %zu\n", max_chain);
}
```

## 总结

调试 COW 和异常处理需要耐心和细致。以下是一些通用建议：

1. **从小测试开始**：先测试简单的 COW 场景，再逐步增加复杂度
2. **使用日志**：在关键路径添加日志，帮助理解执行流程
3. **验证假设**：用 GDB 验证你的假设，不要猜测
4. **保存状态**：在触发异常前保存关键状态，方便事后分析
5. **隔离问题**：把问题分解，一次只测试一个功能
