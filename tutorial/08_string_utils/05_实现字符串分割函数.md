# 实现字符串分割函数

我们来到字符串工具库的最后一部分 —— 字符串分割函数。

`strtok` 是 C 标准库中最"臭名昭著"的函数之一。它非常强大，但也有不少陷阱。不过只要理解了它的工作原理，你就能在解析命令、配置文件等场景下得心应手。

---

## 第一步：理解 strtok 的行为

`strtok` 的作用是按照指定的分隔符把字符串分割成多个 token（标记）。

**基本用法**：

```c
char str[] = "hello world from CCOS";
char* token;

// 第一次调用：传入要分割的字符串
token = strtok(str, " ");
// token 指向 "hello"

// 后续调用：传入 NULL
token = strtok(NULL, " ");
// token 指向 "world"

token = strtok(NULL, " ");
// token 指向 "from"

token = strtok(NULL, " ");
// token 指向 "CCOS"

token = strtok(NULL, " ");
// token 是 NULL，表示分割完毕
```

⚠️ **陷阱 1：原字符串会被修改**

`strtok` 会**原地修改**输入字符串：

```c
char str[] = "hello world";
strtok(str, " ");
// str 现在是 "hello\0world"
//               ^
//               第一个空格被替换成 '\0'
```

这就是为什么 `strtok` 的第一个参数是 `char*` 而不是 `const char*`。

⚠️ **陷阱 2：非线程安全**

`strtok` 使用静态变量来记住分割进度，所以多个线程同时使用会出问题。

## 第二步：更新头文件

在 `kernel/base/string.h` 中添加：

```c
/* ================= String tokenization ================= */

/**
 * @brief Split string into tokens using delimiters.
 *
 * Breaks the string into a sequence of tokens separated by any of the
 * characters in @p delim. This function is NOT thread-safe.
 *
 * @warning The input string is modified in-place!
 *
 * @param str Input string (modified in-place). Pass NULL to continue.
 * @param delim Delimiter character set.
 * @return Pointer to next token, or NULL if no more tokens.
 */
char* strtok(char* str, const char* delim);

/**
 * @brief Reentrant string tokenizer.
 *
 * Thread-safe version of strtok using caller-provided context.
 *
 * @param str Input string (modified in-place). Pass NULL to continue.
 * @param delim Delimiter character set.
 * @param saveptr Pointer to tokenizer state.
 * @return Pointer to next token, or NULL if no more tokens.
 */
char* strtok_r(char* str, const char* delim, char** saveptr);
```

## 第三步：实现 strtok

`strtok` 的核心思想是：
1. 记住当前扫描的位置（静态变量）
2. 跳过前导分隔符
3. 找到下一个分隔符，将其替换为 `'\0'`
4. 返回当前 token 的起始位置

```c
/* 在 kernel/base/string.c 中添加 */

char* strtok(char* str, const char* delim)
{
    static char* last = NULL;  // 记住上次的位置

    if (str != NULL) {
        last = str;
    }

    if (last == NULL) {
        return NULL;  // 没有更多 token 了
    }

    // 跳过前导分隔符
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (!is_delim) {
            break;
        }
        last++;
    }

    if (*last == '\0') {
        return NULL;  // 全是分隔符，没有 token
    }

    // 记住 token 的起始位置
    char* token = last;

    // 找到下一个分隔符
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (is_delim) {
            *last = '\0';  // 替换分隔符为 '\0'
            last++;        // 移动到下一个位置
            break;
        }
        last++;
    }

    return token;
}
```

**让我们走一遍这个算法**：

```
输入字符串: "  hello   world  "
分隔符: " "

步骤 1: 跳过前导空格
        "  hello   world  "
         ^
        last 跳过了两个空格

步骤 2: 记录 token 起始位置
        "  hello   world  "
           ^
        token = last

步骤 3: 找到下一个空格
        "  hello   world  "
               ^
        找到了！替换成 '\0'
        "  hello\0  world  "
               ^
        last++ 移动到下一个字符

步骤 4: 返回 token
        返回指向 "hello" 的指针
```

## 第四步：实现 strtok_r

`strtok_r` 是 `strtok` 的可重入版本，它使用调用者提供的指针来记住状态，而不是静态变量：

```c
char* strtok_r(char* str, const char* delim, char** saveptr)
{
    if (saveptr == NULL) {
        return NULL;
    }

    if (str != NULL) {
        *saveptr = str;
    }

    if (*saveptr == NULL) {
        return NULL;
    }

    // 跳过前导分隔符
    char* last = *saveptr;
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (!is_delim) {
            break;
        }
        last++;
    }

    if (*last == '\0') {
        *saveptr = NULL;
        return NULL;
    }

    // 记住 token 的起始位置
    char* token = last;

    // 找到下一个分隔符
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (is_delim) {
            *last = '\0';
            last++;
            *saveptr = last;
            return token;
        }
        last++;
    }

    // 到达字符串末尾
    *saveptr = NULL;
    return token;
}
```

**使用示例**：

```c
char str[] = "hello world from CCOS";
char* saveptr;
char* token;

token = strtok_r(str, " ", &saveptr);
// token = "hello"

token = strtok_r(NULL, " ", &saveptr);
// token = "world"

token = strtok_r(NULL, " ", &saveptr);
// token = "from"

token = strtok_r(NULL, " ", &saveptr);
// token = "CCOS"

token = strtok_r(NULL, " ", &saveptr);
// token = NULL
```

## 第五步：实际应用示例

### 解析命令行

```c
void parse_command(char* cmd)
{
    char* saveptr;
    char* token;

    token = strtok_r(cmd, " ", &saveptr);
    if (token == NULL) {
        return;  // 空命令
    }

    // token 是命令名
    if (strcmp(token, "echo") == 0) {
        // 获取参数
        char* arg = strtok_r(NULL, " ", &saveptr);
        if (arg != NULL) {
            // 打印参数
        }
    } else if (strcmp(token, "help") == 0) {
        // 显示帮助
    }
}

// 使用
parse_command("echo hello");
```

### 解析配置文件行

```c
void parse_config_line(char* line)
```

## 完整的分割函数代码

```c
/* ========== String tokenization ========== */

char* strtok(char* str, const char* delim)
{
    static char* last = NULL;

    if (str != NULL) {
        last = str;
    }

    if (last == NULL) {
        return NULL;
    }

    // Skip leading delimiters
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (!is_delim) {
            break;
        }
        last++;
    }

    if (*last == '\0') {
        return NULL;
    }

    char* token = last;

    // Find next delimiter
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (is_delim) {
            *last = '\0';
            last++;
            return token;
        }
        last++;
    }

    return token;
}

char* strtok_r(char* str, const char* delim, char** saveptr)
{
    if (saveptr == NULL) {
        return NULL;
    }

    if (str != NULL) {
        *saveptr = str;
    }

    if (*saveptr == NULL) {
        return NULL;
    }

    // Skip leading delimiters
    char* last = *saveptr;
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (!is_delim) {
            break;
        }
        last++;
    }

    if (*last == '\0') {
        *saveptr = NULL;
        return NULL;
    }

    char* token = last;

    // Find next delimiter
    while (*last != '\0') {
        const char* d = delim;
        bool is_delim = false;
        while (*d != '\0') {
            if (*last == *d) {
                is_delim = true;
                break;
            }
            d++;
        }
        if (is_delim) {
            *last = '\0';
            last++;
            *saveptr = last;
            return token;
        }
        last++;
    }

    *saveptr = NULL;
    return token;
}
```

## 验证

创建测试程序：

```c
/* test_strtok.c */

#include <stdio.h>
#include <stdbool.h>

// 函数实现...

int main(void)
{
    printf("=== Testing strtok ===\n");
    char str1[] = "  hello   world  from CCOS  ";
    printf("Original: \"%s\"\n", str1);

    char* token = strtok(str1, " ");
    int count = 1;
    while (token != NULL) {
        printf("Token %d: \"%s\"\n", count++, token);
        token = strtok(NULL, " ");
    }

    printf("\n=== Testing strtok_r ===\n");
    char str2[] = "one,two;three:four";
    printf("Original: \"%s\"\n", str2);

    char* saveptr;
    token = strtok_r(str2, ",;:", &saveptr);
    count = 1;
    while (token != NULL) {
        printf("Token %d: \"%s\"\n", count++, token);
        token = strtok_r(NULL, ",;:", &saveptr);

    printf("\n=== Testing edge cases ===\n");
    char str3[] = ",,,";
    token = strtok_r(str3, ",", &saveptr);
    printf("All delimiters: %p (expected: NULL)\n", (void*)token);

    char str4[] = "abc";
    token = strtok_r(str4, ",", &saveptr);
    printf("No delimiters: \"%s\" (expected: \"abc\")\n", token);
```
