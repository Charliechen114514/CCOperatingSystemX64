# 主机环境测试框架

断言系统搞定了，现在我们来解决一个开发效率问题。

你有没有发现，每次改点代码都要：
1. 重新编译内核
2. 启动 QEMU
3. 等待启动...
4. 才能验证功能

这套流程下来，改一行代码测试一下的时间都够煮碗泡面了。

今天我们要实现一个主机环境的测试框架 —— 几秒钟就能得到反馈。

---

## 为什么要主机测试？

**主机环境**就是你平时写代码用的操作系统（Linux/Windows/macOS）。**内核环境**是你正在开发的操作系统。

| 特性 | 主机测试 | 内核测试 |
|------|----------|----------|
| 编译时间 | 秒级 | 分钟级 |
| 启动时间 | 毫秒级 | 秒十级 |
| 调试工具 | GDB、Valgrind、 sanitizer | GDB 远程调试 |
| 测试成本 | 极低 | 高 |

**策略**：先用主机环境快速验证逻辑，再在内核环境测试集成。

---

## 第一步：创建测试目录

```bash
# 测试代码放在独立的目录
mkdir -p test

# 创建测试文件
touch test/test_string.c
touch test/host_support.h
```

---

## 第二步：理解测试框架的设计

我们的测试框架要：

1. **在主机环境编译运行** - 不依赖内核代码
2. **提供断言宏** - `TEST_ASSERT_EQ`、`TEST_ASSERT_STR_EQ` 等
3. **统计测试结果** - 通过/失败数量
4. **清晰的输出** - 哪个测试通过了，哪个失败了

**核心思想**：测试程序是**独立的可执行文件**，不链接任何内核代码。它只测试我们实现的字符串函数。

## 第三步：实现 host_support.h

创建 `test/host_support.h`，提供测试宏和主机环境的替代定义：

```c
/**
 * @file host_support.h
 * @brief Support macros and definitions for hosted test environment.
 *
 * This header provides testing utilities and type definitions for
 * running kernel string functions in a hosted environment (regular
 * user-space program).
 */
#pragma once

#include <stdio.h>
#include <stdbool.h>
#include <stddef.h>

/* ========== Type definitions ========== */
/* In freestanding kernel environment, these come from kernel/defines/types.h */

typedef enum {
    false = 0,
    true  = 1
} bool;

/* ========== Test counters ========== */

extern int g_test_passed;
extern int g_test_failed;

/* ========== Test assertion macros ========== */

/**
 * @brief Assert that two integers are equal.
 */
#define TEST_ASSERT_EQ(actual, expected, msg) \
    do { \
        if ((actual) != (expected)) { \
            printf("  [FAIL] %s: %d != %d\n", (msg), (int)(actual), (int)(expected)); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while (0)

/**
 * @brief Assert that two strings are equal.
 */
#define TEST_ASSERT_STR_EQ(actual, expected, msg) \
    do { \
        const char* a = (actual); \
        const char* e = (expected); \
        if (a == NULL || e == NULL || strcmp(a, e) != 0) { \
            printf("  [FAIL] %s: \"%s\" != \"%s\"\n", (msg), \
                   a ? a : "(null)", e ? e : "(null)"); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while (0)

/**
 * @brief Assert that a condition is true.
 */
#define TEST_ASSERT_TRUE(condition, msg) \
    do { \
        if (!(condition)) { \
            printf("  [FAIL] %s\n", (msg)); \
            g_test_failed++; \
        } else { \
            g_test_passed++; \
        } \
    } while (0)

/**
 * @brief Print informational message.
 */
#define TEST_INFO(msg) \
    printf("\n--- %s\n", (msg))

/**
 * @brief Mark a test group as passed.
 */
#define TEST_PASS(msg) \
    printf("  [PASS] %s\n", (msg))

/* ========== Utility functions ========== */

/**
 * @brief Print test summary.
 */
static inline void print_test_summary(void)
{
    printf("\n");
    printf("========================================\n");
    printf("Test Results:\n");
    printf("  Passed: %d\n", g_test_passed);
    printf("  Failed: %d\n", g_test_failed);
    printf("========================================\n");

    if (g_test_failed > 0) {
        printf("\n⚠️  SOME TESTS FAILED!\n");
    } else {
        printf("\n✅ All tests passed!\n");
    }
}
```

## 第四步：编写测试程序

创建 `test/test_string.c`：

```c
/**
 * @file test_string.c
 * @brief Comprehensive tests for kernel string utility functions.
 *
 * This program tests the string functions in a hosted environment
 * (regular user-space) for rapid development and verification.
 *
 * Compile with:
 *   gcc -o test_string test_string.c ../kernel/base/string.c -I ../kernel/base
 *
 * Run with:
 *   ./test_string
 */

/* Standard library for hosted environment */
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

/* Forward declare string functions to test */
/* We don't include string.h directly because of size_t conflicts */

extern size_t strlen(const char* s);
extern size_t strnlen(const char* s, size_t maxlen);
extern char* strcpy(char* dest, const char* src);
extern char* strncpy(char* dest, const char* src, size_t n);
extern int strcmp(const char* s1, const char* s2);
extern int strncmp(const char* s1, const char* s2, size_t n);
extern int strcasecmp(const char* s1, const char* s2);
extern int strncasecmp(const char* s1, const char* s2, size_t n);
extern char tolower(char c);
extern char* strchr(const char* s, int c);
extern char* strrchr(const char* s, int c);
extern char* strstr(const char* haystack, const char* needle);
extern char* strpbrk(const char* s, const char* accept);
extern size_t strspn(const char* s, const char* accept);
extern size_t strcspn(const char* s, const char* reject);
extern char* strtok_r(char* str, const char* delim, char** saveptr);
extern char* strtok(char* str, const char* delim);

/* Test counters */
int g_test_passed = 0;
int g_test_failed = 0;

/* Include test support macros */
#include "host_support.h"

/* ============================================================================
 * Test Functions
 * ============================================================================ */

static void test_strlen(void)
{
    TEST_INFO("Testing strlen...");

    TEST_ASSERT_EQ(strlen(""), (size_t)0, "empty string");
    TEST_ASSERT_EQ(strlen("a"), (size_t)1, "single character");
    TEST_ASSERT_EQ(strlen("hello"), (size_t)5, "normal string");
    TEST_ASSERT_EQ(strlen("hello world"), (size_t)11, "string with space");

    TEST_PASS("strlen: all tests passed");
}

static void test_strnlen(void)
{
    TEST_INFO("Testing strnlen...");

    TEST_ASSERT_EQ(strnlen("", 100), (size_t)0, "empty string, large maxlen");
    TEST_ASSERT_EQ(strnlen("hello", 100), (size_t)5, "normal string, large maxlen");
    TEST_ASSERT_EQ(strnlen("hello", 3), (size_t)3, "maxlen < length");
    TEST_ASSERT_EQ(strnlen("hello", 5), (size_t)5, "maxlen == length");
    TEST_ASSERT_EQ(strnlen("hello", 0), (size_t)0, "maxlen = 0");

    TEST_PASS("strnlen: all tests passed");
}

static void test_strcpy(void)
{
    TEST_INFO("Testing strcpy...");

    char buffer[100];

    strcpy(buffer, "hello");
    TEST_ASSERT_STR_EQ(buffer, "hello", "basic copy");

    strcpy(buffer, "");
    TEST_ASSERT_EQ(buffer[0], '\0', "empty string copy");

    strcpy(buffer, "a");
    TEST_ASSERT_STR_EQ(buffer, "a", "single character");

    TEST_PASS("strcpy: all tests passed");
}

static void test_strcmp(void)
{
    TEST_INFO("Testing strcmp...");

    TEST_ASSERT_TRUE(strcmp("", "") == 0, "empty strings equal");
    TEST_ASSERT_TRUE(strcmp("abc", "abc") == 0, "identical strings");
    TEST_ASSERT_TRUE(strcmp("abc", "abd") < 0, "abc < abd");
    TEST_ASSERT_TRUE(strcmp("abd", "abc") > 0, "abd > abc");
    TEST_ASSERT_TRUE(strcmp("abc", "abcd") < 0, "prefix is less");

    TEST_PASS("strcmp: all tests passed");
}

static void test_strchr(void)
{
    TEST_INFO("Testing strchr...");

    char* p;

    p = strchr("hello", 'e');
    TEST_ASSERT_TRUE(p != NULL && p[0] == 'e', "find 'e' in 'hello'");

    p = strchr("hello", 'x');
    TEST_ASSERT_TRUE(p == NULL, "'x' not in 'hello'");

    p = strchr("hello", 'h');
    TEST_ASSERT_TRUE(p != NULL && p[0] == 'h', "find first char");

    p = strchr("hello", 'o');
    TEST_ASSERT_TRUE(p != NULL && p[0] == 'o', "find last char");

    TEST_PASS("strchr: all tests passed");
}

static void test_strstr(void)
{
    TEST_INFO("Testing strstr...");

    char* p;

    p = strstr("hello world", "world");
    TEST_ASSERT_TRUE(p != NULL && strcmp(p, "world") == 0, "find substring");

    p = strstr("hello world", "xyz");
    TEST_ASSERT_TRUE(p == NULL, "substring not found");

    p = strstr("hello", "");
    TEST_ASSERT_TRUE(p != NULL, "empty needle returns string");

    TEST_PASS("strstr: all tests passed");
}

static void test_strtok(void)
{
    TEST_INFO("Testing strtok_r...");

    char str[] = "one,two;three:four";
    char* saveptr;
    char* token;
    int count = 0;

    token = strtok_r(str, ",;:", &saveptr);
    TEST_ASSERT_TRUE(token != NULL && strcmp(token, "one") == 0, "first token");
    if (token) count++;

    token = strtok_r(NULL, ",;:", &saveptr);
    TEST_ASSERT_TRUE(token != NULL && strcmp(token, "two") == 0, "second token");
    if (token) count++;

    token = strtok_r(NULL, ",;:", &saveptr);
    TEST_ASSERT_TRUE(token != NULL && strcmp(token, "three") == 0, "third token");
    if (token) count++;

    token = strtok_r(NULL, ",;:", &saveptr);
    TEST_ASSERT_TRUE(token != NULL && strcmp(token, "four") == 0, "fourth token");
    if (token) count++;

    token = strtok_r(NULL, ",;:", &saveptr);
    TEST_ASSERT_TRUE(token == NULL, "no more tokens");

    TEST_ASSERT_EQ(count, 4, "found 4 tokens");

    TEST_PASS("strtok_r: all tests passed");
}

/* ============================================================================
 * Main
 * ============================================================================ */

int main(void)
{
    printf("╔════════════════════════════════════════════════════════════╗\n");
    printf("║           CCOS String Library Test Suite                  ║\n");
    printf("╚════════════════════════════════════════════════════════════╝\n\n");

    /* Run all tests */
    test_strlen();
    test_strnlen();
    test_strcpy();
    test_strcmp();
    test_strchr();
    test_strstr();
    test_strtok();

    /* Print summary */
    print_test_summary();

    /* Return exit code based on results */
    return g_test_failed > 0 ? 1 : 0;
}
```

## 第六步：添加更多测试

一个好的测试框架应该覆盖各种边界情况。继续添加更多测试：

```c
static void test_strncpy(void)
{
    TEST_INFO("Testing strncpy...");

    char buffer[10];

    // 正常复制
    strncpy(buffer, "hello", sizeof(buffer));
    buffer[9] = '\0';  // 确保结尾
    TEST_ASSERT_STR_EQ(buffer, "hello", "basic strncpy");

    // 限制长度
    strncpy(buffer, "hello world", 5);
    buffer[5] = '\0';  // 手动添加结尾
    TEST_ASSERT_STR_EQ(buffer, "hello", "truncated copy");

    // 复制空字符串
    strncpy(buffer, "", sizeof(buffer));
    TEST_ASSERT_EQ(buffer[0], '\0', "copy empty string");

    TEST_PASS("strncpy: all tests passed");
}

static void test_strcasecmp(void)
{
    TEST_INFO("Testing strcasecmp...");

    TEST_ASSERT_TRUE(strcasecmp("hello", "HELLO") == 0, "case insensitive equal");
    TEST_ASSERT_TRUE(strcasecmp("hello", "HELLP") < 0, "case insensitive compare");
    TEST_ASSERT_TRUE(strcasecmp("Hello", "hello") == 0, "mixed case equal");

    TEST_PASS("strcasecmp: all tests passed");
}
```
