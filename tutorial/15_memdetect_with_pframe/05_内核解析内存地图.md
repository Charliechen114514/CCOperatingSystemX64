# 05 - 内核解析内存地图

说实话，Bootloader 那边的 16 位汇编代码写完，感觉像是打完了一场硬仗。现在回到内核这边用 C 语言写代码，突然觉得轻松了不少。但我们不能掉以轻心，解析内存地图这件事，每一步都要小心翼翼。

---

## 我们现在要做什么

现在 Bootloader 已经把内存地图存储在固定地址 `0x6000` 了。我们的任务是在内核中读取这些数据，并提供一组 API 供后续代码使用。

具体来说，我们需要实现：
1. 从 `0x6000` 读取内存地图
2. 解析条目数据
3. 提供查询 API（如获取条目数量、获取特定条目）
4. 提供统计 API（如总内存、可用内存）
5. 提供验证 API（如检查某个地址范围是否可用）

我们会在 `kernel/mm/memory_detect/` 目录下创建这些代码。

## 第一步 —— 创建目录和头文件

### 创建头文件 `kernel/mm/memory_detect/e820.h`

这个头文件定义了 E820 相关的数据结构和 API：

```c
/**
 * @file e820.h
 * @brief E820 Memory Map Parser
 */

#pragma once

#include "defines/types.h"

/* ==============================================================================
 * E820 Memory Types
 * ============================================================================== */
typedef enum {
    E820_TYPE_USABLE = 1,       /* 可用 RAM */
    E820_TYPE_RESERVED = 2,     /* 保留区域 */
    E820_TYPE_ACPI_RECLAIM = 3, /* ACPI 可回收 */
    E820_TYPE_NVS = 4,          /* ACPI NVS */
    E820_TYPE_UNUSABLE = 5,     /* 不可用 */
} e820_mem_type_t;

/* ==============================================================================
 * Detection Method
 * ============================================================================== */
typedef enum {
    MEM_DETECT_E820 = 0,    /* E820 详细地图 */
    MEM_DETECT_E801 = 1,    /* E801 两区域 */
    MEM_DETECT_88H = 2,     /* INT 15h/88h */
    MEM_DETECT_UNKNOWN = 3, /* 无可用方法 */
} mem_detect_method_t;

/* ==============================================================================
 * E820 Entry Structure (must match bootloader layout)
 * ============================================================================== */
typedef struct PACKED {
    uint64_t base;       /* 基地址 */
    uint64_t length;     /* 长度 (字节) */
    uint32_t type;       /* 类型 */
    uint32_t acpi_attrs; /* ACPI 属性 */
} e820_entry_t;

/* ==============================================================================
 * Memory Statistics
 * ============================================================================== */
typedef struct {
    uint64_t total_bytes;    /* 总内存 (字节) */
    uint64_t usable_bytes;   /* 可用内存 (字节) */
    uint64_t reserved_bytes; /* 保留内存 (字节) */
    uint32_t total_mb;       /* 总内存 (MB) */
    uint32_t usable_mb;      /* 可用内存 (MB) */
    uint32_t entry_count;    /* 条目数量 */
} mem_stats_t;

/* ==============================================================================
 * API: Initialization
 * ============================================================================== */
/**
 * @brief 初始化 E820 内存地图解析器
 *
 * 从 Bootloader 存储的固定地址读取内存地图。
 * 必须在任何其他 E820 函数之前调用。
 */
void e820_init(void);

/* ==============================================================================
 * API: Query Functions
 * ============================================================================== */

/**
 * @brief 获取实际使用的内存检测方法
 */
mem_detect_method_t e820_get_detect_method(void);

/**
 * @brief 获取检测方法的名称字符串
 */
const char* e820_get_detect_method_name(mem_detect_method_t method);

/**
 * @brief 获取 E820 条目数量
 */
uint32_t e820_get_entry_count(void);

/**
 * @brief 获取指定索引的 E820 条目
 *
 * @param entry 输出条目数据的指针
 * @return true 条目存在，false 索引超出范围
 */
bool e820_get_entry(uint32_t index, e820_entry_t* entry);

/* ==============================================================================
 * API: Statistics Functions
 * ============================================================================== */

/**
 * @brief 计算并返回内存统计信息
 */
void e820_get_stats(mem_stats_t* stats);

/**
 * @brief 查找指定地址以上最大的可用内存区域大小
 */
uint64_t e820_get_usable_memory_above(uint64_t min_base);

/**
 * @brief 统计可用内存区域数量
 */
uint32_t e820_count_usable_regions(void);

/* ==============================================================================
 * API: Usability Check
 * ============================================================================== */

/**
 * @brief 检查指定内存范围是否完全可用
 *
 * @param base 基地址
 * @param length 长度
 * @return true 范围完全可用，false 包含不可用区域
 */
bool e820_is_range_usable(uint64_t base, uint64_t length);

/**
 * @brief 查找满足条件的可用内存区域
 *
 * @param min_base 最小基地址
 * @param min_length 最小长度要求
 * @param out_base 输出找到的基地址
 * @param out_length 输出找到的长度
 * @return true 找到，false 未找到
 */
bool e820_find_usable_range(uint64_t min_base, uint64_t min_length,
                            uint64_t* out_base, uint64_t* out_length);

/* ==============================================================================
 * API: Debug Functions
 * ============================================================================== */

/**
 * @brief 打印格式化的内存地图
 */
void e820_dump_map(void);

/**
 * @brief 获取 E820 类型的名称
 */
const char* e820_get_type_name(uint32_t type);
```

注意这个头文件使用了 `#pragma once`，这是现代编译器的标准做法。还注意 `e820_entry_t` 结构使用了 `PACKED` 属性，确保它与 Bootloader 中的布局完全一致。

## 第二步 —— 实现初始化函数

现在我们来实现 `e820.c`，首先是初始化函数。

### 包含头文件和定义常量

```c
/* ==============================================================================
 * CCOS - E820 Memory Map Parser Implementation
 * ============================================================================== */

#include "e820.h"
#include "base/memory.h"
#include "klogs/kprintf.h"
#include "klogs/ksnprintf.h"

/* ==============================================================================
 * Internal State
 * ============================================================================== */

/* Memory map storage - cached from bootloader */
static e820_entry_t s_e820_entries[128];
static uint32_t s_e820_entry_count = 0;
static mem_detect_method_t s_detect_method = MEM_DETECT_UNKNOWN;
static bool s_initialized = false;
```

### 实现 e820_init

`e820_init` 函数从 Bootloader 存储的固定地址读取内存地图：

```c
void e820_init(void) {
    if (s_initialized) {
        klog_warn("[E820] Already initialized\n");
        return;
    }

    /* Pointer to bootloader's memory map storage */
    volatile uint8_t* mem_map_ptr = (volatile uint8_t*)0x6000;

    /* Read detection method (first byte) */
    uint8_t method_byte = mem_map_ptr[0];
    s_detect_method = (mem_detect_method_t)method_byte;

    /* Read entry count (bytes 1-2, little endian 16-bit) */
    uint8_t byte1 = mem_map_ptr[1];
    uint8_t byte2 = mem_map_ptr[2];
    s_e820_entry_count = ((uint32_t)byte2 << 8) | (uint32_t)byte1;

    /* Validate entry count */
    if (s_e820_entry_count > 128) {
        klog_error("[E820] Invalid entry count: %u (max: 128)\n",
                   s_e820_entry_count);
        s_e820_entry_count = 0;
        s_detect_method = MEM_DETECT_UNKNOWN;
        return;
    }

    /* Copy entries from bootloader storage */
    /* Layout: [method (1 byte)][entry_count (2 bytes)][entries (24 bytes each)] */
    /* Entries start at offset 3 */
    volatile e820_entry_t* src_entries = (volatile e820_entry_t*)(mem_map_ptr + 3);

    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        /* Copy entry data */
        s_e820_entries[i].base = src_entries[i].base;
        s_e820_entries[i].length = src_entries[i].length;
        s_e820_entries[i].type = src_entries[i].type;
        s_e820_entries[i].acpi_attrs = src_entries[i].acpi_attrs;

        /* Skip zero-length entries */
        if (s_e820_entries[i].length == 0) {
            klog_warn("[E820] Entry %u: length is 0, skipping\n", i);
            s_e820_entries[i].type = 0;
            continue;
        }
    }

    s_initialized = true;

    klog_trace("[E820] Initialized: method=%s, entries=%u\n",
               e820_get_detect_method_name(s_detect_method),
               s_e820_entry_count);
}
```

### 读取顺序的坑

这里有个细节需要注意：我们使用 `volatile` 指针来读取内存。这是因为这个地址不在我们的数据段中，而是在固定的物理地址。`volatile` 告诉编译器不要优化掉这些读取操作。

另一个细节是字节序。Bootloader 存储的是 little-endian 格式，所以我们需要按照 `byte2 << 8 | byte1` 的顺序来组合 16 位值。

### 获取条目

```c
uint32_t e820_get_entry_count(void) {
    return s_e820_entry_count;
}

bool e820_get_entry(uint32_t index, e820_entry_t* entry) {
    if (!s_initialized) {
        klog_error("[E820] Not initialized\n");
        return false;
    }

    if (entry == NULL) {
        return false;
    }

    if (index >= s_e820_entry_count) {
        return false;
    }

    /* Copy entry data to caller's buffer */
    entry->base = s_e820_entries[index].base;
    entry->length = s_e820_entries[index].length;
    entry->type = s_e820_entries[index].type;
    entry->acpi_attrs = s_e820_entries[index].acpi_attrs;

    return true;
}
```

## 第四步 —— 实现统计函数

统计函数提供内存使用情况的概览。

### 获取内存统计

```c
void e820_get_stats(mem_stats_t* stats) {
    if (!s_initialized || stats == NULL) {
        return;
    }

    /* Initialize stats to zero */
    memset(stats, 0, sizeof(mem_stats_t));
    stats->entry_count = s_e820_entry_count;

    /* Calculate statistics by iterating through all entries */
    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        const e820_entry_t* entry = &s_e820_entries[i];

        stats->total_bytes += entry->length;

        switch (entry->type) {
            case E820_TYPE_USABLE:
                stats->usable_bytes += entry->length;
                break;
            case E820_TYPE_RESERVED:
            case E820_TYPE_UNUSABLE:
                stats->reserved_bytes += entry->length;
                break;
            /* ACPI reclaimable and NVS counted as reserved for safety */
            case E820_TYPE_ACPI_RECLAIM:
            case E820_TYPE_NVS:
                stats->reserved_bytes += entry->length;
                break;
            default:
                break;
        }
    }

    /* Convert to MB */
    stats->total_mb = (uint32_t)(stats->total_bytes / (1024 * 1024));
    stats->usable_mb = (uint32_t)(stats->usable_bytes / (1024 * 1024));
}
```

### 其他统计函数

```c
uint64_t e820_get_usable_memory_above(uint64_t min_base) {
    if (!s_initialized) {
        return 0;
    }

    uint64_t max_usable = 0;

    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        const e820_entry_t* entry = &s_e820_entries[i];

        if (entry->type != E820_TYPE_USABLE) {
            continue;
        }

        /* Check if entry is at or above min_base */
        if ((entry->base + entry->length) > min_base) {
            if (entry->length > max_usable) {
                max_usable = entry->length;
            }
        }
    }

    return max_usable;
}

uint32_t e820_count_usable_regions(void) {
    if (!s_initialized) {
        return 0;
    }

    uint32_t count = 0;

    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        if (s_e820_entries[i].type == E820_TYPE_USABLE) {
            count++;
        }
    }

    return count;
}
```

## 第五步 —— 实现可用性检查函数

这些函数允许内核检查某个地址范围是否可用。

### 检查范围是否可用

```c
bool e820_is_range_usable(uint64_t base, uint64_t length) {
    if (!s_initialized) {
        return false;
    }

    if (length == 0) {
        return true; /* Zero-length range is trivially usable */
    }

    /* Check each entry for overlap */
    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        const e820_entry_t* entry = &s_e820_entries[i];
        uint64_t entry_start = entry->base;
        uint64_t entry_end = entry->base + entry->length;

        /* Check for overlap */
        uint64_t range_end = base + length;

        /* No overlap */
        if (range_end <= entry_start || base >= entry_end) {
            continue;
        }

        /* If this entry overlaps and is NOT usable, range is not fully usable */
        if (entry->type != E820_TYPE_USABLE) {
            return false;
        }
    }

    return true;
}
```

### 查找可用范围

```c
bool e820_find_usable_range(uint64_t min_base, uint64_t min_length,
                            uint64_t* out_base, uint64_t* out_length) {
    if (!s_initialized || out_base == NULL || out_length == NULL) {
        return false;
    }

    /* Find the largest usable region at or above min_base */
    uint64_t best_base = 0;
    uint64_t best_length = 0;

    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        const e820_entry_t* entry = &s_e820_entries[i];

        if (entry->type != E820_TYPE_USABLE) {
            continue;
        }

        uint64_t entry_end = entry->base + entry->length;

        /* Check if entry is at or above min_base */
        if (entry_end <= min_base) {
            continue;
        }

        /* Calculate effective base (max of entry base and min_base) */
        uint64_t effective_base = (entry->base > min_base) ? entry->base : min_base;

        /* Check if we're past the entry */
        if (effective_base >= entry_end) {
            continue;
        }

        /* Calculate effective length */
        uint64_t effective_length = entry_end - effective_base;

        /* Check if this meets minimum length requirement */
        if (effective_length >= min_length && effective_length > best_length) {
            best_base = effective_base;
            best_length = effective_length;
        }
    }

    if (best_length > 0) {
        *out_base = best_base;
        *out_length = best_length;
        return true;
    }

    return false;
}
```

## 第六步 —— 实现调试函数

最后，我们实现一些调试函数，方便在开发过程中查看内存地图。

### 打印内存地图

```c
const char* e820_get_type_name(uint32_t type) {
    switch (type) {
        case E820_TYPE_USABLE:
            return "Usable";
        case E820_TYPE_RESERVED:
            return "Reserved";
        case E820_TYPE_ACPI_RECLAIM:
            return "ACPI Reclaim";
        case E820_TYPE_NVS:
            return "ACPI NVS";
        case E820_TYPE_UNUSABLE:
            return "Unusable";
        default:
            return "Unknown";
    }
}

static void format_size(uint64_t bytes, char* buf, size_t buf_size) {
    if (bytes >= 1024 * 1024 * 1024) {
        ksnprintf(buf, buf_size, "%llu GB", bytes / (1024 * 1024 * 1024));
    } else if (bytes >= 1024 * 1024) {
        ksnprintf(buf, buf_size, "%llu MB", bytes / (1024 * 1024));
    } else if (bytes >= 1024) {
        ksnprintf(buf, buf_size, "%llu KB", bytes / 1024);
    } else {
        ksnprintf(buf, buf_size, "%llu B", bytes);
    }
}

void e820_dump_map(void) {
    if (!s_initialized) {
        klog_error("[E820] Not initialized\n");
        return;
    }

    klog_info("[E820] Memory Map (method: %s, entries: %u)\n",
              e820_get_detect_method_name(s_detect_method),
              s_e820_entry_count);
    klog_info("[E820] %-18s %-18s %-12s\n", "Base", "Length", "Type");

    for (uint32_t i = 0; i < s_e820_entry_count; i++) {
        const e820_entry_t* entry = &s_e820_entries[i];
        char size_buf[32];
        char base_buf[32];

        format_size(entry->length, size_buf, sizeof(size_buf));
        ksnprintf(base_buf, sizeof(base_buf), "0x%016llX", entry->base);

        klog_info("[E820] %s  %s  %-12s\n",
                  base_buf, size_buf, e820_get_type_name(entry->type));
    }

    /* Print summary */
    mem_stats_t stats;
    e820_get_stats(&stats);
    klog_info("[E820] Summary: Total=%u MB, Usable=%u MB\n",
              stats.total_mb, stats.usable_mb);
}
```

## 常见问题

### Q: 为什么条目数量是 0？

A: 可能是以下原因：
1. Bootloader 的内存检测失败了
2. 存储地址 `0x6000` 不正确
3. 内核和 Bootloader 之间的数据格式不匹配

检查一下 Bootloader 的串口输出，确认内存检测是否成功。

### Q: 为什么打印的内存类型都是 Unknown？

A: 这可能是类型值不匹配。检查一下 Bootloader 和内核中类型常量的定义是否一致。

### Q: 为什么总内存大小不对？

A: 这可能是统计逻辑的问题。检查一下 `e820_get_stats` 函数，确保正确累加了所有条目的长度。
