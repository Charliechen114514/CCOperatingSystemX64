# 07 - 物理帧分配器实战

说实话，之前几篇文档都是在打基础，现在我们终于要实现真正的内存管理了。物理帧分配器是内核内存管理的基础，一旦实现完成，我们就能够动态分配物理内存了。这是一个重要的里程碑。

---

## 我们现在要做什么

我们需要实现一个完整的物理帧分配器，它能够：

1. 根据 E820 内存地图初始化位图
2. 标记保留区域为已用
3. 提供分配/释放物理帧的 API
4. 提供统计和调试功能

我们会在 `kernel/mm/pframe/` 目录下创建这些代码。

---

## 第一步 —— 创建目录和头文件

首先，我们需要创建物理帧分配器的目录结构。

```bash
mkdir -p kernel/mm/pframe
```

### 创建头文件 `kernel/mm/pframe/pframe.h`

这个头文件定义了物理帧分配器的数据结构和 API：

```c
/**
 * @file pframe.h
 * @brief Physical Frame Allocator
 */

#pragma once

#include "defines/types.h"

/* ==============================================================================
 * Constants
 * ============================================================================== */
#define PAGE_SIZE       4096    /* 4KB 页大小 */
#define PAGE_SHIFT      12      /* log2(PAGE_SIZE) */
#define PFRAME_MIN_ALLOC_BASE 0x100000  /* 1MB - 最小分配地址 */

/* ==============================================================================
 * Return Codes
 * ============================================================================== */
typedef enum {
    PFRAME_OK = 0,              /* 成功 */
    PFRAME_ERR_NOT_INIT = -1,   /* 未初始化 */
    PFRAME_ERR_OOM = -2,        /* 内存不足 */
    PFRAME_ERR_INVALID = -3,    /* 无效参数 */
    PFRAME_ERR_ALIGN = -4,      /* 对齐不支持 */
} pframe_result_t;

/* ==============================================================================
 * Statistics
 * ============================================================================== */
typedef struct {
    uint64_t total_frames;      /* 总帧数 */
    uint64_t free_frames;       /* 空闲帧数 */
    uint64_t allocated_frames;  /* 已分配帧数 */
    uint64_t reserved_frames;   /* 保留帧数 */
    uint64_t bitmap_size_bytes; /* 位图大小 (字节) */
    uint64_t managed_start;     /* 管理区起始地址 */
    uint64_t managed_end;       /* 管理区结束地址 */
} pframe_stats_t;

/* ==============================================================================
 * API: Initialization
 * ============================================================================== */
/**
 * @brief 初始化物理帧分配器
 *
 * @return PFRAME_OK 成功，错误码 失败
 *
 * 执行流程：
 * 1. 解析 E820 内存地图
 * 2. 计算位图大小
 * 3. 初始化位图（全部设为 1 = 保留）
 * 4. 标记可用内存区域为 0
 * 5. 标记内核和保留区域为 1
 */
pframe_result_t pframe_init(void);

/* ==============================================================================
 * API: Allocation
 * ============================================================================== */
/**
 * @brief 分配单个物理帧（4KB）
 *
 * @param out_addr 输出分配的物理地址
 * @return PFRAME_OK 成功，错误码 失败
 */
pframe_result_t pframe_alloc(physical_addr_t* out_addr);

/**
 * @brief 分配多个连续物理帧
 *
 * @param out_addr 输出分配的基地址
 * @param frame_count 帧数量
 * @return PFRAME_OK 成功，错误码 失败
 */
pframe_result_t pframe_alloc_n(physical_addr_t* out_addr, uint64_t frame_count);

/* ==============================================================================
 * API: Deallocation
 * ============================================================================== */
/**
 * @brief 释放单个物理帧
 *
 * @param addr 要释放的物理地址
 * @return PFRAME_OK 成功，错误码 失败
 */
pframe_result_t pframe_free(physical_addr_t addr);

/**
 * @brief 释放多个连续物理帧
 *
 * @param addr 基地址
 * @param frame_count 帧数量
 * @return PFRAME_OK 成功，错误码 失败
 */
pframe_result_t pframe_free_n(physical_addr_t addr, uint64_t frame_count);

/* ==============================================================================
 * API: Query
 * ============================================================================== */
/**
 * @brief 检查地址是否已分配
 */
bool pframe_is_allocated(physical_addr_t addr);

/**
 * @brief 获取总帧数
 */
uint64_t pframe_get_total_frames(void);

/**
 * @brief 获取空闲帧数
 */
uint64_t pframe_get_free_frames(void);

/**
 * @brief 获取详细统计信息
 */
pframe_result_t pframe_get_stats(pframe_stats_t* stats);

/* ==============================================================================
 * API: Debug
 * ============================================================================== */
/**
 * @brief 打印帧分配器状态
 */
void pframe_dump(void);
```

## 第二步 —— 实现初始化函数

初始化函数是物理帧分配器最复杂的部分。它需要根据 E820 内存地图来初始化位图，并标记各种保留区域。

### 内部状态和辅助函数

```c
/* ==============================================================================
 * CCOS - Physical Frame Allocator Implementation
 * ============================================================================== */

#include "pframe.h"
#include "assert/assert.h"
#include "base/memory.h"
#include "bitmap/bitmap.h"
#include "klogs/kprintf.h"
#include "klogs/ksnprintf.h"
#include "memory_detect/e820.h"

/* ==============================================================================
 * Constants
 * ============================================================================== */

/* 最大位图大小（支持 64GB RAM = 2MB 位图） */
#define PFRAME_MAX_BITMAP_SIZE (2 * 1024 * 1024)

/* 保留内存区域 */
#define PFRAME_BIOS_END 0x100000                 /* 1MB */
#define PFRAME_E820_STORAGE_ADDR 0x6000
#define PFRAME_PAGE_TABLE_START 0x9000
#define PFRAME_PAGE_TABLE_END 0xBFFF
#define PFRAME_KERNEL_LOAD_ADDR 0x10000
#define PFRAME_KERNEL_MAX_SIZE (1 * 1024 * 1024) /* 1MB */

/* ==============================================================================
 * Internal State
 * ============================================================================== */

/* 静态位图存储 */
static byte_t s_frame_bitmap_storage[PFRAME_MAX_BITMAP_SIZE];

/* 帧分配器状态 */
typedef struct {
    bitmap frame_map;         /* 位图 */
    uint64_t total_frames;    /* 总帧数 */
    uint64_t managed_base;    /* 管理区起始地址 */
    uint64_t managed_end;     /* 管理区结束地址 */
    uint64_t last_alloc_hint; /* 分配提示（用于局部性） */
    bool initialized;
} pframe_state_t;

static pframe_state_t s_pframe_state = {0};

/* ==============================================================================
 * Internal Helper Functions
 * ============================================================================== */

/* 地址转帧号 */
static inline uint64_t addr_to_frame(physical_addr_t addr) {
    return addr >> PAGE_SHIFT;
}

/* 帧号转地址 */
static inline physical_addr_t frame_to_addr(uint64_t frame) {
    return frame << PAGE_SHIFT;
}

/* 检查页对齐 */
static inline bool is_page_aligned(physical_addr_t addr) {
    return (addr & (PAGE_SIZE - 1)) == 0;
}

/* 标记区域为已用 */
static void mark_region_reserved(uint64_t base, uint64_t length) {
    if (length == 0) return;

    /* 对齐到页边界 */
    uint64_t start_frame = addr_to_frame((physical_addr_t)align_up(base, PAGE_SHIFT));
    uint64_t end_frame = addr_to_frame((physical_addr_t)align_down(base + length, PAGE_SHIFT));

    /* 限制在管理范围内 */
    if (start_frame >= s_pframe_state.total_frames) return;

    if (end_frame > s_pframe_state.total_frames) {
        end_frame = s_pframe_state.total_frames;
    }

    uint64_t frame_count = end_frame - start_frame;
    bitmap_set_range(&s_pframe_state.frame_map, start_frame, frame_count);
}

/* 标记区域为空闲 */
static void mark_region_free(uint64_t base, uint64_t length) {
    if (length == 0) return;

    /* 对齐到页边界 */
    uint64_t start_frame = addr_to_frame((physical_addr_t)align_up(base, PAGE_SHIFT));
    uint64_t end_frame = addr_to_frame((physical_addr_t)align_down(base + length, PAGE_SHIFT));

    /* 限制在管理范围内 */
    if (start_frame >= s_pframe_state.total_frames) return;

    if (end_frame > s_pframe_state.total_frames) {
        end_frame = s_pframe_state.total_frames;
    }

    uint64_t frame_count = end_frame - start_frame;
    bitmap_clear_range(&s_pframe_state.frame_map, start_frame, frame_count);
}
```

### 实现 pframe_init

```c
pframe_result_t pframe_init(void) {
    if (s_pframe_state.initialized) {
        klog_warn("[PFRAME] Already initialized\n");
        return PFRAME_OK;
    }

    /* 获取 E820 统计信息 */
    mem_stats_t e820_stats;
    e820_get_stats(&e820_stats);

    if (e820_stats.usable_bytes == 0) {
        klog_error("[PFRAME] No usable memory found\n");
        return PFRAME_ERR_INVALID;
    }

    /* 找到最高的可用内存地址 */
    uint64_t max_addr = 0;
    uint32_t entry_count = e820_get_entry_count();

    for (uint32_t i = 0; i < entry_count; i++) {
        e820_entry_t entry;
        if (e820_get_entry(i, &entry)) {
            uint64_t entry_end = entry.base + entry.length;
            /* 只考虑可用内存 */
            if (entry.type == E820_TYPE_USABLE && entry_end > max_addr) {
                max_addr = entry_end;
            }
        }
    }

    /* 对齐到页边界 */
    max_addr = align_down(max_addr, PAGE_SHIFT);

    /* 计算总帧数 */
    s_pframe_state.total_frames = bytes_to_pages(max_addr);
    s_pframe_state.managed_base = PFRAME_MIN_ALLOC_BASE;
    s_pframe_state.managed_end = max_addr;
    s_pframe_state.last_alloc_hint = addr_to_frame(PFRAME_MIN_ALLOC_BASE);

    /* 计算位图大小 */
    size_t bitmap_bytes = (s_pframe_state.total_frames + 7) / 8;

    if (bitmap_bytes > PFRAME_MAX_BITMAP_SIZE) {
        klog_error("[PFRAME] Bitmap size %zu exceeds maximum %u\n",
                   bitmap_bytes, PFRAME_MAX_BITMAP_SIZE);
        return PFRAME_ERR_INVALID;
    }

    /* 初始化位图（使用静态存储） */
    bitmap_init(&s_pframe_state.frame_map,
                s_frame_bitmap_storage,
                s_pframe_state.total_frames);

    /* 先将所有帧标记为已用 */
    bitmap_set_range(&s_pframe_state.frame_map, 0, s_pframe_state.total_frames);

    /* 标记所有 E820 可用区域为空闲 */
    for (uint32_t i = 0; i < entry_count; i++) {
        e820_entry_t entry;
        if (e820_get_entry(i, &entry)) {
            if (entry.type == E820_TYPE_USABLE) {
                mark_region_free(entry.base, entry.length);
            }
        }
    }

    /* 标记保留区域为已用 */
    mark_region_reserved(0, PFRAME_BIOS_END);                           /* BIOS 区 */
    mark_region_reserved(PFRAME_E820_STORAGE_ADDR, PAGE_SIZE);          /* E820 存储 */
    mark_region_reserved(PFRAME_PAGE_TABLE_START,                        /* 页表 */
                         PFRAME_PAGE_TABLE_END - PFRAME_PAGE_TABLE_START + 1);
    mark_region_reserved(PFRAME_KERNEL_LOAD_ADDR, PFRAME_KERNEL_MAX_SIZE); /* 内核 */
    mark_region_reserved((uint64_t)s_frame_bitmap_storage, PFRAME_MAX_BITMAP_SIZE); /* 位图本身 */

    s_pframe_state.initialized = true;

    /* 打印初始化摘要 */
    pframe_stats_t stats;
    pframe_get_stats(&stats);

    klog_info("[PFRAME] Initialized: %llu frames (%llu MB managed)\n",
              stats.total_frames,
              (stats.managed_end - stats.managed_start) / (1024 * 1024));
    klog_info("[PFRAME] Free: %llu, Reserved: %llu, Allocated: %llu\n",
              stats.free_frames, stats.reserved_frames, stats.allocated_frames);

    return PFRAME_OK;
}
```

## 第三步 —— 实现分配函数

分配函数是物理帧分配器的核心。我们需要实现单个帧和多个连续帧的分配。

### 分配单个帧

```c
pframe_result_t pframe_alloc(physical_addr_t* out_addr) {
    CCOS_ASSERT(out_addr != NULL);

    if (!s_pframe_state.initialized) {
        klog_error("[PFRAME] Allocator not initialized\n");
        return PFRAME_ERR_NOT_INIT;
    }

    /* 从上次分配位置开始查找空闲帧 */
    ssize_t frame_idx = bitmap_find_next_zero(&s_pframe_state.frame_map,
                                               s_pframe_state.last_alloc_hint);

    /* 如果没找到，从头再找 */
    if (frame_idx < 0) {
        frame_idx = bitmap_find_first_zero(&s_pframe_state.frame_map);
    }

    if (frame_idx < 0) {
        klog_warn("[PFRAME] Out of memory: no free frames\n");
        return PFRAME_ERR_OOM;
    }

    /* 标记为已用 */
    bitmap_set(&s_pframe_state.frame_map, frame_idx);

    /* 更新提示（提高局部性） */
    s_pframe_state.last_alloc_hint = frame_idx + 1;
    if (s_pframe_state.last_alloc_hint >= s_pframe_state.total_frames) {
        s_pframe_state.last_alloc_hint = addr_to_frame(PFRAME_MIN_ALLOC_BASE);
    }

    *out_addr = frame_to_addr(frame_idx);

    return PFRAME_OK;
}
```

### 分配多个连续帧

```c
pframe_result_t pframe_alloc_n(physical_addr_t* out_addr, uint64_t frame_count) {
    CCOS_ASSERT(out_addr != NULL);

    if (!s_pframe_state.initialized) {
        klog_error("[PFRAME] Allocator not initialized\n");
        return PFRAME_ERR_NOT_INIT;
    }

    if (frame_count == 0) {
        return PFRAME_ERR_INVALID;
    }

    if (frame_count > s_pframe_state.total_frames) {
        return PFRAME_ERR_OOM;
    }

    /* 搜索连续的空闲帧 */
    uint64_t start_frame = s_pframe_state.last_alloc_hint;
    bool found = false;
    uint64_t run_start = 0;
    uint64_t run_length = 0;

    /* 从提示位置到末尾搜索 */
    for (uint64_t i = start_frame; i < s_pframe_state.total_frames;) {
        if (!bitmap_test(&s_pframe_state.frame_map, i)) {
            /* 找到空闲帧 */
            if (run_length == 0) {
                run_start = i;
            }
            run_length++;

            if (run_length >= frame_count) {
                found = true;
                break;
            }

            i++;
        } else {
            /* 帧已用，重置计数 */
            run_length = 0;
            i++;
        }
    }

    /* 如果没找到，从头搜索到提示位置 */
    if (!found) {
        run_length = 0;
        uint64_t min_frame = addr_to_frame(PFRAME_MIN_ALLOC_BASE);

        for (uint64_t i = min_frame; i < start_frame && i < s_pframe_state.total_frames;) {
            if (!bitmap_test(&s_pframe_state.frame_map, i)) {
                if (run_length == 0) {
                    run_start = i;
                }
                run_length++;

                if (run_length >= frame_count) {
                    found = true;
                    break;
                }

                i++;
            } else {
                run_length = 0;
                i++;
            }
        }
    }

    if (!found) {
        klog_warn("[PFRAME] Out of memory: cannot allocate %llu contiguous frames\n",
                  frame_count);
        return PFRAME_ERR_OOM;
    }

    /* 标记为已用 */
    bitmap_set_range(&s_pframe_state.frame_map, run_start, frame_count);

    /* 更新提示 */
    s_pframe_state.last_alloc_hint = run_start + frame_count;
    if (s_pframe_state.last_alloc_hint >= s_pframe_state.total_frames) {
        s_pframe_state.last_alloc_hint = addr_to_frame(PFRAME_MIN_ALLOC_BASE);
    }

    *out_addr = frame_to_addr(run_start);

    return PFRAME_OK;
}
```

## 第四步 —— 实现释放函数

释放函数相对简单，只需要将对应的位清除即可。但我们需要做一些检查，防止错误释放。

### 释放单个帧

```c
pframe_result_t pframe_free(physical_addr_t addr) {
    if (!s_pframe_state.initialized) {
        klog_error("[PFRAME] Allocator not initialized\n");
        return PFRAME_ERR_NOT_INIT;
    }

    if (!is_page_aligned(addr)) {
        klog_error("[PFRAME] Invalid address to free: not page aligned (0x%X)\n", addr);
        return PFRAME_ERR_INVALID;
    }

    uint64_t frame = addr_to_frame(addr);

    if (frame >= s_pframe_state.total_frames) {
        klog_error("[PFRAME] Invalid address to free: out of range (0x%X)\n", addr);
        return PFRAME_ERR_INVALID;
    }

    /* 检查双倍释放 */
    if (!bitmap_test(&s_pframe_state.frame_map, frame)) {
        klog_warn("[PFRAME] Double-free detected: 0x%X\n", addr);
        return PFRAME_ERR_INVALID;
    }

    bitmap_clear(&s_pframe_state.frame_map, frame);

    return PFRAME_OK;
}
```

### 释放多个连续帧

```c
pframe_result_t pframe_free_n(physical_addr_t addr, uint64_t frame_count) {
    if (!s_pframe_state.initialized) {
        klog_error("[PFRAME] Allocator not initialized\n");
        return PFRAME_ERR_NOT_INIT;
    }

    if (frame_count == 0) {
        return PFRAME_ERR_INVALID;
    }

    if (!is_page_aligned(addr)) {
        klog_error("[PFRAME] Invalid address to free: not page aligned (0x%X)\n", addr);
        return PFRAME_ERR_INVALID;
    }

    uint64_t start_frame = addr_to_frame(addr);

    if (start_frame >= s_pframe_state.total_frames) {
        klog_error("[PFRAME] Invalid address to free: out of range (0x%X)\n", addr);
        return PFRAME_ERR_INVALID;
    }

    if (start_frame + frame_count > s_pframe_state.total_frames) {
        klog_error("[PFRAME] Free range exceeds managed memory\n");
        return PFRAME_ERR_INVALID;
    }

    /* 清除位 */
    bitmap_clear_range(&s_pframe_state.frame_map, start_frame, frame_count);

    return PFRAME_OK;
}
```

## 第五步 —— 实现查询和调试函数

最后，我们实现一些查询和调试函数，方便查看分配器的状态。

### 查询函数

```c
bool pframe_is_allocated(physical_addr_t addr) {
    if (!s_pframe_state.initialized) {
        return false;
    }

    uint64_t frame = addr_to_frame(addr);

    if (frame >= s_pframe_state.total_frames) {
        return false;
    }

    return bitmap_test(&s_pframe_state.frame_map, frame);
}

uint64_t pframe_get_total_frames(void) {
    if (!s_pframe_state.initialized) {
        return 0;
    }
    return s_pframe_state.total_frames;
}

uint64_t pframe_get_free_frames(void) {
    if (!s_pframe_state.initialized) {
        return 0;
    }

    pframe_stats_t stats;
    pframe_get_stats(&stats);
    return stats.free_frames;
}

pframe_result_t pframe_get_stats(pframe_stats_t* stats) {
    if (!s_pframe_state.initialized) {
        return PFRAME_ERR_NOT_INIT;
    }

    if (stats == NULL) {
        return PFRAME_ERR_INVALID;
    }

    stats->total_frames = s_pframe_state.total_frames;
    stats->managed_start = s_pframe_state.managed_base;
    stats->managed_end = s_pframe_state.managed_end;
    stats->bitmap_size_bytes = (s_pframe_state.total_frames + 7) / 8;

    /* 统计已分配和空闲帧数 */
    size_t allocated = bitmap_weight(&s_pframe_state.frame_map);
    stats->allocated_frames = allocated;
    stats->free_frames = s_pframe_state.total_frames - allocated;

    /* 保留帧数是 1MB 以下的帧数 */
    uint64_t min_alloc_frame = addr_to_frame(PFRAME_MIN_ALLOC_BASE);
    stats->reserved_frames = min_alloc_frame;

    return PFRAME_OK;
}
```

### 调试函数

```c
void pframe_dump(void) {
    if (!s_pframe_state.initialized) {
        klog_error("[PFRAME] Allocator not initialized\n");
        return;
    }

    pframe_stats_t stats;
    pframe_get_stats(&stats);

    klog_info("[PFRAME] Allocator State:\n");
    klog_info("[PFRAME]   Total frames:    %llu (%llu MB)\n",
              stats.total_frames,
              pages_to_bytes(stats.total_frames) / (1024 * 1024));
    klog_info("[PFRAME]   Free frames:     %llu\n", stats.free_frames);
    klog_info("[PFRAME]   Allocated:       %llu\n", stats.allocated_frames);
    klog_info("[PFRAME]   Reserved:        %llu\n", stats.reserved_frames);
    klog_info("[PFRAME]   Bitmap size:     %llu bytes\n", stats.bitmap_size_bytes);
    klog_info("[PFRAME]   Managed range:   0x%llX - 0x%llX\n",
              stats.managed_start, stats.managed_end);
}
```

## 第六步 —— 在内核中调用 pframe_init

现在我们需要在内核初始化时调用 `pframe_init`。

### 修改 kernel_init.c

在 `kernel/kernel_init.c` 中的 `kernel_init` 函数里添加物理帧分配器初始化：

```c
void kernel_init(void) {
    /* ... 现有初始化代码 ... */

    /* === 内存检测 (Stage 15) === */
    e820_init();
    e820_dump_map();

    /* === 物理帧分配器 (Stage 15) === */
    pframe_init();
    pframe_dump();

    /* ... 继续其他初始化 ... */
}
```

### 添加到 CMakeLists.txt

在 `kernel/mm/CMakeLists.txt` 中添加 pframe 模块：

```cmake
# Physical frame allocator
add_subdirectory(pframe)

target_link_libraries(kernel PUBLIC kernel_pframe)
```

创建 `kernel/mm/pframe/CMakeLists.txt`：

```cmake
add_library(kernel_pframe OBJECT
    pframe.c
)

target_include_directories(kernel_pframe PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(kernel_pframe PUBLIC
    kernel_bitmap
    kernel_memory_detect
```

## 常见问题

### Q: 为什么 reserved_frames 是 256？

A: 因为我们保留低 1MB（256 个 4KB 帧）不参与分配。这 256 帧包括 BIOS 数据区、页表、内核等。

### Q: 为什么 allocated_frames 也是 256？

A: 初始化时，我们标记了内核、页表、E820 存储区、位图本身等为已用。这些加起来大约是 256 帧（1MB）。

### Q: 为什么 managed_start 是 0x100000？

A: 这是我们配置的最小分配基址（1MB）。低 1MB 有很多保留区域，我们从 1MB 以上开始分配内存。

## 到这里我们完成了什么

很好，到这里我们已经完成了物理帧分配器的实现：

1. **实现了初始化函数** - 根据 E820 内存地图初始化位图，标记保留区域
2. **实现了分配函数** - 支持单个帧和多个连续帧的分配
3. **实现了释放函数** - 支持单个帧和多个连续帧的释放
4. **实现了查询函数** - 检查地址是否已分配、获取统计信息
5. **实现了调试函数** - 打印分配器状态

现在我们的内核终于有了物理内存管理能力！我们可以动态分配和释放物理内存了。
