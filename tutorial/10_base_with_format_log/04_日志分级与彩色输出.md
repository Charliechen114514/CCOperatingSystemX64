# 日志分级与彩色输出 —— Stage 10 专业日志系统实战指南

## 前言

写内核的时候，你会遇到各种调试信息：
- "正在初始化内存管理..."
- "页表地址：0x100000"
- "错误：无法分配内存"
- "警告：内存不足，使用备用方案"

这些信息的优先级不一样。有时候你只想看错误信息，不想被调试日志淹没；有时候你想看所有详细信息。

这就是日志分级系统的意义。通过给日志打上"等级"标签，你可以控制看到什么级别的信息。

更棒的是，我们可以给不同级别的日志加上颜色。这样在串口终端里一眼就能看出哪些是错误，哪些是警告。

准备好了吗？我们开始。

## 第一步：定义日志级别

首先，我们来定义日志级别。我们模仿 Linux kernel 的设计，使用 5 个级别：

| 级别 | 数值 | 用途 | 颜色 |
|------|------|------|------|
| TRACE | 0 | 最详细的调试信息 | 灰色 |
| DEBUG | 1 | 调试信息 | 紫色 |
| INFO | 2 | 一般信息 | 绿色 |
| WARN | 3 | 警告信息 | 黄色 |
| ERROR | 4 | 错误信息 | 红色 |

更新 `kernel/klogs/kprintf.h`：

```c
/**
 * @file kprintf.h
 * @brief 内核格式化输出系统（带日志分级）
 * @date 2026-02-16
 */

#pragma once

#include "base/varargs.h"
#include "defines/types.h"

/**
 * @brief 日志级别枚举
 *
 * 级别数值越小，优先级越低
 * 只有级别 >= 当前过滤级别的日志才会被输出
 */
typedef enum {
    KLOG_LEVEL_TRACE = 0,  // 最详细的调试信息
    KLOG_LEVEL_DEBUG = 1,  // 调试信息
    KLOG_LEVEL_INFO  = 2,  // 一般信息（默认级别）
    KLOG_LEVEL_WARN  = 3,  // 警告信息
    KLOG_LEVEL_ERROR = 4,  // 错误信息
} klog_level_t;

/**
 * @brief 后端类型枚举
 */
typedef enum {
    KLOG_BACKEND_NONE = 0,
    KLOG_BACKEND_SERIAL,
    KLOG_BACKEND_VGA,
    KLOG_BACKEND_COUNT
} klog_backend_t;

// ===== kprintf 函数 =====

/**
 * @brief 使用指定后端进行格式化输出
 */
void kprintf(klog_backend_t backend, const char* format, ...);

/**
 * @brief 使用 va_list 进行格式化输出
 */
void kvprintf(klog_backend_t backend, const char* format, va_list args);

// ===== 日志级别管理 =====

/**
 * @brief 设置日志过滤级别
 *
 * 只有级别 >= filtered_level 的日志才会被输出
 *
 * @param filtered_level 最小输出级别
 */
void klog_set_level(klog_level_t filtered_level);

/**
 * @brief 获取当前日志过滤级别
 *
 * @return 当前过滤级别
 */
klog_level_t klog_get_level(void);

/**
 * @brief 获取日志级别名称
 *
 * @param level 日志级别
 * @return 级别名称字符串
 */
const char* klog_level_name(klog_level_t level);

// ===== 分级日志宏 =====

/**
 * @brief TRACE 级别日志
 */
#define klog_trace(format, ...) klog_log(KLOG_LEVEL_TRACE, format, ##__VA_ARGS__)

/**
 * @brief DEBUG 级别日志
 */
#define klog_debug(format, ...) klog_log(KLOG_LEVEL_DEBUG, format, ##__VA_ARGS__)

/**
 * @brief INFO 级别日志
 */
#define klog_info(format, ...) klog_log(KLOG_LEVEL_INFO, format, ##__VA_ARGS__)

/**
 * @brief WARN 级别日志
 */
#define klog_warn(format, ...) klog_log(KLOG_LEVEL_WARN, format, ##__VA_ARGS__)

/**
 * @brief ERROR 级别日志
 */
#define klog_error(format, ...) klog_log(KLOG_LEVEL_ERROR, format, ##__VA_ARGS__)

// ===== 内部函数 =====

/**
 * @brief 分级日志输出（内部函数）
 *
 * @param level 日志级别
 * @param format 格式化字符串
 * @param ... 可变参数
 */
void klog_log(klog_level_t level, const char* format, ...);
```

## 第二步：理解 ANSI 转义序列

在实现颜色支持之前，我们需要先理解 ANSI 转义序列。

**什么是 ANSI 转义序列？**

ANSI 转义序列是一种控制终端显示的标准。它以 `ESC` 字符（ASCII 27，八进制 033）开头，后面跟着 `[` 和一些参数。

**基本格式**：

```
ESC[参数m
```

**常用颜色代码**：

| 代码 | 颜色 | 用途 |
|------|------|------|
| 30 | 黑色 | 前景 |
| 31 | 红色 | 前景（ERROR） |
| 32 | 绿色 | 前景（INFO） |
| 33 | 黄色 | 前景（WARN） |
| 34 | 蓝色 | 前景 |
| 35 | 紫色 | 前景（DEBUG） |
| 36 | 青色 | 前景 |
| 37 | 白色 | 前景 |
| 90 | 黑色（亮） | 前景（TRACE） |
| 90-97 | 亮色版本 | 前景 |
| 0 | 重置 | 恢复默认 |

**示例**：

```c
// 红色文本
"\033[31m这是红色文本\033[0m"

// 绿色文本
"\033[32m这是绿色文本\033[0m"
```

⚠️ **注意**

不是所有终端都支持 ANSI 颜色。但现代串口工具（如 picocom、minicom）基本都支持。

## 第三步：实现 ANSI 颜色支持

现在我们来创建一个颜色支持模块。

创建 `kernel/driver/serial/serial_color.h`：

```c
/**
 * @file serial_color.h
 * @brief ANSI 颜色代码定义
 * @date 2026-02-16
 */

#pragma once

/**
 * @brief ANSI 颜色枚举
 */
typedef enum {
    // 标准颜色
    ANSI_COLOR_BLACK   = 30,
    ANSI_COLOR_RED     = 31,
    ANSI_COLOR_GREEN   = 32,
    ANSI_COLOR_YELLOW  = 33,
    ANSI_COLOR_BLUE    = 34,
    ANSI_COLOR_MAGENTA = 35,
    ANSI_COLOR_CYAN    = 36,
    ANSI_COLOR_WHITE   = 37,

    // 亮色版本
    ANSI_COLOR_BRIGHT_BLACK   = 90,
    ANSI_COLOR_BRIGHT_RED     = 91,
    ANSI_COLOR_BRIGHT_GREEN   = 92,
    ANSI_COLOR_BRIGHT_YELLOW  = 93,
    ANSI_COLOR_BRIGHT_BLUE    = 94,
    ANSI_COLOR_BRIGHT_MAGENTA = 95,
    ANSI_COLOR_BRIGHT_CYAN    = 96,
    ANSI_COLOR_BRIGHT_WHITE   = 97,

    // 特殊
    ANSI_COLOR_RESET = 0,
} ansi_color_t;

/**
 * @brief 获取 ANSI 颜色转义序列
 *
 * @param color 颜色代码
 * @param buf 输出缓冲区
 * @param size 缓冲区大小
 * @return 写入的字节数
 */
int serial_color_ansi(ansi_color_t color, char* buf, int size);

/**
 * @brief 根据日志级别获取 ANSI 颜色
 *
 * @param level 日志级别
 * @return 对应的 ANSI 颜色
 */
ansi_color_t serial_color_for_log_level(klog_level_t level);
```

创建 `kernel/driver/serial/serial_color.c`：

```c
/**
 * @file serial_color.c
 * @brief ANSI 颜色代码生成
 * @date 2026-02-16
 */

#include "serial_color.h"
#include "klogs/kprintf.h"
#include "base/strhelpers.h"

int serial_color_ansi(ansi_color_t color, char* buf, int size) {
    if (size < 5) {
        return 0;  // 缓冲区太小
    }

    // ANSI 转义序列格式：ESC[Nm
    buf[0] = '\033';  // ESC 字符
    buf[1] = '[';

    // 将颜色数字转换为字符串
    itoa((int)color, buf + 2, 10);

    // 找到字符串结尾
    int len = 0;
    while (buf[2 + len] != '\0') {
        len++;
    }

    // 添加 'm' 结束符
    buf[2 + len] = 'm';
    buf[2 + len + 1] = '\0';

    return 2 + len + 1;
}

ansi_color_t serial_color_for_log_level(klog_level_t level) {
    switch (level) {
        case KLOG_LEVEL_TRACE:
            return ANSI_COLOR_BRIGHT_BLACK;  // 灰色
        case KLOG_LEVEL_DEBUG:
            return ANSI_COLOR_MAGENTA;       // 紫色
        case KLOG_LEVEL_INFO:
            return ANSI_COLOR_GREEN;         // 绿色
        case KLOG_LEVEL_WARN:
            return ANSI_COLOR_YELLOW;        // 黄色
        case KLOG_LEVEL_ERROR:
            return ANSI_COLOR_RED;           // 红色
        default:
            return ANSI_COLOR_WHITE;
    }
}
```

## 第四步：更新串口后端支持颜色

现在我们需要更新串口后端，让它支持彩色输出。

更新 `kernel/klogs/backends/serial_backends.c`：

```c
/**
 * @file serial_backends.c
 * @brief 串口后端实现（带颜色支持）
 * @date 2026-02-16
 */

#include "serial_backends.h"
#include "driver/serial/serial.h"
#include "driver/serial/serial_color.h"
#include "klogs/kprintf.h"

// 串口就绪标志
static bool g_serial_ready = false;

// 颜色缓冲区
static char g_color_buf[16];

// 串口输出函数（带颜色）
static void serial_process(const char* str, int level) {
    if (!g_serial_ready) {
        return;  // 串口未初始化
    }

    // 如果有级别，添加颜色
    if (level >= 0) {
        ansi_color_t color = serial_color_for_log_level(level);

        // 输出颜色代码
        int len = serial_color_ansi(color, g_color_buf, sizeof(g_color_buf));
        for (int i = 0; i < len; i++) {
            serial_write(g_color_buf[i]);
        }
    }

    // 输出字符串
    while (*str != '\0') {
        serial_write(*str);
        str++;
    }

    // 如果有级别，重置颜色
    if (level >= 0) {
        int len = serial_color_ansi(ANSI_COLOR_RESET, g_color_buf, sizeof(g_color_buf));
        for (int i = 0; i < len; i++) {
            serial_write(g_color_buf[i]);
        }
    }
}

// 检查串口是否就绪
static bool serial_is_ready(void) {
    return g_serial_ready;
}

// 后端操作接口
static const KLogBackendOps serial_backend_ops = {
    .process = serial_process,
    .is_ready = serial_is_ready,
};

bool klog_serial_backend_init(void) {
    serial_init();
    g_serial_ready = true;
    return true;
}

const KLogBackendOps* klog_serial_backend_get_ops(void) {
    return &serial_backend_ops;
}
```

**注意变化**

1. **检查 level 参数**：如果是 -1，表示不需要颜色（普通 `kprintf`）
2. **输出颜色代码**：在字符串前输出 ANSI 颜色序列
3. **重置颜色**：在字符串后输出重置序列，避免影响后续输出

## 第五步：实现分级日志系统

现在我们在 `kprintf.c` 中实现分级日志功能。

更新 `kernel/klogs/kprintf.c`：

```c
/**
 * @file kprintf.c
 * @brief kprintf 核心实现（带日志分级）
 * @date 2026-02-16
 */

#include "kprintf.h"
#include "kprintf_backends.h"
#include "kprintf_config.h"
#include "base/strhelpers.h"

// 格式化缓冲区
static char g_buffer[KPRINTF_BUFFER_SIZE];

// 数字转换缓冲区
static char g_format_buffer[32];

// 当前日志过滤级别
static klog_level_t g_log_level = KPRINTF_DEFAULT_FILTERED_LOGLEVEL;

// format_string 函数保持不变
static int format_string(char* buffer, size_t size, const char* format, va_list args) {
    // ... 和之前一样的实现 ...
}

void kvprintf(klog_backend_t backend, const char* format, va_list args) {
    const KLogBackendOps* ops = klog_get_backend_ops(backend);
    if (ops == NULL || !ops->is_ready()) {
        return;
    }

    format_string(g_buffer, KPRINTF_BUFFER_SIZE, format, args);
    // level = -1 表示无级别
    ops->process(g_buffer, -1);
}

void kprintf(klog_backend_t backend, const char* format, ...) {
    va_list args;
    va_start(args, format);
    kvprintf(backend, format, args);
    va_end(args);
}

// ===== 日志级别管理 =====

void klog_set_level(klog_level_t filtered_level) {
    g_log_level = filtered_level;
}

klog_level_t klog_get_level(void) {
    return g_log_level;
}

const char* klog_level_name(klog_level_t level) {
    switch (level) {
        case KLOG_LEVEL_TRACE: return "TRACE";
        case KLOG_LEVEL_DEBUG: return "DEBUG";
        case KLOG_LEVEL_INFO:  return "INFO ";
        case KLOG_LEVEL_WARN:  return "WARN ";
        case KLOG_LEVEL_ERROR: return "ERROR";
        default:               return "?????";
    }
}

// ===== 分级日志输出 =====

void klog_log(klog_level_t level, const char* format, ...) {
    // 检查日志级别过滤
    if (level < g_log_level) {
        return;  // 级别太低，不输出
    }

    klog_backend_t backend = klog_get_default_backend();
    const KLogBackendOps* ops = klog_get_backend_ops(backend);
    if (ops == NULL || !ops->is_ready()) {
        return;
    }

    // 格式：[LEVEL] message\n
    int pos = 0;
    g_buffer[pos++] = '[';

    // 添加级别名称
    const char* level_str = klog_level_name(level);
    for (int i = 0; i < 5 && level_str[i] != '\0'; i++) {
        g_buffer[pos++] = level_str[i];
    }

    g_buffer[pos++] = ']';
    g_buffer[pos++] = ' ';

    // 格式化用户消息
    va_list args;
    va_start(args, format);
    int len = format_string(g_buffer + pos, KPRINTF_BUFFER_SIZE - pos - 2, format, args);
    va_end(args);

    pos += len;

    // 添加换行
    g_buffer[pos++] = '\n';
    g_buffer[pos] = '\0';

    // 输出（带级别颜色）
    ops->process(g_buffer, level);
}
```

## 踩坑预警

### 坑 1：颜色代码不生效

如果你的串口工具不支持 ANSI 颜色，你可能看到这样的输出：

```
[31m[ERROR][0m This is an error
```

**解决方案**：
- 使用支持 ANSI 颜色的串口工具（picocom、minicom）
- 或者添加一个配置选项，禁用颜色输出

### 坑 2：级别名称对齐

```c
// 注意这里
case KLOG_LEVEL_INFO:  return "INFO ";  // 有空格！
case KLOG_LEVEL_WARN:  return "WARN ";  // 有空格！
```

我们用空格让所有级别名称都是 5 个字符，这样输出时 `[INFO ]` 和 `[ERROR]` 对齐，更美观。

### 坑 3：缓冲区大小不足

```c
// 危险
char small_buf[10];
klog_info("This is a very long message that will overflow the buffer...");
```

确保 `KPRINTF_BUFFER_SIZE` 足够大（我们用的是 512 字节）。

### 坑 4：忘记重置颜色

```c
// 错误写法
serial_color_ansi(ANSI_COLOR_RED, buf, sizeof(buf));
serial_write_string(buf);
serial_write_string("Error!");
// 忘记重置，后续所有输出都是红色！
```

一定要在输出后重置颜色：`serial_color_ansi(ANSI_COLOR_RESET, ...)`
