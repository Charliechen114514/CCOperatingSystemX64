# 支持大内核加载

## 前言：18 字节玩够了

如果你跟着前面的教程做，现在应该有一个能工作的 20 字节内核。但说实话，这没什么用 —— 一个真正的内核起码要几 KB，甚至几十 KB。

### 当前限制

我们的 `load_kernel` 函数只读取 1 个扇区（512 字节）：

```nasm
mov ah, 0x02            ; 读取功能
mov al, 0x01            ; 读取 1 个扇区 ← 问题在这
```

这意味着内核不能超过 512 字节。这显然不够。

### 解决方案

今天我们要：
1. 实现 LBA Extended Read，支持多扇区读取
2. 自动计算内核大小，动态读取
3. 添加 LBA/CHS 自动降级，确保兼容性

---

## 第一步：理解 LBA Extended Read

LBA Extended Read（INT 13h, AH=42h）是 BIOS 扩展磁盘读取功能。

### DAP 结构

DAP（Disk Address Packet）是 LBA 读取使用的数据结构：

```nasm
struct DAP {
    uint8_t  size;        // 数据包大小（16）
    uint8_t  reserved;    // 保留（0）
    uint16_t block_count; // 要读取的块数
    uint16_t offset;      // 目标缓冲区偏移
    uint16_t segment;     // 目标缓冲区段
    uint64_t lba;         // 起始 LBA
};
```

### 示例

```nasm
; 读取从 LBA 3 开始的 127 个扇区到 0x10000

mov si, dap              ; DS:SI 指向 DAP
mov ah, 0x42             ; 扩展读取功能
mov dl, 0x80             ; 第一个硬盘
int 0x13                 ; 调用 BIOS

; DAP 内容：
dap:
    db 16                ; size = 16
    db 0                 ; reserved = 0
    dw 127               ; block_count = 127
    dw 0x0000            ; offset = 0
    dw 0x1000            ; segment = 0x1000 (地址 = 0x10000)
    dq 3                 ; lba = 3
```

## 第三步：实现多扇区读取

### LBA 读取函数

```nasm
; ============================================================================
; read_sectors_lba - 使用 LBA 扩展读取
; 输入：EAX = 起始 LBA
;      CX  = 要读取的扇区数
;      ES:BX = 目标缓冲区
; 输出：CF=0 成功，CF=1 失败
; ============================================================================
bits 16
read_sectors_lba:
    pusha

    ; 验证扇区数（最多 127）
    cmp cx, 0
    je .error
    cmp cx, 127
    jbe .count_ok
    mov cx, 127             ; 最多读 127 个扇区
.count_ok:

    ; 设置 DS 指向我们的代码段
    push ax
    mov ax, cs
    mov ds, ax
    pop ax

    ; 填充 DAP
    mov byte [dap + 2], cl      ; block count (低字节)
    mov byte [dap + 3], 0       ; block count (高字节)

    mov [dap + 4], bx           ; offset
    mov word [dap + 6], es      ; segment

    mov dword [dap + 8], eax    ; LBA (低 32 位)
    mov dword [dap + 12], 0     ; LBA (高 32 位) = 0

    ; 执行 LBA 读取
    mov si, dap
    mov dl, 0x80
    mov ah, 0x42
    int 0x13

    ; 恢复 DS
    push ax
    xor ax, ax
    mov ds, ax
    pop ax

    jc .error

    popa
    clc
    ret

.error:
    ; 恢复 DS
    push ax
    xor ax, ax
    mov ds, ax
    pop ax
    popa
    stc
    ret
```

### DAP 结构

```nasm
; ============================================================================
; DAP (Disk Address Packet)
; ============================================================================
align 4
dap:
    db 16                   ; 数据包大小
    db 0                    ; 保留
    dw 0                    ; 块数（运行时填充）
    dw 0                    ; 目标偏移（运行时填充）
    dw 0                    ; 目标段（运行时填充）
    dq 0                    ; 起始 LBA（运行时填充）
```

## 第四步：实现循环读取

如果内核很大（超过 127 扇区），我们需要循环读取。

```nasm
; ============================================================================
; load_kernel_lba - 使用 LBA 加载内核（支持多扇区）
; 输入：使用 KERNEL_LBA_START 和 KERNEL_SECTOR_COUNT
; 输出：CF=0 成功，CF=1 失败
; ============================================================================
bits 16
load_kernel_lba:
    pusha

    ; 设置目标地址
    mov ax, KERNEL_LOAD_SEGMENT
    mov es, ax
    mov bx, KERNEL_LOAD_OFFSET

    ; 初始化跟踪变量
    mov di, KERNEL_SECTOR_COUNT     ; DI = 剩余扇区数
    mov si, KERNEL_LBA_START        ; SI = 当前 LBA

.read_loop:
    ; 检查是否读完
    cmp di, 0
    je .read_complete

    ; 计算本次读取的扇区数（最多 127）
    mov cx, di
    cmp cx, 127
    jbe .sectors_ok
    mov cx, 127
.sectors_ok:

    ; 保存扇区数
    mov bp, cx

    ; 调用 LBA 读取
    mov ax, si
    call read_sectors_lba
    jc .read_error

    ; 更新跟踪变量
    sub di, bp              ; 减少剩余扇区数
    add si, bp              ; 前进 LBA

    ; 前进缓冲区指针（ES:BX += BP * 512）
    push ax
    push dx
    mov ax, bp
    xor dx, dx
    mov cx, 512
    mul cx                  ; DX:AX = 读取的字节数
    add bx, ax              ; 前进 BX
    pop dx
    pop ax

    jmp .read_loop

.read_complete:
    popa
    clc
    ret

.read_error:
    popa
    stc
    ret
```

## 第五步：添加 CHS 降级

如果 LBA 不支持，我们需要回退到 CHS。

### LBA 转 CHS

```nasm
; ============================================================================
; lba_to_chs - 将 LBA 转换为 CHS 地址
; 输入：AX = LBA 地址（0-based）
; 输出：CH = 柱面，CL = 扇区（1-based，bits 0-5），DH = 磁头
; ============================================================================
bits 16
lba_to_chs:
    push bx

    ; 保存 LBA
    mov bx, ax

    ; 计算 LBA / SECTORS_PER_TRACK
    xor dx, dx
    mov ax, bx
    mov cx, SECTORS_PER_TRACK
    div cx                  ; AX = LBA / SPT, DX = LBA % SPT

    ; 保存余数（扇区索引 0-based）
    push dx

    ; 计算 柱面 = (LBA / SPT) / HEADS
    xor dx, dx
    mov cx, HEADS
    div cx                  ; AX = 柱面, DX = 磁头

    mov ch, al              ; CH = 柱面（低 8 位）
    mov dh, dl              ; DH = 磁头

    ; 计算 扇区 = (LBA % SPT) + 1
    pop dx                  ; 恢复扇区索引
    mov cl, dl              ; CL = 扇区（0-based）
    add cl, 1               ; 转换为 1-based

    pop bx
    ret
```

### CHS 读取

```nasm
; ============================================================================
; load_kernel_chs - 使用 CHS 加载内核（支持多扇区）
; 输入：使用 KERNEL_LBA_START 和 KERNEL_SECTOR_COUNT
; 输出：CF=0 成功，CF=1 失败
; ============================================================================
bits 16
load_kernel_chs:
    pusha

    ; 设置目标地址
    mov ax, KERNEL_LOAD_SEGMENT
    mov es, ax
    mov bx, KERNEL_LOAD_OFFSET

    ; 初始化跟踪变量
    mov di, KERNEL_SECTOR_COUNT
    mov si, KERNEL_LBA_START

.read_loop:
    ; 检查是否读完
    cmp di, 0
    je .read_complete

    ; 计算本次读取的扇区数
    mov cx, di
    cmp cx, 127
    jbe .sectors_ok
    mov cx, 127
.sectors_ok:

    ; 保存扇区数
    mov bp, cx

    ; 转换 LBA 到 CHS
    mov ax, si
    call lba_to_chs         ; CH=柱面, DH=磁头, CL=扇区(1-based)

    ; 执行 CHS 读取
    mov ah, 0x02
    mov al, bl              ; BL = 扇区数（从 CX 复制的）
    mov dl, 0x80
    int 0x13
    jc .read_error

    ; 验证扇区数
    cmp al, bl
    jne .read_mismatch

    ; 更新跟踪变量
    sub di, bp
    add si, bp

    ; 前进缓冲区
    push ax
    push dx
    mov ax, bp
    xor dx, dx
    mov cx, 512
    mul cx
    add bx, ax
    pop dx
    pop ax

    jmp .read_loop

.read_complete:
    popa
    clc
    ret

.read_error:
.read_mismatch:
    popa
    stc
    ret
```

## 第六步：自动选择 LBA/CHS

```nasm
; ============================================================================
; load_kernel_auto - 自动选择 LBA/CHS 加载内核
; 输出：CF=0 成功，CF=1 失败
; ============================================================================
bits 16
load_kernel_auto:
    pusha

    ; 先尝试 LBA
    call check_lba_support
    jc .try_chs

    ; LBA 可用
    call load_kernel_lba
    jnc .success

    ; LBA 失败，回退到 CHS
.try_chs:
    call load_kernel_chs
    jc .error

.success:
    popa
    clc
    ret

.error:
    popa
    stc
    ret
```

## 第七步：添加消息输出

为了让调试更方便，我们添加串口和 VGA 输出。

### 串口初始化

```nasm
; ============================================================================
; Serial Port Constants
; ============================================================================
SERIAL_PORT    equ 0x3F8
SERIAL_THR     equ 0x03F8
SERIAL_LCR     equ 0x03FB
SERIAL_DLAB    equ 0x80
LCR_8BIT       equ 0x03

; ============================================================================
; serial_init - 初始化串口
; ============================================================================
bits 16
serial_init:
    push dx

    ; 设置波特率（115200）
    mov dx, SERIAL_LCR
    mov al, SERIAL_DLAB | LCR_8BIT
    out dx, al

    mov dx, 0x3F8
    mov al, 0x01
    out dx, al

    mov dx, 0x3F9
    mov al, 0x00
    out dx, al

    ; 设置 8N1
    mov dx, SERIAL_LCR
    mov al, LCR_8BIT
    out dx, al

    pop dx
    ret

; ============================================================================
; serial_write_char - 写一个字符到串口
; ============================================================================
bits 16
serial_write_char:
    push dx
    push ax
.write_loop:
    mov dx, 0x3FD
    in al, dx
    test al, 0x20
    jz .write_loop
    pop ax
    mov dx, SERIAL_THR
    out dx, al
    pop dx
    ret
```

### 打印数字

```nasm
; ============================================================================
; print_decimal - 打印 AX 为十进制
; ============================================================================
bits 16
print_decimal:
    push bx
    push cx
    push dx

    mov bx, 10
    xor cx, cx
.div_loop:
    xor dx, dx
    div bx
    push dx
    inc cx
    test ax, ax
    jnz .div_loop
.print_loop:
    pop ax
    add al, '0'
    call serial_write_char
    loop .print_loop

    pop dx
    pop cx
    pop bx
    ret
```

## 第九步：自动化配置

为了让内核大小自动更新，我们用脚本生成配置。

### 生成脚本

```bash
#!/bin/bash
# gen_boot_config.sh

KERNEL_BIN="build/kernel.bin"
BOOTLOADER_BIN="build/bootloader.bin"
OUTPUT_FILE="boot/boot_config.inc"

# 计算内核扇区数
KERNEL_SIZE=$(stat -c%s "$KERNEL_BIN" 2>/dev/null || echo 0)
KERNEL_SECTORS=$(( (KERNEL_SIZE + 511) / 512 ))

# 计算 bootloader 扇区数
BOOTLOADER_SIZE=$(stat -c%s "$BOOTLOADER_BIN" 2>/dev/null || echo 0)
BOOTLOADER_SECTORS=$(( (BOOTLOADER_SIZE + 511) / 512 ))

# 生成配置文件
cat > "$OUTPUT_FILE" << EOF
; CCOS Bootloader Configuration
; AUTO-GENERATED - DO NOT EDIT

; 内核位置
%define KERNEL_LBA_START          $BOOTLOADER_SECTORS
%define KERNEL_SECTOR_COUNT       $KERNEL_SECTORS

; 内存布局
%define KERNEL_LOAD_SEGMENT       0x1000
%define KERNEL_LOAD_OFFSET        0x0000

; Bootloader 大小
%define BOOTLOADER_SECTORS        $BOOTLOADER_SECTORS

; CHS 几何
%define SECTORS_PER_TRACK         63
%define HEADS                     16
EOF

echo "Generated boot_config.inc:"
echo "  Kernel: $KERNEL_SIZE bytes = $KERNEL_SECTORS sectors"
echo "  Bootloader: $BOOTLOADER_SIZE bytes = $BOOTLOADER_SECTORS sectors"
echo "  Kernel LBA start: $BOOTLOADER_SECTORS"
```

### 更新 Makefile

```makefile
# 生成配置
BOOT_CONFIG_INC := $(BOOT_DIR)/boot_config.inc

$(BOOT_CONFIG_INC): $(BOOTLOADER_BIN) $(KERNEL_BIN)
	@echo "Generating boot configuration..."
	@bash scripts/gen_boot_config.sh

$(BOOTLOADER_BIN): $(BOOT_CONFIG_INC) $(BOOTLOADER_ASM) | prepare
	@echo "Assembling bootloader..."
	$(AS) $(ASFLAGS) $(BOOTLOADER_ASM) -o $@
```

## 踩坑预警

⚠️ **注意**：127 扇区限制

INT 13h AH=42h 一次最多读取 127 个扇区。超过这个数需要循环读取。

⚠️ **注意**：缓冲区溢出

多扇区读取会前进缓冲区指针，确保检查 `ES:BX` 不会溢出！

⚠️ **注意**：LBA vs CHS 扇区编号

- LBA：扇区 0 是第一个
- CHS：扇区 1 是第一个

转换时记得 +1！

⚠️ **注意**：配置文件生成顺序

bootloader.asm 依赖 boot_config.inc，而 boot_config.inc 依赖 bootloader.bin... 这是循环依赖！

解决方法：
1. 第一次编译用默认配置
2. 编译后生成正确配置
3. 重新编译 bootloader

或者：让 boot_config.inc 依赖上一次的 kernel.bin，而不是当前的。
