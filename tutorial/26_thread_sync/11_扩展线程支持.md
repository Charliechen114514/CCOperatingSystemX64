# 扩展线程支持 —— 完整的线程管理

## 前言

在之前的文章中，我们实现了完整的同步原语。现在可以利用这些原语来实现真正的线程支持。

线程（thread）是同一进程内的多个执行流，它们共享地址空间，但有独立的栈和寄存器状态。线程比进程更轻量，创建和切换的开销更小。

这篇文章会实现内核线程和用户线程的创建，以及 join/detach 机制。

## 第一步——扩展 PCB 结构

我们需要在进程控制块（PCB）中添加线程相关字段：

```c
// 在 kernel/process/process.h 中扩展 pcb_t

typedef struct pcb {
    // 原有字段...

    // 线程标识
    int32_t tgid;              /* 线程组 ID（主线程的 PID） */
    bool is_thread;            /* 是否为线程 */

    // 线程组织
    list_head thread_list;     /* 同组线程链表 */
    list_head thread_group;    /* 线程组链表节点 */

    // 线程入口
    virtual_addr_t thread_entry;
    void* thread_arg;

    // 用户栈
    virtual_addr_t user_stack;
    size_t user_stack_size;

    // Join/Detach
    list_head join_waiters;
    bool detached;
    void* return_value;
} pcb_t;
```

## 第二步——实现内核线程创建

```c
// 在 kernel/process/thread.c 中

pcb_t* proc_create_kernel_thread(thread_fn_t entry, void* arg) {
    pcb_t* current = proc_current();

    // 分配 PCB
    pcb_t* thread = kmalloc(sizeof(pcb_t));
    if (!thread) {
        return NULL;
    }

    // 初始化基本字段
    memset(thread, 0, sizeof(pcb_t));
    thread->pid = alloc_pid();
    thread->tgid = current->tgid;  /* 继承线程组 */
    thread->is_thread = true;
    thread->is_user_mode = false;
    thread->state = PROC_READY;

    // 分配内核栈
    thread->kernel_stack = alloc_kernel_stack();
    if (!thread->kernel_stack) {
        kfree(thread);
        return NULL;
    }

    // 设置线程入口
    thread->thread_entry = (virtual_addr_t)entry;
    thread->thread_arg = arg;

    // 共享地址空间
    thread->mm = current->mm;

    // 加入线程组
    spinlock_flags_t flags;
    spin_lock_irqsave(&current->thread_list_lock, &flags);
    list_add(&thread->thread_group, &current->thread_list);
    spin_unlock_irqrestore(&current->thread_list_lock, flags);

    // 初始化 join 支持
    INIT_LIST_HEAD(&thread->join_waiters);
    thread->detached = false;
    thread->return_value = NULL;

    // 加入调度器
    schedule_add(thread);

    return thread;
}
```

## 第三步——实现用户线程创建

```c
pcb_t* proc_create_user_thread(virtual_addr_t entry,
                                virtual_addr_t stack,
                                void* arg) {
    pcb_t* parent = proc_current();

    pcb_t* thread = kmalloc(sizeof(pcb_t));
    if (!thread) {
        return NULL;
    }

    memset(thread, 0, sizeof(pcb_t));
    thread->pid = alloc_pid();
    thread->tgid = parent->tgid;
    thread->is_thread = true;
    thread->is_user_mode = true;
    thread->state = PROC_READY;

    // 设置用户栈
    if (stack == 0) {
        thread->user_stack = alloc_user_stack(DEFAULT_STACK_SIZE);
        thread->user_stack_size = DEFAULT_STACK_SIZE;
    } else {
        thread->user_stack = stack;
        thread->user_stack_size = 0;
    }

    // 设置线程入口
    thread->thread_entry = entry;
    thread->thread_arg = arg;

    // 共享地址空间
    thread->mm = parent->mm;

    // 设置 trap_frame
    setup_user_thread_trap_frame(thread, entry, stack, arg);

    // 加入线程组
    spinlock_flags_t flags;
    spin_lock_irqsave(&parent->thread_list_lock, &flags);
    list_add(&thread->thread_group, &parent->thread_list);
    spin_unlock_irqrestore(&parent->thread_list_lock, flags);

    INIT_LIST_HEAD(&thread->join_waiters);
    thread->detached = false;
    thread->return_value = NULL;

    schedule_add(thread);

    return thread;
}
```

## 第四步——实现线程退出

```c
void proc_thread_exit(void* retval) {
    pcb_t* current = proc_current();

    current->return_value = retval;

    if (current->detached) {
        // 分离线程，直接清理
        reap_thread(current);
    } else {
        // 设置为僵尸状态，等待 join
        current->state = PROC_ZOMBIE;

        // 唤醒所有 join 等待者
        spinlock_flags_t flags;
        spin_lock_irqsave(&current->state_lock, &flags);
        pcb_t* waiter, *tmp;
        list_for_each_entry_safe(waiter, tmp, &current->join_waiters, wait_list) {
            list_del(&waiter->wait_list);
            waiter->state = PROC_READY;
            schedule_add(waiter);
        }
        spin_unlock_irqrestore(&current->state_lock, flags);
    }

    // 调度出去，永不返回
    schedule();
    __builtin_unreachable();
}
```

## 第五步——实现线程 join

```c
int proc_thread_join(int32_t pid, void** retval) {
    pcb_t* current = proc_current();
    pcb_t* thread = find_thread_by_pid(current->tgid, pid);

    if (!thread || thread->tgid != current->tgid) {
        return -EINVAL;
    }

    if (thread->detached) {
        return -EINVAL;
    }

    // 等待线程退出
    spinlock_flags_t flags;
    spin_lock_irqsave(&thread->state_lock, &flags);

    while (thread->state != PROC_ZOMBIE) {
        // 加入 join 等待队列
        list_add(&current->wait_list, &thread->join_waiters);
        current->state = PROC_BLOCKED;

        spin_unlock_irqrestore(&thread->state_lock, flags);
        schedule();
        spin_lock_irqsave(&thread->state_lock, &flags);
    }

    // 收集返回值
    if (retval) {
        *retval = thread->return_value;
    }

    spin_unlock_irqrestore(&thread->state_lock, flags);

    // 清理线程
    reap_thread(thread);

    return 0;
}
```

## 第六步——实现线程 detach

```c
int proc_thread_detach(pcb_t* thread) {
    if (thread->detached) {
        return -EINVAL;
    }

    spinlock_flags_t flags;
    spin_lock_irqsave(&thread->state_lock, &flags);

    thread->detached = true;

    // 如果已经退出，直接清理
    if (thread->state == PROC_ZOMBIE) {
        spin_unlock_irqrestore(&thread->state_lock, flags);
        reap_thread(thread);
    } else {
        spin_unlock_irqrestore(&thread->state_lock, flags);
    }

    return 0;
}
```

## 使用示例

```c
// 内核线程示例
void my_thread_func(void* arg) {
    int value = *(int*)arg;
    klog_info("Thread running with arg: %d\n", value);

    // 做一些工作...

    proc_thread_exit(NULL);
}

void create_kernel_thread_example(void) {
    int arg = 42;
    pcb_t* thread = proc_create_kernel_thread(my_thread_func, &arg);
    if (!thread) {
        klog_error("Failed to create thread\n");
        return;
    }
}
```
