# 编写同步演示程序 —— 验证所有功能

## 前言

在实现了所有同步原语和线程支持之后，我们需要编写演示程序来验证它们的功能。演示程序不仅能验证正确性，还能展示如何使用这些同步原语。

这篇文章会创建 `kernel/demo/sync/` 目录，编写各种演示程序，包括并发计数器、生产者-消费者、读者-写者等经典场景。

## 第二步——Mutex 并发测试

```c
// kernel/demo/sync/sync_demo.h

#ifndef SYNC_DEMO_H
#define SYNC_DEMO_H

#include "sync/mutex.h"

// 共享计数器和锁
extern mutex_t demo_mutex;
extern int demo_counter;

// 演示函数
void demo_mutex_test(void);

#endif
```

```c
// kernel/demo/sync/sync_demo.c

#include "demo/sync/sync_demo.h"
#include "sync/mutex.h"
#include "process/thread.h"
#include "klogs/kprintf.h"

mutex_t demo_mutex;
int demo_counter = 0;

// 线程函数：增加计数器
void counter_thread(void* arg) {
    for (int i = 0; i < 1000; i++) {
        mutex_lock(&demo_mutex);
        demo_counter++;
        mutex_unlock(&demo_mutex);
    }
    proc_thread_exit(NULL);
}

void demo_mutex_test(void) {
    klog_info("=== Mutex Demo Start ===\n");

    mutex_init(&demo_mutex);
    demo_counter = 0;

    // 创建 10 个线程
    pcb_t* threads[10];
    for (int i = 0; i < 10; i++) {
        threads[i] = proc_create_kernel_thread(counter_thread, NULL);
    }

    // 等待所有线程完成
    for (int i = 0; i < 10; i++) {
        proc_thread_join(threads[i]->pid, NULL);
    }

    klog_info("Final counter: %d (expected: 10000)\n", demo_counter);

    if (demo_counter == 10000) {
        klog_info("Mutex test PASSED\n");
    } else {
        klog_error("Mutex test FAILED\n");
    }
}
```

## 第三步——信号量生产者-消费者

```c
// 在 sync_demo.c 中添加

#define BUFFER_SIZE 5
static int buffer[BUFFER_SIZE];
static int in = 0, out = 0;

static semaphore_t empty_slots;
static semaphore_t full_slots;
static mutex_t buffer_mutex;

void producer_thread(void* arg) {
    int id = *(int*)arg;

    for (int i = 0; i < 10; i++) {
        int item = id * 100 + i;

        sem_wait(&empty_slots);
        mutex_lock(&buffer_mutex);

        buffer[in] = item;
        in = (in + 1) % BUFFER_SIZE;

        klog_info("Producer %d: produced %d\n", id, item);

        mutex_unlock(&buffer_mutex);
        sem_post(&full_slots);
    }

    proc_thread_exit(NULL);
}

void consumer_thread(void* arg) {
    int id = *(int*)arg;

    for (int i = 0; i < 10; i++) {
        sem_wait(&full_slots);
        mutex_lock(&buffer_mutex);

        int item = buffer[out];
        out = (out + 1) % BUFFER_SIZE;

        klog_info("Consumer %d: consumed %d\n", id, item);

        mutex_unlock(&buffer_mutex);
        sem_post(&empty_slots);
    }

    proc_thread_exit(NULL);
}

void demo_semaphore_test(void) {
    klog_info("=== Semaphore Demo Start ===\n");

    sem_init(&empty_slots, BUFFER_SIZE);
    sem_init(&full_slots, 0);
    mutex_init(&buffer_mutex);
    in = out = 0;

    // 创建生产者和消费者
    pcb_t* threads[4];
    int ids[2] = {1, 2};

    threads[0] = proc_create_kernel_thread(producer_thread, &ids[0]);
    threads[1] = proc_create_kernel_thread(producer_thread, &ids[1]);
    threads[2] = proc_create_kernel_thread(consumer_thread, &ids[0]);
    threads[3] = proc_create_kernel_thread(consumer_thread, &ids[1]);

    for (int i = 0; i < 4; i++) {
        proc_thread_join(threads[i]->pid, NULL);
    }

    klog_info("Semaphore test PASSED\n");
}
```

## 第四步——读写锁测试

```c
// 在 sync_demo.c 中添加

static rwlock_t rwlock_demo;
static int shared_data = 0;
static int read_count = 0;

void reader_thread(void* arg) {
    for (int i = 0; i < 100; i++) {
        rwlock_read_lock(&rwlock_demo);

        // 多个读者可以同时读取
        int value = shared_data;
        (void)value;  // 避免未使用警告

        rwlock_read_unlock(&rwlock_demo);
    }

    proc_thread_exit(NULL);
}

void writer_thread(void* arg) {
    for (int i = 0; i < 50; i++) {
        rwlock_write_lock(&rwlock_demo);

        shared_data++;

        rwlock_write_unlock(&rwlock_demo);
    }

    proc_thread_exit(NULL);
}

void demo_rwlock_test(void) {
    klog_info("=== RWLock Demo Start ===\n");

    rwlock_init(&rwlock_demo);
    shared_data = 0;

    pcb_t* threads[5];

    // 4 个读者
    for (int i = 0; i < 4; i++) {
        threads[i] = proc_create_kernel_thread(reader_thread, NULL);
    }

    // 1 个写者
    threads[4] = proc_create_kernel_thread(writer_thread, NULL);

    for (int i = 0; i < 5; i++) {
        proc_thread_join(threads[i]->pid, NULL);
    }

    klog_info("Final shared_data: %d (expected: 50)\n", shared_data);
    klog_info("RWLock test PASSED\n");
}
```
