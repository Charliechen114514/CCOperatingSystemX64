# 长模式之路 —— 最终进入 64 位世界

## 前言

终于到了这一刻！

我们从实模式开始，经历了保护模式，现在要踏入 64 位长模式（Long Mode）了。

说实话，这部分内容是整个 Bootloader 开发中最复杂的。你需要：
1. 设置页表（PML4, PDPT, PD, PT）
2. 启用 PAE（物理地址扩展）
3. 设置 EFER.LME 位
4. 启用分页
5. Far Jump 到 64 位代码

一步出错，Triple Fault 等着你。

但别担心，我们一步一步来。

---

## 理解长模式

### 什么是长模式？

长模式（Long Mode）是 x86-64 处理器的原生 64 位模式：
- 64 位地址空间（可以访问 2^64 字节内存）
- 16 个通用寄存器（RAX-R15）
- 扩展和新增的指令
- 兼容模式（可以运行 32 位和 16 位代码）

### 如何进入长模式？

进入长模式的条件：
1. CPU 必须支持 x86-64
2. 必须先进入保护模式
3. 必须启用 PAE（Physical Address Extension）
4. 必须设置页表
5. 必须启用分页

**关键**：启用分页时，CR4.PAE=1 和 EFER.LME=1，CPU 就会进入长模式。

---

## 第一步：理解页表结构

在长模式下，页表是 4 级结构：

```
┌─────────────────────────────────────────────────────┐
│ 48 位虚拟地址                                        │
├──────────┬──────────┬──────────┬──────────┬─────────┤
│ PML4 索引│ PDPT 索引 │  PD 索引  │  PT 索引  │ 偏移    │
│  (9 位)  │  (9 位)  │  (9 位)  │  (9 位)  │(12 位)  │
└──────────┴──────────┴──────────┴──────────┴─────────┘

每个索引选择一个表项，每个表项 8 字节。
每个表 512 项（2^9 = 512）。
每个页 4KB（2^12 = 4096）。
```

**简单映射**：为了快速开始，我们先做一个简单的恒等映射（虚拟地址 = 物理地址）：

```
虚拟地址    物理地址    大小
0x000000    0x000000    2MB
0x200000    0x200000    2MB
...
0xFF800000000 0xFF800000000 2MB  (内核映射)
```

## 第二步：设置页表

在保护模式代码中添加页表设置：

```asm
; ==============================================================================
; 32 位保护模式代码
; ==============================================================================
bits 32

pm_entry:
    ; ========== 设置段寄存器 ==========
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x7E00

    ; ========== 打印保护模式消息 ==========
    mov esi, pm_success_msg
    call print_string_pm

    ; ========== 设置页表 ==========
    call setup_page_tables

    ; ========== 打印页表设置消息 ==========
    mov esi, page_tables_ok_msg
    call print_string_pm

    ; ========== 启用 PAE ==========
    mov eax, cr4
    or eax, 1 << 5      ; 设置位 5 (PAE)
    mov cr4, eax

    ; ========== 加载 PML4 到 CR3 ==========
    mov eax, 0x9000    ; PML4 地址
    mov cr3, eax

    ; ========== 设置 EFER.LME ==========
    mov ecx, 0xC0000080    ; EFER MSR 地址
    rdmsr                  ; 读取 MSR 到 EDX:EAX
    or eax, 1 << 8         ; 设置位 8 (LME)
    wrmsr                  ; 写回 MSR

    ; ========== 启用分页（同时进入长模式）==========
    mov eax, cr0
    or eax, 1 << 31     ; 设置位 31 (PG)
    mov cr0, eax

    ; ========== Far Jump 到 64 位代码 ==========
    jmp 0x18:long_mode   ; 段选择子 0x18 = 描述符 3

; ========== 这里不会到达 ==========
.hang:
    hlt
    jmp .hang
```

## 第三步：编写页表设置函数

```asm
; ==============================================================================
; setup_page_tables - 设置页表
; 输入: 无
; 修改: EAX, EDI, ECX
; ==============================================================================
setup_page_tables:
    pusha

    ; ========== 清空 PML4 ==========
    mov edi, 0x9000       ; PML4 地址
    mov ecx, 4096 / 4     ; 1024 个双字
    xor eax, eax
    rep stosd             ; 填充 0

    ; ========== 清空 PDPT ==========
    mov edi, 0xA000       ; PDPT 地址
    mov ecx, 4096 / 4
    xor eax, eax
    rep stosd

    ; ========== 清空 PD ==========
    mov edi, 0xB000       ; PD 地址
    mov ecx, 4096 / 4
    xor eax, eax
    rep stosd

    ; ========== 设置 PML4[0] -> PDPT ==========
    mov dword [0x9000], 0x0000A003
    ; Base = 0xA000, Present, Writable
    mov dword [0x9004], 0x00000000

    ; ========== 设置 PML4[511] -> PDPT（高地址映射）==========
    mov dword [0x9FF8], 0x0000A003
    mov dword [0x9FFC], 0x00000000

    ; ========== 设置 PDPT[0] -> PD ==========
    mov dword [0xA000], 0x0000B003
    mov dword [0xA004], 0x00000000

    ; ========== 设置 PD[0] -> 2MB 页 ==========
    mov dword [0xB000], 0x00000083
    ; Base = 0x00000, Present, Writable, Huge page (2MB)
    mov dword [0xB004], 0x00000000

    ; ========== 设置 PD[511] -> 2MB 页（高地址映射）==========
    mov dword [0xBFF0], 0x00000083
    mov dword [0xBFF4], 0x00000000

    popa
    ret
```

⚠️ **关键点**

1. **页表必须对齐到 4KB 边界**
   - `0x9000`, `0xA000`, `0xB000` 都是 4KB 对齐的

2. **页表项格式**：
   ```
   63                  52 51   40 39     12 11         0
   ┌────────────────────┬────────┬─────────┬────────────┐
   │    Reserved        │ Available│ Physical │   Flags   │
   │                    │         │  Address │           │
   └────────────────────┴─────────┴─────────┴────────────┘

   Flags:
   - 位 0 (P): Present
   - 位 1 (R/W): Read/Write
   - 位 2 (U/S): User/Supervisor
   - 位 7 (PS): Page Size (1 = 2MB/4MB)
   ```

3. **Huge Page（PS 位）**
   - 在 PD 中设置 PS=1 表示 2MB 页（不需要 PT）

## 第四步：编写 64 位代码

```asm
; ==============================================================================
; 64 位长模式代码
; ==============================================================================
bits 64

long_mode:
    ; ========== 设置段寄存器 ==========
    mov ax, 0x20        ; 64 位数据段选择子
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov rsp, 0x7E00

    ; ========== 清屏（使用新的 VGA 写法）==========
    mov rdi, 0xB8000
    mov rcx, 80 * 25    ; 80 列 × 25 行
    mov rax, 0x0720     ; 空格，白色字符
.clear_loop:
    mov [rdi], rax
    add rdi, 2
    loop .clear_loop

    ; ========== 打印长模式消息 (VGA) ==========
    mov rsi, lm_welcome_msg
    mov rdi, 0xB8000
    mov ah, 0x0F        ; 亮白色字符
.print_loop:
    lodsb               ; 加载 [RSI] 到 AL
    test al, al
    jz .print_done
    stosw               ; 写入显存
    jmp .print_loop
.print_done:

    ; ========== 打印完成消息 ==========
    mov rsi, lm_done_msg
    mov rdi, 0xB8000
    mov ah, 0x0E        ; 黄色字符
.loop2:
    lodsb
    test al, al
    jz .done2
    stosw
    jmp .loop2
.done2:

    ; ========== 停止 ==========
.hang:
    hlt
    jmp .hang
```

## 第八步：验证长模式

你可以用 GDB 验证 CPU 确实在 64 位模式下：

```bash
# 启动 QEMU with GDB
qemu-system-x86_64 \
    -drive format=raw,file=../build/boot.img,if=floppy \
    -nographic \
    -s -S

# 在另一个终端
gdb
(gdb) target remote localhost:1234
(gdb) set architecture i386:x86-64
(gdb) break *0x7E00
(gdb) continue
(gdb) si 500
(gdb) info registers
```

你应该看到：
- `RAX`, `RBX`, `RCX`... 等 64 位寄存器
- `CS` 段选择子 = `0x18`
- `RIP` 在 64 位代码段

## 常见问题

### 问题 1：Triple Fault（无限重启）

**症状**：QEMU 不断重启。

**原因**：
1. 页表设置错误
2. CR4.PAE 或 EFER.LME 没有正确设置
3. GDT 中没有 64 位代码段描述符

**解决**：
- 检查页表地址是否对齐
- 检查描述符 3 的 L 位是否设置
- 用 GDB 单步调试

### 问题 2：Far Jump 后卡住

**原因**：
1. 64 位代码段选择子错误
2. `bits 64` 没有设置
3. 目标地址错误

**检查**：
```asm
jmp 0x18:long_mode   ; 0x18 = 描述符 3 (24 / 8 = 3)
```
