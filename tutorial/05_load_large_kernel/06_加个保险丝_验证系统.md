# 加个保险丝：验证系统

有一次，内核变大了直接把 bootloader 覆盖了，我血压拉满。

---

## 前言 - 一次惨痛的经历

那天我给内核加了一些新功能，编译后一运行，系统直接卡死。

我用 GDB 调试，发现 bootloader 的后半部分被内核覆盖了。

问题出在哪？我计算磁盘布局时出了错，内核的起始扇区号不对，导致内核被写入到了 bootloader 的区域。

从那以后，我发誓要写一个验证脚本，在每次构建后自动检查磁盘布局。

---

## 验证脚本：verify_disk_layout.py

我们的验证脚本要做这些事情：

1. 检查 MBR 签名是否正确
2. 计算 bootloader 和内核的内存布局
3. 检查内存是否有重叠
4. 检查内存间隙是否足够
5. 检查内核大小是否超过限制

```python
#!/usr/bin/env python3
"""
CCOS Disk Layout Verification Script

这个脚本验证 bootloader 和内核在内存中不会重叠，
并检查磁盘布局是否有潜在问题。

用法：
    python3 scripts/verify_disk_layout.py build/bootloader.bin build/kernel.bin
"""

import sys
import struct
from pathlib import Path


# -----------------------------------------------------------------------------
# 第一步：读取二进制文件
# -----------------------------------------------------------------------------

def read_file(path):
    """读取二进制文件的内容。"""
    with open(path, 'rb') as f:
        return f.read()


# -----------------------------------------------------------------------------
# 第二步：检查 MBR 签名
# -----------------------------------------------------------------------------

def check_mbr_signature(data):
    """
    检查数据是否有有效的 MBR 签名（0x55AA 在偏移 510 处）。

    MBR 签名是磁盘启动扇区的最后两个字节，必须是 0x55AA。
    如果不是这个值，BIOS 不会认为这是一个可启动的磁盘。
    """
    if len(data) < 512:
        return False, "File too small for MBR check"

    # struct.unpack 用于解析二进制数据
    # '<H' 表示小端序（little-endian）的无符号短整数（2 字节）
    # data[510:512] 是偏移 510 到 512 的两字节
    signature = struct.unpack('<H', data[510:512])[0]

    if signature != 0xAA55:
        return False, f"Invalid MBR signature: 0x{signature:04X} (expected 0xAA55)"

    return True, "Valid MBR signature"


# -----------------------------------------------------------------------------
# 第三步：分析 bootloader 结构
# -----------------------------------------------------------------------------

def analyze_bootloader(data):
    """
    分析 bootloader 的结构和大小。

    返回一个字典，包含以下信息：
    - size: 文件大小（字节）
    - sectors: 占用的扇区数
    - has_mbr: 是否有 MBR 签名
    - mbr_valid: MBR 签名是否有效
    - stage2_offset: Stage 2 的偏移位置（如果能检测到）
    """
    info = {
        'size': len(data),
        'sectors': (len(data) + 511) // 512,  # 向上取整计算扇区数
        'has_mbr': False,
        'mbr_valid': False,
        'stage2_offset': None,
    }

    # 检查 MBR 签名（在偏移 510 处）
    if len(data) >= 512:
        signature = struct.unpack('<H', data[510:512])[0]
        info['has_mbr'] = True
        info['mbr_valid'] = (signature == 0xAA55)

    # 检测 Stage 2 的位置
    # Stage 2 应该从偏移 512（第 2 扇区）开始
    if len(data) > 512:
        # 检查 GDT 签名或常见的 Stage 2 模式
        # GDT 空描述符是 8 个零字节
        if data[512:520] == b'\x00' * 8:
            info['stage2_offset'] = 512

    return info


# -----------------------------------------------------------------------------
# 第四步：计算内存布局
# -----------------------------------------------------------------------------

def calculate_memory_layout(bootloader_sectors, kernel_sectors):
    """
    计算内存和磁盘布局，检测是否有重叠。

    参数：
        bootloader_sectors: bootloader 占用的扇区数
        kernel_sectors: 内核占用的扇区数

    返回：
        一个字典，包含磁盘布局、内存布局、警告和错误
    """
    layout = {
        'disk': {},
        'memory': {},
        'warnings': [],
        'errors': [],
    }

    # -------------------------------------------------
    # 磁盘布局（LBA）
    # -------------------------------------------------
    layout['disk']['bootloader_start'] = 0      # LBA 0（扇区 1）
    layout['disk']['bootloader_end'] = bootloader_sectors - 1
    layout['disk']['kernel_start'] = bootloader_sectors
    layout['disk']['kernel_end'] = bootloader_sectors + kernel_sectors - 1

    # -------------------------------------------------
    # 内存布局
    # -------------------------------------------------
    # Stage 1: 0x7C00
    # Stage 2: 0x7E00
    # Kernel: 0x10000

    # 计算 Stage 2 的结束地址
    # Stage 2 从 0x7E00 开始，大小为 (bootloader_sectors - 2) * 512
    # 减 2 是因为前 2 个扇区是 Stage 1（0x7C00-0x7E00）
    # 但我们简化计算：假设 bootloader 从 0x7E00 开始，占 bootloader_sectors 个扇区
    stage2_start = 0x7E00
    stage2_end = 0x7E00 + (bootloader_sectors * 512)

    kernel_start = 0x10000
    kernel_end = 0x10000 + (kernel_sectors * 512)

    layout['memory']['stage1'] = (0x7C00, 0x7E00)
    layout['memory']['stage2'] = (stage2_start, stage2_end)
    layout['memory']['kernel'] = (kernel_start, kernel_end)

    # -------------------------------------------------
    # 检查内存重叠
    # -------------------------------------------------
    if stage2_end > kernel_start:
        layout['errors'].append(
            f"Memory overlap: Stage 2 ends at 0x{stage2_end:X}, "
            f"kernel starts at 0x{kernel_start:X}"
        )
    elif (kernel_start - stage2_end) < 512:
        layout['warnings'].append(
            f"Small memory gap: {kernel_start - stage2_end} bytes between "
            f"Stage 2 (0x{stage2_end:X}) and kernel (0x{kernel_start:X})"
        )

    # -------------------------------------------------
    # 检查内核大小（最大 64MB）
    # -------------------------------------------------
    max_kernel_sectors = (64 * 1024 * 1024) // 512  # 64MB = 131072 扇区
    if kernel_sectors > max_kernel_sectors:
        layout['errors'].append(
            f"Kernel too large: {kernel_sectors} sectors ({kernel_sectors * 512 // (1024*1024)}MB), "
            f"max {max_kernel_sectors} sectors (64MB)"
        )

    return layout


# -----------------------------------------------------------------------------
# 第五步：打印布局信息
# -----------------------------------------------------------------------------

def print_layout(layout):
    """以友好的格式打印布局信息。"""
    print("\n" + "=" * 60)
    print("DISK LAYOUT (LBA)")
    print("=" * 60)
    print(f"Bootloader:  Sector 1-{layout['disk']['bootloader_end'] + 1} "
          f"(LBA 0-{layout['disk']['bootloader_end']})")
    print(f"Kernel:      Sector {layout['disk']['kernel_start'] + 1}-"
          f"{layout['disk']['kernel_end'] + 1} "
          f"(LBA {layout['disk']['kernel_start']}-{layout['disk']['kernel_end']})")

    print("\n" + "=" * 60)
    print("MEMORY LAYOUT")
    print("=" * 60)
    stage1 = layout['memory']['stage1']
    stage2 = layout['memory']['stage2']
    kernel = layout['memory']['kernel']
    print(f"Stage 1:     0x{stage1[0]:04X} - 0x{stage1[1]:04X}")
    print(f"Stage 2:     0x{stage2[0]:04X} - 0x{stage2[1]:04X}")
    print(f"Kernel:      0x{kernel[0]:04X} - 0x{kernel[1]:04X}")

    # 计算间隙大小
    stage2_size = stage2[1] - stage2[0]
    gap = kernel[0] - stage2[1]
    print(f"\nStage 2 size: {stage2_size} bytes ({stage2_size // 512} sectors)")
    print(f"Memory gap:  {gap} bytes ({gap // 512} sectors)")

    # 打印警告
    if layout['warnings']:
        print("\n" + "=" * 60)
        print("WARNINGS")
        print("=" * 60)
        for warning in layout['warnings']:
            print(f"  ⚠ {warning}")

    # 打印错误
    if layout['errors']:
        print("\n" + "=" * 60)
        print("ERRORS")
        print("=" * 60)
        for error in layout['errors']:
            print(f"  ✗ {error}")
        return False

    return True


# -----------------------------------------------------------------------------
# 主函数
# -----------------------------------------------------------------------------

def main():
    """主函数：解析参数并执行验证。"""
    if len(sys.argv) < 3:
        print("Usage: python3 verify_disk_layout.py <bootloader.bin> <kernel.bin>")
        sys.exit(1)

    bootloader_path = Path(sys.argv[1])
    kernel_path = Path(sys.argv[2])

    if not bootloader_path.exists():
        print(f"Error: Bootloader not found: {bootloader_path}")
        sys.exit(1)

    if not kernel_path.exists():
        print(f"Error: Kernel not found: {kernel_path}")
        sys.exit(1)

    print("CCOS Disk Layout Verification")
    print("=" * 60)

    # -------------------------------------------------
    # 读取文件
    # -------------------------------------------------
    bootloader_data = read_file(bootloader_path)
    kernel_data = read_file(kernel_path)

    # -------------------------------------------------
    # 分析 bootloader
    # -------------------------------------------------
    print("\nAnalyzing bootloader...")
    boot_info = analyze_bootloader(bootloader_data)
    print(f"  Size: {boot_info['size']} bytes ({boot_info['sectors']} sectors)")
    print(f"  Has MBR: {boot_info['has_mbr']}")
    if boot_info['has_mbr']:
        valid, msg = check_mbr_signature(bootloader_data)
        boot_info['mbr_valid'] = valid
        print(f"  {msg}")

    # -------------------------------------------------
    # 分析内核
    # -------------------------------------------------
    print("\nAnalyzing kernel...")
    kernel_sectors = (len(kernel_data) + 511) // 512
    kernel_size_mb = len(kernel_data) / (1024 * 1024)
    print(f"  Size: {len(kernel_data)} bytes ({kernel_sectors} sectors, {kernel_size_mb:.2f} MB)")

    # -------------------------------------------------
    # 计算布局
    # -------------------------------------------------
    layout = calculate_memory_layout(boot_info['sectors'], kernel_sectors)

    # -------------------------------------------------
    # 打印结果
    # -------------------------------------------------
    success = print_layout(layout)

    # -------------------------------------------------
    # 检查内核的魔法数字
    # -------------------------------------------------
    print("\n" + "=" * 60)
    print("KERNEL MAGIC NUMBERS")
    print("=" * 60)

    # 检查内核开头的魔法数字
    if len(kernel_data) >= 4:
        first_dwords = struct.unpack('<I', kernel_data[:4])[0]
        print(f"  First dword: 0x{first_dwords:08X}")

        # 检查是否是 ELF 文件（如果意外地把 ELF 当成 raw binary）
        if kernel_data[:4] == b'\x7FELF':
            print("  ⚠ Warning: Kernel appears to be an ELF file, should be raw binary")

    # 检查是否是 x86-64 代码
    if len(kernel_data) >= 3:
        if kernel_data[0] == 0x48:  # REX.W 前缀，常见于 x86-64
            print("  ✓ Appears to be x86-64 code (REX prefix found)")

    # -------------------------------------------------
    # 返回结果
    # -------------------------------------------------
    print("\n" + "=" * 60)
    if success:
        print("✓ PASSED: Layout is valid")
        return 0
    else:
        print("✗ FAILED: Layout has errors")
        return 1


if __name__ == '__main__':
    sys.exit(main())
```

### 逐行解释

**`struct.unpack('<H', data[510:512])[0]`**

```python
# '<H' 是格式字符串：
# '<' = 小端序（little-endian）
# 'H' = unsigned short（无符号短整数，2 字节）
# data[510:512] 是从偏移 510 开始的 2 字节
# [0] 是因为 unpack 返回一个元组，我们取第一个元素
signature = struct.unpack('<H', data[510:512])[0]
```

这行代码读取 MBR 签名（最后两个字节）。

**`len(data) + 511) // 512`**

```python
# 向上取整计算扇区数
# 例如：1000 字节 = (1000 + 511) // 512 = 1511 // 512 = 2 扇区
sectors = (len(data) + 511) // 512
```

这是向上取整的标准方法：`(a + b - 1) // b` 等价于 `ceil(a / b)`。

## 集成到 CMake

现在，我们把验证脚本集成到 CMake 的 POST_BUILD 钩子中：

```cmake
# 在 CMakeLists.txt 中

add_custom_command(TARGET boot_img POST_BUILD
    COMMAND ${PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_disk_layout.py
        ${BOOTLOADER_BIN}
        ${KERNEL_BIN}
    COMMENT "Verifying disk layout"
)
```

这样，每次构建 `boot_img` 目标后，验证脚本会自动运行。

如果验证失败，构建也会失败（脚本返回非零退出码）。

## 可能遇到的坑

### 坑 1：Python 版本问题

脚本使用了 f-string（`f"{var:X}"`），需要 Python 3.6+。

```bash
File "verify_disk_layout.py", line 123
    print(f"Memory gap:  {gap} bytes")
                            ^
SyntaxError: invalid syntax
```

**解决方法**：升级 Python 到 3.6 或更高版本。

### 坯 2：路径问题

脚本中的相对路径可能在不同目录下运行时出问题。

**解决方法**：使用绝对路径，或确保从正确的目录运行脚本。

### 坑 3：权限问题

```bash
Permission denied: 'build/bootloader.bin'
```

**解决方法**：检查文件权限，确保可读。
