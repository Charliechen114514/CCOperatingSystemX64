# 实现大内核加载

好，配置文件有了，CHS 的坑也了解了。现在让我们把整个加载功能实现出来。

---

## 前言 - 现在要改 bootloader 了

之前的 `bootloader.asm` 里，内核加载的代码长这样：

```asm
; 旧的加载代码（阶段 04）
load_kernel:
    mov ax, 0x1000
    mov es, ax
    mov bx, 0x0000

    mov ah, 0x02
    mov al, 0x01        ; 硬编码：只读 1 个扇区
    mov ch, 0x00
    mov cl, 0x03        ; 硬编码：从扇区 3 开始
    mov dh, 0x00
    mov dl, 0x80
    int 0x13

    jc disk_error
    ret
```

现在我们要把它改成：**使用配置文件中的参数，支持动态大小的内核**。

---

## 先决定：内核加载到哪里？

在设计加载函数之前，我们先要确定内核加载到内存的哪个位置。

这是一个重要的决策，涉及到内存布局。

### 候选地址

| 地址 | 优点 | 缺点 |
|------|------|------|
| 0x8000 | 紧凑，节省内存 | 可能覆盖 IVT（中断向量表）和 BDA（BIOS 数据区） |
| 0x9000 | 相对安全 | 可能与页表冲突（我们之前把页表放在 0x9000） |
| 0x10000 | 安全隔离，64KB 对齐 | 需要额外的大内核支持代码 |

### 内存布局图

让我们看看当前的内存布局：

```
内存地址        │ 内容              │ 大小
────────────────┼──────────────────┼────────
0x00000 - 0x003FF │ IVT (中断向量表)   │ 1KB
0x00400 - 0x004FF │ BDA (BIOS 数据区)   │ 256B
0x00500 - 0x07BFF │ 可用内存          │ ~28KB
0x07C00 - 0x07DFF │ Stage 1          │ 512B
0x07E00 - 0x083FF │ Stage 2 (部分)    │ 1.5KB
0x08400 - 0x08FFF │ Stage 2 继续      │ 3KB
0x09000 - 0x0AFFF │ 页表              │ 8KB
0x0B000 - 0x0FFFF │ 可用内存          │ ~20KB
0x10000 - ?????   │ 内核加载位置      │ 最大 64MB
```

可以看到，如果我们把内核加载到 0x10000，会留出一个相当大的安全间隙。

### 为什么选择 0x10000？

1. **安全间隙**：Stage 2 结束（约 0x8400）到 内核开始（0x10000）之间有约 30KB 的间隙
2. **64KB 对齐**：0x10000 是 64KB 边界，CPU 访问更高效
3. **扩展空间**：从 0x10000 开始，我们有近 64MB 的空间可以放内核

所以在 `boot_config.inc` 中：

```asm
%define KERNEL_LOAD_SEGMENT       0x1000  ; 段地址
%define KERNEL_LOAD_OFFSET        0x0000  ; 偏移地址
; 物理地址 = 0x1000 × 16 + 0x0000 = 0x10000
```

## 实现 load_kernel_chs 函数

现在，让我们实现完整的 CHS 加载函数：

```asm
; =============================================================================
; load_kernel_chs - 使用 CHS 模式加载内核
; =============================================================================
; 输入：
;   - 使用 boot_config.inc 中的配置
;   - KERNEL_SECTOR_COUNT: 要读取的扇区数
;   - KERNEL_CHS_CYLINDER/HEAD/SECTOR: 内核的 CHS 起始地址
;   - KERNEL_LOAD_SEGMENT/OFFSET: 目标内存地址
;
; 输出：
;   - 内核被加载到 KERNEL_LOAD_SEGMENT:KERNEL_LOAD_OFFSET
;   - CF=0 表示成功，CF=1 表示失败
; =============================================================================

load_kernel_chs:
    pusha                           ; 保存所有通用寄存器
                                    ; 这样我们修改寄存器后可以恢复

    ; -----------------------------------------------------------------
    ; 第一步：设置目标内存地址
    ; -----------------------------------------------------------------
    ; BIOS INT 13h AH=02h 会把数据读到 ES:BX 指向的内存

    mov ax, KERNEL_LOAD_SEGMENT     ; 从配置中读取段地址
                                    ; ax = 0x1000
    mov es, ax                      ; es = 0x1000
                                    ; ES 是"目标段寄存器"

    xor bx, bx                      ; bx = 0
                                    ; 等价于 mov bx, 0，但更短（1 字节 vs 2 字节）
                                    ; xor 是异或操作，任何数和自己异或都是 0
    ; 现在 ES:BX = 0x1000:0x0000
    ; 物理地址 = 0x1000 × 16 + 0x0000 = 0x10000

    ; -----------------------------------------------------------------
    ; 第二步：检查内核大小是否超过 BIOS 限制
    ; -----------------------------------------------------------------
    ; BIOS INT 13h AH=02h 单次最多读取 127 个扇区

    mov ax, KERNEL_SECTOR_COUNT     ; ax = 内核扇区数
    cmp ax, 127                     ; 和 127 比较
    ja .error_too_large             ; ja = jump if above，如果 > 127 则跳转

    ; -----------------------------------------------------------------
    ; 第三步：设置 CHS 参数
    ; -----------------------------------------------------------------

    mov ah, 0x02                    ; BIOS 功能号：读磁盘
    mov al, KERNEL_SECTOR_COUNT     ; al = 要读取的扇区数
                                    ; 注意：这是"要读多少个扇区"，不是"起始扇区"

    mov ch, KERNEL_CHS_CYLINDER     ; ch = 柱面号
                                    ; 对于 LBA 4，柱面号 = 0

    mov cl, KERNEL_CHS_SECTOR       ; cl = 扇区号（1-based！）
                                    ; 对于 LBA 4，扇区号 = 5
                                    ; 注意：cl 的低 6 位是扇区号，高 2 位是柱面号的高位

    mov dh, KERNEL_CHS_HEAD         ; dh = 磁头号
                                    ; 对于 LBA 4，磁头号 = 0

    mov dl, 0x80                    ; dl = 驱动器号
                                    ; 0x80 表示第一个硬盘
                                    ; 0x81 表示第二个硬盘
                                    ; 0x00 表示软驱 A

    ; -----------------------------------------------------------------
    ; 第四步：调用 BIOS 读取磁盘
    ; -----------------------------------------------------------------

    int 0x13                        ; 调用 BIOS 中断 13h
                                    ; AH=02h 是读扇区功能
                                    ; 执行后：
                                    ;   CF=0 表示成功，AL = 实际读取的扇区数
                                    ;   CF=1 表示失败，AH = 错误码

    jc .error_bios                  ; jc = jump if carry，如果 CF=1 则跳转

    ; -----------------------------------------------------------------
    ; 第五步：验证读取结果
    ; -----------------------------------------------------------------
    ; 虽然 BIOS 说成功，但我们要确认确实读了足够的扇区

    cmp al, KERNEL_SECTOR_COUNT     ; 比较实际读取的扇区数（AL）和期望的扇区数
    jne .error_partial              ; jne = jump if not equal，如果不相等则跳转

    ; -----------------------------------------------------------------
    ; 第六步：成功！
    ; -----------------------------------------------------------------

    clc                             ; clc = clear carry，清零 CF 位
                                    ; 表示成功
    popa                            ; 恢复所有通用寄存器
    ret                             ; 返回到调用者

    ; -----------------------------------------------------------------
    ; 错误处理
    ; -----------------------------------------------------------------

.error_too_large:
    ; 内核太大，超过单次 CHS 读取限制（127 扇区）
    mov si, msg_error_too_large     ; si 指向错误消息
    call print_string               ; 打印错误消息
    stc                             ; stc = set carry，设置 CF 位
                                    ; 表示失败
    jmp .done

.error_bios:
    ; BIOS 读取失败
    mov si, msg_error_bios          ; si 指向错误消息
    call print_string               ; 打印错误消息

    ; 打印 BIOS 错误码（AH 寄存器）
    mov al, ah                      ; al = BIOS 错误码
    call print_hex_byte             ; 打印十六进制
    call print_newline              ; 打印换行

    stc                             ; 设置 CF 位
    jmp .done

.error_partial:
    ; 只读取了部分扇区
    mov si, msg_error_partial       ; si 指向错误消息
    call print_string               ; 打印错误消息
    stc                             ; 设置 CF 位
    jmp .done

.done:
    popa                            ; 恢复所有通用寄存器
    ret
```

### 逐行解释关键指令

**`pusha` 和 `popa`**

```asm
pusha                    ; 把 AX, CX, DX, BX, SP, BP, SI, DI 依次压栈
; ... 你的代码 ...
popa                     ; 按相反顺序弹出 DI, SI, BP, SP, BX, DX, CX, AX
```

这样可以保护调用者的寄存器不被我们修改。

**`xor bx, bx` vs `mov bx, 0`**

```asm
xor bx, bx    ; 1 字节机器码，执行时间：1 时钟周期
mov bx, 0     ; 3 字节机器码，执行时间：1-2 时钟周期（取决于 CPU）
```

`xor` 是"异或"操作，任何数和自身异或结果都是 0。这是一个常见的汇编优化技巧。

**`clc` 和 `stc`**

```asm
clc    ; Clear Carry Flag，清零 CF 位（表示成功）
stc    ; Set Carry Flag，设置 CF 位（表示失败）
```

我们使用 CF 位作为返回值：CF=0 成功，CF=1 失败。这是很多 BIOS 函数和汇编函数的惯例。

## 实现辅助函数：print_hex_byte

上面的代码用到了 `print_hex_byte`，这个函数打印一个字节的十六进制值：

```asm
; =============================================================================
; print_hex_byte - 打印 AL 寄存器的十六进制值
; =============================================================================
; 输入：AL = 要打印的字节
; 输出：在屏幕上打印两个十六进制字符
; =============================================================================

print_hex_byte:
    pusha                           ; 保存寄存器

    ; 先打印高 4 位
    mov ah, al                      ; 复制 AL 到 AH
    shr al, 4                       ; 右移 4 位，保留高 4 位
    call print_hex_nibble           ; 打印高 4 位

    ; 再打印低 4 位
    mov al, ah                      ; 恢复 AL
    and al, 0x0F                    ; 只保留低 4 位
    call print_hex_nibble           ; 打印低 4 位

    popa                            ; 恢复寄存器
    ret

; =============================================================================
; print_hex_nibble - 打印 4 位十六进制数
; =============================================================================
; 输入：AL 的低 4 位 = 要打印的数（0-15）
; =============================================================================

print_hex_nibble:
    pusha                           ; 保存寄存器

    and al, 0x0F                    ; 确保只有低 4 位
    cmp al, 9                       ; 和 9 比较
    jbe .digit                      ; 如果 <= 9，是数字

    ; 否则是 A-F
    add al, 'A' - 10                ; 转换成 'A'-'F'
    jmp .done

.digit:
    add al, '0'                     ; 转换成 '0'-'9'

.done:
    mov ah, 0x0E                    ; BIOS 打印字符功能
    mov bh, 0                       ; 页号
    mov bl, 0x07                    ; 颜色（白色）
    int 0x10                        ; 调用 BIOS

    popa                            ; 恢复寄存器
    ret
```

### 逐行解释

**`shr al, 4`**

```asm
shr al, 4       ; Shift Right，右移 4 位
```

例如，如果 AL = 0xAB：

```
右移前：AL = 0b10101011 = 0xAB
右移后：AL = 0b00001010 = 0x0A
```

这样可以提取高 4 位。

**`and al, 0x0F`**

```asm
and al, 0x0F    ; AND 操作，只保留低 4 位
```

例如，如果 AL = 0xAB：

```
AND 前：AL = 0b10101011 = 0xAB
AND 后：AL = 0b00001011 = 0x0B
```

这样可以提取低 4 位。

## 错误消息字符串

别忘了在数据段定义错误消息：

```asm
; =============================================================================
; 数据段 - 字符串常量
; =============================================================================

section .data

msg_error_too_large:
    db '[E3] Kernel too large for single CHS read (max 127 sectors)', 0x0D, 0x0A, 0

msg_error_bios:
    db '[E2] BIOS disk read failed, error code: ', 0

msg_error_partial:
    db '[E4] Partial read: got ', 0

; ... 其他消息 ...
```

注意字符串末尾的 `0`，这是空字符，表示字符串结束。`0x0D, 0x0A` 是回车换行符。

## 调用 load_kernel_chs

现在，我们的主加载函数可以简化为：

```asm
load_kernel:
    pusha                           ; 保存寄存器

    ; 显示"正在加载内核"消息
    mov si, msg_loading_kernel
    call print_string

    ; 调用 CHS 加载函数
    call load_kernel_chs

    ; 检查是否成功
    jc .error                       ; 如果 CF=1，说明失败了

    ; 成功！
    mov si, msg_kernel_loaded
    call print_string

    popa
    ret

.error:
    ; 失败
    mov si, msg_kernel_error
    call print_string

    popa
    jmp $                           ; 死循环
```

## 可能遇到的坑

### 坑 1：ES:BX 指针溢出

如果你读取的扇区数很大，BX 寄存器可能会溢出（超过 0xFFFF）。

```asm
add bx, 512         ; 如果 BX = 0xFF00，+512 后会溢出
```

对于我们的实现，因为我们是一次性读取（最多 127 个扇区），BX 不会溢出。但如果你实现分批读取，需要注意这个问题。

### 坑 2：BIOS 错误码

BIOS INT 13h 在失败时会返回错误码（在 AH 寄存器）：

| 错误码 | 含义 |
|--------|------|
| 0x01 | 无效命令 |
| 0x02 | 找不到地址标记 |
| 0x04 | 扇区没找到 |
| 0x09 | DMA 越界 |
| 0x0C | 无效的介质类型 |
| 0x20 | 控制器失败 |

如果你的代码报错 `[E2] BIOS disk read failed, error code: XX`，可以查这个表。

### 坑 3：内核加载地址冲突

如果你把内核加载到 0x10000，但你的页表也在这个位置附近，就会冲突。

**解决方法**：确保页表在其他位置（比如 0x9000），或者调整内核加载地址。
