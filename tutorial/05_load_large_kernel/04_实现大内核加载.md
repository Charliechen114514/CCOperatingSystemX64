# 实现大内核加载

好，配置文件有了，CHS 的坑也了解了。现在让我们把整个加载功能实现出来。

---

## 前言 - 现在要改 bootloader 了

之前的 `bootloader.asm` 里，内核加载的代码长这样：

```asm
; 旧的加载代码（阶段 04）
load_kernel:
    mov ax, 0x1000
    mov es, ax
    mov bx, 0x0000

    mov ah, 0x02
    mov al, 0x01        ; 硬编码：只读 1 个扇区
    mov ch, 0x00
    mov cl, 0x03        ; 硬编码：从扇区 3 开始
    mov dh, 0x00
    mov dl, 0x80
    int 0x13

    jc disk_error
    ret
```

现在我们要把它改成：**使用配置文件中的参数，支持动态大小的内核**。

## 环境说明

在我们开始编码之前，确认一下环境和工具：

**开发平台**：WSL2 (Ubuntu 22.04) + QEMU 7.0+
**工具链**：NASM 2.16.01，GDB 12.1
**目标架构**：x86_64，实模式 + 长模式
**当前状态**：boot_config.inc 已创建，bootloader 能读取小内核

接下来我们要做的是实现一个支持大内核加载的完整函数。

## 先决定：内核加载到哪里？

在设计加载函数之前，我们先要确定内核加载到内存的哪个位置。这是一个重要的决策，涉及到内存布局。

我们有几个候选地址：0x8000 很紧凑节省内存，但可能会覆盖 IVT（中断向量表）和 BDA（BIOS 数据区）；0x9000 相对安全，但可能会与我们之前放置页表的位置冲突；0x10000 是安全隔离且 64KB 对齐的，但需要额外的大内核支持代码。

让我们看看当前的内存布局：从 0x00000 到 0x003FF 是 IVT（1KB），然后是 BDA（256B），接着是一大块可用内存（约 28KB）。0x07C00 开始是 Stage 1（512B），0x07E00 是 Stage 2 的一部分（1.5KB），0x08400 之后继续是 Stage 2 的剩余部分（约 3KB），0x09000 到 0x0AFFF 是页表（8KB），然后又是一大块可用内存，最后 0x10000 开始是内核加载位置（最大 64MB）。

可以看到，如果我们把内核加载到 0x10000，会留出一个相当大的安全间隙。选择 0x10000 的原因有几个：首先是安全间隙，Stage 2 结束（约 0x8400）到内核开始（0x10000）之间有约 30KB 的间隙；其次是 64KB 对齐，0x10000 是 64KB 边界，CPU 访问更高效；最后是扩展空间，从 0x10000 开始，我们有近 64MB 的空间可以放内核。

⚠️ **千万别**
为了省一点内存就把内核加载地址改得很低。我见过有人把内核加载到 0x9000，结果和页表冲突，系统启动后直接卡死。这种问题很难调试，因为你完全不知道是哪里出了问题。

## 爆改开始：实现 load_kernel_chs 函数

现在，让我们实现完整的 CHS 加载函数：

```asm
; =============================================================================
; load_kernel_chs - 使用 CHS 模式加载内核
; =============================================================================
; 输入：
;   - 使用 boot_config.inc 中的配置
;   - KERNEL_SECTOR_COUNT: 要读取的扇区数
;   - KERNEL_CHS_CYLINDER/HEAD/SECTOR: 内核的 CHS 起始地址
;   - KERNEL_LOAD_SEGMENT/OFFSET: 目标内存地址
;
; 输出：
;   - 内核被加载到 KERNEL_LOAD_SEGMENT:KERNEL_LOAD_OFFSET
;   - CF=0 表示成功，CF=1 表示失败
; =============================================================================

load_kernel_chs:
    pusha                           ; 保存所有通用寄存器
                                    ; 这样我们修改寄存器后可以恢复

    ; -----------------------------------------------------------------
    ; 第一步：设置目标内存地址
    ; -----------------------------------------------------------------
    ; BIOS INT 13h AH=02h 会把数据读到 ES:BX 指向的内存

    mov ax, KERNEL_LOAD_SEGMENT     ; 从配置中读取段地址
                                    ; ax = 0x1000
    mov es, ax                      ; es = 0x1000
                                    ; ES 是"目标段寄存器"

    xor bx, bx                      ; bx = 0
                                    ; 等价于 mov bx, 0，但更短（1 字节 vs 2 字节）
                                    ; xor 是异或操作，任何数和自己异或都是 0
    ; 现在 ES:BX = 0x1000:0x0000
    ; 物理地址 = 0x1000 × 16 + 0x0000 = 0x10000

    ; -----------------------------------------------------------------
    ; 第二步：检查内核大小是否超过 BIOS 限制
    ; -----------------------------------------------------------------
    ; BIOS INT 13h AH=02h 单次最多读取 127 个扇区

    mov ax, KERNEL_SECTOR_COUNT     ; ax = 内核扇区数
    cmp ax, 127                     ; 和 127 比较
    ja .error_too_large             ; ja = jump if above，如果 > 127 则跳转

    ; -----------------------------------------------------------------
    ; 第三步：设置 CHS 参数
    ; -----------------------------------------------------------------

    mov ah, 0x02                    ; BIOS 功能号：读磁盘
    mov al, KERNEL_SECTOR_COUNT     ; al = 要读取的扇区数
                                    ; 注意：这是"要读多少个扇区"，不是"起始扇区"

    mov ch, KERNEL_CHS_CYLINDER     ; ch = 柱面号
                                    ; 对于 LBA 4，柱面号 = 0

    mov cl, KERNEL_CHS_SECTOR       ; cl = 扇区号（1-based！）
                                    ; 对于 LBA 4，扇区号 = 5
                                    ; 注意：cl 的低 6 位是扇区号，高 2 位是柱面号的高位

    mov dh, KERNEL_CHS_HEAD         ; dh = 磁头号
                                    ; 对于 LBA 4，磁头号 = 0

    mov dl, 0x80                    ; dl = 驱动器号
                                    ; 0x80 表示第一个硬盘
                                    ; 0x81 表示第二个硬盘
                                    ; 0x00 表示软驱 A

    ; -----------------------------------------------------------------
    ; 第四步：调用 BIOS 读取磁盘
    ; -----------------------------------------------------------------

    int 0x13                        ; 调用 BIOS 中断 13h
                                    ; AH=02h 是读扇区功能
                                    ; 执行后：
                                    ;   CF=0 表示成功，AL = 实际读取的扇区数
                                    ;   CF=1 表示失败，AH = 错误码

    jc .error_bios                  ; jc = jump if carry，如果 CF=1 则跳转

    ; -----------------------------------------------------------------
    ; 第五步：验证读取结果
    ; -----------------------------------------------------------------
    ; 虽然 BIOS 说成功，但我们要确认确实读了足够的扇区

    cmp al, KERNEL_SECTOR_COUNT     ; 比较实际读取的扇区数（AL）和期望的扇区数
    jne .error_partial              ; jne = jump if not equal，如果不相等则跳转

    ; -----------------------------------------------------------------
    ; 第六步：成功！
    ; -----------------------------------------------------------------

    clc                             ; clc = clear carry，清零 CF 位
                                    ; 表示成功
    popa                            ; 恢复所有通用寄存器
    ret                             ; 返回到调用者

    ; -----------------------------------------------------------------
    ; 错误处理
    ; -----------------------------------------------------------------

.error_too_large:
    ; 内核太大，超过单次 CHS 读取限制（127 扇区）
    mov si, msg_error_too_large     ; si 指向错误消息
    call print_string               ; 打印错误消息
    stc                             ; stc = set carry，设置 CF 位
                                    ; 表示失败
    jmp .done

.error_bios:
    ; BIOS 读取失败
    mov si, msg_error_bios          ; si 指向错误消息
    call print_string               ; 打印错误消息

    ; 打印 BIOS 错误码（AH 寄存器）
    mov al, ah                      ; al = BIOS 错误码
    call print_hex_byte             ; 打印十六进制
    call print_newline              ; 打印换行

    stc                             ; 设置 CF 位
    jmp .done

.error_partial:
    ; 只读取了部分扇区
    mov si, msg_error_partial       ; si 指向错误消息
    call print_string               ; 打印错误消息
    stc                             ; 设置 CF 位
    jmp .done

.done:
    popa                            ; 恢复所有通用寄存器
    ret
```

我来解释一下这段代码的关键部分。pusha 和 popa 用于保存和恢复所有通用寄存器，这样我们的函数不会影响调用者的寄存器状态。xor bx, bx 是一个常见的汇编技巧，用异或操作把 bx 清零，比 mov bx, 0 短一个字节。ja 是 "jump if above"的缩写，用于无符号数的比较。jc 和 jne 分别是 "jump if carry" 和 "jump if not equal" 的缩写。clc 和 stc 分别用于清零和设置 CF 位，我们用 CF 位作为返回值：CF=0 成功，CF=1 失败。

## 实现辅助函数：print_hex_byte

上面的代码用到了 `print_hex_byte`，这个函数打印一个字节的十六进制值：

```asm
; =============================================================================
; print_hex_byte - 打印 AL 寄存器的十六进制值
; =============================================================================
; 输入：AL = 要打印的字节
; 输出：在屏幕上打印两个十六进制字符
; =============================================================================

print_hex_byte:
    pusha                           ; 保存寄存器

    ; 先打印高 4 位
    mov ah, al                      ; 复制 AL 到 AH
    shr al, 4                       ; 右移 4 位，保留高 4 位
    call print_hex_nibble           ; 打印高 4 位

    ; 再打印低 4 位
    mov al, ah                      ; 恢复 AL
    and al, 0x0F                    ; 只保留低 4 位
    call print_hex_nibble           ; 打印低 4 位

    popa                            ; 恢复寄存器
    ret

; =============================================================================
; print_hex_nibble - 打印 4 位十六进制数
; =============================================================================
; 输入：AL 的低 4 位 = 要打印的数（0-15）
; =============================================================================

print_hex_nibble:
    pusha                           ; 保存寄存器

    and al, 0x0F                    ; 确保只有低 4 位
    cmp al, 9                       ; 和 9 比较
    jbe .digit                      ; 如果 <= 9，是数字

    ; 否则是 A-F
    add al, 'A' - 10                ; 转换成 'A'-'F'
    jmp .done

.digit:
    add al, '0'                     ; 转换成 '0'-'9'

.done:
    mov ah, 0x0E                    ; BIOS 打印字符功能
    mov bh, 0                       ; 页号
    mov bl, 0x07                    ; 颜色（白色）
    int 0x10                        ; 调用 BIOS

    popa                            ; 恢复寄存器
    ret
```

这段代码的原理是把一个字节拆成两个 4 位数，分别转换成十六进制字符打印。shr 是右移指令，and 是按位与指令。如果数字小于等于 9，直接加上 '0' 的 ASCII 值；否则加上 'A' - 10，转换成 A-F。

## 错误消息字符串

别忘了在数据段定义错误消息：

```asm
; =============================================================================
; 数据段 - 字符串常量
; =============================================================================

section .data

msg_error_too_large:
    db '[E3] Kernel too large for single CHS read (max 127 sectors)', 0x0D, 0x0A, 0

msg_error_bios:
    db '[E2] BIOS disk read failed, error code: ', 0

msg_error_partial:
    db '[E4] Partial read: got ', 0

; ... 其他消息 ...
```

注意字符串末尾的 0，这是空字符，表示字符串结束。0x0D, 0x0A 是回车换行符。

## 可能遇到的坑

如果你读取的扇区数很大，BX 寄存器可能会溢出（超过 0xFFFF）。对于我们的实现，因为我们是一次性读取（最多 127 个扇区），BX 不会溢出。但如果你实现分批读取，需要注意这个问题。

BIOS INT 13h 在失败时会返回错误码（在 AH 寄存器）。如果你的代码报错 `[E2] BIOS disk read failed, error code: XX`，可以查这个表：0x01 是无效命令，0x02 是找不到地址标记，0x04 是扇区没找到，0x09 是 DMA 越界，0x0C 是无效的介质类型，0x20 是控制器失败。

如果你把内核加载到 0x10000，但你的页表也在这个位置附近，就会冲突。解决方法是确保页表在其他位置（比如 0x9000），或者调整内核加载地址。
