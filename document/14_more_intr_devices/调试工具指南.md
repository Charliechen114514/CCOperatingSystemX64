## 1. QEMU Monitor 调试

### 1.1 启动 Monitor

```bash
# 方法 1: 标准输入/输出
qemu-system-x86_64 -kernel kernel.elf -monitor stdio

# 方法 2: Telnet 连接
qemu-system-x86_64 -kernel kernel.elf \
    -monitor telnet::4444,server,nowainb

# 方法 3: Unix Socket
qemu-system-x86_64 -kernel kernel.elf \
    -monitor unix:qemu-monitor.sock,server,nowait
```

### 1.2 查看设备状态

```qemu
# 查看 PIC 状态
info pic

# 查看 APIC 状态（如果使用）
info apic

# 查看 IRQ 统计
info irq

# 查看 PCI 设备
info pci

# 查看计时器
info timers
```

### 1.3 内存和寄存器操作

```qemu
# 查看内存
x/10xg 0x100000        # 10 个 64 位字
x/20xb 0x60            # 20 个字节 (键盘端口)

# 写入内存
writeb 0xb8000 0x41    # 写入字节
writew 0xb8000 0x0141  # 写入字

# 查看端口
info port              # 列出所有端口
```

### 1.4 中断相关调试

```qemu
# 暂停执行
stop

# 单步执行
step

# 继续执行
cont

# 查看当前执行位置
info registers
x/10i $rip
```

### 1.5 有用的 Monitor 技巧

```qemu
# 保存/恢复状态
savevm before_test
loadvm before_test

# 查看虚拟设备树
qtree

# 查看最近执行
info jit

# 查看内存映射
info mem
```

## 3. 串口日志分析

### 3.1 日志配置

```bash
# QEMU 串口输出到文件
qemu-system-x86_64 -kernel kernel.elf \
    -serial file:serial.log

# 输出到标准输出
qemu-system-x86_64 -kernel kernel.elf \
    -serial stdio

# 输出到虚拟控制台（monitor + serial）
qemu-system-x86_64 -kernel kernel.elf \
    -serial mon:stdio
```

### 3.2 日志格式

```
[INFO] [KBD] Keyboard initialized
[INFO] [RTC] RTC initialized (BCD mode, 24-hour format)
[INFO] [TIMER] Timer initialized at 1000 Hz
[INFO] [UART] UART interrupt mode initialized
[INFO] [SHELL] Shell started on VGA
```

### 3.3 日志过滤

```bash
# 查看键盘日志
grep "KBD" serial.log

# 查看错误信息
grep "ERROR" serial.log

# 查看中断信息
grep "IRQ" serial.log

# 统计中断次数
grep -c "Keyboard IRQ" serial.log
```

```bash

# 高亮显示错误
tail -f serial.log | grep --color=auto "ERROR\|WARN"
```


### 4.2 RTC 状态

```c
void rtc_dump_status(void) {
    klog_info("=== RTC Status ===\n");
    klog_info("Binary mode: %s\n", rtc_binary_mode ? "Yes" : "No");
    klog_info("Valid: %s\n", rtc_is_valid() ? "Yes" : "No (Battery dead)");
    klog_info("Updating: %s\n", rtc_is_updating() ? "Yes" : "No");
    klog_info("Interrupt count: %lu\n", rtc_get_interrupt_count());

    // 打印寄存器
    klog_info("Reg A: 0x%02X\n", cmos_read(RTC_REG_A));
    klog_info("Reg B: 0x%02X\n", cmos_read(RTC_REG_B));
    klog_info("Reg C: 0x%02X\n", cmos_read(RTC_REG_C));
    klog_info("Reg D: 0x%02X\n", cmos_read(RTC_REG_D));

    // 打印当前时间
    rtc_time_t time;
    if (rtc_get_time(&time) == 0) {
        char buffer[32];
        rtc_format_time(&time, buffer, sizeof(buffer));
        klog_info("Current time: %s\n", buffer);
    }
}
```

### 4.3 定时器状态

```c
void timer_dump_status(void) {
    klog_info("=== Timer Status ===\n");
    klog_info("Frequency: %u Hz\n", timer_frequency);
    klog_info("Ticks: %lu\n", timer_get_ticks());
    klog_info("Interrupt count: %lu\n", timer_irq_desc.invocation_count);

    // 计算实际频率
    static uint64_t last_ticks = 0;
    static uint64_t last_check = 0;
    if (last_check != 0) {
        uint64_t elapsed = timer_get_ticks() - last_ticks;
        klog_info("Ticks since last check: %lu\n", elapsed);
    }
    last_ticks = timer_get_ticks();
    last_check = timer_get_ticks();
}
```

### 4.4 UART 状态

```c
void uart_dump_status(void) {
    const uint16_t base = SERIAL_COM1_BASE;
    char buf[128];

    uint8_t ier = inb(SERIAL_INT_ENABLE_REG(base));
    uint8_t iir = inb(SERIAL_INT_ID_REG(base));
    uint8_t lcr = inb(SERIAL_LINE_CTRL_REG(base));
    uint8_t lsr = inb(SERIAL_LINE_STATUS_REG(base));

    ksnprintf(buf, sizeof(buf), "IER=%02X IIR=%02X LCR=%02X LSR=%02X\n",
              ier, iir, lcr, lsr);
    sync_serial_puts(buf);

    klog_info("TX buffer: head=%u tail=%u\n", tx_buffer.head, tx_buffer.tail);
    klog_info("RX buffer: head=%u tail=%u\n", rx_buffer.head, rx_buffer.tail);
    klog_info("Interrupt count: %lu\n", uart_get_interrupt_count());
}
```

## 5. 性能分析

### 5.1 中断处理时间

```c
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ volatile ("rdtsc" : "=a"(lo), "=d"(hi));
    return ((uint64_t)hi << 32) | lo;
}

// 在中断处理程序中
void keyboard_irq_handler(interrupt_frame_t* frame, void* context) {
    uint64_t start = rdtsc();

    // ... 处理中断 ...

    uint64_t end = rdtsc();
    uint64_t cycles = end - start;

    if (cycles > 10000) {  // 超过 10000 周期警告
        klog_warn("Keyboard handler took %lu cycles\n", cycles);
    }
}
```

### 5.2 中断频率统计

```c
// 每秒打印一次中断统计
static uint64_t last_stats_time = 0;
static uint64_t last_keyboard_count = 0;
static uint64_t last_rtc_count = 0;

void print_interrupt_stats(void) {
    uint64_t now = timer_get_ticks();

    if (now - last_stats_time >= timer_frequency) {
        uint64_t keyboard_now = keyboard_get_interrupt_count();
        uint64_t rtc_now = rtc_get_interrupt_count();

        klog_info("Stats/s: Keyboard=%lu RTC=%lu Timer=%lu\n",
                  keyboard_now - last_keyboard_count,
                  rtc_now - last_rtc_count,
                  timer_get_ticks());

        last_stats_time = now;
        last_keyboard_count = keyboard_now;
        last_rtc_count = rtc_now;
    }
}
```

### 5.3 缓冲区使用率

```c
void print_buffer_usage(void) {
    // 键盘缓冲区
    uint32_t kb_count = (kb_buffer.head - kb_buffer.tail) % KEYBOARD_BUFFER_SIZE;
    klog_info("Keyboard buffer: %u/256 (%.1f%%)\n",
              kb_count, kb_count * 100.0 / KEYBOARD_BUFFER_SIZE);

    // UART 缓冲区
    uint32_t tx_count = (tx_buffer.head - tx_buffer.tail) % UART_TX_BUFFER_SIZE;
    uint32_t rx_count = (rx_buffer.head - rx_buffer.tail) % UART_RX_BUFFER_SIZE;
    klog_info("UART TX: %u/256 RX: %u/256\n", tx_count, rx_count);
}
```

### 6.3 日志分析脚本

```python
#!/usr/bin/env python3
"""
analyze_stage14_log.py - 分析阶段 14 日志文件
"""

import sys
import re
from collections import Counter, defaultdict

def parse_interrupts(log_file):
    """解析中断统计"""
    counts = defaultdict(int)

    with open(log_file, 'r') as f:
        for line in f:
            # 匹配 "Keyboard IRQ: 123" 格式
            m = re.search(r'(\w+) IRQ:\s*(\d+)', line)
            if m:
                device, count = m.groups()
                counts[device] += int(count)

    return counts

def parse_errors(log_file):
    """解析错误信息"""
    errors = []

    with open(log_file, 'r') as f:
        for line in f:
            if 'ERROR' in line or 'WARN' in line:
                errors.append(line.strip())

    return errors

def main():
    if len(sys.argv) < 2:
        print("Usage: analyze_stage14_log.py <log_file>")
        sys.exit(1)

    log_file = sys.argv[1]

    # 打印中断统计
    counts = parse_interrupts(log_file)
    print("Interrupt Statistics:")
    for device, count in sorted(counts.items()):
        print(f"  {device}: {count}")

    # 打印错误
    errors = parse_errors(log_file)
    print(f"\nErrors/Warnings ({len(errors)}):")
    for error in errors[-10:]:  # 只显示最后 10 个
        print(f"  {error}")

if __name__ == '__main__':
    main()
```

```
1. 添加调试日志
   ↓
2. QEMU Monitor 查看状态
   ↓
3. GDB 设置关键断点
   ↓
4. 触发问题
   ↓
5. 检查寄存器和缓冲区
   ↓
6. 定位问题根因
   ↓
7. 修复代码
   ↓
8. 验证修复
```
