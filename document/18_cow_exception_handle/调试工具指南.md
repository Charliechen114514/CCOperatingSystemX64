# 调试工具指南 - Stage 18 写时复制与异常处理

本文档介绍用于调试 CCOS Stage 18 写时复制与异常处理机制的工具和方法。

## 1. 内置调试功能

### 1.1 COW 状态查看

#### 获取 COW 统计信息

```c
#include "mm/vmm/cow.h"

// 在代码中调用
cow_stats_t stats;
if (cow_get_stats(&stats) == COW_OK) {
    klog_info("=== COW Statistics ===\n");
    klog_info("Faults handled:  %llu\n", stats.cow_faults_handled);
    klog_info("Pages allocated: %llu\n", stats.cow_pages_allocated);
    klog_info("Pages freed:     %llu\n", stats.cow_pages_freed);
    klog_info("Current blocks:  %llu\n", stats.cow_current_blocks);
    klog_info("Write faults:    %llu\n", stats.cow_write_faults);
    klog_info("Coalesced:       %llu\n", stats.cow_coalesced);
}
```

#### 检查特定页面的 COW 状态

```c
// 查询物理页的 COW 状态
physical_addr_t phys = 0x100000;
cow_block_t* block = cow_lookup_block(phys);

if (block) {
    uint16_t refcount;
    cow_get_refcount(phys, &refcount);
    klog_info("Page 0x%llx: refcount=%u\n", phys, refcount);
} else {
    klog_info("Page 0x%llx: not COW\n", phys);
}
```

#### 检查虚拟地址的 COW 标志

```c
#include "mm/vmm/page.h"

// 查询虚拟地址映射
virtual_addr_t vaddr = 0x400000;
page_query_result_t query;

if (page_query(vmm_get_current_pml4(), vaddr, &query) == PAGE_OK) {
    klog_info("Vaddr 0x%lx:\n", vaddr);
    klog_info("  Phys: 0x%llx\n", query.phys_addr);
    klog_info("  Flags: 0x%llx\n", query.flags);
    klog_info("  COW: %s\n", cow_is_cow_page(query.flags) ? "yes" : "no");
}
```

### 1.4 哈希表调试

```c
#include "base/hashmap.h"

// 遍历所有 COW 页面
typedef struct {
    int count;
    uint64_t total_refcount;
} cow_iter_context_t;

static bool cow_iterator(const void* key, void* value, void* context) {
    cow_iter_context_t* ctx = (cow_iter_context_t*)context;
    cow_block_t* block = (cow_block_t*)value;

    ctx->count++;
    ctx->total_refcount += block->refcount;

    klog_info("  [0x%llx] refcount=%u\n",
              *(const physical_addr_t*)key, block->refcount);
    return true;  // 继续迭代
}

// 使用示例
void dump_all_cow_pages(void) {
    klog_info("=== All COW Pages ===\n");

    cow_iter_context_t ctx = {0, 0};
    hashmap_t* map = cow_get_page_map();  // 假设提供此函数
    size_t count = hashmap_foreach(map, cow_iterator, &ctx);

    klog_info("Total: %zu pages, avg refcount: %llu\n",
              count, ctx.total_refcount / count);
}
```

## 5. 性能分析

### 5.1 RDTSC 计时

```c
static inline uint64_t rdtsc(void) {
    uint64_t low, high;
    __asm__ volatile("rdtsc" : "=a"(low), "=d"(high));
    return (high << 32) | low;
}

// 使用示例
void time_cow_operation(void) {
    uint64_t start = rdtsc();

    // 执行 COW 操作
    cow_handle_fault(pml4, fault_addr);

    uint64_t end = rdtsc();
    uint64_t cycles = end - start;

    klog_info("COW fault handled in %llu cycles\n", cycles);
}
```

### 5.2 统计收集

```c
// 定义统计收集器
typedef struct {
    uint64_t total_faults;
    uint64_t cow_faults;
    uint64_t cow_hits;
    uint64_t cow_copies;
    uint64_t total_cycles;
} perf_stats_t;

static perf_stats_t perf_stats = {0};

// 在关键位置更新统计
void cow_perf_inc_hit(void) {
    perf_stats.cow_hits++;
}

void cow_perf_inc_copy(void) {
    perf_stats.cow_copies++;
}

// 打印性能报告
void cow_perf_report(void) {
    klog_info("=== COW Performance ===\n");
    klog_info("Total faults: %llu\n", perf_stats.total_faults);
    klog_info("COW faults: %llu (%.1f%%)\n",
              perf_stats.cow_faults,
              (float)perf_stats.cow_faults * 100 / perf_stats.total_faults);
    klog_info("COW hits (no copy): %llu (%.1f%%)\n",
              perf_stats.cow_hits,
              (float)perf_stats.cow_hits * 100 / perf_stats.cow_faults);
    klog_info("COW copies: %llu (%.1f%%)\n",
              perf_stats.cow_copies,
              (float)perf_stats.cow_copies * 100 / perf_stats.cow_faults);

    if (perf_stats.total_faults > 0) {
        klog_info("Avg cycles per fault: %llu\n",
                  perf_stats.total_cycles / perf_stats.total_faults);
    }
}
```

## 7. 常见调试场景

### 7.1 调试 COW 写入失败

```c
// 在写入前验证
void verify_cow_setup(virtual_addr_t vaddr) {
    page_query_result_t query;
    if (page_query(vmm_get_current_pml4(), vaddr, &query) != PAGE_OK) {
        klog_error("Page not mapped!\n");
        return;
    }

    if (!cow_is_cow_page(query.flags)) {
        klog_error("Page not marked as COW!\n");
    }

    cow_block_t* block = cow_lookup_block(query.phys_addr);
    if (!block) {
        klog_error("Physical page not in COW tracking!\n");
    } else {
        klog_info("COW setup OK, refcount=%u\n", block->refcount);
    }
}

// 在写入后验证
void verify_cow_result(virtual_addr_t vaddr) {
    page_query_result_t query;
    page_query(vmm_get_current_pml4(), vaddr, &query);

    if (cow_is_cow_page(query.flags)) {
        klog_error("COW flag still set after write!\n");
    } else {
        klog_info("COW flag cleared, page is writable\n");
    }
}
```

### 7.3 内存转储

```c
// 转储内存区域
void dump_memory(virtual_addr_t addr, size_t size) {
    uint8_t* ptr = (uint8_t*)addr;

    for (size_t i = 0; i < size; i += 16) {
        klog_info("%016lx: ", addr + i);

        // 十六进制
        for (size_t j = 0; j < 16; j++) {
            if (i + j < size) {
                klog_info("%02x ", ptr[i + j]);
            } else {
                klog_info("   ");
            }
        }

        klog_info(" ");

        // ASCII
        for (size_t j = 0; j < 16; j++) {
            if (i + j < size) {
                char c = ptr[i + j];
                klog_info("%c", (c >= 32 && c < 127) ? c : '.');
            }
        }

        klog_info("\n");
    }
}
```
