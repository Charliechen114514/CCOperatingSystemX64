## 进程状态定义

### process_state_t

```c
/**
 * process_state_t - 进程状态枚举
 */
typedef enum {
    PROC_READY    = 0,    /** 就绪态：等待 CPU */
    PROC_RUNNING  = 1,    /** 运行态：正在执行 */
    PROC_BLOCKED  = 2,    /** 阻塞态：等待事件 */
    PROC_ZOMBIE   = 3,    /** 僵尸态：已退出，等待回收 */
} process_state_t;
```

**状态转换图**：

```
                    fork()
    ┌─────────────────────────────────┐
    │                                 │
    ▼                                 │
┌───────┐    schedule()    ┌─────────┐
│ READY │ ────────────────> │ RUNNING │
└────────↑                  └────┬────┘
    │                          │
    │                          │ exit()
    │                          ▼
    │                    ┌─────────┐
    │                    │  ZOMBIE │
    │                    └────┬────┘
    │                          │
    │                          │ wait4()
    │                          ▼
    └────────────────────  (reaped)
```

### 进程创建

#### proc_fork

```c
/**
 * proc_fork - 创建新进程
 *
 * 创建当前进程的副本。子进程获得独立的 PID 和地址空间。
 *
 * @return 子进程 PID（父进程），0（子进程），负数（错误）
 *
 * 错误码：
 * - -1: PID 耗尽
 * - -2: 内存不足
 * - -3: PCB 分配失败
 */
int32_t proc_fork(void);
```

**使用示例**：
```c
int32_t pid = proc_fork();
if (pid > 0) {
    // 父进程
    klog_info("Child PID: %d\n", pid);
} else if (pid == 0) {
    // 子进程
    klog_info("I am the child\n");
    proc_exit(0);
} else {
    // 错误
    klog_error("fork failed: %d\n", pid);
}
```

### 进程退出

#### proc_exit

```c
/**
 * proc_exit - 退出当前进程
 * @param exit_code: 退出状态码
 *
 * 行为：
 * 1. 将进程状态设为 ZOMBIE
 * 2. 从运行队列移除
 * 3. 唤醒等待的父进程
 * 4. 调度下一个进程
 *
 * 注意：此函数不返回
 */
void proc_exit(int exit_code);
```

**使用示例**：
```c
void some_function(void) {
    if (error_occurred) {
        proc_exit(EXIT_FAILURE);  // 退出并返回错误码
    }
    proc_exit(EXIT_SUCCESS);      // 正常退出
}
```

### 进程等待

#### proc_wait4

```c
/**
 * proc_wait4 - 等待子进程退出
 * @param pid: 子进程 PID (-1 表示任意子进程)
 * @param wstatus: 输出退出状态码 (可为 NULL)
 * @param options: 选项 (WNOHANG 表示非阻塞)
 * @return 子进程 PID，0 (非阻塞且无子进程退出)，负数 (错误)
 *
 * 选项：
 * - WNOHANG: 如果没有子进程退出，立即返回 0
 */
int32_t proc_wait4(int32_t pid, int* wstatus, int options);
```

**使用示例**：
```c
int32_t child_pid = proc_fork();
if (child_pid > 0) {
    // 父进程等待子进程
    int status;
    int32_t pid = proc_wait4(child_pid, &status, 0);
    if (pid > 0) {
        klog_info("Child %d exited with status %d\n", pid, status);
    }
} else if (child_pid == 0) {
    // 子进程
    proc_exit(42);
}

// 非阻塞等待
int32_t pid = proc_wait4(-1, NULL, WNOHANG);
if (pid == 0) {
    klog_info("No child exited yet\n");
}
```

## 数据结构定义

### pcb_t - 进程控制块

```c
/**
 * pcb_t - 进程控制块 (Process Control Block)
 *
 * 存储进程的所有状态信息。
 */
typedef struct pcb {
    /* === 进程标识 === */
    int32_t pid;                    /** 进程 ID */
    int32_t ppid;                   /** 父进程 ID */
    char comm[16];                  /** 命令名 */

    /* === 进程状态 === */
    process_state_t state;          /** 当前状态 */
    int32_t exit_code;              /** 退出码 */

    /* === 链表节点 === */
    list_head run_list;             /** 运行队列链表 */
    list_head siblings;             /** 兄弟进程链表 */
    list_head children;             /** 子进程链表 */
    list_head zombie_children;      /** 僵尸子进程链表 */

    /* === 内存上下文 === */
    memory_context_t mm;            /** 地址空间信息 */
    virtual_addr_t kernel_stack;    /** 内核栈顶 */
    virtual_addr_t kernel_stack_base; /** 内核栈底 */

    /* === 执行上下文 === */
    cpu_context_t* cpu_ctx;         /** 保存的 CPU 寄存器 */
    trap_frame_t* trap_frame;       /** 用户陷阱帧 */

    /* === 家族关系 === */
    struct pcb* parent;             /** 父进程指针 */
    uint64_t start_time;            /** 创建时间戳 */
} pcb_t;
```

**内存布局**：
```
┌────────────────────────────────────────┐
│ pcb_t                                  │
├────────────────────────────────────────┤
│ pid:    4 bytes                        │
│ ppid:   4 bytes                        │
│ state:  1 byte                         │
│ padding: 3 bytes                       │
│ exit_code: 4 bytes                     │
│ run_list: 16 bytes (双向链表)          │
│ siblings: 16 bytes                     │
│ children: 16 bytes                     │
│ zombie_children: 16 bytes              │
│ mm:      16 bytes                      │
│ kernel_stack: 8 bytes                  │
│ kernel_stack_base: 8 bytes             │
│ cpu_ctx: 8 bytes                       │
│ trap_frame: 8 bytes                    │
│ parent:  8 bytes                       │
│ start_time: 8 bytes                    │
│ comm:    16 bytes                      │
├────────────────────────────────────────┤
│ Total: ~150 bytes                      │
└────────────────────────────────────────┘
```

### cpu_context_t - CPU 上下文

```c
/**
 * cpu_context_t - 保存的 CPU 寄存器状态
 *
 * 在上下文切换时保存/恢复。
 * 只需保存 callee-saved 寄存器 (ABI 规定)。
 */
typedef struct {
    uint64_t rax;
    uint64_t rbx;
    uint64_t rcx;
    uint64_t rdx;
    uint64_t rbp;
    uint64_t rsi;
    uint64_t rdi;
    uint64_t r8;
    uint64_t r9;
    uint64_t r10;
    uint64_t r11;
    uint64_t r12;
    uint64_t r13;
    uint64_t r14;
    uint64_t r15;
    uint64_t rip;
    uint64_t rsp;
    uint64_t rflags;
} cpu_context_t;
```

**Callee-Saved 寄存器** (需要保存)：
- RBX, RBP, R12, R13, R14, R15
- RSP (栈指针)
- RIP (返回地址)

**Caller-Saved 寄存器** (无需保存)：
- RAX, RCX, RDX, RSI, RDI, R8-R11
- 调用者负责保存

### scheduler_t - 调度器

```c
/**
 * scheduler_t - 调度器状态
 */
typedef struct scheduler {
    /**
     * @brief 运行队列
     *
     * 所有就绪态 (READY) 进程的链表。
     * 使用 Round-Robin 算法选择下一个进程。
     */
    list_head run_queue;

    /**
     * @brief 当前运行进程
     *
     * 指向正在使用 CPU 的进程。
     */
    pcb_t* current;

    /**
     * @brief 空闲进程
     *
     * 当没有就绪进程时运行此进程。
     * 通常执行 idle 或 halt 指令。
     */
    pcb_t* idle;

    /**
     * @brief 就绪进程数量
     */
    uint32_t nr_running;

    /**
     * @brief 需要重新调度标志
     *
     * 设置为 true 时，下次会调用 schedule()。
     */
    bool need_resched;
} scheduler_t;
```

## 算法说明

### 1. PID 分配算法

```c
/**
 * pid_alloc - Bitmap PID 分配
 *
 * 算法：
 * 1. 遍历 bitmap 数组
 * 2. 使用 __builtin_ctzll 找到第一个空闲位
 * 3. 设置该位为已占用
 * 4. 返回对应的 PID
 *
 * 时间复杂度：O(n/64)，n 为最大 PID 数
 * 空间复杂度：O(n/64) bits
 */
int32_t pid_alloc(void) {
    for (int i = 0; i < MAX_PID / 64; i++) {
        if (pid_bitmap[i] != 0xFFFFFFFFFFFFFFFF) {
            // 找到第一个 0 位
            int bit = __builtin_ctzll(~pid_bitmap[i]);
            pid_bitmap[i] |= (1ULL << bit);
            return i * 64 + bit + 1;  // PID 从 1 开始
        }
    }
    return -1;  // 无可用 PID
}
```

### 2. 上下文切换算法

```c
/**
 * switch_context - 进程上下文切换
 *
 * 算法：
 * 1. 保存当前进程的 callee-saved 寄存器
 * 2. 更新 TSS 内核栈指针
 * 3. 切换 CR3 (地址空间)
 * 4. 恢复下一个进程的寄存器
 * 5. 返回到下一个进程
 *
 * @param prev: 当前进程的 cpu_ctx 指针的指针
 * @param next: 下一个进程的 PCB 指针
 *
 * 寄存器约定 (System V AMD64 ABI)：
 * - callee-saved: rbx, rbp, r12-r15
 * - caller-saved: rax, rcx, rdx, rsi, rdi, r8-r11
 */
void switch_context(cpu_context_t** prev, pcb_t* next);
```

**汇编实现要点**：
```assembly
# 1. 保存到 prev->cpu_ctx
mov %rax, (%rdi)
mov %rbx, 8(%rdi)
mov %rbp, 16(%rdi)
mov %r12, 24(%rdi)
mov %r13, 32(%rdi)
mov %r14, 40(%rdi)
mov %r15, 48(%rdi)
movq (%rsp), %rax
mov %rax, 56(%rdi)      # RIP
movq %rsp, 64(%rdi)     # RSP

# 2. 切换 TSS
mov PCB_KERNEL_STACK(%rsi), %rdx
mov %rdx, %gs:0

# 3. 切换 CR3
mov PCB_PML4(%rsi), %rdx
mov %rdx, %cr3

# 4. 从 next->cpu_ctx 恢复
mov 64(%rsi), %rsp
mov 56(%rsi), %rax
push %rax
mov 8(%rsi), %rbx
# ...
ret
```

### 3. 进程调度算法

```c
/**
 * schedule - Round-Robin 调度
 *
 * 算法：
 * 1. 从运行队列头部取出进程
 * 2. 如果队列为空，运行 idle 进程
 * 3. 将当前进程移到队列尾部
 * 4. 切换到选中的进程
 *
 * 时间复杂度：O(1)
 */
void schedule(void) {
    pcb_t* prev = scheduler.current;
    pcb_t* next;

    // 取出队列头部
    if (!list_empty(&scheduler.run_queue)) {
        next = list_first_entry(&scheduler.run_queue, pcb_t, run_list);
        list_del(&next->run_list);

        // 将 prev 加入尾部 (如果不是同一个)
        if (prev != next && prev->state == PROC_READY) {
            list_add_tail(&prev->run_list, &scheduler.run_queue);
        }
    } else {
        next = scheduler.idle;
    }

    // 切换进程
    if (prev != next) {
        scheduler.current = next;
        switch_context(&prev->cpu_ctx, next);
    }
}
```

### 4. fork 算法

```c
/**
 * proc_fork - 创建子进程
 *
 * 算法：
 * 1. 分配新 PID
 * 2. 分配并初始化 PCB
 * 3. 创建新地址空间
 * 4. 分配内核栈
 * 5. 复制父进程的 trap_frame
 * 6. 设置子进程的返回值为 0
 * 7. 建立父子关系
 * 8. 添加到运行队列
 * 9. 返回子进程 PID 给父进程
 *
 * 注意：当前简化实现不复制用户内存
 */
int32_t proc_fork(void);
```

### 5. exit 算法

```c
/**
 * proc_exit - 退出当前进程
 *
 * 算法：
 * 1. 设置进程状态为 ZOMBIE
 * 2. 保存退出码
 * 3. 从运行队列移除
 * 4. 将子进程移交给 init 进程
 * 5. 唤醒等待的父进程
 * 6. 调度下一个进程
 *
 * 注意：PCB 的释放在父进程 wait4 时进行
 */
void proc_exit(int exit_code);
```

### 6. wait4 算法

```c
/**
 * proc_wait4 - 等待子进程
 *
 * 算法：
 * 1. 查找指定子进程（或任意僵尸子进程）
 * 2. 如果子进程未退出且非阻塞模式，返回 0
 * 3. 如果子进程未退出，阻塞当前进程
 * 4. 子进程退出后，收集退出状态
 * 5. 释放子进程的 PCB 和资源
 * 6. 返回子进程 PID 和退出码
 */
int32_t proc_wait4(int32_t pid, int* wstatus, int options);
```

## 系统调用映射

### 进程相关系统调用

| 系统调用 | syscall 号 | 处理函数 | 功能描述 |
|----------|-----------|----------|----------|
| fork | SYS_FORK | sys_fork | 创建新进程 |
| exit | SYS_EXIT | sys_exit | 退出当前进程 |
| wait4 | SYS_WAIT4 | sys_wait4 | 等待子进程 |

| getppid | SYS_GETPPID | sys_getppid | 获取父进程 ID |

### 系统调用实现

```c
/**
 * sys_fork - fork 系统调用实现
 */
int64_t sys_fork(void) {
    return proc_fork();
}

/**
 * sys_exit - exit 系统调用实现
 * @param error_code: 退出码
 */
int64_t sys_exit(int64_t error_code) {
    proc_exit((int)error_code);

    return 0;
}

/**
 * sys_wait4 - wait4 系统调用实现
```

## 寄存器保存约定

### System V AMD64 ABI

**Callee-Saved (需要保存)**：
- RBX, RBP, R12, R13, R14, R15
- RSP (栈指针)

**Caller-Saved (调用者保存)**：
- RAX, RCX, RDX, RSI, RDI, R8, R9, R10, R11

**函数返回值**：
- RAX (整数)
- RAX, RDX (128 位)
- XMM0, XMM1 (浮点)
