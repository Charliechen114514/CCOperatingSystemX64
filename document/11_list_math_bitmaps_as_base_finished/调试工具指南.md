```gdb
define dump_list
```

## 2. 数据结构可视化

### 2.1 链表可视化脚本

```python
# gdb_python_helpers.py
import gdb

class DumpListCommand(gdb.Command):
    """Dump a linked list structure"""

    def __init__(self):
        super(DumpListCommand, self).__init__("dump_list", gdb.COMMAND_DATA)

    def invoke(self, arg, from_tty):
        argv = gdb.string_to_argv(arg)
        if len(argv) < 1:
            print("Usage: dump_list <head_ptr> [<member_type> <member_name>]")
            return

        head = gdb.parse_and_eval(argv[0])
        count = 0

        print(f"List at {head}:")
        current = head['next']
        while current != head.address and count < 1000:
            if len(argv) == 3:
                # 获取包含结构体
                struct_type = gdb.lookup_type(argv[1])
                member_name = argv[2]
                member_offset = struct_type[member_name].bitpos // 8
                struct_ptr = (current.cast(gdb.lookup_type("unsigned long")) - member_offset)
                struct_ptr = struct_ptr.cast(struct_type.pointer())
                print(f"  [{count}] {struct_ptr}")
            else:
                print(f"  [{count}] {current}")
            current = current['next']
            count += 1

        print(f"Total: {count} nodes")

DumpListCommand()
```

使用：
```gdb
(gdb) source gdb_python_helpers.py
(gdb) dump_list task_list
```

### 2.2 位图可视化脚本

```python
class DumpBitmapCommand(gdb.Command):
    """Display bitmap in binary format"""

    def invoke(self, arg, from_tty):
        argv = gdb.string_to_argv(arg)
        if len(argv) < 1:
            print("Usage: dump_bitmap <bitmap_ptr>")
            return

        bm = gdb.parse_and_eval(argv[0])
        bits_ptr = bm['bits']
        nbits = int(bm['nbits'])

        print(f"Bitmap ({nbits} bits):")
        for i in range(0, nbits, 64):
            if i % 64 == 0:
                print(f"\n{i:04x}: ", end="")

            byte_idx = i // 8
            byte_val = bits_ptr[byte_idx]
            bit_val = (int(byte_val) >> (i % 8)) & 1

            if i % 8 == 0 and i != 0:
                print(" ", end="")

            print(bit_val, end="")

        print()
```

### 3.3 检测链表环

```gdb
# Floyd 环检测算法
define detect_list_loop
    set $slow = $arg0->next
    set $fast = $arg0->next->next
    set $has_loop = 0

    while $fast != 0 && $fast != $arg0
        if $slow == $fast
            set $has_loop = 1
            loop_break
        end
        set $slow = $slow->next
        if $fast->next == 0
            loop_break
        end
        set $fast = $fast->next->next
    end

    if $has_loop
        printf "ERROR: Loop detected in list %p\n", $arg0
    else
        printf "OK: No loop in list %p\n", $arg0
    end
end

# 使用
(gdb) detect_list_loop head
```

## 4. 性能分析

### 4.1 链表操作计时

```c
// 在代码中添加计时
#include "klogs/kprintf.h"

static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ __volatile__ (
        "rdtsc"
        : "=a"(lo), "=d"(hi)
    );
    return ((uint64_t)hi << 32) | lo;
}

// 测试链表遍历性能
void benchmark_list_traversal(list_head* head) {
    uint64_t start = rdtsc();

    size_t count = 0;
    list_head* pos;
    list_for_each(pos, head) {
        count++;
    }

    uint64_t end = rdtsc();
    uint64_t cycles = end - start;

    klog_info("Traversed %zu nodes in %lu cycles (%lu cycles/node)\n",
              count, cycles, cycles / count);
}
```

### 4.2 位图操作计时

```c
// 测试位图查找性能
void benchmark_bitmap_find_zero(bitmap* bm) {

    for (int i = 0; i < 1000; i++) {
        bitmap_find_first_zero(bm);
    }

    uint64_t end = rdtsc();
```

### 4.3 位操作基准测试

```c
// 测试 2 的幂次检测性能
void benchmark_is_power_of_2(void) {
    const int ITERATIONS = 1000000;
    uint64_t start, end;
    volatile bool result;

    // 测试 2 的幂次
    start = rdtsc();
    for (int i = 0; i < ITERATIONS; i++) {
        result = is_power_of_2(i);
    }
    end = rdtsc();

    klog_info("is_power_of_2: %lu cycles average\n", (end - start) / ITERATIONS);
}
```

## 6. 自定义调试工具

### 6.1 链表完整性检查

```c
// list_debug.h
#ifndef LIST_DEBUG_H
#define LIST_DEBUG_H

#include "list/list.h"
#include "klogs/kprintf.h"

// 检查链表完整性
static inline bool list_verify(list_head* head) {
    // 检查链表头
    if (head->next->prev != head || head->prev->next != head) {
        klog_error("List head %p has broken links\n", head);
        return false;
    }

    // 检查所有节点
    list_head* pos;
    int count = 0;
    list_for_each(pos, head) {
        // 检查双向链接
        if (pos->next->prev != pos) {
            klog_error("Node %p has broken forward link\n", pos);
            return false;
        }
        if (pos->prev->next != pos) {
            klog_error("Node %p has broken backward link\n", pos);
            return false;
        }
        count++;
        if (count > 100000) {
            klog_error("List appears to have a loop (>%d nodes)\n", count);
            return false;
        }
    }

    return true;
}

// 打印链表统计信息
static inline void list_stats(list_head* head) {
    size_t count = list_count(head);
    bool empty = list_is_empty(head);
    bool singular = list_is_singular(head);

    klog_info("List %p: count=%zu, empty=%d, singular=%d\n",
              head, count, empty, singular);
}

#endif // LIST_DEBUG_H
```

### 6.2 位图调试工具

```c
// bitmap_debug.h
#ifndef BITMAP_DEBUG_H
#define BITMAP_DEBUG_H

#include "bitmap/bitmap.h"
#include "klogs/kprintf.h"

// 打印位图状态
static inline void bitmap_dump(const bitmap* bm) {
    klog_info("Bitmap %p (%zu bits):\n", bm, bm->nbits);

    // 打印十六进制
    size_t bytes = (bm->nbits + 7) / 8;
    for (size_t row = 0; row < bytes; row += 16) {
        klog_info("  %04zx: ", row);
        for (size_t col = 0; col < 16 && row + col < bytes; col++) {
            klog_info("%02x ", bm->bits[row + col]);
        }
        klog_info("\n");
    }
}

// 验证位图一致性
static inline bool bitmap_verify(const bitmap* bm) {
    size_t set_count = 0;
    size_t clear_count = 0;

    for (size_t i = 0; i < bm->nbits; i++) {
        bool is_set = bitmap_test(bm, i);
        if (is_set) {
            set_count++;
        } else {
            clear_count++;
        }
    }

    size_t weight = bitmap_weight(bm);

    if (weight != set_count) {
        klog_error("Bitmap weight mismatch: bitmap_weight=%zu, actual=%zu\n",
                   weight, set_count);
        return false;
    }

    if (set_count + clear_count != bm->nbits) {
        klog_error("Bitmap count mismatch: set=%zu, clear=%zu, total=%zu\n",
                   set_count, clear_count, bm->nbits);
        return false;
    }

    return true;
}

#endif // BITMAP_DEBUG_H
```

### 6.3 性能分析工具

```c
// perf_counter.h
#ifndef PERF_COUNTER_H
#define PERF_COUNTER_H

#include <stdint.h>
#include "klogs/kprintf.h"

// 性能计数器
typedef struct {
    const char* name;
    uint64_t total;
    uint64_t count;
    uint64_t min;
    uint64_t max;
} perf_counter_t;

// 初始化计数器
#define PERF_COUNTER_INIT(name) { name, 0, 0, UINT64_MAX, 0 }

// 开始计时
static inline uint64_t perf_start(void) {
    uint32_t lo, hi;
    __asm__ __volatile__ (
        "rdtsc"
        : "=a"(lo), "=d"(hi)
    );
    return ((uint64_t)hi << 32) | lo;
}

// 结束计时并记录
static inline void perf_end(perf_counter_t* pc, uint64_t start) {
    uint64_t end = perf_start();
    uint64_t elapsed = end - start;

    pc->total += elapsed;
    pc->count++;
    if (elapsed < pc->min) pc->min = elapsed;
    if (elapsed > pc->max) pc->max = elapsed;
}

// 打印统计信息
static inline void perf_print(const perf_counter_t* pc) {
    if (pc->count == 0) {
        klog_info("%s: no samples\n", pc->name);
        return;
    }

    uint64_t avg = pc->total / pc->count;
    klog_info("%s: count=%lu, total=%lu, avg=%lu, min=%lu, max=%lu\n",
              pc->name, pc->count, pc->total, avg, pc->min, pc->max);
}

// 使用示例

// uint64_t t = perf_start();
// list_add(...);
// perf_end(&list_add_counter, t);
// perf_print(&list_add_counter);

#endif // PERF_COUNTER_H
```

## 7. 调试工作流

### 7.1 系统化调试流程

```
1. 复现问题
   ↓
2. 添加断点/日志
   ↓
3. 收集数据
   ↓
4. 分析数据
   ↓
5. 提出假设
   ↓
6. 验证假设
   ↓
7. 修复问题
   ↓
8. 验证修复
```

### 7.2 数据结构调试检查清单

#### 链表检查清单

- [ ] 链表头已初始化
- [ ] 节点指针有效
- [ ] 双向链接完整
- [ ] 遍历能正常结束
- [ ] 没有环或无效指针

#### 位图检查清单

- [ ] 位图已初始化
- [ ] 缓冲区大小足够
- [ ] 位索引在范围内
- [ ] 设置/清除操作生效
- [ ] 查找操作返回正确值

#### 栈回溯检查清单

- [ ] 编译时保留帧指针
- [ ] 栈范围定义正确
- [ ] 返回地址有效
- [ ] 符号表已加载
