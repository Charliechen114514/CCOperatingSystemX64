## 初始化问题

### 1.1 调度类注册失败

**症状**:
```
[SCHED] Failed to register RR class
```

**原因**:
- 调度类已注册
- `policy` 值无效
- `sched_class_init()` 未调用

**解决方案**:
```c
// 确保正确顺序
int proc_init(void) {
    // 1. 先初始化框架
    int ret = sched_class_init();
    if (ret != 0) {
        klog_error("[PROC] Failed to init scheduler classes\n");
        return ret;
    }

    // 2. 再注册调度类
    ret = sched_rr_init();
    if (ret != 0) {
        klog_error("[PROC] Failed to init RR scheduler class\n");
        return ret;
    }

    // 3. 不要重复注册
    ret = sched_prio_init();
    // ...
}
```

**检查**:
- 确认 `sched_class_init()` 在 `sched_rr_init()` 之前调用
- 确认没有多次调用 `sched_rr_init()`
- 检查 `policy` 值是否在 `[0, SCHED_MAX)` 范围内

### 1.2 运行队列未初始化

**症状**:
```
[SCHED] rq is NULL
```

**原因**:
- `sched_class_init()` 未调用或失败
- `scheduler.rq` 未正确设置

**解决方案**:
```c
int sched_class_init(void) {
    // 初始化静态运行队列
    for (int i = 0; i < SCHED_MAX; i++) {
        INIT_LIST_HEAD(&s_run_queues[i].queue);
        s_run_queues[i].sched_class = NULL;
        s_run_queues[i].nr_running = 0;
        s_run_queues[i].class_data = NULL;
        s_classes[i] = NULL;
    }

    // 关键：链接到调度器
    scheduler.rq = s_run_queues;     // ← 必须设置
    scheduler.classes = s_classes;   // ← 必须设置

    return 0;
}
```

### 1.3 Priority class_data 分配失败

**症状**:
```
[SCHED] Failed to allocate priority queue data
```

**原因**:
- 堆内存不足
- `kmalloc` 返回 NULL

**解决方案**:
```c
int sched_prio_init(void) {
    prio_rq_data_t* data = (prio_rq_data_t*)kmalloc(sizeof(prio_rq_data_t));
    if (!data) {
        klog_error("[SCHED] Failed to allocate priority queue data\n");
        return -1;
    }

    // 初始化队列...
    for (int i = 0; i < PRIO_LEVELS; i++) {
        INIT_LIST_HEAD(&data->active[i]);
        INIT_LIST_HEAD(&data->expired[i]);
    }

    // 必须设置 class_data
    extern scheduler_t scheduler;
    sched_rq_t* rq = &scheduler.rq[SCHED_PRIORITY];
    rq->class_data = data;  // ← 关键

    return sched_class_register(&prio_sched_class, SCHED_PRIORITY);
}
```

## 调度问题

### 2.1 任务不被调度

**症状**:
- 进程创建后不运行
- 调度器总是选择 idle
- 运行队列显示有任务但不执行

**原因**:
1. 任务未加入运行队列
2. 任务的 `sched_class` 为 NULL
3. 所有队列显示为空

**诊断**:
```c
// 检查运行队列状态
void debug_check_queues(void) {
    extern scheduler_t scheduler;

    klog_info("=== Queue Status ===\n");
    klog_info("RR queue nr_running: %u\n", scheduler.rq[SCHED_NORMAL].nr_running);
    klog_info("Prio queue nr_running: %u\n", scheduler.rq[SCHED_PRIORITY].nr_running);

    // 检查当前任务
    if (scheduler.current) {
        klog_info("Current task: PID=%d, state=%d, time_slice=%u\n",
                  scheduler.current->pid,
                  scheduler.current->state,
                  scheduler.current->sched_entity.time_slice);
        klog_info("  sched_class=%p, policy=%d\n",
                  scheduler.current->sched_entity.sched_class,
                  scheduler.current->sched_entity.policy);
    }
}
```

**解决方案**:
```c
// 确保正确创建和入队
int32_t proc_fork(void) {
    pcb_t* child = proc_alloc_pcb();

    // 1. 设置调度策略
    int ret = sched_set_policy(child, SCHED_NORMAL, 0);
    if (ret != 0) {
        klog_error("[PROC] Failed to set scheduling policy\n");
        // ...
    }

    // 2. 检查 sched_class 是否设置
    if (child->sched_entity.sched_class == NULL) {
        klog_error("[PROC] sched_class is NULL!\n");
        // 重新注册调度类或检查初始化顺序
    }

    // 3. 加入运行队列
    sched_enqueue_task(child, false);

    return child->pid;
}
```

### 2.2 时间片不递减

**症状**:
- 任务时间片始终不变化
- 任务从不被抢占
- `task_tick` 未被调用

**原因**:
1. 定时器回调未注册
2. `current->sched_class` 为 NULL
3. 中断未启用

**诊断**:
```c
// 添加调试代码到 task_tick
static void rr_task_tick(struct sched_rq* rq, struct pcb* pcb) {
    klog_debug("[RR] task_tick: PID=%d, time_slice=%u -> ",
               pcb->pid, pcb->sched_entity.time_slice);

    if (pcb->sched_entity.time_slice > 0) {
        pcb->sched_entity.time_slice--;
    }

    klog_debug("%u\n", pcb->sched_entity.time_slice);

    if (pcb->sched_entity.time_slice == 0) {
        klog_debug("[RR] Time slice expired, setting resched\n");
        sched_set_resched();
    }
}
```

**解决方案**:
```c
// 确保定时器回调已注册
int sched_class_init(void) {
    // ...

    // 注册定时器回调
    timer_set_callback(sched_timer_tick_handler);  // ← 关键

    return 0;
}

// 定时器处理函数
static void sched_timer_tick_handler(uint64_t ticks) {
    if (scheduler.current && scheduler.rq != NULL) {
        if (scheduler.current->sched_entity.sched_class) {
            sched_rq_t* rq = &scheduler.rq[scheduler.current->sched_entity.policy];
            scheduler.current->sched_entity.sched_class->task_tick(rq, scheduler.current);
        } else {
            klog_warn("[SCHED] current task has no sched_class!\n");
        }
    }
}
```

### 2.3 优先级不生效

**症状**:
- 高优先级任务不优先运行
- 低优先级任务抢占高优先级任务

**原因**:
1. `pick_next_task` 未正确检查优先级
2. Priority 队列为空
3. 抢占判断逻辑错误

**诊断**:
```c
// 检查优先级队列状态
void debug_prio_queue(void) {
    extern scheduler_t scheduler;
    prio_rq_data_t* data = prio_rq_data(&scheduler.rq[SCHED_PRIORITY]);

    klog_info("=== Priority Queue ===\n");
    klog_info("nr_active: %u, nr_expired: %u\n", data->nr_active, data->nr_expired);
    klog_info("highest_prio: %d\n", data->highest_prio);

    // 遍历所有优先级
    for (int i = 0; i < PRIO_LEVELS; i++) {
        if (!list_is_empty(&data->active[i])) {
            klog_info("  Prio %d: %u tasks\n", i, count_tasks(&data->active[i]));
        }
    }
}
```

**解决方案**:
```c
// 确保 pick_next_task 正确检查优先级队列
struct pcb* sched_pick_next_task(void) {
    pcb_t* prev = scheduler.current;

    // ... 当前任务检查 ...

    // 优先检查 Priority 队列
    sched_rq_t* prio_rq = &scheduler.rq[SCHED_PRIORITY];
    if (prio_rq->nr_running > 0 && prio_rq->sched_class) {
        struct pcb* next = prio_rq->sched_class->pick_next_task(prio_rq, prev);
        if (next) {
            klog_debug("[SCHED] Selected priority task: PID=%d, prio=%d\n",
                      next->pid, next->sched_entity.priority);
            return next;
        }
    }

    // 回退到 RR 队列
    // ...
}
```

## 抢占问题

### 3.1 高优先级任务不抢占

**症状**:
- 高优先级任务就绪后不立即运行
- 低优先级任务继续运行

**原因**:
1. `should_preempt` 返回 false
2. 抢占检查点缺失
3. `need_resched` 标志未触发

**诊断**:
```c
// 测试抢占逻辑
void test_preemption(void) {
    pcb_t high, low;

    // 高优先级任务
    high.sched_entity.priority = 10;
    high.sched_entity.time_slice = 10;

    // 低优先级任务
    low.sched_entity.priority = 100;
    low.sched_entity.time_slice = 10;

    sched_class_t* prio_class = sched_prio_get_class();

    bool should = prio_class->should_preempt(&high, &low);
    klog_info("Should high(%d) preempt low(%d)? %s\n",
              high.sched_entity.priority,
              low.sched_entity.priority,
              should ? "YES" : "NO");

    // 应该输出 YES
}
```

**解决方案**:
```c
// 确保 should_preempt 正确实现
static bool prio_should_preempt(struct pcb* p, struct pcb* curr) {
    // 高优先级（数值小）抢占低优先级
    if (p->sched_entity.priority < curr->sched_entity.priority) {
        return true;
    }

    // 同优先级：仅当当前时间片用完
    if (p->sched_entity.priority == curr->sched_entity.priority) {
        return (curr->sched_entity.time_slice == 0);
    }

    return false;
}
```

### 3.2 need_resched 标志未检查

**症状**:
- `need_resched` 被设置但不发生调度
- 调度延迟到下一个系统调用

**原因**:
- 中断返回未检查 `need_resched`
- 抢占点缺失

**解决方案**:
```c
// 在中断返回时检查（trap handler）
void trap_handler(trap_frame_t* frame) {
    // ... 处理中断 ...

    // 检查是否需要调度
    if (sched_needs_reschedule()) {
        schedule();
    }
}
```

## 内存问题

### 4.1 class_data 空指针访问

**症状**:
```
Page fault at 0x10 (trying to access NULL->active)
```

**原因**:
- `class_data` 未分配
- 错误的运行队列被访问

**解决方案**:
```c
// 访问前检查
static inline prio_rq_data_t* prio_rq_data(struct sched_rq* rq) {
    if (rq->class_data == NULL) {
        klog_error("[SCHED] class_data is NULL!\n");
        // 返回一个安全的默认值或处理错误
        return NULL;
    }
    return (prio_rq_data_t*)rq->class_data;
}

// 使用
static void prio_enqueue_task(struct sched_rq* rq, struct pcb* pcb, bool head) {
    prio_rq_data_t* data = prio_rq_data(rq);
    if (!data) {
        klog_error("[PRIO] Cannot enqueue: class_data is NULL\n");
        return;
    }
    // ...
}
```

### 4.2 队列为空时访问

**症状**:
```
list_first_entry on empty list
```

**原因**:
- 未检查队列是否为空
- `nr_running` 与实际队列状态不同步

**解决方案**:
```c
// 总是先检查
static struct pcb* prio_pick_next_task(struct sched_rq* rq, struct pcb* prev) {
    prio_rq_data_t* data = prio_rq_data(rq);

    // 检查 active 队列
    if (data->nr_active == 0) {
        if (data->nr_expired > 0) {
            prio_swap_active_expired(data);
        } else {
            return NULL;  // 没有任务
        }
    }

    // 再次检查（防御性编程）
    int highest = find_highest_prio(data);
    if (highest >= PRIO_LEVELS || list_is_empty(&data->active[highest])) {
        return NULL;
    }

    return list_first_entry(&data->active[highest], struct pcb, sched_entity.run_list);
}
```

### 4.3 链表损坏

**症状**:
```
corrupted list head
list_add corruption
```

**原因**:
- 同一节点被多次加入队列
- 队列操作时未正确初始化链表头

**解决方案**:
```c
// 确保正确初始化
static pcb_t* proc_alloc_pcb(void) {
    pcb_t* pcb = kmalloc(sizeof(pcb_t));

    // 初始化所有链表头
    INIT_LIST_HEAD(&pcb->run_list);
    INIT_LIST_HEAD(&pcb->siblings);
    INIT_LIST_HEAD(&pcb->children);
    INIT_LIST_HEAD(&pcb->zombie_children);
    INIT_LIST_HEAD(&pcb->sched_entity.run_list);  // ← 调度实体链表

    return pcb;
}

// 出队时正确处理
static void rr_dequeue_task(struct sched_rq* rq, struct pcb* pcb) {
    list_del_init(&pcb->sched_entity.run_list);  // ← 使用 del_init
    rq->nr_running--;
}
```

### 5.3 统计信息收集

```c
typedef struct sched_stats {
    uint64_t schedule_count;
    uint64_t rr_scheduled;
    uint64_t prio_scheduled;
    uint64_t idle_time;
    uint64_t context_switches;
} sched_stats_t;

sched_stats_t sched_stats;

void schedule(void) {
    sched_stats.schedule_count++;

    pcb_t* next = sched_pick_next_task();

    if (next->sched_entity.policy == SCHED_NORMAL) {
        sched_stats.rr_scheduled++;
    } else if (next->sched_entity.policy == SCHED_PRIORITY) {
        sched_stats.prio_scheduled++;
    }

    // ...
}

void print_sched_stats(void) {

    klog_info("Priority scheduled: %llu\n", sched_stats.prio_scheduled);
```
