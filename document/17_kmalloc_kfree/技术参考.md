## 堆内存布局

### 1. 虚拟地址空间中的位置

```c
// 内核虚拟地址空间布局
#define KERNEL_TEXT_BASE   0xFFFFFFFF80000000ULL  // 内核代码
#define KERNEL_DATA_BASE   0xFFFFFFFF80200000ULL  // 内核数据
#define KERNEL_HEAP_BASE   0xFFFFFFFF81000000ULL  // 内核堆起始
#define KERNEL_HEAP_MAX    0xFFFFFFFF89000000ULL  // 内核堆结束 (128MB)
```

### 2. 堆区域结构

```
0xFFFFFFFF81000000  KERNEL_HEAP_BASE
    │
    ├─ 块 0 (已使用)
    ├─ 块 1 (空闲)
    ├─ 块 2 (已使用)
    ├─ 块 3 (空闲)
    ├─ ...
    │
0xFFFFFFFF89000000  KERNEL_HEAP_MAX (128MB 后)
```

### 3. 单个块的布局

```
┌────────────────────────────────────────┐
│ heap_block_t (16 字节)                 │
├────────────────────────────────────────┤
│ size:    8 字节 (总大小)               │
│ used:    1 字节 (使用状态)             │
│ padding: 3 字节                        │
│ prev:    8 字节 (前块指针)             │
│ next:    8 字节 (下块指针，空闲链表)    │
│ magic:   4 字节 (魔数 0x00114514)      │
│ _pad:    4 字节 (对齐填充)             │
├────────────────────────────────────────┤
│ 用户数据 (size - 16 字节)              │
│ ...                                    │
└────────────────────────────────────────┘
```

### 4. 与其他模块的关系

```
用户调用 kmalloc(size)
    │
    v
heap.c: kmalloc()
    │ (需要更多内存时)
    v
vmm.c: vmm_alloc_pages_at()
    │
    v
pframe.c: pframe_alloc()
    │
    v
物理内存
```

## Heap API 参考

### 初始化

#### heap_init

```c
/**
 * heap_init - 初始化堆分配器
 *
 * 必须在 vmm_init() 之后调用。
 *
 * @return HEAP_OK 成功
 *         HEAP_ERR_OOM 内存不足
 *         HEAP_ERR_VMM VMM 调用失败
 */
heap_result_t heap_init(void);
```

**使用示例**：
```c
void kernel_init(void) {
    // 确保 VMM 已初始化
    vmm_init();

    // 初始化堆
    heap_result_t result = heap_init();
    if (result != HEAP_OK) {
        klog_error("Failed to initialize heap: %d\n", result);
    }
}
```

### 基本分配

#### kmalloc

```c
/**
 * kmalloc - 从堆中分配内存
 * @param size: 请求的字节数
 * @return 成功返回指向内存的指针，失败返回 NULL
 *
 * 特性：
 * - 最小分配 16 字节
 * - 实际分配大小向上舍入到 16 字节对齐
 * - 包含 16 字节头部开销
 * - 内存未初始化，内容不确定
 */
void* kmalloc(size_t size);
```

**使用示例**：
```c
// 分配 100 字节
void* ptr = kmalloc(100);
if (ptr == NULL) {
    // 处理内存不足
}

// 使用内存
memset(ptr, 0, 100);

// 释放
kfree(ptr);
```

#### kfree

```c
/**
 * kfree - 释放之前分配的内存
 * @param ptr: 要释放的内存指针（可以为 NULL）
 *
 * 注意事项：
 * - 传递 NULL 是安全的，不会做任何操作
 * - 双重释放会被检测到并记录错误
 * - 释放后应将指针置 NULL 以避免野指针
 * - 对齐分配的内存使用普通 kfree 释放
 */
void kfree(void* ptr);
```

**使用示例**：
```c
void* ptr = kmalloc(1024);
// ... 使用 ...
kfree(ptr);
ptr = NULL;  // 避免野指针

// NULL 是安全的
kfree(NULL);
```

### 重新分配

#### krealloc

```c
/**
 * krealloc - 重新分配内存块
 * @param ptr: 原内存指针（可以为 NULL）
 * @param new_size: 新的大小
 * @return 新内存指针，失败返回 NULL
 *
 * 行为：
 * - ptr == NULL: 等同于 kmalloc(new_size)
 * - new_size == 0: 等同于 kfree(ptr)，返回 NULL
 * - 新大小 <= 旧大小: 缩小块
 * - 新大小 > 旧大小: 扩展块（可能需要移动）
 *
 * 注意：
 * - 可能返回不同的地址
 * - 原内容会被复制到新位置
 * - 失败时原内存块不变
 * - 返回 NULL 时需要检查是否保留原指针
 */
void* krealloc(void* ptr, size_t new_size);
```

**使用示例**：
```c
// 初始分配
char* str = (char*)kmalloc(16);
strcpy(str, "Hello");

// 扩展
str = (char*)krealloc(str, 64);
if (str != NULL) {
    strcat(str, ", World!");
}

// 缩小
str = (char*)krealloc(str, 32);
if (str != NULL) {
    // 内容保持不变
}

// 释放
kfree(str);

// 特殊用法
char* new_str = (char*)krealloc(NULL, 32);  // 等同于 kmalloc
krealloc(new_str, 0);  // 等同于 kfree
```

### 对齐分配

#### kmalloc_aligned

```c
/**
 * kmalloc_aligned - 分配对齐的内存
 * @param size: 请求的字节数
 * @param alignment: 对齐要求（必须是 2 的幂）
 * @return 对齐的内存指针，失败返回 NULL
 *
 * 特性：
 * - alignment 必须是 2 的幂（16, 32, 64, 4096, ...）
 * - 最小对齐是 16 字节
 * - 返回的地址保证 alignment 对齐
 * - 使用普通 kfree() 释放（不是特殊函数）
 *
 * 用途：
 * - DMA 缓冲区（通常需要 4KB 对齐）
 * - SIMD 数据（16/32 字节对齐）
 * - 页表项（4KB 对齐）
 */
void* kmalloc_aligned(size_t size, size_t alignment);
```

**使用示例**：
```c
// DMA 缓冲区 (4KB 对齐)
void* dma_buffer = kmalloc_aligned(8192, 4096);
if (dma_buffer != NULL) {
    // dma_buffer 保证 4KB 对齐
    // ... 使用 ...
    kfree(dma_buffer);  // 使用普通 kfree
}

// SIMD 数据 (32 字节对齐用于 AVX)
float* simd_data = (float*)kmalloc_aligned(256, 32);
if (simd_data != NULL) {
    // 可以使用 AVX 指令
    kfree(simd_data);
}
```

## 数据结构定义

### heap_block_t

```c
/**
 * heap_block_t - 堆块头部结构
 *
 * 每个分配的内存块前都有这个头部。
 */
typedef struct heap_block {
    /**
     * @brief 总大小（包含头部），16 字节对齐
     *
     * 包括 heap_block_t 和用户数据的总字节数。
     * 总是 16 的倍数。
     */
    uint64_t size;

    /**
     * @brief 使用状态
     * true = 已使用，false = 空闲
     */
    bool used;

    /**
     * @brief 填充字段（对齐）
     */
    uint8_t _padding[3];

    /**
     * @brief 前一个块（内存中相邻）
     *
     * 指向内存中紧邻的前一个块，用于块合并。
     */
    struct heap_block* prev;

    /**
     * @brief 下一个块（空闲链表）
     *
     * 空闲时指向空闲链表中的下一个块（按地址排序）。
     * 已使用时此字段无意义。
     */
    struct heap_block* next;

    /**
     * @brief 魔数，用于检测内存损坏
     *
     * 值应为 0x00114514 (114514 的变体)
     */
    uint32_t magic;

    /**
     * @brief 对齐填充
     *
     * 确保结构体大小为 16 字节。
     */
    uint32_t _padding2;
} __attribute__((aligned(16))) heap_block_t;
```

### heap_state_t

```c
/**
 * heap_state_t - 堆全局状态
 */
typedef struct {
    /** 堆起始虚拟地址 */
    virtual_addr_t heap_start;

    /** 堆结束虚拟地址（最大值） */
    virtual_addr_t heap_end;

    /** 当前堆顶（动态增长） */
    virtual_addr_t heap_brk;

    /** 空闲块链表（按地址排序） */
    heap_block_t* free_list;

    /** 统计信息 */
    heap_stats_t stats;

    /** 初始化标志 */
    bool initialized;
} heap_state_t;
```

### heap_stats_t

```c
/**
 * heap_stats_t - 堆统计信息
 */
typedef struct {
    /** 总字节数（包括已用和空闲） */
    uint64_t total_bytes;

    /** 已使用字节数（不包括头部） */
    uint64_t used_bytes;

    /** 空闲字节数（不包括头部） */
    uint64_t free_bytes;

    /** 总块数 */
    uint64_t total_blocks;

    /** 已使用块数 */
    uint64_t used_blocks;

    /** 空闲块数 */
    uint64_t free_blocks;

    /** 累计分配次数 */
    uint64_t alloc_count;

    /** 累计释放次数 */
    uint64_t free_count;

    /** 堆扩展次数 */

} heap_stats_t;
```

## 算法说明

### 1. Best-Fit 搜索算法

```c
/**
 * find_best_fit - 查找最适合的空闲块
 * @param size: 需要的大小
 * @return 最合适的块，NULL 表示没有足够大的块
 *
 * 算法：
 * 1. 遍历空闲链表
 * 2. 对于每个 >= size 的空闲块
 * 3. 选择其中最小的
 * 4. 如果找到精确匹配，提前返回
 */
static heap_block_t* find_best_fit(size_t size) {
    heap_block_t* best = NULL;
    heap_block_t* current = s_heap.free_list;

    while (current != NULL) {
        if (!current->used && current->size >= size) {
            if (best == NULL || current->size < best->size) {
                best = current;
                // 精确匹配，立即返回
                if (current->size == size) {
                    break;
                }
            }
        }
        current = current->next;
    }

    return best;
}
```

**时间复杂度**：O(n)，n 为空闲块数

**空间复杂度**：O(1)

### 2. 块分割算法

```c
/**
 * split_block - 将大块分割成两个块
 * @param block: 要分割的块
 * @param new_size: 第一个块的新大小
 *
 * 前提：
 * - block->size >= new_size + 头部大小 + 最小块大小
 *
 * 结果：
 * - block 的大小变为 new_size
 * - 创建新块在 block 之后
 * - 新块加入空闲链表
 */
static void split_block(heap_block_t* block, size_t new_size) {
    size_t old_size = block->size;

    // 计算新块位置
    heap_block_t* new_block = (heap_block_t*)
        ((uint8_t*)block + new_size);

    // 设置新块
    new_block->size = old_size - new_size;
    new_block->used = false;
    new_block->prev = block;
    new_block->next = block->next;
    new_block->magic = HEAP_MAGIC;

    // 更新旧块
    block->size = new_size;
    block->next = new_block;

    // 更新新块的下一个块的前向指针
    if (new_block->next != NULL) {
        new_block->next->prev = new_block;
    }

    // 将新块加入空闲链表
    add_to_free_list(new_block);
}
```

### 3. 块合并算法

```c
/**
 * coalesce_blocks - 合并相邻的空闲块
 * @param block: 要合并的块（必须已标记为空闲）
 *
 * 算法：
 * 1. 检查下一个块，如果空闲则合并
 * 2. 检查前一个块，如果空闲则合并到前一个块
 */
static void coalesce_blocks(heap_block_t* block) {
    // 向后合并
    if (block->next != NULL && !block->next->used) {
        // 从空闲链表移除下一个块
        remove_from_free_list(block->next);

        // 合并大小
        block->size += block->next->size;

        // 更新链接
        block->next = block->next->next;
        if (block->next != NULL) {
            block->next->prev = block;
        }
    }

    // 向前合并
    if (block->prev != NULL && !block->prev->used) {
        // 从空闲链表移除当前块
        remove_from_free_list(block);

        // 合并到前一个块
        block->prev->size += block->size;
        block->prev->next = block->next;
        if (block->next != NULL) {
            block->next->prev = block->prev;
        }
        // block->prev 已经在空闲链表中，无需添加
    }
}
```

### 4. 堆扩展算法

```c
/**
 * expand_heap - 扩展堆大小
 * @param min_expansion: 最小需要扩展的字节数
 * @return HEAP_OK 成功，HEAP_ERR_OOM 失败
 *
 * 算法：
 * 1. 计算需要的页数（至少 HEAP_MIN_EXPANSION_PAGES）
 * 2. 检查是否超过堆最大值
 * 3. 使用 VMM 分配虚拟页
 * 4. 清零新内存
 * 5. 创建新块并合并到空闲链表
 * 6. 更新堆顶指针
 */
static heap_result_t expand_heap(size_t min_expansion) {
    // 计算页数
    uint64_t page_count = (min_expansion + PAGE_SIZE - 1) / PAGE_SIZE;
    if (page_count < HEAP_MIN_EXPANSION_PAGES) {
        page_count = HEAP_MIN_EXPANSION_PAGES;
    }

    // 检查边界
    virtual_addr_t new_brk = s_heap.heap_brk + page_count * PAGE_SIZE;
    if (new_brk > s_heap.heap_end) {
        return HEAP_ERR_OOM;
    }

    // 分配虚拟页
    if (vmm_alloc_pages_at(s_heap.heap_brk, page_count,
                          VMAP_FLAG_WRITE) != VMM_OK) {
        return HEAP_ERR_OOM;
    }

    // 清零
    memzero((void*)s_heap.heap_brk, page_count * PAGE_SIZE);

    // 创建块
    heap_block_t* new_block = (heap_block_t*)s_heap.heap_brk;
    new_block->size = page_count * PAGE_SIZE;
    new_block->used = false;
    new_block->prev = get_last_block();
    new_block->next = NULL;
    new_block->magic = HEAP_MAGIC;

    add_to_free_list(new_block);

    // 更新状态
    s_heap.heap_brk = new_brk;
    s_heap.stats.total_bytes += page_count * PAGE_SIZE;
    s_heap.stats.expand_count++;

    return HEAP_OK;
}
```

### 5. 对齐分配算法

```c
/**
 * kmalloc_aligned - 对齐分配实现
 *
 * 算法：
 * 1. 分配额外空间：alignment + sizeof(void*)
 * 2. 找到对齐的地址
 * 3. 在对齐地址前存储原始指针
 * 4. 返回对齐地址
 *
 * 释放时：
 * 1. 从对齐地址前获取原始指针
 * 2. 使用原始指针调用 kfree
 */
void* kmalloc_aligned(size_t size, size_t alignment) {
    // 分配额外空间
    size_t total_size = total_block_size(size) + alignment + sizeof(void*);

    void* raw_ptr = kmalloc(total_size);
    if (raw_ptr == NULL) {
        return NULL;
    }

    // 计算对齐地址
    virtual_addr_t raw = (virtual_addr_t)raw_ptr;
    virtual_addr_t aligned = ALIGN_UP(raw + sizeof(void*), alignment);

    // 存储原始指针
    *(virtual_addr_t*)(aligned - sizeof(void*)) = raw;

    return (void*)aligned;
}
```

## 返回码定义

### heap_result_t

| 值 | 名称 | 说明 |
|----|------|------|
| 0 | HEAP_OK | 操作成功 |
| -1 | HEAP_ERR_NOT_INIT | 堆未初始化 |
| -2 | HEAP_ERR_OOM | 内存不足 |
| -3 | HEAP_ERR_INVALID | 无效参数 |
| -4 | HEAP_ERR_CORRUPTED | 堆损坏（魔数错误） |
| -5 | HEAP_ERR_DOUBLE_FREE | 双重释放 |

### 使用示例

```c
heap_result_t result = heap_init();
switch (result) {
    case HEAP_OK:
        klog_trace("Heap initialized\n");
        break;
    case HEAP_ERR_OOM:
        klog_error("Out of memory during heap init\n");
        break;
    case HEAP_ERR_NOT_INIT:
        klog_error("VMM not initialized\n");
        break;
    default:
        klog_error("Unknown heap error: %d\n", result);
}
```
