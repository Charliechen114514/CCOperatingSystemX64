### 1. heap_dump()

**功能**：转储堆的完整状态到日志

**原型**：
```c
void heap_dump(void);
```

```c
// 在任何位置调用
```

### 2. heap_get_stats()

**功能**：获取堆统计信息

**原型**：
```c
heap_result_t heap_get_stats(heap_stats_t* stats);
```

**使用示例**：
```c
heap_stats_t stats;
if (heap_get_stats(&stats) == HEAP_OK) {
    klog_trace("=== Heap Statistics ===\n");
    klog_trace("Total bytes:   %llu\n", stats.total_bytes);
    klog_trace("Used bytes:    %llu\n", stats.used_bytes);
    klog_trace("Free bytes:    %llu\n", stats.free_bytes);
    klog_trace("Used blocks:   %llu\n", stats.used_blocks);
    klog_trace("Free blocks:   %llu\n", stats.free_blocks);
    klog_trace("Alloc count:   %llu\n", stats.alloc_count);
    klog_trace("Free count:    %llu\n", stats.free_count);
    klog_trace("Expand count:  %llu\n", stats.expand_count);

    // 计算使用率
    if (stats.total_bytes > 0) {
        uint64_t usage = (stats.used_bytes * 100) / stats.total_bytes;
        klog_trace("Usage:         %llu%%\n", usage);
    }
}
```

## GDB 调试

### 1. 常用断点

```gdb
# 在关键函数设置断点
b kmalloc
b kfree
b krealloc
b heap_init
b heap_dump

# 在内部函数设置断点
b find_best_fit
b coalesce_blocks
b expand_heap
```

### 2. 检查堆状态

```gdb
# 查看全局堆状态
print s_heap

# 查看特定字段
print s_heap.heap_start
print s_heap.heap_brk
print s_heap.free_list
print s_heap.stats

# 查看统计信息
print s_heap.stats.used_bytes
print s_heap.stats.free_bytes
print s_heap.stats.used_blocks
```

### 3. 遍历空闲链表

```gdb
# 定义遍历空闲链表的宏
define dump_free_list
    set $current = s_heap.free_list
    printf "Free List:\n"
    while $current != 0
        printf "  [%p] size=%llu, used=%d\n", $current, $current->size, $current->used
        set $current = $current->next
    end
end

# 使用
dump_free_list
```

### 4. 检查块结构

```gdb
# 打印块详情
define print_block
    set $block = $arg0
    printf "Block at %p:\n", $block
    printf "  Size:   %llu\n", $block->size
    printf "  Used:   %d\n", $block->used
    printf "  Magic:  0x%X\n", $block->magic
    printf "  Prev:   %p\n", $block->prev
    printf "  Next:   %p\n", $block->next
    printf "  Data:   %p\n", $block + 1
end

# 使用
print_block s_heap.heap_start
```

### 5. 验证魔数

```gdb
# 检查所有块的魔数
define verify_magic
    set $current = s_heap.heap_start
    set $end = s_heap.heap_brk
    printf "Verifying magic numbers...\n"
    while $current < $end
        if $current->magic != 0x00114514
            printf "BAD MAGIC at %p: 0x%X\n", $current, $current->magic
        end
        set $current = (void*)((char*)$current + $current->size)
    end
    printf "Done\n"
end

# 使用
verify_magic
```

### 6. 分配跟踪

```gdb
# 在 kmalloc 设置条件断点
b kmalloc
condition 1 size > 4096  # 只在分配 > 4KB 时停止

# 在特定调用者处停止
b kmalloc
condition 1 $rip == 0xFFFFFFFF80001234
```

### 7. 调试双重释放

```gdb
# 在 kfree 设置断点
b kfree

# 当停止时，检查块状态
print ptr
print ((heap_block_t*)((char*)ptr - 32))->used
print ((heap_block_t*)((char*)ptr - 32))->magic

# 如果 used == 0，说明是双重释放
```

## 性能分析

### 1. 分配性能测试

```c
void benchmark_alloc(void) {
    const int COUNT = 1000;
    void* ptrs[COUNT];
    uint64_t start, end;

    // 测试分配
    __asm__ volatile("rdtsc" : "=A"(start));
    for (int i = 0; i < COUNT; i++) {
        ptrs[i] = kmalloc(64);
    }
    __asm__ volatile("rdtsc" : "=A"(end));

    klog_trace("1000 x kmalloc(64): %llu cycles (%llu per alloc)\n",
              end - start, (end - start) / COUNT);

    // 清理
    for (int i = 0; i < COUNT; i++) {
        kfree(ptrs[i]);
    }
}
```

### 2. 释放性能测试

```c
void benchmark_free(void) {
    const int COUNT = 1000;
    void* ptrs[COUNT];
    uint64_t start, end;

    // 预先分配
    for (int i = 0; i < COUNT; i++) {
        ptrs[i] = kmalloc(64);
    }

    // 测试释放
    __asm__ volatile("rdtsc" : "=A"(start));
    for (int i = 0; i < COUNT; i++) {
        kfree(ptrs[i]);
    }
    __asm__ volatile("rdtsc" : "=A"(end));

    klog_trace("1000 x kfree: %llu cycles (%llu per free)\n",
              end - start, (end - start) / COUNT);
}
```

### 3. 碎片化分析

```c
void analyze_fragmentation(void) {
    heap_stats_t stats;
    heap_get_stats(&stats);

    // 外部碎片率
    uint64_t external_frag = 0;
    if (stats.free_bytes > 0) {
        // 计算最大连续空闲块
        heap_block_t* current = s_heap.free_list;
        uint64_t max_free = 0;
        while (current != NULL) {
            if (!current->used && current->size > max_free) {
                max_free = current->size - sizeof(heap_block_t);
            }
            current = current->next;
        }
        external_frag = 100 - (max_free * 100 / stats.free_bytes);
    }

    klog_trace("External fragmentation: %llu%%\n", external_frag);
}
```

## 常见调试场景

### 场景 1: 内存泄漏检测

**目标**：查找未释放的内存

**步骤**：

1. 在代码中添加检查点：
```c
heap_stats_t stats_checkpoint;
heap_get_stats(&stats_checkpoint);

// ... 执行可能泄漏的操作 ...

heap_stats_t stats_current;
heap_get_stats(&stats_current);

if (stats_current.used_bytes > stats_checkpoint.used_bytes) {
    klog_warn("Memory leak detected!\n");
    klog_warn("Leaked: %llu bytes\n",
             stats_current.used_bytes - stats_checkpoint.used_bytes);

    // 转储当前状态
    heap_dump();
}
```

2. 使用 GDB 设置条件断点：
```gdb
b heap_get_stats
commands
    printf "Used bytes: %llu\n", s_heap.stats.used_bytes
    continue
end
```

### 场景 2: 双重释放调试

**目标**：定位双重释放的位置

**步骤**：

1. 在 kfree 中添加检测：
```c
void kfree(void* ptr) {
    if (ptr == NULL) return;

    heap_block_t* block = (heap_block_t*)((uint8_t*)ptr - sizeof(heap_block_t));

    // 添加栈跟踪
    if (!block->used) {
        klog_error("[HEAP] Double free detected!\n");
        klog_error("[HEAP] ptr = %p\n", ptr);
        klog_error("[HEAP] block = %p\n", block);
        print_stack_trace();  // 需要实现
        return;
    }

    // ...
}
```

2. 使用 GDB 回溯：
```gdb
b kfree
commands
    if $arg0 != 0
        set $block = (void*)((char*)$arg0 - 32)
        if $block->used == 0
            bt
        end
    end
    continue
end
```

### 场景 3: 堆损坏调试

**目标**：查找损坏堆的代码

**步骤**：

1. 在关键位置添加验证：
```c
void verify_heap(void) {
    heap_block_t* current = (heap_block_t*)s_heap.heap_start;

    while ((virtual_addr_t)current < s_heap.heap_brk) {
        if (current->magic != HEAP_MAGIC) {
            klog_error("Heap corruption at %p\n", current);
            klog_error("Expected magic: 0x%X, got: 0x%X\n",
                      HEAP_MAGIC, current->magic);
            heap_dump();
            return;
        }
        current = (heap_block_t*)((uint8_t*)current + current->size);
    }
}

// 在怀疑的位置调用
verify_heap();
```

2. 使用 GDB 写观察点：
```gdb
# 在块头设置写观察点
watch 0xFFFF81000000
# 当该位置被写入时停止
```

### 场景 4: 性能优化

**目标**：找出慢速分配/释放

**步骤**：

1. 测量每个分配的时间：
```c
uint64_t alloc_start, alloc_end;

__asm__ volatile("rdtsc" : "=A"(alloc_start));
void* ptr = kmalloc(size);
__asm__ volatile("rdtsc" : "=A"(alloc_end));

klog_trace("kmalloc(%zu) took %llu cycles\n", size, alloc_end - alloc_start);
```

2. 在 GDB 中分析：
```gdb
b find_best_fit
commands
    set $start_count = 0
    while $current != 0
        set $start_count = $start_count + 1
        set $current = $current->next
    end
    printf "Searched %d blocks\n", $start_count
    continue
end
```
