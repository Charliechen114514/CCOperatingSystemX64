## 自旋锁 API

### 初始化

```c
void spin_init(spinlock_t* lock);
```

### 基本操作

```c
// 获取锁（忙等待）
void spin_lock(spinlock_t* lock);

// 尝试获取锁（非阻塞）
// 返回 true 成功，false 失败
bool spin_trylock(spinlock_t* lock);

// 释放锁
void spin_unlock(spinlock_t* lock);
```

### 中断安全版本

```c
// 保存中断状态并获取锁
void spin_lock_irqsave(spinlock_t* lock, unsigned long* flags);

// 释放锁并恢复中断状态
void spin_unlock_irqrestore(spinlock_t* lock, unsigned long flags);
```

## 互斥锁 API

### 初始化

```c
void mutex_init(mutex_t* mutex);
```

### 基本操作

```c
// 获取锁（可能阻塞）
void mutex_lock(mutex_t* mutex);

// 尝试获取锁（非阻塞）
// 返回 0 成功，非 0 失败
int mutex_trylock(mutex_t* mutex);

// 释放锁
void mutex_unlock(mutex_t* mutex);
```

### 使用示例

```c
mutex_t lock;
mutex_init(&lock);

// 线程 1
mutex_lock(&lock);
critical_section();
mutex_unlock(&lock);

// 线程 2
mutex_lock(&lock);  // 如果线程 1 持有锁，则阻塞
critical_section();
mutex_unlock(&lock);
```

## 信号量 API

### 初始化

```c
// value: 初始值（0 表示无资源）
void sem_init(semaphore_t* sem, int value);
```

### 基本操作

```c
// 等待信号（P 操作）
// 如果 count <= 0，阻塞当前线程
void sem_wait(semaphore_t* sem);

// 尝试等待（非阻塞）
// 返回 0 成功，非 0 失败
int sem_trywait(semaphore_t* sem);

// 发送信号（V 操作）
// 增加计数，唤醒等待者
void sem_post(semaphore_t* sem);

// 获取当前计数值
int sem_getvalue(semaphore_t* sem);
```

### 使用示例

#### 生产者-消费者

```c
semaphore_t empty, full, mutex;

sem_init(&empty, BUFFER_SIZE);  // 空槽位数
sem_init(&full, 0);             // 满槽位数
sem_init(&mutex, 1);            // 二值信号量作为互斥锁

// 生产者
void producer(void) {
    sem_wait(&empty);
    sem_wait(&mutex);
    buffer.push(item);
    sem_post(&mutex);
    sem_post(&full);
}

// 消费者
void consumer(void) {
    sem_wait(&full);
    sem_wait(&mutex);
    item = buffer.pop();
    sem_post(&mutex);
    sem_post(&empty);
}
```

## 条件变量 API

### 初始化

```c
void condvar_init(condvar_t* cond);
```

### 基本操作

```c
// 等待条件（自动释放并重新获取 mutex）
void condvar_wait(condvar_t* cond, mutex_t* mutex);

// 等待条件（带超时）
// 返回 0 成功，非 0 超时
int condvar_timedwait(condvar_t* cond, mutex_t* mutex,
                      uint64_t timeout_ms);

// 唤醒一个等待者
void condvar_signal(condvar_t* cond);

// 唤醒所有等待者
void condvar_broadcast(condvar_t* cond);
```

### 使用示例

```c
mutex_t mutex;
condvar_t cond;
bool condition = false;

// 等待线程
void waiter(void) {
    mutex_lock(&mutex);
    while (!condition) {
        condvar_wait(&cond, &mutex);
    }
    // 条件满足，执行操作
    mutex_unlock(&mutex);
}

// 通知线程
void notifier(void) {
    mutex_lock(&mutex);
    condition = true;
    condvar_signal(&cond);  // 或 condvar_broadcast(&cond)
    mutex_unlock(&mutex);
}
```

## 读写锁 API

### 初始化

```c
void rwlock_init(rwlock_t* lock);
```

### 读操作

```c
// 获取读锁
void rwlock_read_lock(rwlock_t* lock);

// 尝试获取读锁（非阻塞）
int rwlock_read_trylock(rwlock_t* lock);

// 释放读锁
void rwlock_read_unlock(rwlock_t* lock);
```

### 写操作

```c
// 获取写锁
void rwlock_write_lock(rwlock_t* lock);

// 尝试获取写锁（非阻塞）
int rwlock_write_trylock(rwlock_t* lock);

// 释放写锁
void rwlock_write_unlock(rwlock_t* lock);
```

### 降级操作

```c
// 将写锁降级为读锁
void rwlock_write_downgrade(rwlock_t* lock);
```

### 使用示例

```c
rwlock_t rwlock;
rwlock_init(&rwlock);

// 读者线程
void reader(void) {
    rwlock_read_lock(&rwlock);
    read_data();
    rwlock_read_unlock(&rwlock);
}

// 写者线程
void writer(void) {
    rwlock_write_lock(&rwlock);
    write_data();
    rwlock_write_unlock(&rwlock);
}
```

## 原子操作 API

### 初始化

```c
// 设置值
void atomic_set(atomic_t* v, int i);

// 读取值
int atomic_read(atomic_t* v);
```

### 算术操作

```c
// 增加
void atomic_inc(atomic_t* v);
void atomic_add(atomic_t* v, int i);

// 减少
void atomic_dec(atomic_t* v);
void atomic_sub(atomic_t* v, int i);
```

### 带返回值的操作

```c
// 增加并返回新值
int atomic_inc_return(atomic_t* v);
int atomic_add_return(atomic_t* v, int i);

// 减少并返回新值
int atomic_dec_return(atomic_t* v);
int atomic_sub_return(atomic_t* v, int i);
```

### 条件操作

```c
// 减少并测试是否为 0
// 返回 true 如果结果为 0
bool atomic_dec_and_test(atomic_t* v);

// 减少并测试是否为负
// 返回 true 如果结果 < 0
bool atomic_dec_and_negative(atomic_t* v);

// 增加并测试是否为负
bool atomic_inc_and_test(atomic_t* v);
```

### CAS 操作

```c
// 比较并交换
// 如果 *oldval == 当前值，则设置为 newval
// 返回 true 如果交换成功
bool atomic_compare_and_exchange(atomic_t* v,
                                  int* oldval,
                                  int newval);
```

### 位操作

```c
// 设置位
void atomic_set_bit(int nr, atomic_t* v);

// 清除位
void atomic_clear_bit(int nr, atomic_t* v);

// 测试并设置位
// 返回位是否被设置
bool atomic_test_and_set_bit(int nr, atomic_t* v);

// 测试并清除位
bool atomic_test_and_clear_bit(int nr, atomic_t* v);
```

### 内存屏障

```c
// 完整屏障
#define mb()  __asm__ volatile("mfence" ::: "memory")

// 读屏障
#define rmb() __asm__ volatile("lfence" ::: "memory")

// 写屏障
#define wmb() __asm__ volatile("sfence" ::: "memory")

// 编译器屏障
#define barrier() __asm__ volatile("" ::: "memory")
```

### 使用示例

```c
atomic_t counter;
atomic_set(&counter, 0);

// 多线程安全计数
atomic_inc(&counter);
int value = atomic_read(&counter);

// CAS 实现
int expected = 1;
int desired = 2;
if (atomic_compare_and_exchange(&counter, &expected, desired)) {
    // 交换成功
}
```

## 等待队列 API

### 初始化

```c
void wait_queue_init(wait_queue_head_t* q);
```

### 基本操作

```c
// 阻塞当前进程直到被唤醒
void sleep_on(wait_queue_head_t* q);

// 阻塞当前进程直到条件满足
int sleep_on_timeout(wait_queue_head_t* q, uint64_t timeout);

// 唤醒一个等待者
void wake_up(wait_queue_head_t* q);

// 唤醒所有等待者
void wake_up_all(wait_queue_head_t* q);

// 唤醒独占等待者
void wake_up_exclusive(wait_queue_head_t* q);
```

### 使用示例

```c
wait_queue_head_t wq;
wait_queue_init(&wq);

// 等待线程
void waiter(void) {
    // 阻塞直到被唤醒
    sleep_on(&wq);
    // 被唤醒后继续执行
}

// 唤醒线程
void waker(void) {
    wake_up(&wq);  // 唤醒一个
    // 或
    wake_up_all(&wq);  // 唤醒所有
}
```

## 线程 API

### 创建线程

```c
// 创建内核线程
pcb_t* proc_create_kernel_thread(thread_entry_t entry, void* arg);

// 创建用户线程
pcb_t* proc_create_user_thread(virtual_addr_t entry,
                                 virtual_addr_t stack,
                                 void* arg);
```

### 线程退出

```c
// 退出当前线程
void proc_thread_exit(void* retval);
```

### 线程等待

```c
// 等待指定线程结束
// retval: 接收线程返回值
// 返回 0 成功，非 0 失败
int proc_thread_join(int32_t pid, void** retval);
```

### 线程分离

```c
// 分离线程（使其自动回收）
// 返回 0 成功，非 0 失败
int proc_thread_detach(pcb_t* thread);
```

### 线程查询

```c
// 获取当前线程
pcb_t* proc_current(void);

// 获取线程组 ID
int32_t proc_get_tgid(pcb_t* thread);

// 查找同组线程
pcb_t* proc_find_thread_in_group(int32_t tgid, int32_t pid);
```

### 使用示例

```c
// 线程函数
void my_thread(void* arg) {
    int value = *(int*)arg;
    klog_info("Thread value: %d\n", value);
    proc_thread_exit(NULL);
}

// 主线程
void main(void) {
    int arg = 42;

    // 创建线程
    pcb_t* thread = proc_create_kernel_thread(my_thread, &arg);

    // 等待线程结束
    void* retval;
    proc_thread_join(thread->pid, &retval);

    klog_info("Thread finished\n");
}
```

## 数据结构定义

### 自旋锁

```c
typedef struct {
    volatile int locked;
} spinlock_t;
```

### 互斥锁

```c
typedef struct {
    atomic_t locked;       // 锁状态
    pcb_t* owner;          // 所有者进程
    uint32_t count;        // 递归计数
    spinlock_t wait_lock;  // 保护等待列表
    list_head wait_list;   // 等待进程列表
} mutex_t;
```

### 信号量

```c
typedef struct {
    atomic_t count;        // 资源计数
    spinlock_t lock;       // 保护状态
    list_head wait_list;   // 等待进程列表
} semaphore_t;
```

### 条件变量

```c
typedef struct {
    spinlock_t lock;       // 保护等待者
    list_head waiters;     // 等待进程列表
} condvar_t;
```

### 读写锁

```c
typedef struct {
    atomic_t readers;      // 活跃读者数
    atomic_t writers;      // 活跃写者数 (0 或 1)
    spinlock_t lock;       // 内部锁
} rwlock_t;
```

### 原子变量

```c
typedef struct {
    volatile int counter;
} atomic_t;
```

### 等待队列

```c
typedef struct wait_queue_entry {
    list_head list;
    pcb_t* task;
    process_state_t old_state;
    bool exclusive;
} wait_queue_entry_t;

typedef struct {
    list_head task_list;
    spinlock_t lock;
} wait_queue_head_t;
```

## 同步原语选择指南

| 场景 | 推荐原语 | 理由 |
|------|---------|------|
| 短临界区 (< 微秒) | spinlock | 忙等待快，避免调度开销 |
| 长临界区 | mutex | 可睡眠，不浪费 CPU |
| 资源计数 | semaphore | 天然支持计数 |
| 条件等待 | condvar + mutex | 标准模式 |
| 读多写少 | rwlock | 允许并发读 |
| 简单计数 | atomic | 无锁，最快 |
| 复杂等待 | waitqueue | 灵活可控 |
