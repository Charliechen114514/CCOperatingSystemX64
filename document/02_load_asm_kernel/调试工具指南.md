## 1. 汇编与构建

### 1.1 NASM - 汇编器

#### 基本信息

```bash
# 版本检查
nasm --version
# NASM version 2.15.05

# 基本语法
nasm [options] input.asm -o output.bin
```

#### CCOS 使用的选项

```bash
# 二进制格式 (原始字节)
nasm -f bin kernel.asm -o kernel.bin

# ELF 格式 (带调试符号)
nasm -f elf64 kernel.asm -o kernel.o
ld kernel.o -o kernel.elf

# 生成列表文件
nasm -f bin kernel.asm -o kernel.bin -l kernel.lst
```

#### 条件汇编

```asm
; 调试开关
%ifdef DEBUG
    mov si, debug_msg
    call print_string
%endif

; 编译时启用
nasm -D DEBUG kernel.asm -o kernel.bin
```

### 1.2 Make - 构建工具

#### CCOS Makefile 结构

```makefile
# 变量定义
AS = nasm
ASFLAGS = -f bin
QEMU = qemu-system-x86_64

# 目录
SRC_DIR = boot
KERNEL_DIR = kernel
BUILD_DIR = build

# 文件
BOOT_ASM = $(SRC_DIR)/boot.asm
KERNEL_ASM = $(KERNEL_DIR)/kernel.asm
KERNEL_BIN = $(BUILD_DIR)/kernel.bin

# 目标
.PHONY: all clean run debug

all: $(BOOT_IMG)

# 构建内核
$(KERNEL_BIN): $(KERNEL_ASM) | prepare
	@echo "Building Kernel..."
	$(AS) $(ASFLAGS) $< -o $@
	@ls -lh $@

# 运行
run: $(BOOT_IMG)
	$(QEMU) -drive format=raw,file=$(BOOT_IMG),if=ide -nographic
```

#### 使用 Make

```bash
# 构建全部
make

# 清理
make clean

# 运行
make run

# 调试
make debug
```

## 2. 反汇编工具

### 2.1 ndisasm - NASM 反汇编器

#### 基本用法

```bash
# 自动检测格式
ndisasm boot.bin

# 指定模式
ndisasm -b 16 boot.bin    # 16 位
ndisasm -b 32 boot2.bin   # 32 位
ndisasm -b 64 kernel.bin  # 64 位

# 指定起始地址
ndisasm -o 0x7C00 boot.bin

# 输出到文件
ndisasm -b 64 -o 0x10000 kernel.bin > kernel.disasm
```

#### CCOS 实用示例

```bash
# 反汇编 Stage 1
ndisasm -b 16 -o 0x7C00 build/boot.bin | head -20
# 00007C00  FA              cli
# 00007C01  31C0            xor ax, ax
# ...

# 查找特定模式
ndisasm -b 64 build/kernel.bin | grep "0xB8000"
# 00010000  48BF0005B8B8    mov rdi, 0xB8000+160*5

# 对比前后版本
ndisasm -b 32 boot2_old.bin > boot2_old.txt
ndisasm -b 32 boot2_new.bin > boot2_new.txt
diff boot2_old.txt boot2_new.txt
```

### 2.2 objdump - 目标文件反汇编

#### ELF 文件

```bash
# 基本反汇编
objdump -d kernel.elf

# Intel 语法
objdump -d -M intel kernel.elf

# 带符号
objdump -d -S kernel.elf

# 查看所有信息
objdump -x kernel.elf
```

#### 节信息

```bash
# 查看节表
objdump -h kernel.elf

# 输出示例:
# Sections:
# Idx Name          Size      VMA               LMA
#   0 .text         00000012  0000000000001000  0000000000001000
#   1 .data         00000000  0000000000002000  0000000000002000
```

### 2.3 输出分析

```bash
# 美化输出
ndisasm -b 64 kernel.bin | \
  awk '{printf "%08x: %s %s %s %s %s\n", $1, $2, $3, $4, $5, $6}'

# 统计指令数
ndisasm -b 64 kernel.bin | wc -l

# 查找 far jump
ndisasm -b 16 boot.bin | grep -E "(jmp|call) far"

# 查找模式切换点
ndisasm -b 32 boot2.bin | grep -E "(cr0|cr3|cr4|efr)"
```

## 3. 十六进制查看

### 3.1 xxd - 十六进制查看器

#### 基本用法

```bash
# 查看文件
xxd boot.img

# 限制行数
xxd boot.img | head -20

# 跳过部分
xxd -s 0x7C00 boot.img

# 限制长度
xxd -l 512 boot.img

# 只显示十六进制 (不含 ASCII)
xxd -p boot.img
```

#### CCOS 应用场景

```bash
# 查看 MBR 签名
xxd -s 510 -l 2 boot.img
# 000001fe: 55aa

# 查看 GDT 内容
xxd -s 0x3 -l 40 boot2.bin

# 查看内核开头
xxd -s 0x10000 -l 32 boot.img

# 验证磁盘布局
xxd boot.img | grep -E "^0000[01][0-9a-f]{2}:"
# 00000000: eb3c (boot1 开始)
# 00000200: fa (boot2 开始)
# 00000600: b8 (kernel 开始)
```

### 3.2 hexdump - 另一种查看器

```bash
# 十六进制 + ASCII 显示
hexdump -C boot.img

# 十六进制显示
hexdump -X boot.img

# 指定偏移和长度
hexdump -C -s 0x7C00 -n 512 boot.img

# 格式化输出
hexdump -e '1/4 "%08x " "\n"' boot.img
```

### 3.3 对比工具

```bash
# 直接对比
xxd boot1.bin > boot1.hex
xxd boot2.bin > boot2.hex
diff boot1.hex boot2.hex

# 可视化对比
vimdiff boot1.hex boot2.hex

# 使用 cmp
cmp boot1.bin boot2.bin
# boot1.bin boot2.bin differ: byte 3, line 1

# 使用 diff
diff -y boot1.hex boot2.hex --suppress-common-lines
```

## 4. 调试器

### 4.1 GDB - GNU 调试器

#### 启动和连接

```bash
# 启动 QEMU 调试模式
qemu-system-x86_64 -drive format=raw,file=boot.img,if=ide -nographic -s -S

# 连接 GDB (另一个终端)
gdb
(gdb) target remote :1234
```

#### 基本命令

```gdb
# 加载符号
file build/kernel.bin

# 设置断点
b *0x7E00          # 地址断点
b *0x10000         # 内核入口

# 运行和单步
c                 # continue
si                # step instruction
ni                # next instruction

# 查看代码
x/10i $pc         # 反汇编当前 PC
x/10i 0x7E00      # 反汇编指定地址
disas 0x7E00 0x7E20
```

#### 查看寄存器

```gdb
# 所有寄存器
info registers
info all-registers

# 特定寄存器
print $rax
print $eflags

# 控制寄存器
print $cr0
print $cr3
print $cr4

# 段寄存器
print $cs
print $ds
print $ss
```

#### 查看内存

```gdb
# 十六进制查看
x/10x 0xB8000      # 10 个 64 位值
x/10wx 0x9000      # 10 个 32 位值
x/10hw 0x7E00      # 10 个 16 位值

# 反汇编查看
x/10i 0x7E00

# 字符串查看
x/s 0x7E00

# 持续查看
x/20gx 0xB8000     # 20 个 8 字节值
```

#### GDB 脚本

```gdb
# gdb.txt
target remote :1234
set architecture i386:x86-64
b *0x7E00
b *0x10000
c
```

```bash
# 执行脚本
gdb -x gdb.txt
```

### 4.2 条件断点和观察点

```gdb
# 条件断点
b *0x7E00 if $rax == 0x1234

# 观察点
watch *0xB8000     # 内存变化时停止
rwatch *0xB8000    # 读时停止
awatch *0xB8000    # 读写时停止

# 忽略 N 次
b *0x7E00
ignore 1 100       # 忽略前 100 次
```

### 4.3 修改状态

```gdb
# 修改寄存器
set $rax = 0x10000
set $eflags = 0x202

# 修改内存
set {uint16_t}0xB8000 = 0x1F58
set *(uint16_t*)0xB8000 = 0x1F58

# 写入汇编代码
assemble 0x7E00
cli
nop
nop
end
```

## 5. 模拟器

### 5.1 QEMU 基础

#### 启动选项

```bash
# 基本启动
qemu-system-x86_64 -drive format=raw,file=boot.img,if=ide

# 无输出模式
qemu-system-x86_64 -drive format=raw,file=boot.img,if=ide -nographic

# 调试模式
qemu-system-x86_64 -drive format=raw,file=boot.img,if=ide -nographic -s -S

# 串口输出
qemu-system-x86_64 -drive format=raw,file=boot.img,if=ide -serial stdio
```

#### CCOS Makefile 目标

```makefile
run: $(BOOT_IMG)
	@echo "Running with QEMU..."
	$(QEMU) -drive format=raw,file=$(BOOT_IMG),if=ide -nographic

debug: $(BOOT_IMG)
	@echo "Running with QEMU (debug)..."
	$(QEMU) -drive format=raw,file=$(BOOT_IMG),if=ide -nographic -s -S
```

### 5.2 QEMU 监控

#### 访问监控

```
# 在 QEMU 窗口按 Ctrl+Alt+2 (图形模式)
# 或使用 -monitor stdio (命令行)

(qemu) info registers
(qemu) info mem
(qemu) x/10i 0x7E00
(qemu) xp/10x 0xB8000
```

#### 有用的监控命令

```
# 寄存器信息
info registers
info cpustats
info lapic

# 内存信息
info mem
info tlb
info mtree

# 执行控制
c              # 继续
s              # 单步
stop           # 暂停
quit           # 退出
```

### 5.3 QEMU 日志

#### 调试选项

```bash
# 完整调试输出
qemu-system-x86_64 \
  -d in_asm,int,cpu_reset,guest_errors \
  -D debug.log \
  -drive format=raw,file=boot.img,if=ide

# 特定组件
qemu-system-x86_64 \
  -d trace:crs \        # 控制寄存器
  -d trace:mmu \        # 内存管理单元
  -d trace:port \       # 端口 I/O
  -drive format=raw,file=boot.img,if=ide
```

#### 日志分析

```bash
# 查看指令执行
cat debug.log | grep "IN:"
cat debug.log | grep "0x0000000000007E00:"

# 查找异常
cat debug.log | grep "Exception"
cat debug.log | grep "Triple Fault"

# 统计
cat debug.log | wc -l
cat debug.log | grep "jmp" | wc -l
```

## 6. 其他工具

### 6.1 文件操作

#### dd - 磁盘操作

```bash
# 提取特定扇区
dd if=boot.img bs=512 skip=3 count=1 of=kernel_sector.bin

# 合并文件
cat boot1.bin boot2.bin > combined.bin

# 填充到指定大小
dd if=boot.bin of=boot_padded.bin bs=512 count=1 conv=sync

# 转换大小写
dd if=boot.img of=boot.img conv=ucase
```

#### 文件信息

```bash
# 文件大小
ls -lh boot.img

# 扇区大小
stat boot.img

# 文件类型
file boot.img
file boot.bin
```

### 6.2 文本处理

#### grep - 搜索模式

```bash
# 搜索特定指令
ndisasm boot.bin | grep "jmp"

# 搜索地址
ndisasm boot2.bin | grep "7E00"

# 反向搜索
ndisasm boot.bin | grep -B2 -A2 "hlt"

# 计数
ndisasm boot.bin | grep "int" | wc -l
```

#### awk - 文本处理

```bash
# 提取地址
ndisasm boot.bin | awk '{print $1}'

# 格式化输出
ndisasm boot.bin | awk '{printf "%-20s %s\n", $1, $2}'

# 筛选特定范围
ndisasm boot.bin | awk '$1 >= /^00007c00/ && $1 <= /^00007d00/'
```

### 6.3 二进制编辑

#### printf - 生成二进制

```bash
# 生成简单文件
printf "\x55\xAA" > mbr_sig.bin

# 生成模式
printf '\x90%.0s' {1..510} > padding.bin
printf '\x55\xAA' >> mbr.bin
```

#### dd + printf - 创建文件

```bash
# 创建 512 字节的空文件
dd if=/dev/zero of=empty.img bs=512 count=1

# 写入 MBR 签名
printf '\x55\xAA' | dd of=empty.img bs=1 seek=510 conv=notrunc
```

## 7. 工作流示例

### 7.1 完整调试会话

```bash
# 1. 编译
make clean && make

# 2. 检查输出大小
ls -lh build/*.bin

# 3. 反汇编验证
ndisasm -b 64 build/kernel.bin | head

# 4. 启动调试
make debug

# 5. 连接 GDB
gdb -x gdb_script.txt

# 6. 分析问题
# (在 GDB 中单步执行)

# 7. 修复问题
vim kernel.asm

# 8. 重新测试
make && make run
```

### 7.2 自动化测试脚本

```bash
#!/bin/bash
# test.sh

set -e

echo "Building..."
make clean && make

echo "Checking sizes..."
[ -f build/boot.bin ] && [ $(stat -c %s build/boot.bin) -eq 512 ]
[ -f build/kernel.bin ]

echo "Disassembling..."
ndisasm -b 64 -o 0x10000 build/kernel.bin | head -5

echo "Running test..."
timeout 5 make run || true

echo "Done!"
```

### 7.3 性能分析

```bash
# QEMU 内置分析
qemu-system-x86_64 \
  -d in_asm \
  -d cpu \
  -D profile.log \
  -drive format=raw,file=boot.img,if=ide

# 分析执行路径
cat profile.log | grep "IN:" | wc -l

# 热点分析
cat profile.log | grep "IN:" | sort | uniq -c | sort -rn | head -10
```

| 任务 | 命令 |
