## 1. GDB 调试

### 1.1 启动 GDB 调试

#### 基本命令

```bash
# 终端 1: 启动 QEMU 并等待 GDB 连接
qemu-system-x86_64 -kernel kernel.elf -s -S

# 终端 2: 启动 GDB
gdb kernel.elf

# 在 GDB 中连接到 QEMU
(gdb) target remote :1234
(gdb) continue
```

#### VSCode 调试配置

```json
{
    "version": "0.2.0",
    "configurations": [
        {
            "type": "cppdbg",
            "request": "launch",
            "name": "Debug CCOS",
            "program": "${workspaceFolder}/build/kernel/kernel.elf",
            "miDebuggerServerAddress": "localhost:1234",
            "miDebuggerPath": "/usr/bin/gdb",
            "MIMode": "gdb",
            "setupCommands": [
                {
                    "description": "Enable pretty-printing",
                    "text": "-enable-pretty-printing",
                    "ignoreFailures": true
                }
            ]
        }
    ]
}
```

### 1.2 断点设置

#### 函数断点

```gdb
# 在 klog_log 函数设置断点
(gdb) break klog_log

# 在 format_string 函数设置断点
(gdb) break format_string
```

#### 条件断点

```gdb
# 只在特定级别时中断
(gdb) break klog_log if level == 4

# 只在特定字符串时中断
(gdb) break format_string if *p == '%'
```

#### 查看断点

```gdb
# 列出所有断点
(gdb) info breakpoints

# 删除断点
(gdb) delete 1

# 禁用/启用断点
(gdb) disable 1
(gdb) enable 1
```

### 1.3 变量检查

#### 查看局部变量

```gdb
# 在 klog_log 函数中
(gdb) print level
(gdb) print format

# 查看缓冲区内容
(gdb) print buffer
(gdb) print buffer[0]@100  # 打印前 100 个字符
```

#### 查看静态变量

```gdb
# 查看全局日志级别
(gdb) print g_log_level

# 查看后端操作表
(gdb) print g_backends[0]

# 检查后端是否为空
(gdb) print g_backends[0] == 0
```

#### 查看内存

```gdb
# 查看指定地址的内存
(gdb) x/10s 0x7c00

# 查看格式化缓冲区
(gdb) x/512s buffer

# 十六进制查看
(gdb) x/32xw buffer
```

### 1.4 调用栈

```gdb
# 查看调用栈
(gdb) backtrace

# 查看特定帧
(gdb) frame 2

# 切换帧
(gdb) up
(gdb) down
```

### 1.5 单步执行

```gdb
# 单步执行 (进入函数)
(gdb) step

# 单步执行 (跳过函数)
(gdb) next

# 继续执行
(gdb) continue

# 执行到当前函数返回
(gdb) finish
```

## 2. 串口监控工具

### 2.1 picocom

#### 安装

```bash
# Ubuntu/Debian
sudo apt-get install picocom

# Fedora/RHEL
sudo dnf install picocom
```

#### 使用

```bash
# 基本连接 (115200 波特率)
picocom -b 115200 /dev/ttyS0

# 指定行设置
picocom -b 115200 --flow h /dev/ttyS0

# 添加回显
picocom -b 115200 --echo /dev/ttyS0

# 退出: Ctrl+A, Ctrl+X
```

#### 有用选项

| 选项 | 说明 |
|------|------|
| `-b <rate>` | 波特率 (默认 9600) |
| `--flow h` | 硬件流控 |
| `--flow n` | 无流控 |
| `--echo` | 本地回显 |
| `--noinit` | 不初始化端口 |

### 2.2 minicom

#### 安装

```bash
# Ubuntu/Debian
sudo apt-get install minicom

# Fedora/RHEL
sudo dnf install minicom
```

#### 配置

```bash
# 进入配置模式
sudo minicom -s

# 设置:
# - Serial port: /dev/ttyS0
# - Baud rate: 115200
# - Data bits: 8
# - Parity: None
# - Stop bits: 1

# 保存为默认
```

#### 使用

```bash
# 直接启动
minicom

# 使用指定配置
minicom -o serial

# 退出: Ctrl+A, Q
```

### 2.3 screen

```bash
# 连接到串口
screen /dev/ttyS0 115200

# 断开: Ctrl+A, K (确认)
# 退出: Ctrl+A, \
```

### 2.4 Windows 工具

#### PuTTY

1. 打开 PuTTY
2. 选择 "Serial" 连接类型
3. 设置 Serial line: COM1
4. 设置 Speed: 115200
5. 点击 "Open"

#### Tera Term

1. 文件 → 新建连接
2. 选择 "Serial"
3. 设置端口: COM1
4. 设置波特率: 115200
5. 点击 "OK"

## 4. 内存分析

### 4.1 检查缓冲区溢出

#### 使用金丝雀值

```c
// 在 kprintf.c 中添加
#define CANARY 0xDEADBEEF

static char buffer[KPRINTF_BUFFER_SIZE];

void check_buffer(void) {
    uint32_t* canary = (uint32_t*)(buffer + KPRINTF_BUFFER_SIZE - 4);
    if (*canary != CANARY) {
        klog_error("Buffer overflow detected!\n");
        while(1) { /* halt */ }
    }
}

// 初始化
void init_buffer(void) {
    uint32_t* canary = (uint32_t*)(buffer + KPRINTF_BUFFER_SIZE - 4);
    *canary = CANARY;
}
```

## 5. 性能分析

### 5.1 使用 RDTSC

```c
// 定义 RDTSC 宏
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ __volatile__ (
        "rdtsc"
        : "=a"(lo), "=d"(hi)
    );
    return ((uint64_t)hi << 32) | lo;
}

// 使用示例
uint64_t start = rdtsc();
klog_info("Test message\n");
uint64_t end = rdtsc();
uint64_t cycles = end - start;

klog_info("Cycles: %lu\n", cycles);
```

### 5.2 性能测试

```c
// 测试格式化性能
void benchmark_format(void) {
    const int ITERATIONS = 1000;
    uint64_t start, end;

    start = rdtsc();
    for (int i = 0; i < ITERATIONS; i++) {
        char buf[64];
        itoa(i, buf, 10);
    }
    end = rdtsc();

    uint64_t total = end - start;
    uint64_t avg = total / ITERATIONS;

    klog_info("itoa average: %lu cycles\n", avg);
}
```

## 6. QEMU 调试


```qemu
# 查看信息
info mem
info cpus
info registers

# 暂停/继续
stop
cont

# 单步执行
step

# 查看虚拟内存
x /10xg 0x7c00

# 设置断点
pb *0x7c00
```

## 7. 日志分析

### 7.1 日志过滤

```bash
# 只显示 ERROR 级别
grep "ERROR" serial.log

# 统计各级别数量
grep "TRACE" serial.log | wc -l
grep "DEBUG" serial.log | wc -l
grep "INFO" serial.log | wc -l
grep "WARN" serial.log | wc -l
grep "ERROR" serial.log | wc -l
```

### 7.2 日志时间戳

```c
// 添加时间戳支持
uint64_t get_timestamp(void) {
    return rdtsc();
}

void klog_log_ts(klog_level_t level, const char* format, ...) {
    uint64_t ts = get_timestamp();
    klog_log(level, "[%lu] ", ts);
    // ... 原有代码
}
```

## 8. 常见调试场景

### 8.1 格式化输出错误

```gdb
# 在 format_string 设置断点
break format_string

# 运行到格式化时检查
run

# 检查参数
print format
print args

# 单步执行
next

# 检查缓冲区
print buffer
x/10s buffer
```

### 8.2 后端未输出

```gdb
# 在 klog_serial_backend_process 设置断点
break klog_serial_backend_process

# 运行
run

# 检查是否到达断点
# 如果未到达，检查后端注册

# 检查后端操作表
print g_backends[KLOG_BACKEND_SERIAL]
```

### 8.3 日志级别过滤问题

```gdb
# 在 klog_log 函数开头设置断点
break klog_log

# 检查级别比较
print level
print g_log_level

# 检查是否被过滤
print level < g_log_level
```

## 9. 工作流建议

### 9.1 调试流程

```
1. 复现问题
   ↓
2. 添加日志 (klog_trace/debug/info)
   ↓
3. 使用 GDB 设置断点
   ↓
4. 单步执行，检查变量
   ↓
5. 定位问题根因
   ↓
6. 修复代码
   ↓
7. 验证修复
```

### 9.2 日志策略

```
开发阶段:
  klog_set_level(KLOG_LEVEL_TRACE)

测试阶段:
  klog_set_level(KLOG_LEVEL_DEBUG)

生产阶段:
  klog_set_level(KLOG_LEVEL_INFO)
```
