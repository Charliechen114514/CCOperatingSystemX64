## QEMU 调试

### 1. QEMU Monitor 命令

#### 查看内存信息

```bash
# 在 QEMU Monitor 中 (Ctrl+Alt+2 或 -monitor stdio)

# 查看虚拟内存映射
info mem

# 查看物理内存
xp/512x 0x1000      # 查看物理地址 0x1000 的内容
xp/16gx 0xB8000     # 查看 VGA 缓冲区

# 查看页表
info tlb            # 查看 TLB 内容（有限）
```

#### 查看寄存器

```bash
# 查看所有寄存器
info registers

# 查看特定寄存器
info registers cr3
info registers cr2

# 手动检查 CR3
print $cr3
```

#### 断点和执行

```bash
# 继续执行
continue

# 单步执行
stepi             # 指令级单步
nexti             # 指令级单步（跳过函数调用）

# 查看当前位置
x/10i $pc         # 查看当前指令
```

### 2. QEMU 启动选项

#### 调试模式

```bash
# 启用 GDB 服务器
qemu-system-x86_64 -kernel build/kernel.bin -s -S

# -s: GDB server on localhost:1234
# -S: 暂停启动，等待 GDB 连接

# 增加内存
qemu-system-x86_64 -kernel build/kernel.bin -m 2G

# 启用串口日志
qemu-system-x86_64 -kernel build/kernel.bin -serial stdio

# Monitor 在 stdio
qemu-system-x86_64 -kernel build/kernel.bin -monitor stdio
```

## GDB 调试

### 1. 基础 GDB 配置

创建 `.gdbinit` 文件：

```gdb
# .gdbinit
target remote localhost:1234
set architecture i386:x86-64
set disassembly-flavor intel

# 加载符号
file build/kernel.elf

# 常用断点
b pf_handler
b page_map_page
b page_init
b vmm_init

# 显示设置
set pagination off
```

### 2. 页表相关 GDB 命令

#### 检查 CR3 和页表

```gdb
# 查看 CR3 (PML4 物理地址)
print $cr3

# 计算 PML4 虚拟地址
print $cr3 & 0x000FFFFFFFFFF000

# 假设直接映射在 0xFFFF800000000000
print 0xFFFF800000000000 + ($cr3 & 0x000FFFFFFFFFF000)

# 查看前 16 个 PML4 项
x/16gx 0xFFFF800000009000
```

#### 检查页表项格式

```gdb
# 定义页表项格式解析宏
define pte_flags
    if $arg0 & 1
        printf "P "
    end
    if $arg0 & 2
        printf "W "
    end
    if $arg0 & 4
        printf "U "
    end
    if $arg0 & 0x80
        printf "PS "
    end
    if $arg0 & 0x8000000000000000
        printf "NX "
    end
    printf "\n"
end

# 使用
print $rax
pte_flags($rax)
```

#### 查看完整页表路径

```gdb
# 定义查找页表路径宏
define walk_page_table
    set $vaddr = $arg0
    set $pml4 = 0xFFFF800000009000

    # PML4 索引
    set $pml4_idx = ($vaddr >> 39) & 0x1FF
    printf "PML4[%d]: ", $pml4_idx
    x/1gx $pml4 + $pml4_idx * 8

    # 假设非零，继续...
end
```

### 3. 页错误调试

```gdb
# 在页错误处理器设置断点
b pf_handler

# 当断点触发时，查看参数
frame
info registers

# 查看 CR2 (故障地址)
print $cr2
x/1i $rip

# 解析错误码
define parse_pf_error
    if $arg0 & 1
        printf "Protection fault\n"
    else
        printf "Page not present\n"
    end
    if $arg0 & 2
        printf "Write operation\n"
    else
        printf "Read operation\n"
    end
    if $arg0 & 4
        printf "User mode\n"
    else
        printf "Kernel mode\n"
    end
end

# 使用
parse_pf_error($rbx)  # 假设 error_code 在 rbx
```

### 4. 变量检查

```gdb

print s_direct_map_established
print s_initialized

# 查看统计信息
print s_stats
print s_pf_stats

# 查看内存区域

print s_region_count
```

## 内核调试工具

### 1. 页表转储工具

内核提供了页表转储函数：

```c
// 转储单个地址的完整页表路径
page_dump_mapping(0xFFFFFFFF80000000, 4);

// 输出示例：
// [PML4] Index: 511 (0x1FF)
//        Entry: 0x00000000000A0063
//        PDPT Phys: 0xA000
//        Flags: P W
// [PDPT] Index: 510 (0x1FE)
//        Entry: 0x00000000000B0063
//        PD Phys: 0xB000
//        Type: 2MB Huge Page
//        Mapped: 0xFFFFFFFF80000000 -> 0x0
```

### 2. PML4 摘要转储

```c
// 转储 PML4 的所有非空项
page_dump_pml4(page_get_pml4());

// 输出示例：
// === PML4 Dump (Phys: 0x9000) ===
// Total PML4 entries present: 3/512
// [PML4[  0] -> PDPT at 0xA000 (Low Memory)
// [PML4[256] -> PDPT at 0xC000 (Direct Map)
// [PML4[511] -> PDPT at 0xB000 (Kernel High Half)
```

### 3. VMM 转储工具

```c
// 转储 VMM 状态
vmm_dump();

// 输出示例：
// [VMM] Virtual Memory Manager State:
// [VMM]   Kernel PML4:     0x9000
// [VMM]   Total pages:     16
// [VMM]   Mapped pages:    16
// [VMM]   Kernel pages:    16
// [VMM]   User pages:      0
// [VMM]   Tracked regions: 6
```

### 4. 内存映射转储

```c
// 转储虚拟-物理映射
vmm_dump_memory_map(0, 64);

// 输出示例：
// ========================================
//     Virtual to Physical Memory Map
// ========================================
// PML4: 0x9000
// Found 48 mapping entries:
//
// Virtual Address              | Physical    | Type   | Flags
// ----------------------------|-------------|--------|-------
// 0x0000000000000000 | 0x00000000 | 2MB    | PW
// 0x0000000000200000 | 0x00200000 | 2MB    | PW
// 0xFFFF800000000000 | 0x00000000 | 2MB    | PW
// 0xFFFFFFFF80000000 | 0x00000000 | 2MB    | PW
// ========================================
```

### 5. 查询 API

```c
// 查询特定地址的映射
page_query_result_t result;
if (page_query(page_get_pml4(), vaddr, &result) == PAGE_OK) {
    klog_trace("Present: %s\n", result.present ? "yes" : "no");
    klog_trace("Physical: 0x%X\n", result.phys_addr);
    klog_trace("Flags: 0x%llX\n", result.flags);
    klog_trace("Huge: %s\n", result.is_huge ? "yes" : "no");
}

// 虚拟地址转物理地址
physical_addr_t paddr;
if (page_virt_to_phys(page_get_pml4(), vaddr, &paddr) == PAGE_OK) {
    klog_trace("0x%llX -> 0x%X\n", vaddr, paddr);
}
```

## 页表分析

### 1. 手动页表遍历

```c
// 手动遍历页表（调试用）
void debug_walk_page_table(virtual_addr_t vaddr) {
    physical_addr_t pml4_phys = page_get_pml4();
    pml4_t* pml4 = (pml4_t*)phys_to_virt(pml4_phys);

    // PML4
    page_table_entry_t* pml4e = &pml4->entries[PML4_INDEX(vaddr)];
    klog_trace("[PML4] Entry: 0x%016llX\n", pml4e->value);
    if (!pml4e->bits.present) return;

    // PDPT
    pdpt_t* pdpt = (pdpt_t*)phys_to_virt(pml4e->bits.frame << PAGE_SHIFT);
    page_table_entry_t* pdpte = &pdpt->entries[PDPT_INDEX(vaddr)];
    klog_trace("[PDPT] Entry: 0x%016llX\n", pdpte->value);
    if (!pdpte->bits.present) return;

    // 检查 1GB 巨大页
    if (pdpte->bits.pat) {
        klog_trace("[PDPT] 1GB Huge Page\n");
        return;
    }

    // PD
    pd_t* pd = (pd_t*)phys_to_virt(pdpte->bits.frame << PAGE_SHIFT);
    page_table_entry_t* pde = &pd->entries[PD_INDEX(vaddr)];
    klog_trace("[PD] Entry: 0x%016llX\n", pde->value);
    if (!pde->bits.present) return;

    // 检查 2MB 巨大页
    if (pde->bits.pat) {
        klog_trace("[PD] 2MB Huge Page\n");
        return;
    }

    // PT
    pt_t* pt = (pt_t*)phys_to_virt(pde->bits.frame << PAGE_SHIFT);
    page_table_entry_t* pte = &pt->entries[PT_INDEX(vaddr)];
    klog_trace("[PT] Entry: 0x%016llX\n", pte->value);

    klog_trace("[PT] Mapped: 0x%llX -> 0x%X\n",
              vaddr, pte->bits.frame << PAGE_SHIFT);
}
```

### 2. 验证映射一致性

```c
// 验证虚拟地址到物理地址再转换回来
void verify_mapping(virtual_addr_t vaddr) {
    physical_addr_t paddr;

    if (page_virt_to_phys(page_get_pml4(), vaddr, &paddr) == PAGE_OK) {
        // 写测试模式
        volatile uint64_t* vptr = (volatile uint64_t*)vaddr;
        uint64_t test_pattern = 0xDEADBEEFCAFEBABEULL;

        vptr[0] = test_pattern;

        // 通过直接映射验证
        volatile uint64_t* direct_ptr = (volatile uint64_t*)phys_to_virt(paddr);
        if (direct_ptr[0] == test_pattern) {
            klog_trace("Mapping verified: 0x%llX <-> 0x%X\n", vaddr, paddr);
        } else {
            klog_error("Mapping mismatch!\n");
        }
    }
}
```

## 性能分析

### 1. TLB 性能

```c
// 测试页表遍历性能
void benchmark_page_walk(virtual_addr_t vaddr) {
    uint64_t start, end;
    physical_addr_t paddr;

    // 使用 RDTSC 读取时间戳
    __asm__ volatile("rdtsc" : "=A"(start));

    for (int i = 0; i < 1000; i++) {
        page_virt_to_phys(page_get_pml4(), vaddr + i * PAGE_SIZE, &paddr);
    }

    __asm__ volatile("rdtsc" : "=A"(end));

    klog_trace("1000 translations: %llu cycles\n", end - start);
}
```

### 2. 分配性能

```c
// 测试页分配性能
void benchmark_allocation(void) {
    uint64_t start, end;
    virtual_addr_t pages[100];

    __asm__ volatile("rdtsc" : "=A"(start));

    for (int i = 0; i < 100; i++) {
        pages[i] = vmm_alloc_pages(1, VMAP_FLAG_WRITE);
    }

    __asm__ volatile("rdtsc" : "=A"(end));

    klog_trace("100 allocations: %llu cycles\n", end - start);

    // 清理
    for (int i = 0; i < 100; i++) {
        if (pages[i] != 0) {
            vmm_free_pages(pages[i], 1);
        }
    }
}
```

## 常见场景

### 场景 1: 调试页错误

```
1. 在 pf_handler 设置断点
2. 触发页错误
3. 查看寄存器状态
4. 解析 CR2 和错误码
5. 检查页表路径
6. 确定错误原因
```

GDB 命令序列：
```gdb
b pf_handler
continue
# 断点触发后
info registers
print $cr2
parse_pf_error($r9)  # 假设 error_code 在 r9
page_dump_mapping($cr2, 4)
```

### 场景 2: 验证映射建立

```c
// 1. 建立映射
virtual_addr_t vaddr = vmm_alloc_pages(1, VMAP_FLAG_WRITE);

// 2. 验证映射
page_query_result_t result;
page_query(page_get_pml4(), vaddr, &result);

// 3. 写入测试
volatile uint64_t* ptr = (volatile uint64_t*)vaddr;
ptr[0] = 0x123456789ABCDEF0ULL;

// 4. 通过直接映射验证
physical_addr_t paddr;
page_virt_to_phys(page_get_pml4(), vaddr, &paddr);
volatile uint64_t* direct = (volatile uint64_t*)phys_to_virt(paddr);
assert(direct[0] == 0x123456789ABCDEF0ULL);
```

### 场景 3: 内存泄漏检测

```c
// 1. 记录初始状态
vmm_stats_t stats_before;
vmm_get_stats(&stats_before);

// 2. 执行操作
some_operation();

// 3. 记录结束状态
vmm_stats_t stats_after;
vmm_get_stats(&stats_after);

// 4. 检查差异
if (stats_after.mapped_pages != stats_before.mapped_pages) {
    klog_warn("Possible leak: %llu pages allocated\n",
             stats_after.mapped_pages - stats_before.mapped_pages);
}
```

### 场景 4: 调试用户空间映射

```c
// 1. 创建用户地址空间
physical_addr_t user_pml4;
vmm_create_user_space(&user_pml4);

// 2. 映射页到用户空间
physical_addr_t paddr;
pframe_alloc(&paddr);
vmm_map_to_user(user_pml4, 0x400000, paddr, 1,
               VMAP_FLAG_WRITE | VMAP_FLAG_USER);

// 3. 验证映射（从内核 PML4 查询用户 PML4）
page_query_result_t result;
page_query(user_pml4, 0x400000, &result);
assert(result.present == true);
assert((result.flags & PAGE_USER) != 0);
```
