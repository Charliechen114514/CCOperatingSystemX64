## 1. QEMU 磁盘调试

### 1.1 添加磁盘镜像

#### 方法 1: 命令行

```bash
# 创建测试镜像
dd if=/dev/zero of=disk.img bs=1M count=128

# Primary Master (index=0)
qemu-system-x86_64 \
    -drive file=disk.img,format=raw,index=0,media=disk \
    -kernel kernel.elf \
    ...

# 多个磁盘
qemu-system-x86_64 \
    -drive file=disk1.img,format=raw,index=0,media=disk \
    -drive file=disk2.img,format=raw,index=1,media=disk \
    ...
```

#### 方法 2: 使用构建脚本

修改 `scripts/build_helpers/build_normal_run.sh`:

```bash
QEMU="qemu-system-x86_64"
KERNEL="build/kernel/kernel.elf"

$QEMU \
    -drive file=disk.img,format=raw,index=0,media=disk \
    -kernel $KERNEL \
    -serial stdio \
    -m 128M
```

## 4. ATA 端口监控

### 4.1 状态寄存器监控

```c
// 实时监控状态变化
void monitor_status(uint16_t io_base) {
    uint8_t last_status = 0;
    uint8_t status;

    while (1) {
        status = inb(io_base + ATA_REG_STATUS);

        if (status != last_status) {
            klog_info("Status: 0x%02X ", status);

            if (status & ATA_STATUS_BSY) klog_trace("[BSY] ");
            if (status & ATA_STATUS_DRDY) klog_trace("[DRDY] ");
            if (status & ATA_STATUS_DRQ) klog_trace("[DRQ] ");
            if (status & ATA_STATUS_ERR) klog_trace("[ERR] ");

            klog_trace("\n");
            last_status = status;
        }

        ata_delay();
    }
}
```

### 4.2 完整寄存器转储

```c
void dump_all_registers(uint16_t io_base) {
    klog_info("=== ATA Register Dump @ 0x%04X ===\n", io_base);

    // 读取命令块寄存器
    uint8_t data_lo = inb(io_base + ATA_REG_DATA);
    uint8_t data_hi = inb(io_base + ATA_REG_DATA + 1);
    uint8_t error = inb(io_base + ATA_REG_ERROR);
    uint8_t seccount = inb(io_base + ATA_REG_SECCOUNT);
    uint8_t lba_lo = inb(io_base + ATA_REG_LBA_LO);
    uint8_t lba_mid = inb(io_base + ATA_REG_LBA_MID);
    uint8_t lba_hi = inb(io_base + ATA_REG_LBA_HI);
    uint8_t device = inb(io_base + ATA_REG_DEVICE);
    uint8_t status = inb(io_base + ATA_REG_STATUS);

    klog_info("Data:    0x%04X\n", (data_hi << 8) | data_lo);
    klog_info("Error:   0x%02X\n", error);
    klog_info("SecCnt:  0x%02X\n", seccount);
    klog_info("LBA:     0x%02X 0x%02X 0x%02X (0x%02X)\n",
              lba_lo, lba_mid, lba_hi, device);
    klog_info("Status:  0x%02X\n", status);

    // 解析状态位
    klog_info("  BSY=%d DRDY=%d DWF=%d DSC=%d DRQ=%d CORR=%d IDX=%d ERR=%d\n",
              !!(status & 0x80), !!(status & 0x40), !!(status & 0x20),
              !!(status & 0x10), !!(status & 0x08), !!(status & 0x04),
              !!(status & 0x02), !!(status & 0x01));
}
```

## 6. 调试脚本

### 6.1 QEMU 启动脚本

```bash
#!/bin/bash
# debug_ata_qemu.sh

QEMU="qemu-system-x86_64"
KERNEL="build/kernel/kernel.elf"
DISK="disk.img"

# 检查磁盘镜像
if [ ! -f "$DISK" ]; then
    echo "Creating test disk image..."
    dd if=/dev/zero of="$DISK" bs=1M count=128
fi

$QEMU \
    -kernel $KERNEL \
    -drive file="$DISK",format=raw,index=0,media=disk \
    -monitor telnet::4444,server,nowait \
    -serial stdio \
    -m 128M \
    -d int,cpu_reset \
    2>&1 | tee ata_debug.log
```

### 6.2 GDB 自动化脚本

```gdb
# debug_ata.gdb
target remote :1234

# 设置断点
break ata_device_present
break ata_identify_device
break ata_read_pio
break ata_write_pio

# 定义宏
define dump_ata_controller
    set $ctrl = $arg0
    printf "Controller @ %p:\n", $ctrl
    printf "  IO Base:    0x%04X\n", $ctrl->io_base
    printf "  IRQ:        %d\n", $ctrl->irq
    printf "  Master:     %s\n", $ctrl->master_present ? "Present" : "Absent"
    printf "  Slave:      %s\n", $ctrl->slave_present ? "Present" : "Absent"
    printf "  Read Cnt:   %llu\n", $ctrl->read_count
    printf "  Write Cnt:  %llu\n", $ctrl->write_count
    printf "  Errors:     %llu\n", $ctrl->error_count
end

define hexdump_buffer
    set $buf = $arg0
    set $len = $arg1
    printf "Buffer @ %p (%d bytes):\n", $buf, $len
    set $i = 0
    while $i < $len
        printf "%04X: ", $i
        set $j = 0
        while $j < 16 && $i + $j < $len
            printf "%02X ", *(uint8_t*)($buf + $i + $j)
            set $j = $j + 1
        end
        printf "\n"
        set $i = $i + 16
    end
end

# 继续执行
continue
```

使用方式：
```bash
gdb -x debug_ata.gdb build/kernel/kernel.elf
```

### 6.3 Python 辅助脚本

```python
#!/usr/bin/env python3
"""
parse_ata_log.py - 解析 ATA 驱动日志
"""

import sys
import re

def parse_ata_log(log_file):
    """解析 ATA 驱动日志"""

    stats = {
        'reads': 0,
        'writes': 0,
        'errors': 0,
        'devices_detected': 0
    }

    with open(log_file, 'r') as f:
        for line in f:
            # 统计设备检测
            if 'Present' in line and 'ATA' in line:
                stats['devices_detected'] += 1

            # 统计读取
            if 'Successfully read' in line:
                stats['reads'] += 1

            # 统计写入
            if 'Successfully write' in line:
                stats['writes'] += 1

            # 统计错误
            if 'ATA ERROR' in line:
                stats['errors'] += 1

    # 打印统计
    print("=== ATA Driver Statistics ===")
    print(f"Devices detected: {stats['devices_detected']}")
    print(f"Read operations:  {stats['reads']}")
    print(f"Write operations: {stats['writes']}")
    print(f"Errors:           {stats['errors']}")

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: parse_ata_log.py <ata_debug.log>")
        sys.exit(1)

    parse_ata_log(sys.argv[1])
```

```

   ↓

```
