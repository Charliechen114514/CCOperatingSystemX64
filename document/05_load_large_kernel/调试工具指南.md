## 1. 磁盘布局验证工具

### 1.1 verify_disk_layout.py

#### 基本信息

- **路径**: [scripts/verify_disk_layout.py](../../../scripts/verify_disk_layout.py)
- **功能**: 验证磁盘和内存布局，检测重叠
- **依赖**: Python 3.6+

#### 基本用法

```bash
# 验证构建产物
python3 scripts/verify_disk_layout.py \
    build/bootloader.bin \
    build/kernel.bin

# 输出示例
CCOS Disk Layout Verification
============================================================
Analyzing bootloader...
  Size: 1398 bytes (3 sectors)
  Has MBR: True
  Valid MBR signature

Analyzing kernel...
  Size: 512 bytes (1 sectors, 0.00 MB)

============================================================
DISK LAYOUT (LBA)
============================================================
Bootloader:  Sector 1-3 (LBA 0-2)
Kernel:      Sector 4-4 (LBA 3-3)

============================================================
MEMORY LAYOUT
============================================================
Stage 1:     0x7C00 - 0x7E00
Stage 2:     0x7E00 - 0x8400
Kernel:      0x10000 - 0x10200

Stage 2 size: 1536 bytes (3 sectors)
Memory gap:  30720 bytes (60 sectors)

✓ PASSED: Layout is valid
```

#### 功能详解

##### 1. MBR 签名验证

```python
def check_mbr_signature(data):
    """检查 MBR 签名 (0x55AA)"""
    if len(data) < 512:
        return False, "File too small for MBR check"

    signature = struct.unpack('<H', data[510:512])[0]
    if signature != 0xAA55:
        return False, f"Invalid MBR signature: 0x{signature:04X}"

    return True, "Valid MBR signature"
```

##### 2. 内存重叠检测

```python
def calculate_memory_layout(bootloader_sectors, kernel_sectors):
    """计算内存布局并检测重叠"""
    stage2_end = 0x7E00 + (bootloader_sectors * 512)
    kernel_start = 0x10000

    if stage2_end > kernel_start:
        return "ERROR: Memory overlap detected"

    gap = kernel_start - stage2_end
    if gap < 512:
        return f"WARNING: Small memory gap ({gap} bytes)"

    return "OK"
```

##### 3. 内核魔数检查

```python
# 检查内核魔数
if len(kernel_data) >= 4:
    first_dword = struct.unpack('<I', kernel_data[:4])[0]
    print(f"  First dword: 0x{first_dword:08X}")

    # 检查 ELF
    if kernel_data[:4] == b'\x7FELF':
        print("  ⚠ Warning: Kernel appears to be an ELF file")

    # 检查 x86-64 代码
    if len(kernel_data) >= 3 and kernel_data[0] == 0x48:
        print("  ✓ Appears to be x86-64 code (REX prefix found)")
```

#### 返回码

```bash
# 成功返回 0
python3 scripts/verify_disk_layout.py ... && echo "Valid"

# 失败返回 1
python3 scripts/verify_disk_layout.py ... || echo "Invalid"

# 在 CMake 中使用
add_custom_command(TARGET boot_img POST_BUILD
    COMMAND ${PYTHON} ${CMAKE_SOURCE_DIR}/scripts/verify_disk_layout.py ...
    # 验证失败会中断构建
)
```

## 3. 配置生成脚本

### 3.1 GenerateKernelSize.cmake

#### 基本信息

- **路径**: [cmake/GenerateKernelSize.cmake](../../../cmake/GenerateKernelSize.cmake)
- **功能**: 根据内核大小生成 boot_config.inc
- **使用方式**: CMake 脚本模式

#### 独立运行

```bash
# 手动运行配置生成
cmake \
    -DKERNEL_BIN=build/kernel.bin \
    -DOUTPUT_FILE=build/boot_config.inc \
    -DBOOTLOADER_SECTORS=3 \
    -P cmake/GenerateKernelSize.cmake
```

#### 输出示例

```
-- Generated boot configuration:
--   Kernel: 512 bytes = 1 sectors
--   LBA start: 4 (sector 4, 1-based)
--   CHS: C=0 H=0 S=5
--   Load address: 0x10000
--   Output: build/boot_config.inc
```

### 3.2 脚本工作原理

#### 文件大小获取

```cmake
file(SIZE "${KERNEL_BIN}" KERNEL_SIZE_BYTES)
message(STATUS "Kernel size: ${KERNEL_SIZE_BYTES} bytes")
```

#### 扇区数计算

```cmake
# 向上取整
math(EXPR KERNEL_SECTORS "(${KERNEL_SIZE_BYTES} + 511) / 512")
message(STATUS "Kernel sectors: ${KERNEL_SECTORS}")
```

#### CHS 参数计算

```cmake
set(SECTORS_PER_TRACK 63)
set(HEADS 16)

math(EXPR KERNEL_LBA_ZERO "${KERNEL_LBA_START} - 1")
math(EXPR CHS_CYLINDER "${KERNEL_LBA_ZERO} / (${SECTORS_PER_TRACK} * ${HEADS})")
math(EXPR CHS_HEAD "(${KERNEL_LBA_ZERO} / ${SECTORS_PER_TRACK}) % ${HEADS}")
math(EXPR CHS_SECTOR "(${KERNEL_LBA_ZERO} % ${SECTORS_PER_TRACK}) + 1")
```

### 3.3 调试配置生成

```bash
# 显示生成的配置
cmake ... -P GenerateKernelSize.cmake && cat build/boot_config.inc

# 比较配置差异
diff boot/boot_config.inc build/boot_config.inc

# 验证生成的宏
grep "define.*KERNEL" build/boot_config.inc
```

## 4. 十六进制分析

### 4.1 检查 boot_config.inc

#### 查看配置文件

```bash
# 直接查看
cat build/boot/boot_config.inc

# 格式化显示
grep "%define" build/boot/boot_config.inc | column -t
```

#### 输出示例

```
%define KERNEL_LBA_START          4
%define KERNEL_SECTOR_COUNT       1
%define KERNEL_LOAD_SEGMENT       0x1000
%define KERNEL_LOAD_OFFSET        0x0000
%define BOOTLOADER_SECTORS        3
%define KERNEL_MAX_SECTORS        131072
%define SECTORS_PER_TRACK         63
%define HEADS                     16
%define KERNEL_CHS_CYLINDER       0
%define KERNEL_CHS_HEAD           0
%define KERNEL_CHS_SECTOR         5
```

### 4.2 检查编译后的 bootloader

#### 查找配置使用

```bash
# 反汇编查找 load_kernel_chs
ndisasm -b 16 build/bootloader.bin | grep -A 20 "load_kernel_chs"

# 查找配置引用
ndisasm -b 16 build/bootloader.bin | grep -E "mov.*,0x1000|mov.*,0x0000"
```

#### 验证 CHS 参数

```bash
# 查找 CHS 相关指令
ndisasm -b 16 build/bootloader.bin | grep -E "mov ch,|mov cl,|mov dh,"

# 输出应该类似：
#   00000150  B400              mov ah, 0x00  ; 检查这里
#   00000152  B001              mov al, 0x01  ; 检查这里
#   00000154  B005              mov cl, 0x05  ; CHS 扇区号
```

### 4.3 检查磁盘镜像

```bash
# 查看磁盘布局
xxd build/boot.img | head -40

# 检查 MBR 签名
xxd -s 510 -l 2 build/boot.img
# 输出: 000001fe: 55aa

# 检查内核起始位置（LBA 4 = 偏移 2048）
xxd -s 2048 -l 64 build/boot.img
```

## 5. QEMU 调试技巧

### 5.1 监控磁盘读取

#### QEMU Monitor 命令

```
# 启动 QEMU 后按 Ctrl+Alt+2 进入 monitor

# 查看磁盘状态
info block

# 查看驱动器
info blockdrv

# 查看物理内存
xp/256x 0x10000  # 查看内核加载位置
```

### 5.2 调试内核加载

#### 添加调试输出

```asm
; 在 load_kernel_chs 中添加
call load_kernel_chs
jc .error

; 添加成功消息
pusha
mov si, msg_kernel_loaded
call print_bios
popa

; 添加地址检查
mov ax, es
mov di, ax
call print_hex
```

#### 使用 GDB 单步

```bash
# 启动 QEMU with GDB
qemu-system-x86_64 -drive format=raw,file=build/boot.img -nographic -s -S

# 连接 GDB
gdb
(gdb) target remote localhost:1234
(gdb) set architecture i8086

# 设置断点在 load_kernel_chs
# 需要先查找函数地址
(gdb) break *0x7E00
(gdb) continue

# 单步执行
(gdb) si
(gdb) info registers
```

### 5.3 测试不同内核大小

#### 生成测试内核

```bash
# 1 扇区内核
dd if=/dev/zero of=kernel_test.bin bs=512 count=1

# 127 扇区内核（最大单次读取）
dd if=/dev/zero of=kernel_test.bin bs=512 count=127

# 128 扇区内核（需要分批读取）
dd if=/dev/zero of=kernel_test.bin bs=512 count=128
```

#### 构建和测试

```bash
# 替换内核
cp kernel_test.bin build/kernel.bin

# 重新生成 boot.img
dd if=build/bootloader.bin of=build/boot.img bs=512 count=1 conv=notrunc
dd if=build/kernel.bin of=build/boot.img bs=512 seek=3 conv=notrunc

# 运行测试
timeout 5 qemu-system-x86_64 -drive format=raw,file=build/boot.img -nographic
```

### 5.4 常见 QEMU 选项

```bash
# 禁用 KVM（软件模拟，更易调试）
qemu-system-x86_64 -no-kvm -drive ...

# 增加 CPU 核心数（用于多核调试）
qemu-system-x86_64 -smp 2 -drive ...

# 指定内存大小
qemu-system-x86_64 -m 128M -drive ...

# 记录执行日志
qemu-system-x86_64 -d in_asm,cpu,exec -drive ... 2> qemu.log

# 检查 INT 13h 调用
qemu-system-x86_64 -d int -drive ... 2> qemu.log
```

## 6. 工作流最佳实践

### 6.1 完整调试流程

```bash
# 1. 清理构建
rm -rf build && mkdir build && cd build

# 2. 配置（详细输出）
cmake .. -DCMAKE_VERBOSE_MAKEFILE=ON

# 3. 构建
cmake --build . --verbose

# 4. 验证布局
python3 ../scripts/verify_disk_layout.py \
    bootloader.bin \
    kernel.bin

# 5. 检查配置
cat boot/boot_config.inc

# 6. 反汇编关键部分
ndisasm -b 16 bootloader.bin | grep -A 20 "load_kernel"

# 7. 运行测试
timeout 5 qemu-system-x86_64 -drive format=raw,file=boot.img -nographic
```

### 6.2 快速检查清单

```bash
# ✓ 检查 boot_config.inc 存在
ls -la build/boot/boot_config.inc

# ✓ 检查 MBR 签名
xxd -s 510 -l 2 build/bootloader.bin | grep "55 aa"

# ✓ 检查内核大小
ls -la build/kernel.bin

# ✓ 检查内存布局
python3 ../scripts/verify_disk_layout.py \
    build/bootloader.bin \
    build/kernel.bin
```

### 6.3 自动化测试脚本

```bash
#!/bin/bash
# test_build.sh

set -e

echo "=== Building CCOS ==="
mkdir -p build && cd build
cmake .. -DCMAKE_BUILD_TYPE=Debug
cmake --build . --target boot_img

echo "=== Verifying layout ==="
python3 ../scripts/verify_disk_layout.py \
    bootloader.bin \
    kernel.bin

echo "=== Running in QEMU ==="
timeout 5 qemu-system-x86_64 \
    -drive format=raw,file=boot.img \
    -nographic \
    -no-reboot \
    2>&1 | tee run.log

echo "=== Check for kernel output ==="
if grep -q "X" run.log; then
    echo "✓ Kernel executed successfully"
else
    echo "✗ Kernel did not execute"
    exit 1
fi
```
