## 1. QEMU Monitor 调试

### 1.1 启动 Monitor

#### 方法 1: 标准输入/输出

```bash
qemu-system-x86_64 -kernel kernel.elf -monitor stdio
```

#### 方法 2: Telnet 连接

```bash
# 启动 QEMU 时开启 telnet monitor
qemu-system-x86_64 -kernel kernel.elf \
    -monitor telnet::4444,server,nowait \
    -serial stdio

# 在另一个终端连接
telnet localhost 4444
```

#### 方法 3: Unix Socket

```bash
qemu-system-x86_64 -kernel kernel.elf \
    -monitor unix:qemu-monitor.sock,server,nowait

# 使用 socat 连接
socat unix-connect:qemu-monitor.sock readline
```

### 1.2 PIC 相关命令

```qemu
# 查看 PIC 状态
info pic

# 输出示例:
# pic: vector=0x20 irr=0x00 isr=0x00
# pic: vector=0x28 irr=0x00 isr=0x00

# 查看中断状态
info irq

# 查看 APIC 状态 (如果使用 APIC)
info apic
```

### 1.3 寄存器和内存

```qemu
# 查看所有寄存器
info registers

# 查看特定寄存器
info registers rax
info registers rsp

# 查看内存
x/10xg 0x7c00        # 10 个 64 位字
x/20i 0x10000        # 20 条指令

# 查看栈
x/20gx $rsp

# 查看虚拟内存映射
info mem
```

### 1.4 执行控制

```qemu
# 暂停执行
stop

# 继续执行
cont

# 单步执行
step

# 单步执行 (跳过函数调用)
next

# 执行到指定地址
jump 0x10000
```

### 1.5 有用的 Monitor 技巧

```qemu
# 保存 VM 状态
savevm state1

# 恢复 VM 状态
loadvm state1

# 查看保存的状态
info snapshots

# 查看虚拟设备
qtree

# 查看计时器
info timers
```

## 2. GDB 中断调试

### 2.1 启动 GDB 调试

#### 基本启动

```bash
# 终端 1: 启动 QEMU 等待 GDB
qemu-system-x86_64 -kernel kernel.elf -s -S

# 终端 2: 启动 GDB
gdb kernel.elf

# 在 GDB 中连接
(gdb) target remote :1234
(gdb) continue
```

#### 使用构建脚本

```bash
# 使用项目的调试脚本
./scripts/build_helpers/build_release_qemu_monitor_run.sh
```

### 2.2 中断相关断点

```gdb
# 在中断处理程序设置断点
(gdb) break interrupt_handler

# 在定时器处理程序设置断点
(gdb) break timer_handler

# 在汇编入口设置断点
(gdb) break interrupt_common

# 在特定 ISR stub 设置断点
(gdb) break isr32
```

### 2.3 查看中断状态

```gdb
# 查看调用栈
(gdb) backtrace

# 查看当前帧
(gdb) info frame

# 查看寄存器
(gdb) info registers

# 查看特定寄存器
(gdb) print $rax
(gdb) print $rsp
(gdb) print $rbp

# 查看 RFLAGS (包括 IF 位)
(gdb) print $rflags
(gdb) print/t $rflags  # 以二进制显示
```

### 2.4 检查中断帧

```gdb
# 在 interrupt_handler 中断后
# 查看中断帧结构
(gdb) print *frame

# 查看特定字段
(gdb) print frame->rip
(gdb) print frame->cs
(gdb) print frame->rflags
(gdb) print frame->error_code

# 检查中断是否启用
(gdb) print/t ($rflags & 0x200)
```

### 2.5 单步执行中断

```gdb
# 指令级单步
(gdb) stepi

# 跳过函数调用
(gdb) nexti

# 执行直到返回
(gdb) finish

# 执行到指定地址
(gdb) advance *0x10000
```

### 2.6 条件断点

```gdb
# 只在特定中断向量时中断
(gdb) break interrupt_handler if vector == 32

# 只在错误码非零时中断
(gdb) break interrupt_handler if error_code != 0

# 只在第 N 次调用时中断
(gdb) break timer_handler if timer_ticks == 100
```

### 2.7 GDB 宏

```gdb
# 打印中断状态
define print_intr_state
    printf "Interrupts: %s\n", ($rflags & 0x200) ? "Enabled" : "Disabled"
    printf "RIP: 0x%016lx\n", $rip
    printf "RSP: 0x%016lx\n", $rsp
    printf "RBP: 0x%016lx\n", $rbp
end

# 检查栈对齐
define check_alignment
    set $rsp_aligned = ($rsp & 0xF)
    if $rsp_aligned == 0
        printf "RSP is 16-byte aligned\n"
    else
        printf "RSP is NOT aligned (offset = %d)\n", $rsp_aligned
    end
end

# 打印 PIC 状态 (需要读取端口)
define dump_pic
    printf "PIC1: ISR=0x%02x IRR=0x%02x\n", $isr1, $irr1
end
```

## 3. 串口日志分析

### 3.1 串口配置

#### QEMU 串口输出

```bash
qemu-system-x86_64 -kernel kernel.elf \
    -serial file:serial.log

```

```
[INFO] [TIMER] Tick #2 received
```

```bash
tail -f serial.log | grep --color=auto "ERROR\|EXCEPTION"
```

## 4. PIC 状态查看

### 4.1 读取 PIC 寄存器

```c
// 添加到内核代码
void dump_pic_status(void) {
    // 读取 In-Service Register
    outb(PIC1_CMD, PIC_READ_ISR);
    uint8_t isr1 = inb(PIC1_CMD);

    outb(PIC2_CMD, PIC_READ_ISR);
    uint8_t isr2 = inb(PIC2_CMD);

    // 读取 Interrupt Request Register
    outb(PIC1_CMD, PIC_READ_IRR);
    uint8_t irr1 = inb(PIC1_CMD);

    outb(PIC2_CMD, PIC_READ_IRR);
    uint8_t irr2 = inb(PIC2_CMD);

    // 读取中断屏蔽字
    uint8_t mask1 = inb(PIC1_DATA);
    uint8_t mask2 = inb(PIC2_DATA);

    klog_info("=== PIC Status ===\n");
    klog_info("PIC1: ISR=0x%02x IRR=0x%02x MASK=0x%02x\n",
              isr1, irr1, mask1);
    klog_info("PIC2: ISR=0x%02x IRR=0x%02x MASK=0x%02x\n",
              isr2, irr2, mask2);

    // 打印每个 IRQ 的状态
    for (int i = 0; i < 16; i++) {
        uint8_t pic = (i < 8) ? 1 : 2;
        uint8_t bit = (i < 8) ? i : (i - 8);
        uint8_t isr = (i < 8) ? isr1 : isr2;
        uint8_t irr = (i < 8) ? irr1 : irr2;
        uint8_t mask = (i < 8) ? mask1 : mask2;

        klog_info("IRQ %2d: %s %s %s\n",
                  i,
                  (isr & (1 << bit)) ? "[IN_SERVICE]" : "           ",
                  (irr & (1 << bit)) ? "[PENDING]"   : "         ",
                  (mask & (1 << bit)) ? "[MASKED]"    : "        ");
    }
}
```

### 4.2 解释 PIC 状态

```
PIC1: ISR=0x01 IRR=0x00 MASK=0x00
│     │     │      │      │
│     │     │      │      └─ 中断屏蔽字 (0=启用, 1=屏蔽)
│     │     │      └──────── 中断请求寄存器 (1=有请求)
│     │     └─────────────── 服务中寄存器 (1=正在服务)
│     └───────────────────── PIC 编号
└─────────────────────────── 固定文本
```

### 4.3 常见状态模式

| ISR | IRR | MASK | 状态 |
|-----|-----|------|------|
| 0 | 0 | 0 | 空闲，IRQ 启用 |
| 0 | 1 | 0 | 有中断请求，等待处理 |
| 1 | 0 | 0 | 正在处理中断 |
| 0 | 0 | 1 | IRQ 被屏蔽 |

```c
// 检查 IDT 条目
void verify_idt_entry(uint8_t vector) {
    idt_entry_t* entry = &idt[vector];

    uint64_t handler = (uint64_t)(entry->offset_high << 32 |
                                  entry->offset_middle << 16 |
                                  entry->offset_low);

    klog_info("IDT[%d]:\n", vector);
    klog_info("  Handler: 0x%016lx\n", handler);
    klog_info("  Segment: 0x%04x\n", entry->segment_selector);
    klog_info("  Type:    0x%02x\n", entry->type_attr);
    klog_info("  IST:     0x%02x\n", entry->ist);

    // 验证类型属性
    if (entry->type_attr & 0x80) {
        klog_info("  Present: Yes\n");
    } else {
        klog_error("  Present: No (NOT ACTIVE!)\n");
    }
}
```

## 6. 性能分析

### 6.1 测量中断处理时间

```c
// RDTSC 辅助函数
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ volatile ("rdtsc" : "=a"(lo), "=d"(hi));
    return ((uint64_t)hi << 32) | lo;
}

// 中断处理时间统计
static uint64_t intr_times[256] = {0};
static uint64_t intr_counts[256] = {0};

void interrupt_handler(uint64_t vector, uint64_t error_code,
                      interrupt_frame_t* frame) {
    uint64_t start = rdtsc();

    // 处理中断...
    if (custom_handlers[vector]) {
        custom_handlers[vector](frame, error_code);
    }

    uint64_t end = rdtsc();
    uint64_t elapsed = end - start;

    intr_times[vector] += elapsed;
    intr_counts[vector]++;

    // 每 1000 次打印统计
    if (intr_counts[vector] % 1000 == 0) {
        uint64_t avg = intr_times[vector] / intr_counts[vector];
        klog_info("IRQ %d: avg %lu cycles\n", vector, avg);
    }
}
```

### 6.2 中断频率分析

```c
// 定时器中断间隔测量
static uint64_t last_timer_tick = 0;
static uint64_t timer_intervals[100];
static int interval_index = 0;

void timer_handler(interrupt_frame_t* frame, uint64_t error_code) {
    uint64_t now = rdtsc();

    if (last_timer_tick != 0) {
        uint64_t interval = now - last_timer_tick;
        timer_intervals[interval_index % 100] = interval;
        interval_index++;

        if (interval_index % 100 == 0) {
            uint64_t sum = 0;
            for (int i = 0; i < 100; i++) {
                sum += timer_intervals[i];
            }
            klog_info("Timer interval: avg %lu cycles\n", sum / 100);
        }
    }

    last_timer_tick = now;
    timer_ticks++;
    pic_send_eoi(0);
}
```

### 7.3 Python 辅助脚本

```python
#!/usr/bin/env python3
"""
parse_qemu_log.py - 解析 QEMU 日志提取中断信息
"""

import sys
import re

def parse_interrupts(log_file):
    """解析 QEMU 日志中的中断信息"""
    intr_count = {}
    current_intr = None

    with open(log_file, 'r') as f:
        for line in f:
            # 查找中断行
            match = re.search(r'Interrupt (\d+)', line)
            if match:
                intr_num = int(match.group(1))
                intr_count[intr_num] = intr_count.get(intr_num, 0) + 1

    # 打印统计
    print("Interrupt Statistics:")
    for intr in sorted(intr_count.keys()):
        print(f"  IRQ {intr - 32}: {intr_count[intr]} times")

if __name__ == '__main__':
    if len(sys.argv) < 2:
        print("Usage: parse_qemu_log.py <qemu.log>")
        sys.exit(1)

    parse_interrupts(sys.argv[1])
```

## 9. 工作流建议

### 9.1 中断调试流程

```
1. 启用详细日志
   ↓
2. QEMU Monitor 查看状态
   ↓
3. GDB 设置关键断点
   ↓
4. 触发问题
   ↓
5. 检查寄存器和栈
   ↓
6. 定位问题根因
   ↓
7. 修复代码
   ↓
8. 验证修复
```
