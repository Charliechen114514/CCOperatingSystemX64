## 1. NASM Section 机制

### 1.1 基本概念

#### Section 声明

```asm
section name [options]
    ; 代码或数据
```

#### 常用选项

| 选项 | 说明 | 示例 |
|------|------|------|
| `align=N` | 对齐到 N 字节边界 | `section .data align=16` |
| `vstart=addr` | 虚拟起始地址 | `section .stage2 vstart=0x7E00` |
| `follows=sec` | 紧跟指定 section | `section .text follows=.mbr` |

### 1.2 org 指令

#### 作用

```asm
org 0x7C00
start:
    ; 此处的代码地址从 0x7C00 开始计算
```

**重要**: `org` 设置代码期望加载的地址，但不影响输出文件的位置

#### org 与 vstart 的区别

```asm
; 示例 1: 只有 org
section .mbr
    org 0x7C00
label1:
    db 0x01
; label1 的值 = 0x7C00

; 示例 2: org + vstart
section .mbr
    org 0x7C00
label1:
    db 0x01

section .stage2 vstart=0x7E00
label2:
    dd label1    ; 值 = 0x7C00 (正确的物理地址)
```

### 1.3 vstart 指令

#### 作用

```asm
section .stage2 vstart=0x7E00
gdt_start:
    dq 0

gdt_ptr:
    dd gdt_start    ; 正确! = 0x7E00 + 偏移
```

**vstart 的作用**:
- 告诉 NASM 此 section 中的标签应该加上这个偏移
- 用于计算跨 section 的引用

#### vstart 使用场景

```asm
; 场景 1: Stage 2 引用自身的 GDT
section .stage2 vstart=0x7E00
gdt_ptr:
    dw gdt_end - gdt_start - 1
    dd gdt_start    ; 需要 vstart 才能正确计算

; 场景 2: 在 32 位代码中引用 16 位数据
section .stage2 vstart=0x7E00
    bits 16
msg:
    db "Hello", 0

section .stage2 vstart=0x7E00
    bits 32
pm_entry:
    mov esi, msg    ; msg = 0x7E00 + 偏移 (正确)
```

### 1.4 Section 在输出文件中的位置

#### NASM 默认行为

```asm
section .mbr
    org 0x7C00
    db 0xAA

section .stage2
    org 0x7E00
    db 0xBB

section .text
    db 0xCC
```

**输出文件布局**:
```
偏移 0:    0xAA    (.mbr)
偏移 1:    0xBB    (.stage2)
偏移 2:    0xCC    (.text)
```

#### 控制 Section 顺序

```asm
; 方法 1: 按声明顺序
section .mbr
    ; ...

; 方法 2: 使用 follows
section .stage2 follows=.mbr
    ; ...
```

## 2. 单文件多阶段架构

### 2.1 架构设计

#### 文件结构

```asm
; bootloader.asm
; ==============================================================================
; Section 1: MBR (Stage 1)
; ==============================================================================
section .mbr
    org 0x7C00
    bits 16

start:
    ; Stage 1 代码
    jmp 0x7E00    ; 跳转到 Stage 2

times 510-($-$$) db 0
dw 0xAA55

; ==============================================================================
; Section 2: Stage 2
; ==============================================================================
section .stage2 vstart=0x7E00
    bits 16

stage2_entry:
    ; Stage 2 代码
```

### 2.2 Stage 1 与 Stage 2 的通信

#### 数据传递方式

```asm
; 方式 1: 通过固定内存位置
section .mbr
    org 0x7C00
start:
    mov word [0x7E00], 0x1234    ; 传递数据

; 方式 2: 通过寄存器
section .mbr
    org 0x7C00
start:
    mov ax, 0x1234
    jmp 0x7E00    ; AX 保持不变

section .stage2 vstart=0x7E00
stage2_entry:
    ; AX = 0x1234
```

### 2.3 跨 Stage 跳转

#### Near Jump (同一 section 内)

```asm
section .mbr
    org 0x7C00
start:
    jmp main    ; near jump
main:
    ; ...
```

#### Far Jump (跨 section)

```asm
section .mbr
    org 0x7C00
start:
    jmp 0x7E00    ; far jump 到 Stage 2

section .stage2 vstart=0x7E00
stage2_entry:
    ; ...
```

#### 段间跳转 (保护模式)

```asm
bits 16
    ; 设置 CR0.PE
    mov eax, cr0
    or eax, 1
    mov cr0, eax

    ; Far jump 到 32 位代码
    db 0x66         ; operand-size prefix
    db 0xEA         ; jmp far opcode
    dd pm_entry     ; 32 位偏移
    dw 0x08         ; 段选择子

bits 32
pm_entry:
    ; 32 位代码
```

## 3. 库函数接口规范

### 3.1 命名约定

#### 函数命名

| 模式 | 说明 | 示例 |
|------|------|------|
| `action` | 16 位版本 | `print_string` |
| `action_N` | N 位版本 | `print_string64` |
| `action_device` | 特定设备 | `serial_write_char` |

#### 变量命名

```asm
; 常量 (全大写)
SERIAL_PORT    equ 0x3F8
BOOTLOADER_SECTORS equ 2

; 变量 (小写下划线)
kernel_size    dd 0
buffer_ptr     dw 0

; 标签 (小写点开头)
.loop:
.done:
.error:
```

### 3.2 调用约定

#### 标准格式

```asm
; function_name - 函数描述
; Input: 寄存器/内存参数
; Output: 返回值寄存器
; Clobbers: 破坏的寄存器列表
function_name:
    ; 函数体
    ret
```

#### 示例

```asm
; print_string - 使用 BIOS 打印字符串
; Input: SI = null-terminated string pointer
; Output: none
; Clobbers: AX, SI
print_string:
    pusha
.loop:
    lodsb
    test al, al
    jz .done
    mov ah, 0x0E
    int 0x10
    jmp .loop
.done:
    popa
    ret
```

### 3.3 错误处理约定

#### Carry Flag 表示错误

```asm
; 成功时 CLC (Clear Carry)
; 失败时 STC (Set Carry)

function_name:
    pusha
    ; ... 操作
    jc .error    ; 检测到错误
    popa
    clc         ; 成功
    ret
.error:
    popa
    stc         ; 失败
    ret
```

#### 调用示例

```asm
call load_kernel_auto
jc load_error    ; CF=1 表示失败
; 继续执行
```

### 3.4 库函数文档

#### bios_string.asm

```asm
; print_string - 打印字符串 (BIOS INT 10h)
; Input: SI = 字符串指针
; Output: none
; Clobbers: AX, SI

; print_bios - print_string 的别名
; Input: SI = 字符串指针
; Output: none
; Clobbers: AX, SI
```

#### disk_io.asm

```asm
; load_second_stage - 加载第二阶段 bootloader
; Input: none
; Output: 加载到 0x7E00
; Clobbers: AX, BX, CX, DX

; load_kernel_auto - 自动选择 LBA/CHS 加载内核
; Input: none (使用 boot_config.inc 中的配置)
; Output: 内核加载到 KERNEL_LOAD_SEGMENT:KERNEL_LOAD_OFFSET
; Clobbers: AX, BX, CX, DX, SI, DI, BP
; Returns: CF=0 成功, CF=1 失败
```

#### pmode.asm

```asm
; setup_page_tables - 设置长模式页表
; Input: none
; Output: 页表在 0x9000 (PML4), 0xA000 (PDPT), 0xB000 (PD)
; Clobbers: EAX, ECX, EDI
```

#### lmode.asm

```asm
; clear_screen64 - 清屏 (64 位)
; Input: none
; Output: VGA 文本缓冲区清空
; Clobbers: RAX, RDI, RCX

; print_string64 - 打印字符串到 VGA (64 位)
; Input: RSI = 字符串指针
; Output: none
; Clobbers: RAX, RDI, RSI
```

## 4. 内存布局

### 4.1 物理内存映射

#### Bootloader 使用的内存

```
地址          大小     用途              来源
─────────────────────────────────────────────────
0x00000       1KB      BIOS 数据区        硬件
0x00400       256B     BIOS BDA           硬件
0x007C00      512B     Stage 1 (MBR)      磁盘扇区 1
0x007E00      ~1KB     Stage 2 代码       磁盘扇区 2+
0x009000      4KB      PML4 页表          运行时创建
0x00A000      4KB      PDPT 页表          运行时创建
0x00B000      4KB      PD 页表            运行时创建
0x010000      变化     内核映像           磁盘加载
0x0B8000      32KB     VGA 文本缓冲区     硬件固定
```

### 4.2 Stage 内存布局

#### Stage 1 (MBR)

```
偏移      内容                    说明
────────────────────────────────────────
0x0000    代码入口 (start)        BIOS 跳转到这里
0x01FE    MBR 签名 (0xAA55)      必须在 510-511
```

#### Stage 2

```
偏移      内容                    说明
────────────────────────────────────────
0x0000    stage2_entry            入口点
0x0003    GDT                     5 个描述符
0x002B    gdt_ptr                 GDT 指针
0x0030    代码段                  实模式代码
0x0100+   数据段                  消息字符串
```

### 4.3 虚拟地址映射 (长模式)

#### 页表结构

```
PML4[0] -> PDPT[0] -> PD[0] -> 0x00000000 (2MB, identity)
                           -> 0x00200000 (2MB)

PML4[511] -> PDPT[0] -> PD[511] -> 0xFFFF800000000000 (2MB, higher-half)
                                -> 0xFFFF800000020000 (2MB)
```

## 5. 磁盘布局

### 5.1 扇区分配

#### 统一 Bootloader 布局

```
LBA    扇区号        内容              大小
─────────────────────────────────────────────
0      1             Stage 1 (MBR)    512 bytes
1      2             Stage 2 部分 1   512 bytes
2      3             Stage 2 部分 2   512 bytes
3+     4+            内核映像         变化
```

### 5.2 Bootloader 文件结构

```
bootloader.bin (编译输出)
├── Offset 0-511:     Stage 1 (MBR)
│   ├── 代码:         0-509
│   ├── 填充:         510-509->510
│   └── 签名:         510-511 (0xAA55)
└── Offset 512+:      Stage 2
    ├── GDT:          0-39
    ├── 代码:         40+
    └── 数据:         消息字符串
```

### 5.3 内核加载

#### LBA 地址

```asm
; boot_config.inc
KERNEL_LBA_START      equ 2     ; 内核起始 LBA (0-based)
                                    ; = 扇区 3 (1-based)
KERNEL_SECTOR_COUNT   equ 1     ; 默认 1 个扇区
```

#### 加载地址

```asm
; boot_config.inc
KERNEL_LOAD_SEGMENT   equ 0x1000
KERNEL_LOAD_OFFSET    equ 0x0000
; 最终物理地址: 0x10000
```

## 6. 构建系统

### 6.1 Makefile 结构

#### 目标定义

```makefile
# 工具
AS      = nasm
ASFLAGS = -f bin

# 源文件
BOOTLOADER_ASM = boot/bootloader.asm
KERNEL_ASM      = kernel/kernel.asm

# 输出文件
BOOTLOADER_BIN  = build/bootloader.bin
KERNEL_BIN      = build/kernel.bin
BOOT_IMG        = build/boot.img

# 默认目标
all: $(BOOT_IMG)
```

#### 编译规则

```makefile
# 编译 bootloader
$(BOOTLOADER_BIN): $(BOOTLOADER_ASM) | prepare
	@echo "Building unified bootloader..."
	$(AS) $(ASFLAGS) $< -o $@
	@echo "Bootloader size:"
	@ls -lh $@

# 编译内核
$(KERNEL_BIN): $(KERNEL_ASM) | prepare
	@echo "Building kernel..."
	$(AS) $(ASFLAGS) $< -o $@
```

#### 镜像生成

```makefile
# 生成启动镜像
$(BOOT_IMG): $(BOOTLOADER_BIN) $(KERNEL_BIN)
	@echo "Combining bootloader and kernel..."
	dd if=$(BOOTLOADER_BIN) of=$@ bs=512 count=2 conv=notrunc
	dd if=$(KERNEL_BIN) of=$@ bs=512 seek=2 conv=notrunc
	@echo "Done! Boot image: $@"
```

### 6.2 配置文件

#### boot_config.inc

```asm
; Bootloader 配置
; 由 Makefile 或构建脚本生成

; 扇区配置
BOOTLOADER_SECTORS    equ 2     ; bootloader 占用的扇区数

; 内核加载配置
KERNEL_LBA_START      equ 2     ; 内核起始 LBA (扇区 3)
KERNEL_SECTOR_COUNT   equ 1     ; 默认内核扇区数
KERNEL_LOAD_SEGMENT   equ 0x1000
KERNEL_LOAD_OFFSET    equ 0x0000

; 磁盘几何参数 (CHS 回退用)
SECTORS_PER_TRACK     equ 18
HEADS                 equ 2
```

### 6.3 调试构建

#### 调试符号

```makefile
# 添加调试信息
ASFLAGS_DEBUG = -g -F dwarf

$(BOOTLOADER_BIN): $(BOOTLOADER_ASM)
	$(AS) $(ASFLAGS_DEBUG) -f bin $< -o $@
```

#### 列表文件

```makefile
# 生成汇编列表
$(BOOTLOADER_BIN): $(BOOTLOADER_ASM)
	$(AS) $(ASFLAGS) -l $(@:.bin=.lst) $< -o $@
```

## 7. 附录

### 7.1 完整的 Section 示例

```asm
; ==============================================================================
; CCOS Unified Bootloader - 完整结构示例
; ==============================================================================

; 包含配置
%include "boot_config.inc"

; ==============================================================================
; Section 1: MBR (Stage 1)
; ==============================================================================
section .mbr
    org 0x7C00
    bits 16

start:
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00
    sti

    ; 加载 Stage 2
    mov ax, 0x7E0
    mov es, ax
    xor bx, bx
    mov ah, 0x02
    mov al, BOOTLOADER_SECTORS - 1
    mov ch, 0x00
    mov cl, 0x02
    mov dh, 0x00
    mov dl, 0x80
    int 0x13
    jc load_error

    ; 跳转到 Stage 2
    jmp 0x7E00

load_error:
    mov si, error_msg
    call print_string
.hang:
    hlt
    jmp .hang

print_string:
    pusha
.loop:
    lodsb
    test al, al
    jz .done
    mov ah, 0x0E
    int 0x10
    jmp .loop
.done:
    popa
    ret

error_msg:
    db "Boot error!", 0

times 510-($-$$) db 0
dw 0xAA55

; ==============================================================================
; Section 2: Stage 2
; ==============================================================================
section .stage2 vstart=0x7E00
    bits 16

stage2_entry:
    jmp short stage2_main
    nop

; GDT
gdt_start:
    dq 0x0000000000000000
gdt_code:
    dq 0x00CF9A000000FFFF
gdt_data:
    dq 0x00CF92000000FFFF
gdt_code64:
    dq 0x00AF9A000000FFFF
gdt_end:

gdt_ptr:
    dw gdt_end - gdt_start - 1
    dd gdt_start    ; 需要 vstart 才能正确计算!

stage2_main:
    ; Stage 2 代码
    ; ...

bits 32
pm_entry:
    ; 保护模式代码
    ; ...

bits 64
long_mode:
    ; 长模式代码
    ; ...
```

### 7.2 常用 NASM 指令

| 指令 | 说明 | 示例 |
|------|------|------|
| `section` | 声明 section | `section .text` |
| `org` | 设置起始地址 | `org 0x7C00` |
| `bits` | 设置指令模式 | `bits 16` |
| `times` | 重复指令 | `times 510-($-$$) db 0` |
| `equ` | 定义常量 | `PORT equ 0x3F8` |
| `%include` | 包含文件 | `%include "lib/string.asm"` |
| `db/dw/dd` | 定义数据 | `db 0xAA`, `dw 0xAABB` |
