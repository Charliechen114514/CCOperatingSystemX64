## 1. GDB 调试

### 1.1 启动 GDB 调试

#### 基本命令

```bash
# 终端 1: 启动 QEMU 并等待 GDB 连接
qemu-system-x86_64 -kernel kernel.elf -s -S

# 终端 2: 启动 GDB
gdb kernel.elf

# 在 GDB 中连接到 QEMU
(gdb) target remote :1234
(gdb) continue
```

### 1.2 栈帧调试

#### 查看栈帧

```gdb

(gdb) backtrace
#0  kernel_main () at kernel_main.c:42
#1  0x0000000000007c00 in kernel_start ()
#2  0x0000000000000000 in ?? ()

# 查看指定帧
(gdb) frame 1
#1  0x0000000000007c00 in kernel_start ()

# 查看局部变量
(gdb) info locals
```

#### 检查帧指针

```gdb

(gdb) x/2gx $rbp
```

### 1.3 断点设置

#### 函数断点

```gdb
# 在 dump_stack 函数设置断点
(gdb) break dump_stack

# 在 find_symbol 函数设置断点
(gdb) break find_symbol
```

#### 条件断点

```gdb
# 只在特定地址时中断
(gdb) break find_symbol if addr == 0x1001a

# 只在第 5 次调用时中断
(gdb) break dump_stack if frame == 5
```

#### 查看断点

```gdb
# 列出所有断点
(gdb) info breakpoints

# 删除断点
(gdb) delete 1
```

## 2. 符号表工具

### 2.1 符号提取脚本

#### 基本用法

```bash
# 提取符号
python scripts/extract_symbols.py build/kernel/kernel.elf

# 指定输出文件
python scripts/extract_symbols.py build/kernel/kernel.elf -o symbols.c

# 显示详细信息
python scripts/extract_symbols.py build/kernel/kernel.elf --verbose
```

#### 脚本选项

| 选项 | 说明 |
|------|------|
| `-o, --output` | 输出文件路径 |
| `-v, --verbose` | 显示详细信息 |
| `-h, --help` | 显示帮助信息 |

### 2.2 符号表验证

#### 验证脚本

```python
#!/usr/bin/env python3
import sys

def verify_symbols(symbol_file):
    """验证符号表的正确性"""
    with open(symbol_file, 'r') as f:
        content = f.read()

    # 检查必要的标记
    checks = [
        '_symbol_table_start' in content,
        '_symbol_string_table' in content,
        '_symbol_text_start' in content,
        '_symbol_text_end' in content,
    ]

    if all(checks):
        print("✓ Symbol table looks valid")
        return True
    else:
        print("✗ Symbol table is invalid")
        return False

if __name__ == '__main__':
    verify_symbols(sys.argv[1])
```

## 3. ELF 分析工具

### 3.1 readelf

#### 查看符号表

```bash
# 查看所有符号
readelf -s kernel.elf

# 只查看函数符号
readelf -s kernel.elf | grep FUNC

# 查看特定符号
readelf -s kernel.elf | grep kernel_main
```

#### 查看段信息

```bash
# 查看所有段
readelf -S kernel.elf

# 查看 .text 段
readelf -S kernel.elf | grep .text

# 输出示例:
[ 1] .text             PROGBITS        0000000000010000  00001000
     0000000000005187  0000000000000000  AX       0     0     16
```

### 3.2 objdump

#### 反汇编

```bash
# 反汇编整个文件
objdump -d kernel.elf

# 反汇编特定函数
objdump -d kernel.elf | grep -A30 "kernel_main>"

# 只显示头部
objdump -h kernel.elf
```

#### 查看调试信息

```bash
# 查看调试信息（如果有）
objdump -g kernel.elf
```

### 3.3 nm

#### 符号列表

```bash
# 按地址排序
nm -n kernel.elf

# 只显示未定义符号
nm -u kernel.elf

# 显示符号类型
nm -t x kernel.elf
```

#### 符号类型

| 类型 | 说明 |
|------|------|
| T | 代码段，全局 |
| t | 代码段，局部 |
| D | 数据段，全局 |
| d | 数据段，局部 |

## 5. 性能分析

### 5.1 使用 RDTSC 计时

```c
// 定义 RDTSC 宏
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    __asm__ __volatile__ (
        "rdtsc"
        : "=a"(lo), "=d"(hi)
    );
    return ((uint64_t)hi << 32) | lo;
}

// 测量栈回溯性能
uint64_t start = rdtsc();
dump_stack_full();
uint64_t end = rdtsc();

uint64_t cycles = end - start;
klog_info("dump_stack took %lu cycles\n", cycles);
```

### 5.2 性能测试

```c
// 测试符号查找性能
void benchmark_symbol_lookup(void) {
    const int ITERATIONS = 1000;
    uint64_t start, end;

    const symbol_table_t* table = get_symbol_table();
    uintptr_t test_addr = table->entries[table->count / 2].address;

    start = rdtsc();
    for (int i = 0; i < ITERATIONS; i++) {
        const char* name;
        uintptr_t offset;
        find_symbol(test_addr, &name, &offset);
    }
    end = rdtsc();

    uint64_t total = end - start;
    uint64_t avg = total / ITERATIONS;

    klog_info("find_symbol average: %lu cycles\n", avg);
}
```

```gdb

(gdb) break dump_stack

# 运行到断点
(gdb) continue

# 单步执行
(gdb) step

# 检查 RBP
(gdb) print $rbp

# 手动检查栈帧
(gdb) x/2gx $rbp
```

### 6.2 符号查找失败

```gdb
# 在 find_symbol 设置断点
(gdb) break find_symbol

# 运行
(gdb) continue

# 检查参数
(gdb) print addr
(gdb) print *table

# 单步执行查找逻辑
(gdb) next

# 检查中间结果
(gdb) print mid
(gdb) print table->entries[mid]
```

### 6.3 验证帧指针链

```python
#!/usr/bin/env python3
"""
GDB Python 脚本：验证帧指针链
"""

import gdb

def validate_frame_chain():
    """验证当前栈的帧指针链"""
    frame = gdb.newest_frame()
    count = 0
    visited = set()

    while frame and count < 100:
        rbp = frame.read_register('rbp')
        rbp_val = int(rbp)

        # 检查循环
        if rbp_val in visited:
            print(f"Frame loop detected at frame {count}")
            break

        visited.add(rbp_val)

        # 检查对齐
        if rbp_val % 8 != 0:
            print(f"Frame {count}: Misaligned RBP 0x{rbp_val:x}")

        # 检查边界
        if rbp_val > 0x80000 or rbp_val < 0x70000:
            print(f"Frame {count}: RBP out of bounds 0x{rbp_val:x}")

        print(f"Frame {count}: RBP=0x{rbp_val:x}, PC={frame.pc():x}")

        frame = frame.older()
        count += 1

validate_frame_chain()
```

使用方式：
```gdb
(gdb) source validate_frames.py
```

## 7. 工作流建议

### 7.1 调试流程

```
1. 启动 QEMU + GDB
   ↓
2. 在关键函数设置断点
   ↓
3. 触发栈回溯
   ↓
4. 单步执行，验证逻辑
   ↓
5. 检查寄存器和内存
   ↓
6. 定位问题根因
   ↓
7. 修复代码
   ↓
8. 重新验证
```

### 7.2 符号表更新流程

```bash
# 1. 清理旧的符号表
rm kernel/stacktrace/symbols.c

# 2. 重新构建
make clean && make

# 3. 验证符号表
readelf -s build/kernel/kernel.elf | wc -l

# 4. 运行测试
qemu-system-x86_64 -kernel build/kernel/kernel.elf
```

## 8. 有用的 GDB 宏

### 8.1 打印当前栈帧

```gdb
define print_frame
    printf "RBP=0x%lx, RSP=0x%lx, RIP=0x%lx\n", $rbp, $rsp, $rip
    x/2gx $rbp
end
```

### 8.2 遍历栈帧链

```gdb
define walk_frames
    set $rbp = $rbp
    set $count = 0
    while ($rbp != 0 && $count < 16)
        printf "Frame %d: RBP=0x%lx\n", $count, $rbp
        x/2gx $rbp
        set $rbp = *(unsigned long*)$rbp
        set $count = $count + 1
    end
end
```
