## VFS API 参考

### 超级块操作

#### vfs_mount

挂载文件系统。

```c
struct vfs_superblock* vfs_mount(const char* dev,
                                  const char* path,
                                  const char* type,
                                  int flags,
                                  const char* opts);
```

**参数**:
- `dev`: 设备名 ("hda", "hdb", 等)
- `path`: 挂载点路径
- `type`: 文件系统类型 ("ext2")
- `flags`: 挂载标志
- `opts`: 挂载选项

**返回值**:
- 成功: 返回超级块指针
- 失败: 返回 NULL

**示例**:
```c
struct vfs_superblock* sb = vfs_mount("hda", "/", "ext2", 0, NULL);
if (!sb) {
    klog_error("Mount failed\n");
}
```

#### vfs_umount

卸载文件系统。

```c
int vfs_umount(const char* path);
```

### Inode 操作

#### vfs_iget

获取 inode（优先从缓存）。

```c
struct vfs_inode* vfs_iget(struct vfs_superblock* sb, uint64_t ino);
```

**参数**:
- `sb`: 超级块
- `ino`: inode 号

**返回值**:
- 成功: 返回 inode 指针
- 失败: 返回 NULL

#### vfs_iput

释放 inode 引用。

```c
void vfs_iput(struct vfs_inode* inode);
```

### 文件操作

#### vfs_open

打开文件。

```c
file_t* vfs_open(const char* path, int flags);
```

**参数**:
- `path`: 文件路径
- `flags`: 打开标志 (O_RDONLY, O_WRONLY, O_RDWR, O_CREAT, O_TRUNC)

**返回值**:
- 成功: 返回文件指针
- 失败: 返回 NULL

#### vfs_close

关闭文件。

```c
int vfs_close(file_t* file);
```

#### vfs_read

从文件读取数据。

```c
ssize_t vfs_read(file_t* file, char* buffer, size_t count);
```

**参数**:
- `file`: 文件指针
- `buffer`: 接收缓冲区
- `count`: 要读取的字节数

**返回值**:
- 成功: 返回读取的字节数
- 失败: 返回负数错误码

#### vfs_write

向文件写入数据。

```c
ssize_t vfs_write(file_t* file, const char* buffer, size_t count);
```

#### vfs_lseek

设置文件位置。

```c
off_t vfs_lseek(file_t* file, off_t offset, int whence);
```

**whence**:
- `SEEK_SET`: 从文件开头
- `SEEK_CUR`: 从当前位置
- `SEEK_END`: 从文件结尾

### 目录操作

#### vfs_readdir

读取目录项。

```c
int vfs_readdir(file_t* dir, struct dirent* entry);
```

**dirent 结构**:
```c
struct dirent {
    uint64_t d_ino;        // inode 号
    char     d_name[256];  // 文件名
    uint8_t  d_type;       // 文件类型
};
```

```c
struct vfs_superblock* ext2_read_superblock(const char* dev);
```

#### ext2_write_superblock

```c
int ext2_write_superblock(struct vfs_superblock* sb);
```

```c
int ext2_read_inode(struct vfs_inode* inode);
```

#### ext2_write_inode

```c
int ext2_write_inode(struct vfs_inode* inode);
```

```c
struct vfs_dentry* ext2_lookup(struct vfs_inode* dir,
```

## 块设备 API 参考

### 块设备管理

#### block_init

初始化块设备层。

```c
int block_init(void);
```

#### block_register_device

注册块设备。

```c
int block_register_device(int dev_id,
                          uint32_t block_size,
                          uint64_t nblocks,
                          block_device_ops_t* ops);
```

### 块 I/O 操作

#### block_read_sync

同步读取块设备。

```c
int block_read_sync(int dev_id,
                    uint64_t block,
                    void* buffer,
                    size_t nblocks);
```

**参数**:
- `dev_id`: 设备 ID (0-3)
- `block`: 起始块号
- `buffer`: 接收缓冲区
- `nblocks`: 要读取的块数

**返回值**:
- 成功: 0
- 失败: 负数错误码

#### block_write_sync

同步写入块设备。

```c
int block_write_sync(int dev_id,
                     uint64_t block,
                     const void* buffer,
                     size_t nblocks);
```

## 数据结构定义

### VFS 超级块

```c
typedef struct vfs_superblock {
    uint32_t s_magic;              // 文件系统魔数
    uint32_t s_blocksize;          // 块大小
    uint32_t s_flags;              // 标志
    uint64_t s_maxbytes;           // 最大文件大小

    struct vfs_inode* s_root;      // 根 inode
    void* s_fs_info;               // 文件系统私有数据

    struct super_operations* s_op; // 超级块操作
    struct file_system_type* s_type;

    struct list_head list;         // 超级块链表
} vfs_superblock_t;
```

### VFS Inode

```c
typedef struct vfs_inode {
    uint64_t i_ino;                // inode 号
    uint32_t i_mode;               // 文件模式和权限
    uint32_t i_uid;                // 所有者
    uint32_t i_gid;                // 组
    uint64_t i_size;               // 文件大小
    uint64_t i_blocks;             // 512字节块数
    uint64_t i_atime;              // 访问时间
    uint64_t i_mtime;              // 修改时间
    uint64_t i_ctime;              // 状态改变时间

    struct vfs_inode* i_parent;    // 父目录
    struct list_head i_hash;       // 哈希表链表
    struct list_head i_list;       // inode 链表
    struct list_head i_dentry;     // 关联的 dentry

    struct inode_operations* i_op; // inode 操作
    struct file_operations* i_fop; // 文件操作

    struct vfs_superblock* i_sb;   // 所属超级块
    void* i_private;               // 私有数据
} vfs_inode_t;
```

### VFS 目录项

```c
typedef struct vfs_dentry {
    char d_name[256];              // 文件名
    struct vfs_inode* d_inode;     // 关联的 inode
    struct vfs_dentry* d_parent;   // 父目录项

    struct list_head d_hash;       // 哈希表链表
    struct list_head d_list;       // LRU 链表
    struct list_head d_alias;      // inode 别名链表

    unsigned int d_flags;          // 状态标志
} vfs_dentry_t;
```

### 文件结构

```c
typedef struct file {
    struct vfs_inode* f_inode;     // 关联的 inode
    struct vfs_dentry* f_dentry;   // 关联的目录项
    uint64_t f_pos;                // 当前位置
    uint32_t f_flags;              // 打开标志

    struct file_operations* f_op;  // 文件操作
    void* f_private;               // 私有数据
} file_t;
```

### EXT2 超级块

```c
typedef struct ext2_superblock {
    uint32_t s_inodes_count;       // inode 总数
    uint32_t s_blocks_count;       // 块总数
    uint32_t s_r_blocks_count;     // 保留块数
    uint32_t s_free_blocks_count;  // 空闲块数
    uint32_t s_free_inodes_count;  // 空闲 inode 数
    uint32_t s_first_data_block;   // 第一个数据块
    uint32_t s_log_block_size;     // 块大小
    uint32_t s_blocks_per_group;   // 每组块数
    uint32_t s_inodes_per_group;   // 每组 inode 数
    uint32_t s_magic;              // 魔数 0xEF53
    uint32_t s_mtime;              // 挂载时间
    uint16_t s_state;              // 文件系统状态
    // ...
} ext2_superblock_t;
```

### EXT2 Inode

```c
typedef struct ext2_inode {
    uint16_t i_mode;               // 文件类型和权限
    uint16_t i_uid;                // 所有者 UID
    uint32_t i_size;               // 文件大小
    uint32_t i_atime;              // 访问时间
    uint32_t i_ctime;              // 创建时间
    uint32_t i_mtime;              // 修改时间
    uint32_t i_dtime;              // 删除时间
    uint16_t i_gid;                // 组 ID
    uint16_t i_links_count;        // 硬链接数
    uint32_t i_blocks;             // 512字节块数
    uint32_t i_flags;              // 文件标志
    uint32_t i_block[15];          // 块指针数组
    // ...
} ext2_inode_t;
```

### 块设备

```c
typedef struct block_device {
    int dev_id;                    // 设备 ID
    uint32_t block_size;           // 块大小
    uint64_t nblocks;              // 块数量
    block_device_ops_t* ops;       // 操作函数
    void* private_data;            // 私有数据
    atomic_t refcnt;               // 引用计数
} block_device_t;
```

## 常量定义

### 文件类型

```c
#define S_IFMT     0170000  // 文件类型掩码
#define S_IFDIR    0040000  // 目录
#define S_IFREG    0100000  // 常规文件
#define S_IFLNK    0120000  // 符号链接
```

### 文件权限

```c
#define S_IRWXU    00700    // 所有者权限
#define S_IRUSR    00400    // 所有者读
#define S_IWUSR    00200    // 所有者写
#define S_IXUSR    00100    // 所有者执行

#define S_IRWXG    00070    // 组权限
#define S_IRWXO    00007    // 其他用户权限
```

### 打开标志

```c
#define O_RDONLY   0        // 只读
#define O_WRONLY   1        // 只写
#define O_RDWR     2        // 读写
#define O_CREAT    0100     // 创建文件
#define O_TRUNC    01000    // 截断文件
#define O_EXCL     0200     // 独占创建
```

### 文件系统魔数

```c
#define EXT2_SUPER_MAGIC  0xEF53
```

### 块大小

```c
#define EXT2_MIN_BLOCK_LOG_SIZE  10  // 1024 字节
#define EXT2_MAX_BLOCK_LOG_SIZE  16  // 65536 字节
```
