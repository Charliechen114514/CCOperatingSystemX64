## 1. 进程状态查看

### 1.1 打印当前进程

```c
#include "process/process.h"

void print_current_process(void) {
    pcb_t* current = proc_current();

    if (!current) {
        klog_info("[PROC] No current process\n");
        return;
    }

    const char* state_names[] = {
        [PROC_READY] = "READY",
        [PROC_RUNNING] = "RUNNING",
        [PROC_BLOCKED] = "BLOCKED",
        [PROC_ZOMBIE] = "ZOMBIE",
    };

    klog_info("[PROC] Current Process:\n");
    klog_info("  PID:      %d\n", current->pid);
    klog_info("  PPID:     %d\n", current->ppid);
    klog_info("  State:    %s\n", state_names[current->state]);
    klog_info("  Command:  %s\n", current->comm);
    klog_info("  PML4:     0x%016llX\n", current->mm.pml4_phys);
    klog_info("  KStack:   0x%016llX\n", current->kernel_stack);
}
```

### 1.2 打印所有进程

```c
void print_all_processes(void) {
    klog_info("[PROC] All Processes:\n");

    // 打印当前进程
    if (scheduler.current) {
        print_current_process();
    }

    // 打印运行队列中的进程
    klog_info("[PROC] Run Queue (%u processes):\n", scheduler.nr_running);

    pcb_t* proc;
    int idx = 0;
    list_for_each_entry(proc, &scheduler.run_queue, run_list) {
        const char* state = (proc->state == PROC_READY) ? "READY" : "UNKNOWN";
        klog_info("  [%d] PID %d, PPID %d, State %s, Comm %s\n",
                  idx++, proc->pid, proc->ppid, state, proc->comm);
    }

    // 打印僵尸进程
    if (scheduler.current) {
        int zombie_count = 0;
        pcb_t* zombie;
        list_for_each_entry(zombie, &scheduler.current->zombie_children, zombie_children) {
            if (zombie_count == 0) {
                klog_info("[PROC] Zombie Children:\n");
            }
            klog_info("  PID %d, Exit Code %d\n", zombie->pid, zombie->exit_code);
            zombie_count++;
        }
    }
}
```

### 1.3 打印进程树

```c
void print_process_tree_r(pcb_t* root, int depth) {
    const char* state_chars[] = {
        [PROC_READY] = "R",
        [PROC_RUNNING] = "P",
        [PROC_BLOCKED] = "S",
        [PROC_ZOMBIE] = "Z",
    };

    for (int i = 0; i < depth; i++) {
        klog_info("  ");
    }

    const char* state = (root->state < 4) ? state_chars[root->state] : "?";
    klog_info("[%d] %s (%s)\n", root->pid, root->comm, state);

    pcb_t* child;
    list_for_each_entry(child, &root->children, siblings) {
        print_process_tree_r(child, depth + 1);
    }
}

void print_process_tree(void) {
    klog_info("\n╔════════════════════════════════════╗\n");
    klog_info("║       Process Tree                  ║\n");
    klog_info("╚════════════════════════════════════╝\n");

    // 从 PID 1 (init) 开始
    pcb_t* init = proc_find(1);
    if (init) {
        print_process_tree_r(init, 0);
    } else {
        klog_info("No init process found\n");
    }

    klog_info("\n");
}
```

## 2. PID 分配器调试

### 2.1 检查 PID 位图

```c
#include "bitmap/bitmap.h"

extern struct bitmap s_pid_bitmap;

void dump_pid_bitmap(void) {
    int allocated = 0;
    int free_count = 0;

    for (int i = 0; i < PID_MAX; i++) {
        if (bitmap_test(&s_pid_bitmap, i)) {
            allocated++;
            if (allocated <= 10) {
                klog_info("  Allocated PID: %d\n", i);
            }
        } else {
            free_count++;
        }
    }

    klog_info("\n╔════════════════════════════════════╗\n");
    klog_info("║       PID Allocator Status         ║\n");
    klog_info("╚════════════════════════════════════╝\n");
    klog_info("Total PIDs:     %d\n", PID_MAX);
    klog_info("Allocated:      %d\n", allocated);
    klog_info("Free:           %d\n", free_count);
    klog_info("Usage:          %d%%\n", (allocated * 100) / PID_MAX);
    klog_info("\n");
}
```

### 2.2 统计已分配 PID

```c
void count_allocated_pids(void) {
    extern struct bitmap s_pid_bitmap;
    int count = 0;

    for (int i = 0; i < PID_MAX; i++) {
        if (bitmap_test(&s_pid_bitmap, i)) {
            count++;
        }
    }

    klog_info("[PID] Allocated: %d / %d\n", count, PID_MAX);

    if (count > scheduler.nr_running) {
        klog_warn("[PID] Possible PID leak!\n");
        klog_warn("      Allocated PIDs > running processes\n");
    }
}
```

## 3. 上下文切换调试

### 3.1 在 switch_context 中设置断点

使用 GDB 在上下文切换处设置断点：

```bash
(gdb) break switch_context
```

```c
void dump_cpu_context(cpu_context_t* ctx) {
    if (!ctx) {

    klog_info("CPU Context:\n");
    klog_info("  RBX: 0x%016llX\n", ctx->rbx);

    klog_info("  R12: 0x%016llX\n", ctx->r12);
```

### 3.3 验证栈指针

```c
void validate_stack_pointer(pcb_t* pcb) {
    if (!pcb) return;

    virtual_addr_t stack_base = pcb->kernel_stack_base;
    virtual_addr_t stack_top = pcb->kernel_stack;
    virtual_addr_t current_sp = pcb->cpu_ctx->rsp;

    klog_info("[STACK] Validation for PID %d:\n", pcb->pid);
    klog_info("  Base:  0x%016llX\n", stack_base);
    klog_info("  Top:   0x%016llX\n", stack_top);
    klog_info("  Current: 0x%016llX\n", current_sp);

    if (current_sp < stack_base || current_sp > stack_top) {
        klog_error("[STACK] Stack pointer out of bounds!\n");
    }

    // 检查对齐
    if ((current_sp & 0xF) != 0) {
        klog_warn("[STACK] Stack misaligned (should be 16-byte)\n");
    }
}
```

### 3.4 跟踪上下文切换

```c
extern void switch_context(pcb_t** prev, pcb_t* next);

void traced_schedule(void) {
    pcb_t* prev = scheduler.current;
    pcb_t* next = NULL;

    if (list_is_empty(&scheduler.run_queue)) {
        klog_warn("[SCHED] Run queue empty!\n");
        return;
    }

    next = list_first_entry(&scheduler.run_queue, pcb_t, run_list);

    klog_info("[SCHED] Context Switch:\n");
    klog_info("  Prev: PID %d (0x%016llX)\n",
              prev ? prev->pid : -1,
              prev ? prev->kernel_stack : 0);
    klog_info("  Next: PID %d (0x%016llX)\n",
              next->pid,
              next->kernel_stack);

    switch_context(&prev, next);
}
```

## 4. TSS 调试

### 4.1 打印 TSS 状态

TSS 模块提供了 `tss_dump()` 函数：

```c
#include "interrupt/tss.h"

void debug_tss(void) {
    klog_info("\n╔════════════════════════════════════╗\n");
    klog_info("║         TSS Status                  ║\n");
    klog_info("╚════════════════════════════════════╝\n");
    tss_dump();

    // 与当前进程对比
    if (scheduler.current) {
        extern tss_t s_tss;
        klog_info("Current process kernel_stack: 0x%016llX\n",
                  scheduler.current->kernel_stack);
        klog_info("TSS RSP0:                  0x%016llX\n",
                  s_tss.rsp0);

        if (s_tss.rsp0 != scheduler.current->kernel_stack) {
            klog_error("TSS RSP0 mismatch!\n");
        }
    }
}
```

### 4.2 检查 RSP0 值

```c
void check_tss_rsp0(void) {

    klog_info("[TSS] RSP2: 0x%016llX\n", s_tss.rsp2);

    // 检查 IST 栈
```

```c
#include "mm/vmm/vmm.h"

void print_current_cr3(void) {
```

```c
void verify_address_space_separation(void) {
    klog_info("\n╔════════════════════════════════════╗\n");
    klog_info("║   Address Space Separation Check   ║\n");
    klog_info("╚════════════════════════════════════╝\n");

    // 检查区域是否不重叠
    if (KERNEL_HEAP_MAX > KERNEL_GENERAL_BASE) {
        klog_error("Heap and General regions overlap!\n");
    } else {
        klog_info("  Heap and General regions: OK\n");
    }

    // 检查区域对齐

        klog_warn("Heap base not 1MB aligned\n");
    }

    klog_info("\n");
}
```

## 8. 日志分析

### 8.1 启用详细日志

```c
#include "klogs/kprintf.h"

void enable_verbose_process_logging(void) {
    klog_set_level(LOG_LEVEL_TRACE);

    klog_trace("[PROC] Verbose logging enabled\n");
    klog_trace("[PROC] PID allocations will be logged\n");
    klog_trace("[PROC] Context switches will be logged\n");
}
```

### 8.2 追踪进程切换

在 `schedule()` 函数中添加日志：

```c
void schedule(void) {
    pcb_t* prev = scheduler.current;
    pcb_t* next = list_first_entry(&scheduler.run_queue, pcb_t, run_list);

    klog_trace("[SCHED] Switch: PID %d -> PID %d\n",
               prev ? prev->pid : -1, next->pid);
    klog_trace("[SCHED] Next RIP: 0x%016llX\n", next->cpu_ctx->rip);
    klog_trace("[SCHED] Next RSP: 0x%016llX\n", next->cpu_ctx->rsp);
    klog_trace("[SCHED] Next PML4: 0x%016llX\n", next->mm.pml4_phys);

    switch_context(&prev, next);

    klog_trace("[SCHED] Returned to PID %d\n", scheduler.current->pid);
}
```

### 8.3 统计调度次数

```c
static uint64_t schedule_count = 0;
static uint64_t context_switch_count = 0;

void schedule_with_stats(void) {
    schedule_count++;

    if (scheduler.current && scheduler.current->pid != 0) {
        context_switch_count++;
        klog_info("[STAT] Schedule #%llu, Switch #%llu\n",
                  schedule_count, context_switch_count);
    }

    schedule();
}
```

## 9. 性能分析

### 9.1 测量上下文切换时间

```c
#include "interrupt/rtc.h"

uint64_t measure_context_switch_time(void) {
    uint64_t start = rtc_get_nanoseconds();

    // 执行多次上下文切换
    for (int i = 0; i < 100; i++) {
        schedule();
    }

    uint64_t end = rtc_get_nanoseconds();
    return (end - start) / 100;  // 平均每次切换时间
}
```

### 9.2 统计调度次数

```c
typedef struct {
    uint64_t total_schedules;
    uint64_t context_switches;
    uint64_t idle_schedules;
    uint64_t proc_fork_calls;
    uint64_t proc_exit_calls;
} proc_stats_t;

static proc_stats_t g_proc_stats = {0};

void print_proc_statistics(void) {
    klog_info("\n╔════════════════════════════════════╗\n");
    klog_info("║     Process Statistics              ║\n");
    klog_info("╚════════════════════════════════════╝\n");
    klog_info("Total Schedules:    %llu\n", g_proc_stats.total_schedules);
    klog_info("Context Switches:   %llu\n", g_proc_stats.context_switches);
    klog_info("Idle Schedules:     %llu\n", g_proc_stats.idle_schedules);
    klog_info("Fork Calls:         %llu\n", g_proc_stats.proc_fork_calls);
    klog_info("Exit Calls:         %llu\n", g_proc_stats.proc_exit_calls);

    if (g_proc_stats.total_schedules > 0) {
        uint64_t switch_percent = (g_proc_stats.context_switches * 100) /
                                  g_proc_stats.total_schedules;
        klog_info("Switch Ratio:        %llu%%\n", switch_percent);
    }
}
```

```c
void print_process_memory_usage(void) {
    size_t pcb_size = sizeof(pcb_t);
    size_t cpu_ctx_size = sizeof(cpu_context_t);
    size_t total_per_process = pcb_size + cpu_ctx_size + KERNEL_STACK_SIZE;

    klog_info("\n╔════════════════════════════════════╗\n");

    klog_info("╚════════════════════════════════════╝\n");
    klog_info("PCB Size:              %zu bytes\n", pcb_size);
    klog_info("CPU Context Size:      %zu bytes\n", cpu_ctx_size);
    klog_info("Kernel Stack Size:     %u bytes\n", KERNEL_STACK_SIZE);
    klog_info("Total Per Process:     %zu bytes\n", total_per_process);
    klog_info("Running Processes:     %u\n", scheduler.nr_running);
    klog_info("Total Process Memory:  %zu KB\n",
              (total_per_process * scheduler.nr_running) / 1024);
}
```
