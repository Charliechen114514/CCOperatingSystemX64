### Ring 3 切换

#### user_switch_to_usermode

```c
/**
 * user_switch_to_usermode - 切换到用户态
 * @param ctx 用户上下文
 *
 * 此函数不返回。它使用 iretq 切换到 Ring 3。
 *
 * user_context_t 结构：
 *   - entry: 用户 RIP
 *   - stack_top: 用户 RSP
 *   - cs: 用户 CS (0x1B)
 *   - ss: 用户 SS (0x23)
 *   - rflags: 用户 RFLAGS (IF=1)
 */
void user_switch_to_usermode(user_context_t* ctx) __attribute__((noreturn));
```

**使用示例**：
```c
user_context_t ctx = {
    .entry = 0x400000,
    .stack_top = user_pcb->user_stack,
    .cs = USER_CS,
    .ss = USER_SS,
    .rflags = 0x202  /* IF = 1 */
};

user_switch_to_usermode(&ctx);  // 不返回
```

## 算法说明

### 1. 用户指针验证算法

```c
/**
 * user_validate_pointer - 验证用户指针
 *
 * 算法：
 * 1. 检查地址是否在用户空间范围
 * 2. 检查加法是否溢出
 * 3. 检查范围是否超出用户空间
 * 4. 检查是否在 NULL 页保护区域
 */
bool user_validate_pointer(const void* ptr, size_t size, bool write) {
    virtual_addr_t addr = (virtual_addr_t)ptr;

    /* 1. 检查用户空间范围 */
    if (!vmm_is_user_addr(addr)) {
        return false;
    }

    /* 2. 检查溢出 */
    if (addr + size < addr) {
        return false;
    }

    /* 3. 检查超出用户空间 */
    if (addr + size > USER_END) {
        return false;
    }

    /* 4. 检查 NULL 页保护 */
    if (addr < USER_BASE) {
        return false;
    }

    return true;
}
```

### 2. Ring 0 到 Ring 3 切换算法

```c
/**
 * user_switch_to_usermode - 切换到用户态
 *
 * 算法：
 * 1. 加载用户上下文到寄存器
 * 2. 设置 iretq 栈帧 (SS, RSP, RFLAGS, CS, RIP)
 * 3. 设置数据段寄存器为用户段
 * 4. 执行 iretq 切换特权级
 *
 * iretq 行为：
 * - 弹出 RIP 到指令指针
 * - 弹出 CS 到代码段 (CPL=3)
 * - 弹出 RFLAGS 到标志寄存器
 * - 弹出 RSP 到栈指针
 * - 弹出 SS 到栈段
 */
```

**汇编实现**：
```assembly
; RDI = ctx
mov rax, [rdi + user_context.entry]      ; 用户 RIP
mov rbx, [rdi + user_context.stack_top]  ; 用户 RSP
mov rcx, [rdi + user_context.cs]         ; 用户 CS
mov rdx, [rdi + user_context.ss]         ; 用户 SS
mov rsi, [rdi + user_context.rflags]     ; 用户 RFLAGS

; iretq 栈帧
push rdx        ; SS
push rbx        ; RSP
push rsi        ; RFLAGS
push rcx        ; CS
push rax        ; RIP

; 数据段
mov ax, 0x23
mov ds, ax
mov es, ax
mov fs, ax
mov gs, ax

iretq
```

### 3. 用户栈分配算法

```c
/**
 * user_create_process - 分配用户栈
 *
 * 算法：
 * 1. 计算栈大小页数 (1MB = 256 页)
 * 2. 对齐栈地址到页边界
 * 3. 循环分配物理帧
 * 4. 映射到用户空间
 * 5. 清零栈页
 * 6. 设置 is_user_mode 标志
 */
int user_create_process(virtual_addr_t entry, pcb_t* pcb) {
    size_t stack_pages = USER_STACK_SIZE / PAGE_SIZE;
    pcb->user_stack = (USER_END - USER_STACK_SIZE) & ~(PAGE_SIZE - 1);

    for (size_t i = 0; i < stack_pages; i++) {
        physical_addr_t paddr;
        pframe_alloc(&paddr);

        vmm_map_to_user(pcb->mm.pml4_phys,
                       pcb->user_stack + (i * PAGE_SIZE),
                       paddr, 1,
                       VMAP_FLAG_WRITE | VMAP_FLAG_USER);

        memset((void*)phys_to_virt_offset(paddr), 0, PAGE_SIZE);
    }

    pcb->is_user_mode = true;
    return 0;
}
```

### 4. brk 算法

```c
/**
 * user_brk - 程序断点管理
 *
 * 算法：
 * 1. NULL 参数查询当前断点
 * 2. 验证新断点在有效范围
 * 3. 缩小时直接更新
 * 4. 扩大时分配新页并映射
 */
virtual_addr_t user_brk(pcb_t* pcb, virtual_addr_t new_brk) {
    /* 1. 查询 */
    if (new_brk == 0) {
        return pcb->mm.brk;
    }

    /* 2. 验证 */
    new_brk = (new_brk + PAGE_SIZE - 1) & ~(PAGE_SIZE - 1);
    if (new_brk < USER_BASE || new_brk >= (USER_END - USER_STACK_SIZE)) {
        return pcb->mm.brk;
    }

    /* 3. 缩小 */
    if (new_brk <= pcb->mm.brk) {
        pcb->mm.brk = new_brk;
        return new_brk;
    }

    /* 4. 扩大 */
    size_t old_brk = pcb->mm.brk ? pcb->mm.brk : (USER_BASE + 16*1024*1024);
    size_t additional_pages = (new_brk - old_brk) / PAGE_SIZE;

    for (size_t i = 0; i < additional_pages; i++) {
        physical_addr_t paddr;
        pframe_alloc(&paddr);

        vmm_map_to_user(pcb->mm.pml4_phys,
                       old_brk + (i * PAGE_SIZE),
                       paddr, 1,
                       VMAP_FLAG_WRITE | VMAP_FLAG_USER);
    }

    pcb->mm.brk = new_brk;
    return new_brk;
}
```

## x86_64 特权级切换参考

### syscall 指令 (用户 → 内核)

```assembly
; 用户态执行
mov rax, 60     ; syscall 号 (SYS_EXIT)
mov rdi, 0      ; 退出码
syscall         ; 切换到 Ring 0

; CPU 自动执行：
; 1. 保存 RCX → 用户 RIP
; 2. 保存 R11 → 用户 RFLAGS
; 3. 从 IA32_STAR MSR 加载内核 CS:RIP
; 4. 切换到 Ring 0
```

### iretq 指令 (内核 → 用户)

```assembly
; 内核态执行
push 0x23       ; 用户 SS
push user_rsp   ; 用户 RSP
push 0x202       ; 用户 RFLAGS (IF=1)
push 0x1B        ; 用户 CS
push user_rip   ; 用户 RIP
iretq           ; 切换到 Ring 3

; CPU 自动执行：
; 1. 弹出 RIP → 用户指令指针
; 2. 弹出 CS → 用户代码段 (CPL=3)
; 3. 弹出 RFLAGS → 用户标志
; 4. 弹出 RSP → 用户栈指针
; 5. 弹出 SS → 用户栈段
```
