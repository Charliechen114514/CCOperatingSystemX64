# 内存操作与串口驱动 - 调试工具指南

本文档详细说明内存操作和串口驱动阶段的调试工具和最佳实践。

---

## 1. 串口调试工具

### 1.1 QEMU 串口重定向

#### 重定向到 stdio（推荐）

```bash
# 最常用的调试配置
qemu-system-x86_64 -drive format=raw,file=build/boot.img \
    -nographic -serial mon:stdio

# 输出示例：
# CCOS Bootloader v1.0
# [Serial] Bootloader initialized
# [Serial] Loading kernel...
# [Serial] Kernel entry at 0x10000
```

#### 分离监控器和串口

```bash
# 监控器在 stdio，串口在文件
qemu-system-x86_64 -drive format=raw,file=build/boot.img \
    -serial mon:stdio -serial file:serial.log

# 监控器在 stdio，串口在另一个终端
qemu-system-x86_64 -drive format=raw,file=build/boot.img \
    -monitor stdio -serial pty
# 然后在另一个终端连接显示的 pty 设备
```

### 1.2 使用 screen/minicom 连接串口

```bash
# 方法 1: 使用 screen
screen /dev/ttyS0 115200
# 退出: Ctrl+A, K

# 方法 2: 使用 minicom
minicom -D /dev/ttyS0 -b 115200
# 退出: Ctrl+A, Q
```

### 1.3 串口输出捕获到文件

```bash
# QEMU 启动时重定向
qemu-system-x86_64 -drive format=raw,file=build/boot.img \
    -serial file:serial.log

# 或使用 tee 同时显示和保存
qemu-system-x86_64 -drive format=raw,file=build/boot.img \
    -serial mon:stdio 2>&1 | tee serial.log

# 分析日志
grep "ERROR" serial.log
grep "Kernel" serial.log
```

## 2. 内存调试工具

### 2.1 GDB 内存检查

```gdb
# 查看内存内容
(gdb) x/20bx 0x7c00        # 20 字节十六进制
(gdb) x/10hd 0x10000       # 10 个半字（2字节）
(gdb) x/5wd 0x20000        # 5 个字（4字节）
(gdb) x/10i 0x7c00         # 10 条指令

# 搜索内存
(gdb) find 0x7c00, +0x10000, "CCOS"
(gdb) find 0x10000, +0x1000, 0x90

# 填充内存
(gdb) set {char[10]}0x10000 = "test"
(gdb) set {int}0x10000 = 0x12345678
```

### 2.2 使用 Valgrind 检测内存问题

```bash
# 内存泄漏检测
valgrind --leak-check=full ./build/test/test_memory

# 检测非法内存访问
valgrind ./build/test/test_memory

# 详细报告
valgrind --leak-check=full --show-leak-kinds=all \
    --track-origins=yes ./build/test/test_memory

# 检测重叠内存操作
valgrind --memcheck:overlap=Yes ./build/test/test_memory
```

### 2.3 内存函数单元测试

```c
// 测试框架使用
static void test_memset_overlap(void) {
    TEST_INFO("Testing memset with overlapping regions");

    char buffer[100];

    // 测试基本填充
    memset(buffer, 0x42, 100);
    for (int i = 0; i < 100; i++) {
        TEST_ASSERT_EQ(buffer[i], 0x42, "memset fill");
    }

    // 测试零长度
    memset(buffer, 0xAA, 0);
    TEST_ASSERT_EQ(buffer[0], 0x42, "memset zero length no change");
}
```

## 3. 硬件调试工具

### 3.1 QEMU Monitor 命令

```

# 查看物理内存
```

### 3.3 端口 I/O 调试

## 4. Bootloader 调试

### 4.1 实模式调试

```gdb
# 启动 GDB 连接到 QEMU
gdb build/kernel.elf
(gdb) target remote :1234
(gdb) set architecture i8086

# 在实模式入口断点
(gdb) break *0x7c00
(gdb) continue

# 查看代码
(gdb) disassemble
(gdb) x/10i $pc

# 查看寄存器
(gdb) info registers
```

### 4.2 模式切换调试

```gdb
# 在保护模式切换点
(gdb) break enable_protected_mode
(gdb) continue

# 检查 CR0 寄存器
(gdb) print/x $cr0
(gdb) print/t $cr0  # 二进制显示

# 检查段寄存器
(gdb) info registers cs ds es fs gs ss

# 在长模式
(gdb) break kernel_entry
(gdb) continue
(gdb) set architecture i386:x86-64
```

```c
    uint64_t tsc;
    __asm__ volatile("rdtsc" : "=A"(tsc));
    return tsc;
}
```

### 5.2 串口输出性能优化

```c
// 批量发送（避免逐字符等待）
void serial_write_chunk(const char* data, size_t len) {
    const uint16_t base = SERIAL_COM1_BASE;

    for (size_t i = 0; i < len; i++) {
        while ((inb(SERIAL_LINE_STATUS_REG(base)) & 0x20) == 0) {
            // 等待就绪
        }
        outb(SERIAL_DATA_REG(base), data[i]);
    }
}

// FIFO 优化（启用硬件缓冲）
void serial_enable_fifo(void) {
    const uint16_t base = SERIAL_COM1_BASE;
    outb(SERIAL_FIFO_CTRL_REG(base), 0xC7);  // 启用 FIFO，清空，14 字节阈值
}
```

## 6. 常见调试场景

### 6.1 调试 memcpy 重叠问题

```gdb
# 在 memcpy 中设置断点
(gdb) break memcpy

# 检查参数
(gdb) print dest
(gdb) print src
(gdb) print n

# 检查是否重叠
(gdb) print dest < src && dest + n > src

# 使用 memmove 临时替换
(gdb) call memmove(dest, src, n)
```

### 6.2 调试串口初始化

```gdb
# 在 serial_init 中单步
(gdb) break serial_init
(gdb) continue

# 逐步检查每个端口写入
(gdb) next
(gdb) print/x inb(0x3FB)  # 读取 LCR
(gdb) print/x inb(0x3F9)  # 读取 IER

# 验证波特率除数
(gdb) print/x inb(0x3F8)  # DLL
(gdb) print/x inb(0x3F9)  # DLM
```

### 6.3 调试 VGA 和串口同步

```c
// 添加同步标记
void debug_sync_puts(const char* str) {
```

## 7. 调试最佳实践

### 7.1 渐进式调试

```
1. 先在主机测试内存函数
   ↓
2. 然后在内核中测试内存函数
   ↓
3. 接着测试串口初始化
   ↓
4. 最后测试串口输出
```

### 7.2 使用颜色编码输出

```c
// 定义颜色宏
#define SERIAL_COLOR_RED     "\033[0;31m"
#define SERIAL_COLOR_GREEN   "\033[0;32m"
#define SERIAL_COLOR_YELLOW  "\033[0;33m"
#define SERIAL_COLOR_RESET   "\033[0m"

// 使用
sync_serial_puts(SERIAL_COLOR_GREEN "[OK]" SERIAL_COLOR_RESET " Test passed\n");
sync_serial_puts(SERIAL_COLOR_RED "[FAIL]" SERIAL_COLOR_RESET " Test failed\n");
sync_serial_puts(SERIAL_COLOR_YELLOW "[WARN]" SERIAL_COLOR_RESET " Warning\n");
```

### 7.3 调试宏

```c
// 条件编译的调试输出
#ifdef SERIAL_DEBUG
    #define SERIAL_DBG(fmt, ...) \
        do { \
```

## 8. 快速参考

### QEMU 串口选项

| 选项 | 功能 |
|------|------|
| `-serial mon:stdio` | 监控器+串口在 stdio |
| `-serial file:log` | 输出到文件 |
| `-serial pty` | 使用伪终端 |
| `-nographic` | 禁用图形输出 |
| `-display none` | 完全无显示 |

### 串口寄存器速查

| 端口 | 读/写 | 功能 |
|------|-------|------|
| 0x3F8 | W | THR (发送保持) |
| 0x3F8 | R | RBR (接收缓冲) |
| 0x3F9 | W | IER (中断使能) |
| 0x3F9 | R | IIR (中断识别) |
| 0x3FA | R | IIR (只读) |
| 0x3FB | W | LCR (线路控制) |
| 0x3FC | W | MCR (调制解调器) |
| 0x3FD | R | LSR (线路状态) |
| 0x3FE | R | MSR (调制解调器状态) |

### LSR 位定义

| 位 | 名称 | 功能 |
|----|------|------|
| 0 | DR | 数据就绪 |
| 5 | THRE | 发送保持为空 |
| 6 | TEMT | 发送器为空 |

### GDB 常用命令

| 命令 | 功能 |
|------|------|
| `target remote :1234` | 连接到 QEMU GDB |
| `break *0xaddress` | 在地址设置断点 |
| `x/20bx addr` | 查看内存字节 |
| `info registers` | 显示寄存器 |
| `set architecture i386:x86-64` | 设置为 64 位模式 |
| `call inb(0x3F8)` | 调用函数 |
